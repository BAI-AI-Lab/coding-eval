class Solution:
    def maxActiveSectionsAfterTrade(self, s: str) -> int:
        t = "1" + s + "1"

        # Compress t into alternating runs of identical characters.
        runs = []
        i = 0
        while i < len(t):
            j = i + 1
            while j < len(t) and t[j] == t[i]:
                j += 1
            runs.append((t[i], j - i))
            i = j

        initial = s.count("1")
        answer = initial

        # An internal '1' run can be removed. Filling the resulting zero run
        # gives a net gain equal to the lengths of its two adjacent zero runs.
        for i in range(1, len(runs) - 1):
            if runs[i][0] == "1":
                gain = runs[i - 1][1] + runs[i + 1][1]
                answer = max(answer, initial + gain)

        return answer

class Solution:
    def maxActiveSectionsAfterTrade(self, s: str) -> int:
        total_ones = s.count('1')
        
        # Augment with '1' at both ends.
        t = '1' + s + '1'
        
        # Run-length encode t.
        runs = []
        prev = t[0]
        count = 1
        
        for ch in t[1:]:
            if ch == prev:
                count += 1
            else:
                runs.append((prev, count))
                prev = ch
                count = 1
        runs.append((prev, count))
        
        best_gain = 0
        
        # An internal '1' block is surrounded by '0' blocks.
        for i in range(1, len(runs) - 1):
            if runs[i][0] == '1':
                left_zero_count = runs[i - 1][1]
                right_zero_count = runs[i + 1][1]
                best_gain = max(best_gain, left_zero_count + right_zero_count)
        
        return total_ones + best_gain

class Solution:
    def maxActiveSectionsAfterTrade(self, s: str) -> int:
        runs = []
        i = 0
        n = len(s)
        while i < n:
            c = s[i]
            j = i
            while j < n and s[j] == c:
                j += 1
            runs.append((c, j - i))
            i = j

        total_ones = 0
        for c, length in runs:
            if c == '1':
                total_ones += length

        best_gain = 0
        m = len(runs)
        for i in range(1, m - 1):
            if runs[i][0] == '1':
                # runs[i-1] and runs[i+1] are guaranteed to be '0' due to alternating runs
                best_gain = max(best_gain, runs[i - 1][1] + runs[i + 1][1])

        return total_ones + best_gain
