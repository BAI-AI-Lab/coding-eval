题面

来源：USACO 官网。模型收到的是同一份题面的文字。

It's the year $3000$, and Bessie became the first cow in space! During her journey between the stars, she found a number line with $N$ ($2 \leq N \leq 5 \cdot 10^5$) points, numbered from $1$ to $N$. All points are initially colored white. She can perform the following operation any number of times.

  • Choose a position $i$ within the number line and a positive integer $x$. Then, color all the points in the interval $[i, i + x - 1]$ red and all points in $[i + x, i + 2x - 1]$ blue. All chosen intervals must be disjoint (i.e. no points in $[i, i + 2x - 1]$ can be already colored red or blue). The entire interval must also fall within the number line (i.e. $1 \leq i \leq i + 2x - 1 \leq N$).

Farmer John gives Bessie a string $s$ of length $N$ consisting of characters $R$, $B$, and $X$. The string represents Farmer John's color preferences for each point: $s_i=R$ means the $i$'th point must be colored red, $s_i = B$ means the $i$'th point must be colored blue, and $s_i = X$ means there is no constraint on the color for the $i$'th point.

Help Bessie count the number of distinct ways for the number line to be colored while satisfying Farmer John's preferences. Two colorings are different if there is at least one corresponding point with a different color. Because the answer may be large, output it modulo $10^9+7$.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains an integer $N$.

The following line contains string $s$.

OUTPUT FORMAT (print output to the terminal / stdout):

Output the number of distinct ways for the number line to be colored while satisfying Farmer John's preferences modulo $10^9+7$.

SAMPLE INPUT:

6
RXXXXB

SAMPLE OUTPUT:

5
Bessie can choose $i=1,x=1$ (i.e. color point $1$ red and point $2$ blue) and $i=3,x=2$ (i.e. color points $3,4$ red and points $5,6$ blue) to produce the coloring $RBRRBB$.

The other colorings are $RRBBRB$, $RBWWRB$, $RRRBBB$, and $RBRBRB$.

SAMPLE INPUT:

6
XXRBXX

SAMPLE OUTPUT:

6
The six colorings are $WWRBWW$, $WWRBRB$, $WRRBBW$, $RBRBWW$, $RBRBRB$, and $RRRBBB$.

SAMPLE INPUT:

12
XBXXXXRXRBXX

SAMPLE OUTPUT:

18

SCORING:

  • Input 4: $N\le 500$
  • Inputs 5-6: $N\le 10^4$
  • Inputs 7-13: All but at most $100$ characters in $s$ are $X$.
  • Inputs 14-23: No additional constraints

#include <bits/stdc++.h>
using namespace std;

static const int MOD = 1000000007;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    string s;
    cin >> N >> s;

    vector<int> nextB(N + 1, N), nextR(N + 1, N), prevR(N, -1);
    for (int i = N - 1; i >= 0; --i) {
        nextB[i] = (s[i] == 'B' ? i : nextB[i + 1]);
        nextR[i] = (s[i] == 'R' ? i : nextR[i + 1]);
    }
    int last = -1;
    for (int i = 0; i < N; ++i) {
        if (s[i] == 'R') last = i;
        prevR[i] = last;
    }

    vector<int> dp(N + 1, 0);
    array<vector<int>, 2> diff;
    diff[0].assign(N + 3, 0);
    diff[1].assign(N + 3, 0);

    long long active[2] = {0, 0};
    dp[0] = 1;

    for (int p = 0; p <= N; ++p) {
        int parity = p & 1;
        active[parity] += diff[parity][p];
        if (active[parity] >= MOD) active[parity] -= MOD;

        int val = dp[p] + (int)active[parity];
        if (val >= MOD) val -= MOD;
        dp[p] = val;

        if (p == N) break;
        if (val == 0) continue;

        if (s[p] == 'X') {
            dp[p + 1] += val;
            if (dp[p + 1] >= MOD) dp[p + 1] -= MOD;
        }

        int U = min((N - p) / 2, nextB[p] - p);
        if (U <= 0) continue;

        auto &D = diff[parity];

        // Add val to dp[p+2*x] for every x in [a,b].
        auto addInterval = [&](int a, int b) {
            if (a > b) return;
            int l = p + 2 * a;
            int r = p + 2 * b;

            int z = D[l] + val;
            if (z >= MOD) z -= MOD;
            D[l] = z;

            z = D[r + 2] - val;
            if (z < 0) z += MOD;
            D[r + 2] = z;
        };

        /*
         * For an R at distance d from p, the operation is invalid exactly
         * for half-lengths x in [d/2+1, d]. Only d <= 2*U-1 matters.
         * Merge these intervals; their union has O(log N) components.
         */
        int lastPos = p + 2 * U - 1;
        int rpos = nextR[p + 1];
        int cursor = 1;

        while (rpos <= lastPos) {
            int d = rpos - p;
            int L = d / 2 + 1;

            addInterval(cursor, L - 1);

            int c = d;
            while (c < U) {
                int far = prevR[p + 2 * c + 1];
                if (far > p + c) {
                    c = far - p;
                } else {
                    break;
                }
            }

            if (c >= U) {
                cursor = U + 1;
                break;
            }

            cursor = c + 1;
            int nextPos = p + 2 * c + 2;
            rpos = (nextPos > lastPos ? N : nextR[nextPos]);
        }

        addInterval(cursor, U);
    }

    cout << dp[N] << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

static constexpr int MOD = 1000000007;

struct Fenwick {
    int n = 0;
    vector<int> tree;

    void init(int size) {
        n = size;
        tree.assign(n + 1, 0);
    }

    // Adds val to source index pos (0-based).
    void add(int pos, int val) {
        for (int i = pos + 1; i <= n; i += i & -i) {
            tree[i] += val;
            if (tree[i] >= MOD) tree[i] -= MOD;
        }
    }

    // Sum of source indices 0 through cnt-1.
    int prefix(int cnt) const {
        long long result = 0;
        for (int i = cnt; i > 0; i -= i & -i) {
            result += tree[i];
        }
        return static_cast<int>(result % MOD);
    }
};

    int activeSum[2] = {0, 0};

            bit[p].add(a, MOD - dp[a]);
            activeSum[p] -= dp[a];
            if (activeSum[p] < 0) activeSum[p] += MOD;
        }

        // A source a becomes available for tokens ending at b = a + 2.
        if (b >= 2) {
            int a = b - 2;
            if (dp[a] != 0) {
                int t = nextB[a + 1];
                long long expiration;
                if (t == N + 1) {
                    expiration = N + 1LL;

                    expiration = 2LL * t - a - 1;
                }

                if (expiration >= b) {
                    int p = a & 1;
                    bit[p].add(a, dp[a]);
                    activeSum[p] += dp[a];

                    if (expiration + 1 <= N) {
                        removals[static_cast<int>(expiration + 1)].push_back(a);
                    }
                }
            }

        int lower = max(0, 2 * lastR - b);

        if (ways < 0) ways += MOD;

        if (s[b - 1] == 'X') {
            ways += dp[b - 1];
            if (ways >= MOD) ways -= MOD;
        }

        dp[b] = static_cast<int>(ways);

#include <bits/stdc++.h>
using namespace std;

const int MOD = 1'000'000'007;

struct Run {
    int a, b, len, val;
};

int N, M;
vector<Run> runs;
vector<int> seg;
vector<int> pref0, pref1;
int curI, curL, curBlock;

void buildSeg(int node, int l, int r) {
    if (l == r) {
        seg[node] = runs[l].val;
        return;
    }
    int mid = (l + r) >> 1;
    buildSeg(node << 1, l, mid);
    buildSeg(node << 1 | 1, mid + 1, r);
    seg[node] = max(seg[node << 1], seg[node << 1 | 1]);
}

void addRun(int idx) {
    const Run &r = runs[idx];

    int lo = max(r.a, curI - curL);
    lo = max(lo, (curI + r.a) / 2);

    int hi = min(r.b, curI - 1);
    if (lo > hi) return;

    int pL = 2 * lo - curI;
    int pH = 2 * hi - curI;

    if (pH < 0) return;
    if (pL < 0) pL = 0;

    int total;
    if (curI & 1) {
        total = pref1[pH];
        if (pL > 0) total -= pref1[pL - 1];
    } else {
        total = pref0[pH];
        if (pL > 0) total -= pref0[pL - 1];
    }

    if (total < 0) total += MOD;

    curBlock += total;
    if (curBlock >= MOD) curBlock -= MOD;
}

void processRange(int node, int l, int r, int ql, int qr, int threshold) {
    if (r < ql || l > qr || seg[node] < threshold) return;

    if (l == r) {
        addRun(l);
        return;
    }

    int mid = (l + r) >> 1;
    processRange(node << 1, l, mid, ql, qr, threshold);
    processRange(node << 1 | 1, mid + 1, r, ql, qr, threshold);
}

#include <bits/stdc++.h>
using namespace std;

static const int MOD = 1000000007;

struct BIT {
    int n;
    vector<int> tree;

    BIT() {}
    BIT(int n) { init(n); }

    void init(int n_) {
        n = n_;
        tree.assign(n + 1, 0);
    }

    void add(int idx, int val) {
        if (val < 0) val += MOD;
        for (; idx <= n; idx += idx & -idx) {
            tree[idx] += val;
            if (tree[idx] >= MOD) tree[idx] -= MOD;
        }
    }

    int sum(int idx) const {
        int res = 0;
        for (; idx > 0; idx -= idx & -idx) {
            res += tree[idx];
            if (res >= MOD) res -= MOD;
        }
        return res;
    }

    int rangeSum(int l, int r) const {
        if (l > r) return 0;
        int res = sum(r) - sum(l - 1);
        if (res < 0) res += MOD;
        return res;
    }
};

        for (int L : removeAt[i]) {

            bit[L & 1].add(L, -w);

            int w = dp[Lins - 1];
            bit[Lins & 1].add(Lins, w);

#include <bits/stdc++.h>
using namespace std;

static const int MOD = 1000000007;

struct Fenwick {
    int n;
    vector<int> bit;

    Fenwick() : n(0) {}
    Fenwick(int n) : n(n), bit(n + 1, 0) {}

    void add(int idx, int val) {
        for (int i = idx + 1; i <= n; i += i & -i) {
            int x = bit[i] + val;
            if (x >= MOD) x -= MOD;
            bit[i] = x;
        }
    }

    int sumPrefix(int idx) const {
        if (idx < 0) return 0;
        if (idx >= n) idx = n - 1;
        int res = 0;
        for (int i = idx + 1; i > 0; i -= i & -i) {
            res += bit[i];
            if (res >= MOD) res -= MOD;
        }
        return res;
    }

    int rangeSum(int l, int r) const {
        if (l > r) return 0;
        int res = sumPrefix(r) - sumPrefix(l - 1);
        if (res < 0) res += MOD;
        return res;
    }
};

    Fenwick fen[2] = {Fenwick(N + 1), Fenwick(N + 1)};

    vector<vector<int>> removals(N + 2);

    int lastR = -1;

    for (int i = 1; i <= N; ++i) {
        // Remove starts that are no longer usable for this end.
        for (int a : removals[i]) {
            int p = a & 1;
            int val = (MOD - dp[a]) % MOD;
            fen[p].add(a, val);
        }

        // Start a = i - 2 first becomes usable for a token of length 2.
        int aNew = i - 2;
        if (aNew >= 0 && expire[aNew] >= i) {
            int p = aNew & 1;
            fen[p].add(aNew, dp[aNew]);

            if (expire[aNew] + 1 <= N) {
                removals[expire[aNew] + 1].push_back(aNew);
            }
        }

        if (s[i - 1] == 'R') {
            lastR = i - 1;
        }

        // Blue half must contain no R.
        int lower = 2 * lastR - i + 1;
        if (lower < 0) lower = 0;

        int upper = i - 2;
        int tokenWays = 0;

        if (lower <= upper) {
            int p = i & 1;
            tokenWays = fen[p].rangeSum(lower, upper);
        }

        int ways = tokenWays;

        // Last atom is a single white point.
        if (s[i - 1] == 'X') {
            ways += dp[i - 1];
            if (ways >= MOD) ways -= MOD;
        }

        dp[i] = ways;
    }

    cout << dp[N] << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

constexpr int MOD = 1'000'000'007;

struct Fenwick {
    int n;
    int total = 0;
    vector<int> bit;

    explicit Fenwick(int size) : n(size), bit(size + 1, 0) {}

    void add(int index, int value) {
        if (value == 0) return;

        total += value;
        if (total >= MOD) total -= MOD;

        for (int i = index; i <= n; i += i & -i) {
            bit[i] += value;
            if (bit[i] >= MOD) bit[i] -= MOD;
        }
    }

    int prefixSum(int index) const {
        int result = 0;
        for (int i = index; i > 0; i -= i & -i) {
            result += bit[i];
            if (result >= MOD) result -= MOD;
        }
        return result;
    }

    int suffixSum(int firstIndex) const {
        if (firstIndex > n) return 0;

        int result = total - prefixSum(firstIndex - 1);
        if (result < 0) result += MOD;
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    string s;
    cin >> N >> s;
    s.insert(s.begin(), ' ');  // 1-indexed

    /*
       For each prefix boundary j, let b_j be the first fixed B strictly
       after j. A block starting at j+1 can end at t only while

           t < 2*b_j - j.

       expiration[j] is therefore the first endpoint where j becomes
       invalid. Expirations are stored in linked-list buckets.
    */
    vector<int> head(N + 2, -1);
    vector<int> nextInBucket(N + 1, -1);

    int nextB = N + 1;
    for (int j = N; j >= 0; --j) {
        // nextB is the first fixed B strictly after j.
        int expiration = 2 * nextB - j;
        if (expiration <= N) {
            nextInBucket[j] = head[expiration];
            head[expiration] = j;
        }

        // Position j becomes available for boundaries strictly before j.
        if (j >= 1 && s[j] == 'B') nextB = j;
    }

    /*
       Separate trees for even and odd prefix boundaries.
       Boundary j is stored at coordinate j/2 + 1 in its parity's tree.
    */
    Fenwick trees[2] = {
        Fenwick(N / 2 + 1),
        Fenwick((N + 1) / 2)
    };

    vector<int> dp(N + 1, 0);
    dp[0] = 1;
    trees[0].add(1, dp[0]);

    int lastR = 0;

    for (int t = 1; t <= N; ++t) {
        // These boundaries have just become invalid.
        for (int j = head[t]; j != -1; j = nextInBucket[j]) {
            if (dp[j] != 0) {
                trees[j & 1].add(j / 2 + 1, MOD - dp[j]);
            }
        }

        if (s[t] == 'R') lastR = t;

        /*
           A block covering positions j+1 ... t is valid only if

               j >= 2*lastR - t.

           Also j and t must have the same parity.
        */
        int lower = 2 * lastR - t;
        if (lower < 0) lower = 0;

        int parity = t & 1;
        if ((lower & 1) != parity) ++lower;

        int firstCoordinate = lower / 2 + 1;
        int ways = trees[parity].suffixSum(firstCoordinate);

        // Position t can instead remain white.
        if (s[t] == 'X') {
            ways += dp[t - 1];
            if (ways >= MOD) ways -= MOD;
        }

        dp[t] = ways;

        // This boundary may be used by future blocks.
        trees[parity].add(t / 2 + 1, dp[t]);
    }

    cout << dp[N] << '\n';
    return 0;
}

        if (k >= 1 && s[k] == 'B') nearest = k;
    }

       A block beginning at k+1 and ending at i has midpoint
       (k+i)/2. If its first fixed B is b, it remains valid with
       respect to B's until i reaches expiry[k] = 2*b-k.
    */
    vector<int> expiry(N + 1);
    vector<int> head(N + 1, -1), link(N + 1, -1);

        int b = nextB[k];
        expiry[k] = 2 * b - k;

            link[k] = head[expiry[k]];
            head[expiry[k]] = k;
        }

    int fenwickSize = N / 2 + 2;
    Fenwick fen[2] = {Fenwick(fenwickSize), Fenwick(fenwickSize)};

        if (i >= 2) {
            int k = i - 2;
            if (expiry[k] > i) {
                int parity = k & 1;
                fen[parity].add(k / 2 + 1, dp[k]);

                total[parity] += dp[k];
                if (total[parity] >= MOD) total[parity] -= MOD;

                active[k] = true;
            }
        }

        // Leave position i white.
        int answer = (s[i] == 'X' ? dp[i - 1] : 0);

        // End a red-blue block at position i.
        if (i >= 2) {
            int parity = i & 1;

            // The blue half contains no fixed R iff k >= 2*lastR-i.
            int lower = 2 * lastR - i;

            answer += contribution;
            if (answer >= MOD) answer -= MOD;

class Fenwick {
    int n;
    vector<int> tree;

public:
    explicit Fenwick(int n) : n(n), tree(n + 1, 0) {}

    void add(int index, int delta) {
        for (int p = index + 1; p <= n; p += p & -p) {
            tree[p] += delta;
            if (tree[p] >= MOD) tree[p] -= MOD;
        }
    }

    // Sum over indices [0, index].
    int prefix(int index) const {
        if (index < 0) return 0;
        index = min(index, n - 1);

        int result = 0;
        for (int p = index + 1; p > 0; p -= p & -p) {
            result += tree[p];
            if (result >= MOD) result -= MOD;
        }
        return result;
    }
};
