#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    vector<int> a(N + 1), cnt(N + 1, 0);
    for (int i = 1; i <= N; ++i) {
        cin >> a[i];
        ++cnt[a[i]];
    }

    // buckets[k] contains all cow numbers having exactly k votes.
    vector<set<int>> buckets(N + 1);
    set<int> levels; // Nonempty positive vote counts, in increasing order.

    int maxVotes = 0;
    for (int i = 1; i <= N; ++i) {
        if (cnt[i] > 0) {
            if (buckets[cnt[i]].empty()) levels.insert(cnt[i]);
            buckets[cnt[i]].insert(i);
            maxVotes = max(maxVotes, cnt[i]);
        }
    }

    auto addToBucket = [&](int cow, int votes) {
        if (buckets[votes].empty()) levels.insert(votes);
        buckets[votes].insert(cow);
    };

    auto removeFromBucket = [&](int cow, int votes) {
        buckets[votes].erase(cow);
        if (buckets[votes].empty()) levels.erase(votes);
    };

    vector<int> values, mnPos, mxPos;
    values.reserve(700);
    mnPos.reserve(700);
    mxPos.reserve(700);

    auto computeAnswer = [&]() -> int {
        values.clear();
        mnPos.clear();
        mxPos.clear();

        for (int k : levels) {
            values.push_back(k);
            mnPos.push_back(*buckets[k].begin());
            mxPos.push_back(*buckets[k].rbegin());
        }

        const int d = (int)values.size();
        int p = d;
        int partnerMin = N + 1;
        int partnerMax = -1;
        int answer = 0;

        // A pair with vote counts v and w is possible exactly when v+w >= maxVotes.
        // For each v, maintain all positions whose counts are at least maxVotes-v.
        for (int i = 0; i < d; ++i) {
            int needed = maxVotes - values[i];
            while (p > 0 && values[p - 1] >= needed) {
                --p;
                partnerMin = min(partnerMin, mnPos[p]);
                partnerMax = max(partnerMax, mxPos[p]);
            }

            answer = max(answer, partnerMax - mnPos[i]);
            answer = max(answer, mxPos[i] - partnerMin);
        }

        return answer;
    };

    while (Q--) {
        int i, x;
        cin >> i >> x;

        int old = a[i];
        a[i] = x;

        if (old != x) {
            int oldCount = cnt[old];
            removeFromBucket(old, oldCount);
            --cnt[old];
            if (cnt[old] > 0) addToBucket(old, cnt[old]);

            int targetOldCount = cnt[x];
            if (targetOldCount > 0) removeFromBucket(x, targetOldCount);
            ++cnt[x];
            addToBucket(x, cnt[x]);

            maxVotes = max(maxVotes, cnt[x]);
            while (maxVotes > 0 && buckets[maxVotes].empty()) --maxVotes;
        }

        cout << computeAnswer() << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

struct SegTree {
    int n, size;
    vector<int> tr;

    SegTree(int n_) : n(n_) {
        size = 1;
        while (size < n) size <<= 1;
        tr.assign(2 * size, 0);
    }

    void setValue(int pos, int value) {
        int p = pos + size - 1;
        tr[p] = value;
        for (p >>= 1; p; p >>= 1)
            tr[p] = max(tr[p << 1], tr[p << 1 | 1]);
    }

    int firstAtLeast(int value) const {
        if (tr[1] < value) return n + 1;
        int p = 1;
        while (p < size) {
            if (tr[p << 1] >= value) p <<= 1;
            else p = p << 1 | 1;
        }
        return p - size + 1;
    }

    int lastAtLeast(int value) const {
        if (tr[1] < value) return 0;
        int p = 1;
        while (p < size) {
            if (tr[p << 1 | 1] >= value) p = p << 1 | 1;
            else p <<= 1;
        }
        return p - size + 1;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    vector<int> a(N + 1), cnt(N + 1, 0);
    map<int, set<int>> byCount;
    SegTree seg(N);

    for (int i = 1; i <= N; ++i) {
        cin >> a[i];
        ++cnt[a[i]];
    }

    for (int x = 1; x <= N; ++x) {
        if (cnt[x] > 0) {
            byCount[cnt[x]].insert(x);
            seg.setValue(x, cnt[x]);
        }
    }

    auto changeCount = [&](int x, int newCount) {
        int oldCount = cnt[x];
        if (oldCount > 0) {
            auto &s = byCount[oldCount];
            s.erase(x);
            if (s.empty()) byCount.erase(oldCount);
        }
        cnt[x] = newCount;
        if (newCount > 0) byCount[newCount].insert(x);
        seg.setValue(x, newCount);
    };

    while (Q--) {
        int i, x;
        cin >> i >> x;

        int old = a[i];
        if (old != x) {
            changeCount(old, cnt[old] - 1);
            changeCount(x, cnt[x] + 1);
            a[i] = x;
        }

        int M = byCount.rbegin()->first;
        int answer = 0;

        auto it = byCount.lower_bound((M + 1) / 2);
        for (; it != byCount.end(); ++it) {
            int c = it->first;
            int needed = max(1, M - c);
            int lo = seg.firstAtLeast(needed);
            int hi = seg.lastAtLeast(needed);

            for (int h : it->second) {
                answer = max(answer, abs(h - lo));
                answer = max(answer, abs(h - hi));
            }
        }

        cout << answer << '\n';
    }

    return 0;
}

            activePosition[f] = (int)activeFrequencies.size();
            activeFrequencies.push_back(f);
        }

        s.insert(label);
        exactMin[f] = min(exactMin[f], label);
        exactMax[f] = max(exactMax[f], label);
    }

    void eraseFromBucket(int label, int f) {
        auto &s = bucket[f];
        auto it = s.find(label);
        s.erase(it);

        if (s.empty()) {
            exactMin[f] = INF;
            exactMax[f] = -INF;

            int pos = activePosition[f];
            int last = activeFrequencies.back();

            activeFrequencies[pos] = last;
            activePosition[last] = pos;
            activeFrequencies.pop_back();
            activePosition[f] = -1;
        } else {
            exactMin[f] = *s.begin();
            exactMax[f] = *s.rbegin();
        }
    }

    void refreshSuffix(int f) {
        suffixMin[f] = min(exactMin[f], suffixMin[f + 1]);
        suffixMax[f] = max(exactMax[f], suffixMax[f + 1]);
    }

    void normalizeMaximum() {
        while (maxFrequency > 0 && bucket[maxFrequency].empty()) {
            --maxFrequency;
        }
    }

public:
    explicit VoteTracker(vector<int> initial)
        : n((int)initial.size() - 1),
          frequency(std::move(initial)),
          bucket(n + 1),
          exactMin(n + 1, INF),
          exactMax(n + 1, -INF),
          suffixMin(n + 2, INF),
          suffixMax(n + 2, -INF),
          activePosition(n + 1, -1) {

        for (int label = 1; label <= n; ++label) {
            if (frequency[label] > 0) {
                auto &s = bucket[frequency[label]];
                s.insert(s.end(), label);
            }
        }

        for (int f = 1; f <= n; ++f) {
            if (!bucket[f].empty()) {
                exactMin[f] = *bucket[f].begin();
                exactMax[f] = *bucket[f].rbegin();

                activePosition[f] = (int)activeFrequencies.size();
                activeFrequencies.push_back(f);
            }
        }

        maxFrequency = activeFrequencies.empty() ? 0 : activeFrequencies.back();

        for (int f = n; f >= 1; --f) {
            refreshSuffix(f);
        }
    }

    void removeOneVote(int label) {
        int oldFrequency = frequency[label];

        eraseFromBucket(label, oldFrequency);
        frequency[label] = oldFrequency - 1;

        if (oldFrequency > 1) {
            insertIntoBucket(label, oldFrequency - 1);
        }

        // This label leaves exactly the set of labels with frequency >= oldFrequency.
        refreshSuffix(oldFrequency);
        normalizeMaximum();
    }

    void addOneVote(int label) {
        int oldFrequency = frequency[label];

        if (oldFrequency > 0) {
            eraseFromBucket(label, oldFrequency);
        }

        frequency[label] = oldFrequency + 1;
        insertIntoBucket(label, oldFrequency + 1);

        // This label enters exactly the set of labels with frequency >= oldFrequency+1.
        refreshSuffix(oldFrequency + 1);

        maxFrequency = max(maxFrequency, oldFrequency + 1);
        normalizeMaximum();
    }

    int bestDiversity() const {
        int answer = 0;

        for (int f : activeFrequencies) {
            // A partner with frequency g must satisfy f+g >= maxFrequency.
            int required = max(1, maxFrequency - f);

            answer = max(answer, suffixMax[required] - exactMin[f]);
            answer = max(answer, exactMax[f] - suffixMin[required]);
        }

        return answer;
    }
};

    vector<int> count(N + 1, 0);

            tracker.removeOneVote(a[i]);
            tracker.addOneVote(x);
            a[i] = x;
        }

#include <bits/stdc++.h>
using namespace std;

class ElectionSolver {
    struct CacheCell {
        int nonEdge = -1;
        int oneEdge = -1;
        uint32_t nonGen1 = 0, nonGen2 = 0;
        uint32_t oneGen1 = 0, oneGen2 = 0;
    };

    int N;
    vector<int> a, cnt;

    // group[c] contains the cows receiving exactly c votes.
    vector<set<int>> group;
    set<int> activeCounts;

    // Slots give active vote-count values compact cache indices.
    vector<int> slotOf;
    vector<int> slotValue;
    vector<uint32_t> slotGeneration;
    vector<int> freeSlots;
    vector<vector<CacheCell>> cacheCells;

    int currentAnswer = 0;

    static vector<int> firstK(const set<int>& s, int k) {
        vector<int> result;
        result.reserve(k);
        for (auto it = s.begin(); it != s.end() && (int)result.size() < k; ++it) {
            result.push_back(*it);
        }
        return result;
    }

    static vector<int> lastKDescending(const set<int>& s, int k) {
        vector<int> result;
        result.reserve(k);
        auto it = s.end();
        while (it != s.begin() && (int)result.size() < k) {
            --it;
            result.push_back(*it);
        }
        return result;
    }

    static vector<int> extremeIds(const set<int>& s) {
        vector<int> result = firstK(s, 3);
        vector<int> last = lastKDescending(s, 3);
        result.insert(result.end(), last.begin(), last.end());
        sort(result.begin(), result.end());
        result.erase(unique(result.begin(), result.end()), result.end());
        return result;
    }

    int allocateSlot(int countValue) {
        int s;
        if (!freeSlots.empty()) {
            s = freeSlots.back();
            freeSlots.pop_back();
            slotValue[s] = countValue;
            ++slotGeneration[s];
        } else {
            s = (int)slotValue.size();
            slotValue.push_back(countValue);
            slotGeneration.push_back(1);
            cacheCells.emplace_back();
            for (auto& row : cacheCells) {
                row.resize(s + 1);
            }
        }
        slotOf[countValue] = s;
        return s;
    }

    void markDirty(int c1, int c2) {
        if (c1 <= 0 || c2 <= 0) return;
        int s1 = slotOf[c1], s2 = slotOf[c2];
        if (s1 < 0 || s2 < 0) return;
        if (s1 > s2) swap(s1, s2);
        cacheCells[s1][s2].nonEdge = -1;
        cacheCells[s1][s2].oneEdge = -1;
    }

    void moveBetweenGroups(int cow, int oldCount, int newCount) {
        if (oldCount == newCount) return;

        if (oldCount > 0) {
            group[oldCount].erase(cow);

            int s = slotOf[oldCount];
            if (s >= 0) ++slotGeneration[s];

            if (group[oldCount].empty()) {
                if (s >= 0) {
                    slotOf[oldCount] = -1;
                    freeSlots.push_back(s);
                }
                activeCounts.erase(oldCount);
            }
        }

        if (newCount > 0) {
            int s = slotOf[newCount];
            if (s < 0) s = allocateSlot(newCount);
            ++slotGeneration[s];

            group[newCount].insert(cow);
            activeCounts.insert(newCount);
        }
    }

    bool adjacent(int x, int y) const {
        return a[x] == y || a[y] == x;
    }

    // For two cows whose counts sum to M, they can be leaders exactly
    // when neither votes for the other.
    int computeNonEdgeBest(int c, int d) {
        const auto& C = group[c];
        const auto& D = group[d];

        int best = 0;
        int minC = *C.begin();
        int minD = *D.begin();

        int limX = (int)min<long long>((long long)d + 2, C.size());
        int limY = (int)min<long long>((long long)c + 2, D.size());

        vector<int> firstX = firstK(C, limX);
        vector<int> lastX = lastKDescending(C, limX);
        vector<int> firstY = firstK(D, limY);
        vector<int> lastY = lastKDescending(D, limY);

        // Cases with the cow from D having the larger index.
        for (int y : lastY) {
            if (y - minC <= best) break;
            for (int x : firstX) {
                if (x >= y) break;
                if (!adjacent(x, y)) {
                    best = max(best, y - x);
                    break;
                }
            }
        }

        // Cases with the cow from C having the larger index.
        for (int x : lastX) {
            if (x - minD <= best) break;
            for (int y : firstY) {
                if (y >= x) break;
                if (!adjacent(x, y)) {
                    best = max(best, x - y);
                    break;
                }
            }
        }

        return best;
    }

    // For two cows whose counts sum to M+1, at most one directed
    // relation between them is allowed.
    bool validOneEdge(int x, int y, int c, int d) const {
        int u = (a[y] == x);
        int v = (a[x] == y);

        int p = c - u;
        int q = d - v;

        return p >= 1 && q >= 1 && u + v <= 1;
    }

    int computeOneEdgeBest(int c, int d) {
        vector<int> C = extremeIds(group[c]);
        vector<int> D = extremeIds(group[d]);

        int best = 0;
        for (int x : C) {
            for (int y : D) {
                if (x == y) continue;
                if (validOneEdge(x, y, c, d)) {
                    best = max(best, abs(x - y));
                }
            }
        }
        return best;
    }

    int getNonEdgeBest(int c, int d) {
        int s1 = slotOf[c], s2 = slotOf[d];
        int lo = min(s1, s2), hi = max(s1, s2);
        CacheCell& cell = cacheCells[lo][hi];

        if (cell.nonEdge < 0 ||
            cell.nonGen1 != slotGeneration[lo] ||
            cell.nonGen2 != slotGeneration[hi]) {
            cell.nonEdge = computeNonEdgeBest(c, d);
            cell.nonGen1 = slotGeneration[lo];
            cell.nonGen2 = slotGeneration[hi];
        }
        return cell.nonEdge;
    }

    int getOneEdgeBest(int c, int d) {
        int s1 = slotOf[c], s2 = slotOf[d];
        int lo = min(s1, s2), hi = max(s1, s2);
        CacheCell& cell = cacheCells[lo][hi];

        if (cell.oneEdge < 0 ||
            cell.oneGen1 != slotGeneration[lo] ||
            cell.oneGen2 != slotGeneration[hi]) {
            cell.oneEdge = computeOneEdgeBest(c, d);
            cell.oneGen1 = slotGeneration[lo];
            cell.oneGen2 = slotGeneration[hi];
        }
        return cell.oneEdge;
    }

    int calculateAnswer() {
        int M = *activeCounts.rbegin();

        vector<int> values(activeCounts.begin(), activeCounts.end());
        int K = (int)values.size();

        vector<int> suffixMin(K), suffixMax(K);
        for (int i = K - 1; i >= 0; --i) {
            int c = values[i];
            int mn = *group[c].begin();
            int mx = *group[c].rbegin();
            if (i + 1 < K) {
                mn = min(mn, suffixMin[i + 1]);
                mx = max(mx, suffixMax[i + 1]);
            }
            suffixMin[i] = mn;
            suffixMax[i] = mx;
        }

        int answer = 0;

        // If the two vote counts sum to at least M+2, the pair is
        // always feasible, regardless of the directed votes between them.
        int firstValid = K;
        for (int i = 0; i < K; ++i) {
            int c = values[i];
            int threshold = M - c + 2;

            while (firstValid > 0 && values[firstValid - 1] >= threshold) {
                --firstValid;
            }
            if (firstValid == K) continue;

            int mn = *group[c].begin();
            int mx = *group[c].rbegin();

            answer = max(answer, mx - suffixMin[firstValid]);
            answer = max(answer, suffixMax[firstValid] - mn);
        }

        // Counts summing to M+1.
        for (int c : values) {
            int d = M + 1 - c;
            if (c > d || d <= 0 || group[d].empty()) continue;
            answer = max(answer, getOneEdgeBest(c, d));
        }

        // Counts summing to M.
        for (int c : values) {
            int d = M - c;
            if (c > d || d <= 0 || group[d].empty()) continue;
            answer = max(answer, getNonEdgeBest(c, d));
        }

        return answer;
    }

public:
    ElectionSolver(int n, vector<int> initialVotes)
        : N(n), a(std::move(initialVotes)), cnt(n + 1, 0),
          group(n + 1), slotOf(n + 1, -1) {
        for (int v : a) ++cnt[v];

        for (int cow = 1; cow <= N; ++cow) {
            if (cnt[cow] > 0) {
                group[cnt[cow]].insert(group[cnt[cow]].end(), cow);
                activeCounts.insert(cnt[cow]);
            }
        }

        for (int c : activeCounts) allocateSlot(c);
        currentAnswer = calculateAnswer();
    }

    int update(int cow, int newVote) {
        int oldVote = a[cow];
        if (oldVote == newVote) return currentAnswer;

        a[cow] = newVote;

        int oldCount = cnt[oldVote];
        --cnt[oldVote];
        moveBetweenGroups(oldVote, oldCount, oldCount - 1);

        int newCount = cnt[newVote];
        ++cnt[newVote];
        moveBetweenGroups(newVote, newCount, newCount + 1);

        // Only relations involving this cow's outgoing vote changed.
        markDirty(cnt[cow], cnt[oldVote]);
        markDirty(cnt[cow], cnt[newVote]);

        currentAnswer = calculateAnswer();
        return currentAnswer;
    }
};

using namespace std;

class MaxSegTree {
    int n, base;
    vector<int> tree;

public:
    MaxSegTree(int n_, const vector<int>& values) : n(n_), base(1) {
        while (base < n) base <<= 1;
        tree.assign(2 * base, 0);

        for (int i = 1; i <= n; ++i) {
            tree[base + i - 1] = values[i];
        }
        for (int p = base - 1; p >= 1; --p) {
            tree[p] = max(tree[2 * p], tree[2 * p + 1]);
        }
    }

    void update(int pos, int value) {
        int p = base + pos - 1;
        tree[p] = value;

        for (p >>= 1; p >= 1; p >>= 1) {
            tree[p] = max(tree[2 * p], tree[2 * p + 1]);
        }
    }

    // Maximum over count values in [pos, n].
    int suffixMax(int pos) const {
        int l = base + pos - 1;
        int r = base + n;
        int result = 0;

        while (l < r) {
            if (l & 1) result = max(result, tree[l++]);
            if (r & 1) result = max(result, tree[--r]);
            l >>= 1;
            r >>= 1;
        }
        return result;
    }
};

    vector<set<int>> bucket(N + 1);

            auto& s = bucket[cnt[cow]];
            s.insert(s.end(), cow);
        }
    }

    vector<int> exactMax(N + 1, 0);

    vector<int> activeCounts;
    vector<int> activePos(N + 1, -1);

            activeCounts.push_back(c);

        int p = activePos[c];
        int last = activeCounts.back();

        activeCounts[p] = last;
        activePos[last] = p;
        activeCounts.pop_back();
        activePos[c] = -1;

            activate(c);
            exactMin[c] = *bucket[c].begin();
            exactMax[c] = *bucket[c].rbegin();

            exactMin[c] = 0;
            exactMax[c] = 0;
        } else {
            exactMin[c] = *s.begin();
            exactMax[c] = *s.rbegin();
