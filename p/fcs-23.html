#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long Tid;
    if (!(cin >> Tid)) return 0;

    cout << "DY=(return)1\n";
    cout << "EY=(return)1\n";
    cout << "FY=(return)1\n";
    cout << "WZ=(return)0\n";

    cout << "S=XYZ\n";

    for (char c : string("abc"))
        cout << c << "XY=XY" << c << '\n';

    for (char p : string("ABC")) {
        for (char c : string("abc"))
            cout << c << p << "K=" << p << "K" << c << '\n';
    }

    for (char p : string("ABC"))
        cout << "Y" << p << "K=" << p << "Y\n";

    cout << "Za=AKZ\n";
    cout << "Zb=BKZ\n";
    cout << "Zc=CKZ\n";

    cout << "Ya=Yg\n";
    cout << "Yb=Yh\n";
    cout << "Yc=Yi\n";

    cout << "Wa=g\n";
    cout << "Wb=h\n";
    cout << "Wc=i\n";

    cout << "Yg=gY\n";
    cout << "Yh=hY\n";
    cout << "Yi=iY\n";

    for (char v : string("ghi")) {
        for (char p : string("ABCDEF"))
            cout << p << v << '=' << v << p << '\n';
    }

    cout << "Xg=X3\n";
    cout << "Xh=X4\n";
    cout << "Xi=X5\n";

    const string states = "ABCDEF";

    for (int q = 0; q < 6; ++q) {
        const int text = q % 3;
        const int carry = q / 3;

        for (int p = 0; p < 6; ++p) {
            const int expected = p % 3;
            const int oldActive = p / 3;
            const int newActive = carry && (text == expected);
            const int newIndex = newActive ? 3 + expected : expected;
            const char newState = states[newIndex];

            cout << char('0' + q) << states[p] << 'Y'
                 << '=' << newState << "YW\n";
        }
    }

    for (int q = 0; q < 6; ++q) {
        const int text = q % 3;
        const int carry = q / 3;

        for (int p = 0; p < 6; ++p) {
            const int expected = p % 3;
            const int oldActive = p / 3;
            const int newActive = carry && (text == expected);
            const int newIndex = newActive ? 3 + expected : expected;
            const char newState = states[newIndex];
            const char nextScanner = char('0' + text + (oldActive ? 3 : 0));

            cout << char('0' + q) << states[p]
                 << '=' << newState << nextScanner << '\n';
        }
    }

    return 0;
}
