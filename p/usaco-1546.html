题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Bessie has challenged Farmer John to a game involving milk buckets! There are $N$ $(2 \leq N \leq 2 \cdot 10^5)$ milk buckets lined up in a row. The $i$-th bucket from the left initially contains $a_i$ $(0 \leq a_i \leq 10^9)$ gallons of milk.

The game consists of two phases:

Phase 1: Farmer John may swap any two adjacent buckets. He may perform as many swaps as he likes, but each swap costs 1 coin.

Phase 2: After swapping, Farmer John performs the following operation until only one bucket is left: Choose two adjacent buckets with milk amounts $a_i$ and $a_{i+1}$, and replace both buckets with one bucket containing $\frac{a_i+a_{i+1}}2$ gallons of milk in their place.

Your goal is to determine the minimum number of coins Farmer John must spend in the swapping phase to maximize the amount of milk in the final bucket after all merges are complete.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains one integer $T$ $(1 \leq T \leq 100)$: the number of independent test cases.

Then, for each test case, the first line contains an integer $N$: the number of milk buckets. The second line contains $N$ integers $a_1, a_2, \dots, a_N$, separated by spaces: the number of gallons of milk in each bucket.

It is guaranteed that the sum of $N$ over all test cases does not exceed $5 \cdot 10^5.$

OUTPUT FORMAT (print output to the terminal / stdout):

For each test case, output the minimum number of coins Farmer John must spend to maximize the amount of milk in the final bucket.

SAMPLE INPUT:

2
3
0 0 1
3
0 1 0

SAMPLE OUTPUT:

0
1

For the first test, we do not need to swap any milk buckets in the first phase. In the second phase, Farmer John can merge the first two buckets and then merge the only two buckets left to achieve a final amount of 0.5. It can be shown that this final amount is maximal.

For the second test, we must perform a singular swap of the first two milk buckets in the first stage to achieve a final amount of 0.5 in the second stage. It can be shown that we cannot achieve a final amount of 0.5 without swaps in the first stage.

SAMPLE INPUT:

4
4
9 4 9 2
6
0 0 2 0 0 0
3
2 0 1
9
3 3 3 10 3 2 13 14 13

SAMPLE OUTPUT:

1
2
0
3

For the first test, Farmer John can swap the second and the third buckets in the first phase. Then, in the second phase, Farmer John can perform the following:

  • $[9,9,4,2]$ -> merge the third and fourth buckets ->
  • $[9,9,3]$ -> merge the second and third buckets ->
  • $[9,6]$ -> merge the first and second buckets ->
  • $[7.5]$

The final amount of milk is 7.5, which is the maximum possible. It can be shown that even with additional swaps, the final amount cannot exceed 7.5, and that with fewer swaps, the final amount cannot reach 7.5.

SCORING:

  • Inputs 3-4: $a_i\le 1$ and $N\le 2000$ (sum of $N\le 5000$)
  • Inputs 5-6: $a_i\le 1$
  • Inputs 7-9: $N\le 2000$ (sum of $N\le 5000$)
  • Inputs 10-14: No additional constraints.

#include <bits/stdc++.h>
using namespace std;

struct Fenwick {
    int n;
    vector<int> bit;
    Fenwick(int n_) : n(n_), bit(n_ + 1, 0) {}

    void add(int idx, int val) {
        for (; idx <= n; idx += idx & -idx) bit[idx] += val;
    }

    int sum(int idx) const {
        int res = 0;
        for (; idx > 0; idx -= idx & -idx) res += bit[idx];
        return res;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        vector<int> a(N);
        vector<int> vals;
        vals.reserve(N);

        for (int &x : a) {
            cin >> x;
            vals.push_back(x);
        }

        sort(vals.begin(), vals.end());
        vals.erase(unique(vals.begin(), vals.end()), vals.end());

        int M = vals.size();
        vector<int> rank(N), freq(M, 0);

        for (int i = 0; i < N; i++) {
            rank[i] = lower_bound(vals.begin(), vals.end(), a[i]) - vals.begin();
            freq[rank[i]]++;
        }

        vector<int> totalSmaller(M);
        int prefix = 0;
        for (int i = 0; i < M; i++) {
            totalSmaller[i] = prefix;
            prefix += freq[i];
        }

        Fenwick fw(M);
        long long answer = 0;

        for (int i = 0; i < N; i++) {
            int r = rank[i];
            int leftSmaller = fw.sum(r); // ranks 0 .. r-1 are strictly smaller
            int rightSmaller = totalSmaller[r] - leftSmaller;
            answer += min(leftSmaller, rightSmaller);
            fw.add(r + 1, 1);
        }

        cout << answer << '\n';
    }

    return 0;
}

    int T;
    cin >> T;
    while (T--) {
        int N;
        cin >> N;

        vector<pair<long long, int>> items(N);

        vector<vector<int>> groups;
        int lowerCount = 0;

        for (int l = 0; l < N; ) {
            int r = l;

            vector<int> costs;
            costs.reserve(r - l);
            for (int i = l; i < r; ++i) {
                int p = items[i].second;
                int before = fw.sum(p - 1);
                int after = lowerCount - before;
                costs.push_back(min(before, after));

        for (int g = (int)groups.size() - 1; g >= 0 && remainingRanks > 0; --g) {
            auto &costs = groups[g];
            int take = min(remainingRanks, (int)costs.size());

            if (take < (int)costs.size()) {
                nth_element(costs.begin(), costs.begin() + take, costs.end());
            }
            for (int i = 0; i < take; ++i) answer += costs[i];
            remainingRanks -= take;
        }

struct Fenwick {
    int n;
    vector<int> bit;
    Fenwick(int n = 0) : n(n), bit(n + 1, 0) {}

    void add(int idx, int val) {
        for (; idx <= n; idx += idx & -idx) bit[idx] += val;
    }

    int sumPrefix(int idx) const {
        int res = 0;
        for (; idx > 0; idx -= idx & -idx) res += bit[idx];
        return res;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        vector<int> a(N);
        vector<int> comp;
        comp.reserve(N);

        for (int i = 0; i < N; i++) {
            cin >> a[i];
            comp.push_back(a[i]);
        }

        sort(comp.begin(), comp.end());
        comp.erase(unique(comp.begin(), comp.end()), comp.end());

        int K = (int)comp.size();
        vector<vector<int>> positions(K);

        for (int i = 0; i < N; i++) {
            int id = lower_bound(comp.begin(), comp.end(), a[i]) - comp.begin();
            positions[id].push_back(i + 1);
        }

        Fenwick fw(N);

        long long answer = 0;
        long long inserted = 0;

        for (int id = 0; id < K; id++) {
            const vector<int>& pos = positions[id];
            int c = (int)pos.size();

            long long smallerCount = inserted;

            vector<int> before(c);
            long long totalBefore = 0;

            for (int i = 0; i < c; i++) {
                before[i] = fw.sumPrefix(pos[i] - 1);
                totalBefore += before[i];
            }

            long long prefix = 0;
            long long best = LLONG_MAX;

            for (int left = 0; left <= c; left++) {
                long long cost =
                    (long long)(c - left) * smallerCount +
                    2LL * prefix -
                    totalBefore;

                best = min(best, cost);

                if (left < c) prefix += before[left];
            }

            answer += best;

            for (int p : pos) fw.add(p, 1);
            inserted += c;
        }

        cout << answer << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>

class Fenwick {
    int n;
    vector<int> bit;

public:
    explicit Fenwick(int n) : n(n), bit(n + 1, 0) {}

    void add(int idx, int value) {
        for (; idx <= n; idx += idx & -idx) {
            bit[idx] += value;
        }
    }

    int sum(int idx) const {
        int result = 0;
        for (; idx > 0; idx -= idx & -idx) {
            result += bit[idx];
        }
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        vector<int> a(N);
        for (int &x : a) cin >> x;

        vector<int> order(N);
        iota(order.begin(), order.end(), 0);
        sort(order.begin(), order.end(), [&](int x, int y) {
            if (a[x] != a[y]) return a[x] < a[y];
            return x < y;
        });

        Fenwick fw(N);

        long long inversions = 0;
        long long adjustment = 0;

        for (int left = 0; left < N; ) {
            int right = left + 1;
            while (right < N && a[order[right]] == a[order[left]]) {
                ++right;
            }

            int groupSize = right - left;
            long long totalSmaller = left;

            vector<long long> b(groupSize);

            for (int t = 0; t < groupSize; ++t) {
                int pos = order[left + t] + 1;

                // The Fenwick tree currently contains only strictly smaller values.
                long long smallerBefore = fw.sum(pos - 1);
                long long equalBefore = t;
                long long smallerAfter = totalSmaller - smallerBefore;

                b[t] = smallerBefore + equalBefore - smallerAfter;

                // Earlier elements greater than a[pos].
                inversions += (pos - 1LL) - smallerBefore - equalBefore;
            }

            sort(b.begin(), b.end());

            long long prefixSum = 0;
            long long best = 0;  // Choosing zero elements for the decreasing side.

            for (int k = 1; k <= groupSize; ++k) {
                prefixSum += b[k - 1];
                long long cost = prefixSum - 1LL * k * (k - 1) / 2;
                best = min(best, cost);
            }

            adjustment += best;

            for (int t = 0; t < groupSize; ++t) {
                fw.add(order[left + t] + 1, 1);
            }

            left = right;
        }

        cout << inversions + adjustment << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

struct Fenwick {
    int n;
    vector<int> bit;

    // Initially every position contains 1.
    explicit Fenwick(int n) : n(n), bit(n + 1) {
        for (int i = 1; i <= n; ++i) {
            bit[i] = i & -i;
        }
    }

    void add(int index, int delta) {
        for (; index <= n; index += index & -index) {
            bit[index] += delta;
        }
    }

    int prefixSum(int index) const {
        int result = 0;
        for (; index > 0; index -= index & -index) {
            result += bit[index];
        }
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        vector<pair<int64, int>> order(N);
        for (int i = 0; i < N; ++i) {
            cin >> order[i].first;
            order[i].second = i + 1;
        }

        // Larger values first; occurrences of the same value are in
        // increasing original-position order.
        sort(order.begin(), order.end(),
             [](const auto& x, const auto& y) {
                 if (x.first != y.first) return x.first > y.first;
                 return x.second < y.second;
             });

        Fenwick fenwick(N);
        vector<int64> smallerBefore;

        int64 answer = 0;
        int active = N;

        for (int start = 0; start < N; ) {
            int finish = start + 1;
            while (finish < N &&
                   order[finish].first == order[start].first) {
                ++finish;
            }

            int count = finish - start;
            int smaller = active - count;

            // After removing this group, the Fenwick tree contains
            // exactly the positions with strictly smaller values.
            for (int i = start; i < finish; ++i) {
                fenwick.add(order[i].second, -1);
            }

            smallerBefore.resize(count);

            int64 currentCost = 0;
            for (int i = 0; i < count; ++i) {
                int position = order[start + i].second;
                smallerBefore[i] = fenwick.prefixSum(position - 1);

                // Initially, place every occurrence on the right side.
                currentCost += static_cast<int64>(smaller) - smallerBefore[i];
            }

            int64 bestCost = currentCost;

            // Move occurrences from the right side to the left side
            // in their original order.
            for (int i = 0; i < count; ++i) {
                currentCost += 2 * smallerBefore[i] - smaller;
                bestCost = min(bestCost, currentCost);
            }

            answer += bestCost;
            active = smaller;
            start = finish;
        }

        cout << answer << '\n';
    }

    return 0;
}

            ++tree[index];
        }
    }

    int prefixSum(int index) const {
        int result = 0;
        for (; index > 0; index -= index & -index) {
            result += tree[index];
        }
        return result;
