class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        def all_substrings(x: str):
            result = [""]
            for start in range(len(x)):
                for end in range(start + 1, len(x) + 1):
                    result.append(x[start:end])
            return result

        s_substrings = all_substrings(s)
        t_substrings = all_substrings(t)

        best = 0
        for a in s_substrings:
            for b in t_substrings:
                length = len(a) + len(b)
                if length <= best:
                    continue

                candidate = a + b
                if candidate == candidate[::-1]:
                    best = length

        return best

class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        def is_palindrome(x: str) -> bool:
            return x == x[::-1]

        substrings_s = [""]
        substrings_t = [""]

        for i in range(len(s)):
            for j in range(i + 1, len(s) + 1):
                substrings_s.append(s[i:j])

        for i in range(len(t)):
            for j in range(i + 1, len(t) + 1):
                substrings_t.append(t[i:j])

        best = 0

        for a in substrings_s:
            for b in substrings_t:
                candidate = a + b
                if is_palindrome(candidate):
                    best = max(best, len(candidate))

        return best

class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        def is_pal(x: str) -> bool:
            return x == x[::-1]
        
        s_subs = [""]

        t_subs = [""]
        for i in range(len(t)):
            for j in range(i + 1, len(t) + 1):
                t_subs.append(t[i:j])
        
        ans = 0
