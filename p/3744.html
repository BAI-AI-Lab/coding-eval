from typing import List

class Solution:
    def minOperations(self, queries: List[List[int]]) -> int:
        answer = 0

        for l, r in queries:
            total_depth = 0
            max_depth = 0
            power = 1

            # A number x needs one operation for each power 1, 4, 16, ... <= x.
            while power <= r:
                max_depth += 1
                total_depth += r - max(l, power) + 1
                power *= 4

            # Each operation handles two unit steps, while the deepest element
            # can be processed at most once per operation.
            answer += max(max_depth, (total_depth + 1) // 2)

        return answer

from typing import List

class Solution:
    def minOperations(self, queries: List[List[int]]) -> int:
        def prefix(n: int) -> int:
            total = 0
            p = 1
            k = 1

            while p <= n:
                end = min(n, p * 4 - 1)
                total += k * (end - p + 1)
                p *= 4
                k += 1

            return total

        def max_steps(x: int) -> int:
            steps = 0
            while x > 0:
                x //= 4
                steps += 1
            return steps

        answer = 0

        for l, r in queries:
            total_steps = prefix(r) - prefix(l - 1)
            max_required = max_steps(r)
            answer += max(max_required, (total_steps + 1) // 2)

        return answer

from typing import List

class Solution:
    def minOperations(self, queries: List[List[int]]) -> int:
        # pref[k] = total base-4 digits of all positive integers
        # with at most k digits (i.e. numbers 1..4^k - 1)
        pref = [0] * 64
        for k in range(1, 64):
            pref[k] = pref[k - 1] + k * 3 * (1 << (2 * (k - 1)))

        def total_digits_upto(n: int) -> int:
            if n <= 0:
                return 0

            # d = number of base-4 digits of n
            d = (n.bit_length() + 1) // 2
            first_d_digit = 1 << (2 * (d - 1))

            return pref[d - 1] + d * (n - first_d_digit + 1)

        ans = 0
        for l, r in queries:
            total_steps = total_digits_upto(r) - total_digits_upto(l - 1)
            ans += (total_steps + 1) // 2

        return ans
