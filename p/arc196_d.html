import sys
from bisect import bisect_left

SHIFT = 20
MASK = (1 << SHIFT) - 1
INF = 1 << 60


class MinSeg:
    __slots__ = ("n", "t")

    def __init__(self, size):
        n = 1
        while n < size:
            n <<= 1
        self.n = n
        self.t = [INF] * (2 * n)

    def update(self, pos, value):
        t = self.t
        p = pos + self.n
        t[p] = value
        p >>= 1
        while p:
            a = t[p << 1]
            b = t[p << 1 | 1]
            t[p] = a if a < b else b
            p >>= 1

    def query(self, left, right):
        t = self.t
        l = left + self.n
        r = right + self.n
        res = INF
        while l < r:
            if l & 1:
                v = t[l]
                if v < res:
                    res = v
                l += 1
            if r & 1:
                r -= 1
                v = t[r]
                if v < res:
                    res = v
            l >>= 1
            r >>= 1
        return res


class MaxSeg:
    __slots__ = ("n", "t")

    def __init__(self, size):
        n = 1
        while n < size:
            n <<= 1
        self.n = n
        self.t = [-1] * (2 * n)

    def update(self, pos, value):
        t = self.t
        p = pos + self.n
        t[p] = value
        p >>= 1
        while p:
            a = t[p << 1]
            b = t[p << 1 | 1]
            t[p] = a if a > b else b
            p >>= 1

    def query(self, left, right):
        t = self.t
        l = left + self.n
        r = right + self.n
        res = -1
        while l < r:
            if l & 1:
                v = t[l]
                if v > res:
                    res = v
                l += 1
            if r & 1:
                r -= 1
                v = t[r]
                if v > res:
                    res = v
            l >>= 1
            r >>= 1
        return res


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, q = data[0], data[1], data[2]
    ptr = 3

    left = [0] * (m + 1)
    right = [0] * (m + 1)
    sign = [0] * (m + 1)  # 0: S < T, 1: S > T

    right_coords = [[], []]
    left_coords = [[], []]

    for i in range(1, m + 1):
        s = data[ptr]
        t = data[ptr + 1]
        ptr += 2
        if s < t:
            left[i] = s
            right[i] = t
            sign[i] = 0
        else:
            left[i] = t
            right[i] = s
            sign[i] = 1
        left_coords[sign[i]].append(left[i])
        right_coords[sign[i]].append(right[i])

    for s in (0, 1):
        left_coords[s] = sorted(set(left_coords[s]))
        right_coords[s] = sorted(set(right_coords[s]))

    left_maps = [
        {v: i for i, v in enumerate(left_coords[0])},
        {v: i for i, v in enumerate(left_coords[1])},
    ]
    right_maps = [
        {v: i for i, v in enumerate(right_coords[0])},
        {v: i for i, v in enumerate(right_coords[1])},
    ]

    pos_left = [0] * (m + 1)
    pos_right = [0] * (m + 1)
    for i in range(1, m + 1):
        s = sign[i]
        pos_left[i] = left_maps[s][left[i]]
        pos_right[i] = right_maps[s][right[i]]

    del left_maps, right_maps

    min_trees = [
        MinSeg(len(right_coords[0])),
        MinSeg(len(right_coords[1])),
    ]
    max_trees = [
        MaxSeg(len(left_coords[0])),
        MaxSeg(len(left_coords[1])),
    ]

    active_left = [0] * (n + 1)
    active_right = [0] * (n + 1)
    bound = [0] * (m + 1)

    window_left = 1

    for cur in range(1, m + 1):
        a = left[cur]
        b = right[cur]
        s = sign[cur]

        while True:
            bad = active_left[a]
            if bad == 0:
                bad = active_right[b]

            if bad == 0:
                coords = right_coords[s]
                lo = bisect_left(coords, a + 1)
                hi = bisect_left(coords, b)
                if lo < hi:
                    key = min_trees[s].query(lo, hi)
                    if (key >> SHIFT) < a:
                        bad = key & MASK

            if bad == 0:
                coords = left_coords[s]
                lo = bisect_left(coords, a + 1)
                hi = bisect_left(coords, b)
                if lo < hi:
                    key = max_trees[s].query(lo, hi)
                    if key >= 0 and (key >> SHIFT) > b:
                        bad = key & MASK

            if bad == 0:
                break

            for j in range(window_left, bad + 1):
                x = left[j]
                y = right[j]
                sj = sign[j]
                active_left[x] = 0
                active_right[y] = 0
                min_trees[sj].update(pos_right[j], INF)
                max_trees[sj].update(pos_left[j], -1)

            window_left = bad + 1

        active_left[a] = cur
        active_right[b] = cur
        min_trees[s].update(pos_right[cur], (a << SHIFT) | cur)
        max_trees[s].update(pos_left[cur], (b << SHIFT) | cur)
        bound[cur] = window_left

    answers = []
    for _ in range(q):
        l = data[ptr]
        r = data[ptr + 1]
        ptr += 2
        answers.append("Yes" if l >= bound[r] else "No")

    sys.stdout.write("\n".join(answers))

    main()

def solve():

    N = next(it)
    M = next(it)
    Q = next(it)

    L = [0] * (M + 1)
    R = [0] * (M + 1)
    Sign = [0] * (M + 1)  # 1: rightward (+), 0: leftward (-)

    for i in range(1, M + 1):
        s = next(it)
        t = next(it)
        if s < t:
            l = s - 1
            r = t - 1
            sign = 1
        else:
            l = t - 1
            r = s - 1
            sign = 0
        L[i] = l
        R[i] = r
        Sign[i] = sign

    queries = []
    for _ in range(Q):
        queries.append((next(it), next(it)))

    f = [0] * (M + 1)

    # Type A: identical interval, opposite signs
    same_interval = {}
    for i in range(1, M + 1):
        key = (L[i], R[i])
        entry = same_interval.get(key)
        if Sign[i] == 1:
            if entry is not None and entry[1] != 0:
                if entry[1] > f[i]:
                    f[i] = entry[1]
            if entry is None:
                same_interval[key] = [i, 0]
            else:
                entry[0] = i
        else:
            if entry is not None and entry[0] != 0:
                if entry[0] > f[i]:
                    f[i] = entry[0]
            if entry is None:
                same_interval[key] = [0, i]
            else:
                entry[1] = i

    top2_idx_l = [0] * N

    minus_intervals = [(i, L[i], R[i]) for i in range(1, M + 1) if Sign[i] == 0]

    def process_structure(intervals, mode, f):
        if not intervals:
            return

        Nm1 = N - 1

        if mode == 1:
            xs = sorted({r for _, _, r in intervals})
        else:
            xs = sorted({l for _, l, _ in intervals})

        K = len(xs)
        size = 1
        while size < K:

        pos_map = {x: i for i, x in enumerate(xs)}
        node_ys = [None] * (2 * size)

        if mode == 1:
            for _, l, r in intervals:
                x = r
                y = l
                pos = pos_map[x]
                node = size + pos
                while node:
                    lst = node_ys[node]
                    if lst is None:

                    else:

                    lst = node_ys[node]
                    if lst is None:
                        node_ys[node] = [y]
                    else:
                        lst.append(y)
                    node >>= 1

            lst = node_ys[i]
            if lst is not None:
                lst.sort()
                m = 0
                for val in lst:
                    if m == 0 or val != lst[m - 1]:
                        lst[m] = val
                        m += 1
                del lst[m:]
                node_ys[i] = lst
                node_bit[i] = [0] * (m + 1)

                xR = r - 1
                Y = l - 1

                    hi = bisect_right(xs, xR) - 1

                        left = size + lo
                        right = size + hi

                                    p = bisect_right(uniq, Y)
                                    while p:
                                        val = bit[p]
                                        if val > res:

                                        p -= p & -p
                                left += 1
                            if not (right & 1):
                                node = right
                                uniq = node_ys[node]
                                if uniq is not None:
                                    bit = node_bit[node]
                                    p = bisect_right(uniq, Y)
                                    while p:
                                        val = bit[p]
                                        if val > res:
                                            res = val
                                        p -= p & -p
                                right -= 1
                            left >>= 1
                            right >>= 1
                        if res > f[idx]:
                            f[idx] = res

                x = r
                y = l
                pos = pos_map[x]
                node = size + pos
                while node:
                    uniq = node_ys[node]
                    if uniq is not None:
                        bit = node_bit[node]
                        yidx = bisect_left(uniq, y) + 1
                        m = len(uniq)
                        p = yidx
                        while p <= m:
                            bit[p] = idx
                            p += p & -p
                    node >>= 1
        else:
            for idx, l, r in intervals:
                xL = l + 1
                xR = r - 1
                Y = Nm1 - (r + 1)

                if xL <= xR:
                    lo = bisect_left(xs, xL)
                    hi = bisect_right(xs, xR) - 1
                    if lo <= hi:
                        res = 0
                        left = size + lo
                        right = size + hi
                        while left <= right:
                            if left & 1:
                                node = left
                                uniq = node_ys[node]
                                if uniq is not None:
                                    bit = node_bit[node]
                                    p = bisect_right(uniq, Y)
                                    while p:
                                        val = bit[p]
                                        if val > res:
                                            res = val
                                        p -= p & -p
                                left += 1
                            if not (right & 1):
                                node = right
                                uniq = node_ys[node]
                                if uniq is not None:
                                    bit = node_bit[node]
                                    p = bisect_right(uniq, Y)
                                    while p:
                                        val = bit[p]
                                        if val > res:
                                            res = val
                                        p -= p & -p
                                right -= 1
                            left >>= 1
                            right >>= 1
                        if res > f[idx]:
                            f[idx] = res

                x = l
                y = Nm1 - r
                pos = pos_map[x]
                node = size + pos
                while node:
                    uniq = node_ys[node]
                    if uniq is not None:
                        bit = node_bit[node]
                        yidx = bisect_left(uniq, y) + 1
                        m = len(uniq)
                        p = yidx
                        while p <= m:
                            bit[p] = idx
                            p += p & -p
                    node >>= 1

    # Type B: crossing intervals with the same sign
    process_structure(plus_intervals, 1, f)
    process_structure(plus_intervals, 2, f)
    process_structure(minus_intervals, 1, f)
    process_structure(minus_intervals, 2, f)

    # Range maximum of f
    size_seg = 1
    while size_seg < M:
        size_seg <<= 1
    seg = [0] * (2 * size_seg)
    for i in range(1, M + 1):
        seg[size_seg + i - 1] = f[i]
    for i in range(size_seg - 1, 0, -1):
        seg[i] = seg[i << 1] if seg[i << 1] > seg[i << 1 | 1] else seg[i << 1 | 1]

    out = []
    for lq, rq in queries:
        left = lq + size_seg - 1
        right = rq + size_seg - 1
        mx = 0
        while left <= right:
            if left & 1:
                if seg[left] > mx:
                    mx = seg[left]
                left += 1
            if not (right & 1):
                if seg[right] > mx:
                    mx = seg[right]
                right -= 1
            left >>= 1
            right >>= 1
        if mx >= lq:
            out.append("No")
        else:
            out.append("Yes")

    sys.stdout.write("\n".join(out))
