#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<vector<int>> positions(N + 1);
    int distinctLabels = 0;

    for (int i = 0; i < N; ++i) {
        int a;
        cin >> a;
        if (positions[a].empty()) ++distinctLabels;
        positions[a].push_back(i);
    }

    vector<int> multiLabels;
    int singletonLabels = 0;

    for (int label = 1; label <= N; ++label) {
        if (positions[label].size() == 1) {
            ++singletonLabels;
        } else if (positions[label].size() > 1) {
            multiLabels.push_back(label);
        }
    }

    int B = max(1, (int)sqrt((double)N));
    int K = (N - 1) / (B + 1) + 1;

    vector<int> answer(N + 1, 0);
    vector<int> delta(N + 2, 0);

    // Directly compute the answers for small x.
    for (int x = 1; x <= B; ++x) {
        int total = singletonLabels;

        for (int label : multiLabels) {
            const vector<int> &v = positions[label];
            int m = (int)v.size();
            int j = 0;

            while (j < m) {
                ++total;
                int start = v[j++];
                while (j < m && v[j] - start <= x) ++j;
            }
        }

        answer[x] = total;
    }

    const int firstLargeX = B + 1;

    // A threshold g contributes one extra group for firstLargeX <= x < g.
    auto addThreshold = [&](int g) {
        if (g > firstLargeX) {
            ++delta[firstLargeX];
            --delta[g];
        }
    };

    /*
     * For each label, let g[k] be the smallest x allowing at most k groups.
     * For large x, only g[1], ..., g[min(m,K)-1] are needed.
     *
     * DP transition, covering the first i occurrences with k groups:
     * dp[i] = min over j of max(previous[j], v[i-1] - v[j]).
     * The monotone pointer finds the best j in amortized O(1).
     */
    for (int label : multiLabels) {
        const vector<int> &v = positions[label];
        int m = (int)v.size();
        int limit = min(m, K);

        if (limit <= 1) continue;

        int g = v.back() - v.front();
        addThreshold(g);

        if (g <= firstLargeX || limit == 2) continue;

        vector<int> previous(m + 1), current(m + 1);

        // DP for one group.
        for (int i = 1; i <= m; ++i) {
            previous[i] = v[i - 1] - v[0];
        }

        for (int k = 2; k < limit; ++k) {
            int ptr = k - 1;

            for (int i = k; i <= m; ++i) {
                int endpoint = v[i - 1];

                while (ptr < i - 1 &&
                       previous[ptr] + v[ptr] < endpoint) {
                    ++ptr;
                }

                int best = previous[ptr];
                if (ptr > k - 1) {
                    best = min(best, endpoint - v[ptr - 1]);
                }

                current[i] = best;
            }

            g = current[m];
            addThreshold(g);

            if (g <= firstLargeX) break;
            swap(previous, current);
        }
    }

    int extraGroups = 0;
    for (int x = firstLargeX; x <= N; ++x) {
        extraGroups += delta[x];
        answer[x] = distinctLabels + extraGroups;
    }

    for (int x = 1; x <= N; ++x) {
        cout << answer[x] << '\n';
    }

    return 0;
}

    int N;
    cin >> N;

    vector<vector<int>> positions(N + 1);
    for (int i = 1; i <= N; i++) {
        int a;
        cin >> a;
        positions[a].push_back(i);
    }

    int distinctLabels = 0;
    for (int label = 1; label <= N; label++) {
        if (!positions[label].empty()) distinctLabels++;
    }

    vector<int> direct(N + 1, 0);
    vector<int> delta(N + 3, 0);

    for (int label = 1; label <= N; label++) {
        const vector<int>& p = positions[label];
        int m = (int)p.size();
        if (m <= 1) continue;

        int L = m - 1;

        if (L == 1) {
            int span = p[1] - p[0];
            if (span > 1) {
                delta[1]++;
                delta[span]--;
            }
            continue;
        }

        vector<int> prefix(L + 1, 0);
        for (int i = 1; i <= L; i++) {
            prefix[i] = p[i] - p[0];
        }

        int span = prefix[L];
        int K = (int)sqrt((double)span);
        K = max(K, 1);
        K = min(K, L);

        vector<int> bottleneck(K + 1, 0);
        vector<int> previous(L + 1, 0), current(L + 1, 0), combined(L + 1, 0);

        for (int i = 1; i <= L; i++) previous[i] = prefix[i];
        bottleneck[1] = span;

        for (int r = 2; r <= K; r++) {
            int low = r - 1;

            for (int i = low; i <= L; i++) {
                combined[i] = previous[i] + prefix[i];
            }

            int pointer = low - 1;

            for (int j = r; j <= L; j++) {
                int high = j - 1;

                while (pointer < high &&
                       combined[pointer + 1] <= prefix[j]) {
                    pointer++;
                }

                int best = -1;

                if (pointer >= low) {
                    best = previous[pointer];
                }

                if (pointer < high) {
                    int candidate = prefix[j] - prefix[pointer + 1];
                    best = max(best, candidate);
                }

                current[j] = best;
            }

            bottleneck[r] = current[L];
            swap(previous, current);
        }

        if (K == L) {
            for (int r = 1; r <= L; r++) {
                int b = bottleneck[r];
                if (b > 1) {
                    delta[1]++;
                    delta[b]--;
                }
            }
        } else {
            int split = bottleneck[K];

            for (int x = 1; x < split; x++) {
                int extraGroups = 0;
                int last = p[0];

                for (int i = 1; i < m; i++) {
                    if (p[i] - last > x) {
                        extraGroups++;
                        last = p[i];
                    }
                }

                direct[x] += extraGroups;
            }

            for (int r = 1; r <= K; r++) {
                if (bottleneck[r] > split) {
                    delta[split]++;
                    delta[bottleneck[r]]--;
                }
            }
        }
    }

    int active = 0;
    for (int x = 1; x <= N; x++) {
        active += delta[x];
        cout << distinctLabels + direct[x] + active << '\n';
    }

    return 0;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1);
    vector<vector<int>> occ(N + 1);
    for (int i = 1; i <= N; i++) {
        cin >> a[i];
        occ[a[i]].push_back(i);
    }

    int B = max(1, (int)sqrt(2.0 * N));
    B = min(B, N);

    vector<int> answer(N + 1, 0);

    // Compute small x directly using the greedy maximum separated set.
    vector<int> last(N + 1, 0), seen(N + 1, 0);
    for (int x = 1; x <= B; x++) {
        int groups = 0;
        for (int i = 1; i <= N; i++) {
            int c = a[i];
            if (seen[c] != x) {
                seen[c] = x;
                last[c] = i;
                groups++;
            } else if (i - last[c] > x) {
                last[c] = i;
                groups++;
            }
        }
        answer[x] = groups;
    }

    // drop[r] is how many groups disappear when x reaches r.
    vector<int> drop(N + 2, 0);

    for (int c = 1; c <= N; c++) {
        const vector<int> &p = occ[c];
        int k = (int)p.size();
        if (k < 2) continue;

        int diameter = p.back() - p.front();
        if (diameter <= B) continue;

        // If R_v > B, then (v-1)*(B+1) <= diameter.
        int V = min(k, 1 + diameter / (B + 1));

        // R_2 is simply the diameter.
        drop[diameter]++;
        if (V == 2) continue;

        // prev[i]: best bottleneck for a chain of the current length
        // starting at p[0] and ending at p[i].
        vector<int> prev(k), cur(k);
        for (int i = 1; i < k; i++) {
            prev[i] = p[i] - p[0]; // Chains of length 2.
        }

        for (int v = 3; v <= V; v++) {
            int L = v - 2;       // First valid endpoint in prev.
            int h = L - 1;

            for (int i = v - 1; i < k; i++) {
                while (h + 1 < i &&
                       p[h + 1] + prev[h + 1] <= p[i]) {
                    h++;
                }

                int best = -1;
                if (h >= L) best = prev[h];

                int j = h + 1;
                if (j < i) best = max(best, p[i] - p[j]);

                cur[i] = best;
            }

            int R = cur[k - 1];
            if (R <= B) break; // Later R values are no larger.

            drop[R]++;
            swap(prev, cur);
        }
    }

    for (int x = B + 1; x <= N; x++) {
        answer[x] = answer[x - 1] - drop[x];
    }

    for (int x = 1; x <= N; x++) {
        cout << answer[x] << '\n';
    }

    return 0;
}

int count_groups_full(const vector<int>& P, int x) {
    int m = (int)P.size();
    int i = 0, cnt = 0;

    if (m > 700) {
        while (i < m) {
            ++cnt;
            int start = P[i];
            i = upper_bound(P.begin() + i + 1, P.end(), start + x) - P.begin();
        }
    } else {
        while (i < m) {
            ++cnt;
            int start = P[i];
            ++i;
            while (i < m && P[i] - start <= x) ++i;
        }
    }

    return cnt;
}

bool groups_leq(const vector<int>& P, int x, int k) {
    int m = (int)P.size();
    int i = 0, cnt = 0;

    if (m > 700) {
        while (i < m) {
            ++cnt;
            if (cnt > k) return false;

            int start = P[i];
            i = upper_bound(P.begin() + i + 1, P.end(), start + x) - P.begin();
        }
        return true;
    } else {
        while (i < m) {
            ++cnt;
            if (cnt > k) return false;

            int start = P[i];
            ++i;
            while (i < m && P[i] - start <= x) ++i;
        }
        return true;
    }
}

                start[label] = pos;
                ++groups;

    if (S < N) {

                        lo = mid;
                    }

                if (L <= R) {
                    int extra = k - 1;
                    diff[L] += extra;
                    diff[R + 1] -= extra;
                }

        for (int x = S + 1; x <= N; ++x) {

    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<vector<int>> pos(N + 1);
    for (int i = 0; i < N; ++i) {
        int a;
        cin >> a;
        pos[a].push_back(i); // 0-indexed positions
    }

    vector<vector<int>> groups;
    groups.reserve(N);
    for (int v = 1; v <= N; ++v) {
        if (!pos[v].empty()) {
            groups.push_back(move(pos[v]));
        }
    }

    // For small x, compute answers directly.
    const int T = min(N, 500);
    vector<long long> base(T + 1, 0);

    for (int x = 1; x <= T; ++x) {
        long long total = 0;

        for (const auto &p : groups) {
            int sz = (int)p.size();
            int i = 0;

            while (i < sz) {
                ++total;

                int limit = p[i] + x;
                int j = i;
                while (j + 1 < sz && p[j + 1] <= limit) {
                    ++j;
                }

                i = j + 1;
            }
        }

        base[x] = total;
    }

    if (T == N) {
        for (int x = 1; x <= N; ++x) {
            cout << base[x] << '\n';
        }
        return 0;
    }

    const int startX = T + 1;
    const int SMALL_K = 3000;

    vector<long long> diff(N + 3, 0);

    for (const auto &p : groups) {
        int k = (int)p.size();
        int span = p.back() - p.front();

        // For all x > T >= span, all occurrences can be in one group.
        if (span <= T) {
            diff[startX] += 1;
            diff[N + 1] -= 1;
            continue;
        }

        if (k <= SMALL_K) {
            vector<int> prev(k), cur(k);
            vector<int> d(k + 1);

            // One group.
            d[1] = span;
            for (int i = 0; i < k; ++i) {
                prev[i] = p[i] - p[0];
            }

            // DP to find d[m] = minimum possible maximum span when using m groups.
            for (int m = 2; m <= k; ++m) {
                int jptr = m - 1;

                for (int i = m - 1; i < k; ++i) {
                    while (jptr + 1 <= i &&
                           prev[jptr] + p[jptr + 1] <= p[i]) {
                        ++jptr;
                    }

                    int best = INT_MAX;

                    {
                        int j = jptr;
                        int A = prev[j - 1];
                        int B = p[i] - p[j];
                        best = max(A, B);
                    }

                    if (jptr + 1 <= i) {
                        int j = jptr + 1;
                        int A = prev[j - 1];
                        int B = p[i] - p[j];
                        best = min(best, max(A, B));
                    }

                    cur[i] = best;
                }

                d[m] = cur[k - 1];
                prev.swap(cur);
            }

            // Add this label's contribution for x > T.
            for (int m = 1; m <= k; ++m) {
                int L = (m == 1 ? d[1] : d[m]);
                int R = (m == 1 ? N : d[m - 1] - 1);

                if (L < startX) L = startX;
                if (R > N) R = N;

                if (L <= R) {
                    diff[L] += m;
                    diff[R + 1] -= m;
                }
            }
        } else {
            // Heavy label: simulate each x > T using binary search jumps.
            int end = min(N, span);

            for (int x = startX; x <= end; ++x) {
                int groupsCount = 0;
                int i = 0;

                while (i < k) {
                    ++groupsCount;

                    int limit = p[i] + x;

                    if (limit >= p.back()) {
                        break;
                    }

                    int lo = i + 1;
                    int hi = k;

                    while (lo < hi) {
                        int mid = (lo + hi) >> 1;
                        if (p[mid] <= limit) {
                            lo = mid + 1;
                        } else {
                            hi = mid;
                        }
                    }

                    i = lo;
                }

                diff[x] += groupsCount;
                diff[x + 1] -= groupsCount;
            }

            // For x > span, all occurrences fit into one group.
            int L = max(startX, span + 1);
            if (L <= N) {
                diff[L] += 1;
                diff[N + 1] -= 1;
            }
        }
    }

    long long curSum = 0;

    for (int x = 1; x <= N; ++x) {
        curSum += diff[x];

        long long ans;
        if (x <= T) {
            ans = base[x] + curSum;
        } else {
            ans = curSum;
        }

        cout << ans << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
const int B = 2000;

int N;
vector<int> pos[MAXN];
long long diffArr[MAXN];

inline void addThreshold(int d) {
    // Add +1 to extra[x] for 1 <= x < d
    if (d > 1) {
        diffArr[1] += 1;
        diffArr[d] -= 1;
    }
}

inline int calcGroups(const vector<int>& p, int x) {
    int m = (int)p.size();
    int idx = 0;
    int groups = 0;

    while (idx < m) {
        ++groups;
        int limit = p[idx] + x;
        idx = int(upper_bound(p.begin() + idx, p.end(), limit) - p.begin());
    }

    return groups;
}

void processSmall(const vector<int>& p) {
    int m = (int)p.size();
    if (m < 2) return;

    int span = p.back() - p.front();
    addThreshold(span);

    if (m == 2) return;

    vector<int> prev(m), cur(m);

    // For selecting 2 points ending at j, best minimum gap is p[j] - p[0].
    for (int j = 1; j < m; ++j) {
        prev[j] = p[j] - p[0];
    }

    // Compute M_k = maximum possible minimum gap when selecting k points.
    for (int k = 3; k <= m; ++k) {
        int L = k - 2;
        int t = L - 1;

        for (int i = k - 1; i < m; ++i) {
            while (t + 1 <= i - 1 &&
                   prev[t + 1] + p[t + 1] < p[i]) {
                ++t;
            }

            int best = 0;

            if (t >= L) best = prev[t];

            int j2 = max(L, t + 1);
            if (j2 <= i - 1) {
                best = max(best, p[i] - p[j2]);
            }

            cur[i] = best;
        }

        addThreshold(cur[m - 1]);
        swap(prev, cur);
    }
}

void solveRange(const vector<int>& p, int l, int r, int vl, int vr) {
    if (l > r) return;

    if (vl == vr) {
        if (vl > 0) {
            diffArr[l] += vl;
            diffArr[r + 1] -= vl;
        }
        return;
    }

    if (l == r) {
        if (vl > 0) {
            diffArr[l] += vl;
            diffArr[l + 1] -= vl;
        }
        return;
    }

    if (l + 1 == r) {
        if (vl > 0) {
            diffArr[l] += vl;
            diffArr[l + 1] -= vl;
        }
        if (vr > 0) {
            diffArr[r] += vr;
            diffArr[r + 1] -= vr;
        }
        return;
    }

    int mid = (l + r) >> 1;
    int vm = calcGroups(p, mid) - 1;

    solveRange(p, l, mid, vl, vm);

    int vm1;
    if (mid + 1 == r) vm1 = vr;
    else vm1 = calcGroups(p, mid + 1) - 1;

    solveRange(p, mid + 1, r, vm1, vr);
}

    int e1 = calcGroups(p, 1) - 1;
    if (e1 <= 0) return;

    int R = p.back() - p.front();
    solveRange(p, 1, R, e1, 0);
}

using namespace std;

static int countGroups(const vector<int>& p, int x) {
    int groups = 1;
    int first = p[0];

    for (int i = 1; i < (int)p.size(); ++i) {
        if (p[i] - first > x) {
            ++groups;
            first = p[i];
        }
    }

    return groups;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<vector<int>> positions(N + 1);
    for (int i = 1; i <= N; ++i) {
        int a;
        cin >> a;
        positions[a].push_back(i);
    }

    const int S = max(1, (int)sqrt((double)N));

    vector<int> answer(N + 1, 0);
    vector<int> drops(N + 1, 0);

    int usedLabels = 0;
    for (int label = 1; label <= N; ++label) {
        if (!positions[label].empty()) {
            ++usedLabels;
        }
    }

    // Every nonempty label contributes at least one group.
    for (int x = 1; x <= S; ++x) {
        answer[x] = usedLabels;
    }

    vector<int> dp, nextDp;

    for (int label = 1; label <= N; ++label) {
        const vector<int>& p = positions[label];
        int m = (int)p.size();

        if (m <= 1) {
            continue;
        }

        int span = p.back() - p.front();

        // Compute all small-x answers directly.
        for (int x = 1; x <= S; ++x) {
            if (span <= x) {
                continue;  // This label already contributes its one baseline group.
            }
            answer[x] += countGroups(p, x) - 1;
        }

        int b = countGroups(p, S);
        if (b <= 1) {
            continue;
        }

        /*
        dp[i] = minimum possible maximum group span when the first i
                occurrences are partitioned into exactly g groups.

        Initially g = 1.
        */
        dp.resize(m + 1);
        nextDp.resize(m + 1);

        for (int i = 1; i <= m; ++i) {
            dp[i] = p[i - 1] - p[0];
        }

        // D_1: all occurrences must be in one group.
        ++drops[span];

        /*
        For each g = 2,...,b-1, compute dp_g[i] in O(m).

        The optimal previous split is found by maintaining the first j for which
        dp_{g-1}[j] >= p[i-1] - p[j].
        */
        for (int g = 2; g < b; ++g) {
            int ptr = g - 1;

            for (int i = g; i <= m; ++i) {
                while (ptr < i - 1 &&
                       dp[ptr] < p[i - 1] - p[ptr]) {
                    ++ptr;
                }

                int best = dp[ptr];

                if (ptr > g - 1) {
                    best = min(best, p[i - 1] - p[ptr - 1]);
                }

                nextDp[i] = best;
            }

            dp.swap(nextDp);

            // D_g = dp_g[m].
            ++drops[dp[m]];
        }
    }

    // For x > S, apply all group-count decreases at their thresholds.
    int current = answer[S];
    for (int x = S + 1; x <= N; ++x) {
        current -= drops[x];
        answer[x] = current;
    }

    for (int x = 1; x <= N; ++x) {
        cout << answer[x] << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<vector<int>> positions(N + 1);
    for (int i = 1; i <= N; ++i) {
        int a;
        cin >> a;
        positions[a].push_back(i);
    }

    int labelCount = 0;
    for (int a = 1; a <= N; ++a) {
        if (!positions[a].empty()) ++labelCount;
    }

    // B is Theta(sqrt(N)); the constant is tuned to balance the two parts.
    int B = min(N, max(1, (int)sqrt(2.0 * N)));
    int K = 1 + (N - 1) / (B + 2);

    // Every nonempty label always contributes at least one group.
    vector<int> answer(N + 1, labelCount);
    vector<int> diff(N + 2, 0);

    vector<int> previous, current, D;

    for (int a = 1; a <= N; ++a) {
        const vector<int>& p = positions[a];
        int m = (int)p.size();

        if (m <= 1) continue;

        int span = p.back() - p.front();

        // Small x: compute the greedy number of groups directly.
        // For x >= span, all occurrences already form one group.
        int smallLimit = min(B, span - 1);
        for (int x = 1; x <= smallLimit; ++x) {
            int groups = 1;
            int groupEnd = p[0] + x;

            for (int i = 1; i < m; ++i) {
                if (p[i] > groupEnd) {
                    ++groups;
                    groupEnd = p[i] + x;
                }
            }

            answer[x] += groups - 1;
        }

        // At x >= B+1, span <= B+1 means this label forms one group.
        if (span <= B + 1) continue;

        int limit = min(m, K);
        if (limit < 2) continue;

        // previous[j] = F_{k-1}(j).
        // N acts as infinity because every distance is at most N-1.
        previous.assign(m, N);
        current.assign(m, 0);
        D.assign(limit + 1, 0);

        int lastComputed = 1;

        for (int k = 2; k <= limit; ++k) {
            int c = k - 2;

            for (int j = k - 1; j < m; ++j) {
                // Find the first crossing:
                // previous[c] >= p[j] - p[c].
                while (c < j && previous[c] + p[c] < p[j]) {
                    ++c;
                }

                int best = 0;

                // Right side of the crossing.
                if (c < j) {
                    best = p[j] - p[c];
                }

                // Left side of the crossing.
                if (c > k - 2) {
                    best = max(best, previous[c - 1]);
                }

                current[j] = best;
            }

            // F_k[j] is nondecreasing in j.
            D[k] = current[m - 1];
            lastComputed = k;

            previous.swap(current);

            // D_k is nonincreasing, so later thresholds cannot contribute.
            if (D[k] <= B + 1) break;
        }

        // D_k > x contributes one additional group exactly when
        // x is in [B+1, D_k-1].
        for (int k = 2; k <= lastComputed; ++k) {
            int d = D[k];
            if (d > B + 1) {
                ++diff[B + 1];
                --diff[d];
            }
        }
    }

    int extraGroups = 0;
    for (int x = B + 1; x <= N; ++x) {
        extraGroups += diff[x];
        answer[x] += extraGroups;
    }

    for (int x = 1; x <= N; ++x) {
        cout << answer[x] << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<vector<int>> positions(N + 1);
    for (int i = 1; i <= N; ++i) {
        int label;
        cin >> label;
        positions[label].push_back(i);
    }

    const int B = max(1, (int)sqrt((double)N));
    const int K = (N + B) / (B + 1);  // ceil(N / (B + 1))

    // For small x, directly run the optimal greedy algorithm.
    vector<int> smallAnswer(B + 1);
    vector<int> groupStart(N + 1);

    for (int x = 1; x <= B; ++x) {
        fill(groupStart.begin(), groupStart.end(), 0);

        int groups = 0;
        for (int i = 1; i <= N; ++i) {
            int label = a_label(i);
            if (groupStart[label] == 0 || i - groupStart[label] > x) {
                ++groups;
                groupStart[label] = i;
            }
        }
        smallAnswer[x] = groups;
    }

    // Difference array for the answers with x > B.
    vector<long long> diff(N + 2, 0);
    vector<int> previous, current;

    for (int label = 1; label <= N; ++label) {
        const vector<int>& p = positions[label];
        const int m = (int)p.size();
        if (m == 0) continue;

        if (m == 1) {
            if (B < N) {
                ++diff[B + 1];
                --diff[N + 1];
            }
            continue;
        }

        /*
          D[k] is the smallest possible maximum group span when the
          occurrences of this label are partitioned into k groups.

          previous[i] is the corresponding optimum for the first i
          occurrences using k-1 groups.
        */
        const int limitK = min(m, K);
        vector<int> D(limitK + 1);
        D[0] = N + 1;

        previous.resize(m + 1);
        current.resize(m + 1);

        // Base case: one group.
        previous[0] = 0;
        for (int i = 1; i <= m; ++i) {
            previous[i] = p[i - 1] - p[0];
        }
        D[1] = previous[m];

        for (int k = 2; k <= limitK; ++k) {
            int t = k - 1;

            for (int i = k; i <= m; ++i) {
                /*
                  Find the first split j for which
                  previous[j] > p[i-1] - p[j].
                */
                while (t < i && previous[t] + p[t] <= p[i - 1]) {
                    ++t;
                }

                if (t == k - 1) {
                    current[i] = previous[t];
                } else if (t == i) {
                    current[i] = 0;
                } else {
                    current[i] = min(previous[t], p[i - 1] - p[t - 1]);
                }
            }

            D[k] = current[m];
            previous.swap(current);
        }

        /*
          The minimum number of groups for a given x is k exactly when
          D[k] <= x < D[k-1].
        */
        for (int k = 1; k <= limitK; ++k) {
            int left = max(B + 1, D[k]);
            int right = min(N, D[k - 1] - 1);

            if (left <= right) {
                diff[left] += k;
                diff[right + 1] -= k;
            }
        }
    }

    vector<long long> answer(N + 1);
    for (int x = 1; x <= B; ++x) {
        answer[x] = smallAnswer[x];
    }

    long long running = 0;
    for (int x = B + 1; x <= N; ++x) {
        running += diff[x];
        answer[x] = running;
    }

    for (int x = 1; x <= N; ++x) {
        cout << answer[x] << '\n';
    }

    return 0;
}
