Lazy Sort

题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Farmer John has $N$ cows ($2 \leq N \leq 5\cdot 10^6$) and is attempting to get them to sort a non-negative integer array $A$ of length $N$ by relying on their laziness. He has a lot of heavy boxes so he lines the cows up one behind another, where cow $i+1$ is behind cow $i$, and gives $a_i$ boxes to cow $i$ ($0\le a_i$).

Cows are inherently lazy so they always look to pass their work off to someone else. From cow $1$ to $N-1$ in order, each cow looks to the cow behind them. If cow $i$ has strictly more boxes than cow $i+1$, cow $i$ thinks this is "unfair" and gives one of its boxes to cow $i+1$. This process repeats until every cow is satisfied.

Farmer John will then note the number of boxes $b_i$ that each cow $i$ is holding and create an array $B$ out of these values. If $B = sorted(A)$, then Farmer John will be happy. Unfortunately, Farmer John forgot all but $Q$ values ($2 \leq Q \leq \min(N, 100)$) in $A$. Luckily, those values include the number of boxes he was going to give to the first and last cow. Each value that FJ remembers is given in the form $c_i \; v_i$ representing that $a_{c_i}=v_i$ ($1 \leq c_i \leq N$, $1\le v_i\le 10^9$). Determine the number of different ways the missing values can be filled in so that he will be happy mod $10^9+7$.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains two space-separated integers $N$ and $Q$ representing the number of cows and queries respectively.

The next $Q$ lines contain two space separated integers $c_i \; v_i$ representing that cow $c_i$ initially holds $v_i$ boxes. It is guaranteed that $c_1 = 1$, $c_Q = N$, and $c_i < c_{i+1}$ (the order of the cows is strictly increasing).

OUTPUT FORMAT (print output to the terminal / stdout):

Print the number of different ways modulo $10^9+7$ that values $a_i$ can be assigned such that Farmer John will be happy after the cows perform the lazy sort. It is guaranteed that there will be at least one valid assignment.

SAMPLE INPUT:

3 2
1 3
3 2

SAMPLE OUTPUT:

2
In this example, FJ remembers the values at the ends of the array. The arrays $[3,2,2]$ and $[3,3,2]$ are the valid arrays that will make FJ happy at the end of the lazy sorting.

SAMPLE INPUT:

6 3
1 1
3 3
6 5

SAMPLE OUTPUT:

89

SCORING:

  • Inputs 3-4: $N,v_i\le 100$
  • Inputs 5-6: $N\le 100$ and $v_i \leq 10^6$
  • Inputs 7-9: $N \leq 2\cdot 10^5$ and $v_i \le 10^6$
  • Inputs 10-12: $N \leq 2\cdot 10^5$
  • Inputs 13-15: No additional constraints.

#include <bits/stdc++.h>
using namespace std;

static const long long MOD = 1000000007LL;
static const long long INV2 = 500000004LL;

vector<int> pow2, inv;

// S(d,h) = sum_r C(d,r) * C(h,r) * 2^(d-r).
long long calcS(int d, long long h) {
    if (h < 0) return 0;
    if (d == 0) return 1;
    if (h == 0) return pow2[d];

    if (h < d) {
        // Evaluate the sum in O(h).
        int k = (int)h;
        long long term = pow2[d];
        long long result = term;

        for (int r = 0; r < k; ++r) {
            long long factor = 1LL * (d - r) * (h - r) % MOD;
            term = term * factor % MOD;
            term = term * inv[r + 1] % MOD;
            term = term * inv[r + 1] % MOD;
            term = term * INV2 % MOD;

            result += term;
            if (result >= MOD) result -= MOD;
        }
        return result;
    }

    // Otherwise use the recurrence obtained from
    // sum_d S(d,h)x^d = (1-x)^h / (1-2x)^(h+1).
    long long hm = h % MOD;
    long long previous = 0, current = 1;

    for (int n = 0; n < d; ++n) {
        long long coefficient = (hm + 3LL * n + 2) % MOD;
        long long next = coefficient * current % MOD;

        if (n > 0) {
            next -= (2LL * n % MOD) * previous % MOD;
            if (next < 0) next += MOD;
        }

        next = next * inv[n + 1] % MOD;
        previous = current;
        current = next;
    }

    return current;
}

    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    vector<int> position(Q);
    vector<long long> value(Q);

    for (int i = 0; i < Q; ++i) {
        cin >> position[i] >> value[i];
    }

    pow2.resize(N + 1);
    inv.resize(N + 1);

    pow2[0] = 1;
    for (int i = 1; i <= N; ++i) {
        pow2[i] = 2LL * pow2[i - 1] % MOD;
    }

    inv[1] = 1;
    for (int i = 2; i <= N; ++i) {
        inv[i] = MOD - 1LL * (MOD / i) * inv[MOD % i] % MOD;
    }

    // After a fixed position with value v, the running maximum is either
    // v (low state) or v+1 (high state).
    long long dpLow = 1, dpHigh = 0;

    for (int i = 0; i + 1 < Q; ++i) {
        int d = position[i + 1] - position[i] - 1;
        long long delta = value[i + 1] - value[i];

        // Cache S(d,delta-1), S(d,delta), S(d,delta+1).
        long long heights[3] = {delta - 1, delta, delta + 1};
        long long cached[3] = {0, 0, 0};
        bool ready[3] = {false, false, false};

        auto getS = [&](int index) -> long long {
            if (index < 0 || index > 2 || heights[index] < 0) return 0;
            if (!ready[index]) {
                cached[index] = calcS(d, heights[index]);
                ready[index] = true;
            }
            return cached[index];
        };

            if (ways == 0) return;

            long long total = getS(totalIndex);
            long long same = getS(sameIndex);

            nextHigh = (nextHigh + ways * higher) % MOD;

        if (delta >= -1) {
            transition(dpLow, 2, 1);

    }

    pow2[0] = 1;
    const long long INV2 = (MOD + 1LL) / 2;

    for (int i = 1; i <= N; ++i) {
        pow2[i] = (2LL * pow2[i - 1]) % MOD;
        if (i == 1) {
            inv[i] = 1;
        } else {
            inv[i] = MOD - (long long)(MOD / i) * inv[MOD % i] % MOD;
        }
        ratio[i] = (long long)inv[i] * INV2 % MOD;
    }

    // Returns U(k,d-1), U(k,d), U(k,d+1), where
    // U(k,n) = sum_j C(k,j) C(n,j) 2^(k-j).
    auto eval3 = [&](int k, long long d) -> array<int, 3> {
        array<int, 3> sum{pow2[k], pow2[k], pow2[k]};
        int lim = (int)min<long long>(k, d + 1);

        int w = pow2[k];
        int cur = 1;       // C(d,j)
        int prevA = 1;     // C(d-1,j-1)
        int prevB = 1;     // C(d,j-1)

        for (int j = 1; j <= lim; ++j) {
            w = (long long)w * (k - j + 1) % MOD;
            w = (long long)w * ratio[j] % MOD;

            long long factor = (d - j + 1) % MOD;
            if (factor < 0) factor += MOD;

            int b = (long long)cur * factor % MOD;
            b = (long long)b * inv[j] % MOD;

            int a = b - prevA;
            if (a < 0) a += MOD;

            int c = b + prevB;
            if (c >= MOD) c -= MOD;

            cur = b;
            prevA = a;
            prevB = b;

            sum[0] += (long long)w * a % MOD;
            if (sum[0] >= MOD) sum[0] -= MOD;
            sum[1] += (long long)w * b % MOD;
            if (sum[1] >= MOD) sum[1] -= MOD;
            sum[2] += (long long)w * c % MOD;
            if (sum[2] >= MOD) sum[2] -= MOD;
        }
        return sum;
    };

    // After processing a fixed value R, the suffix minimum is R or R-1.
    long long alpha = 1; // suffix minimum R
    long long beta = 0;  // suffix minimum R-1

    for (int i = Q - 2; i >= 0; --i) {
        int k = pos[i + 1] - pos[i] - 1;
        long long d = val[i + 1] - val[i];
        long long nextAlpha = 0, nextBeta = 0;

        if (d >= 1) {
            auto u = eval3(k, d);

            nextAlpha = (alpha * u[1] + beta * u[0]) % MOD;

            int diffNext = u[2] - u[1];
            if (diffNext < 0) diffNext += MOD;
            int diffCur = u[1] - u[0];
            if (diffCur < 0) diffCur += MOD;

            nextBeta = (alpha * diffNext + beta * diffCur) % MOD;
        } else if (d == 0) {
            long long p = pow2[k];
            long long diff = 0;
            if (k > 0) {
                diff = (long long)k * pow2[k - 1] % MOD;
            }
            nextAlpha = alpha * p % MOD;
            nextBeta = (alpha * diff + beta * p) % MOD;
        } else if (d == -1) {
            nextAlpha = 0;
            nextBeta = alpha * pow2[k] % MOD;
        }

        alpha = nextAlpha;
        beta = nextBeta;
    }

    cout << (alpha + beta) % MOD << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

static const int MOD = 1000000007;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    vector<int> pos(Q);
    vector<long long> val(Q);
    int maxGap = 0;

    for (int i = 0; i < Q; i++) {
        cin >> pos[i] >> val[i];
        if (i > 0) {
            maxGap = max(maxGap, pos[i] - pos[i - 1] - 1);
        }
    }

    vector<int> pow2(maxGap + 1);
    vector<int> inv(maxGap + 1);

    pow2[0] = 1;
    for (int i = 1; i <= maxGap; i++) {
        pow2[i] = (int)((2LL * pow2[i - 1]) % MOD);
    }

    if (maxGap >= 1) {
        inv[1] = 1;
        for (int i = 2; i <= maxGap; i++) {
            inv[i] = (int)(MOD - (long long)(MOD / i) * inv[MOD % i] % MOD);
        }
    }

    // Counts one segment whose left prefix maximum is M, with k free
    // positions followed by the fixed value y. The returned weights are
    // for ending with prefix maximum y and y+1, respectively.
    auto segmentWeights = [&](int k, long long M, long long y)
        -> pair<int, int> {
        if (M > y + 1) return {0, 0};

        if (M == y + 1) {
            return {0, pow2[k]};
        }

        if (M == y) {
            int higher = 0;
            if (k > 0) {
                higher = (int)((long long)k * pow2[k - 1] % MOD);
            }
            return {pow2[k], higher};
        }

        // Here d = y-M >= 1.
        // Let
        // F(d) = sum_j C(k,j) C(d-1,j-1) 2^(k-j),
        // G(d) = sum_j C(k,j) C(d-1,j)   2^(k-j).
        // The two answers are F(d)+G(d) and F(d+1).
        long long d = y - M;
        long long D1 = d - 1;
        long long D2 = d;

        int limit = (int)min<long long>(k, d + 1);

        long long F1 = 0;
        long long G = pow2[k];
        long long F2 = 0;

        long long chooseK = 1;
        long long p1 = 1; // C(D1, j-1)
        long long p2 = 1; // C(D2, j-1)

        for (int j = 1; j <= limit; j++) {
            chooseK = chooseK * (k - j + 1) % MOD;
            chooseK = chooseK * inv[j] % MOD;

            long long ways = pow2[k - j];

            if (j <= d) {
                long long term = chooseK * p1 % MOD;
                term = term * ways % MOD;
                F1 += term;
                if (F1 >= MOD) F1 -= MOD;
            }

            long long term2 = chooseK * p2 % MOD;
            term2 = term2 * ways % MOD;
            F2 += term2;
            if (F2 >= MOD) F2 -= MOD;

            if (j <= D1) {
                long long current = p1 * (D1 - j + 1) % MOD;
                current = current * inv[j] % MOD;

                long long term = chooseK * current % MOD;
                term = term * ways % MOD;
                G += term;
                if (G >= MOD) G -= MOD;

                p1 = current;
            }

            if (j <= D2) {
                p2 = p2 * (D2 - j + 1) % MOD;
                p2 = p2 * inv[j] % MOD;
            }
        }

        int first = (int)(F1 + G);
        if (first >= MOD) first -= MOD;
        return {first, (int)F2};
    };

    // State 0: prefix maximum at the current remembered position is its value.
    // State 1: prefix maximum is one greater than its value.
    long long dp[2] = {1, 0};

    for (int i = 1; i < Q; i++) {
        int k = pos[i] - pos[i - 1] - 1;
        long long x = val[i - 1];
        long long y = val[i];

        long long next[2] = {0, 0};

        for (int state = 0; state < 2; state++) {
            if (dp[state] == 0) continue;

            pair<int, int> weights = segmentWeights(k, x + state, y);

            next[0] = (next[0] + dp[state] * weights.first) % MOD;
            next[1] = (next[1] + dp[state] * weights.second) % MOD;
        }

        dp[0] = next[0];
        dp[1] = next[1];
    }

    cout << (dp[0] + dp[1]) % MOD << '\n';
    return 0;
}

int64 modPow(int64 a, int64 e) {
    int64 result = 1;
    a %= MOD;
    while (e > 0) {
        if (e & 1) result = result * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return result;
}

struct ExtensionCounts {
    int64 pD;      // P(D)
    int64 pD1;     // P(D + 1)
    int64 atMostD; // Sum_{d=0}^D P(d)
};

/*
    P(d) is the number of ways to fill a gap of length len so that
    the prefix maximum increases by exactly d.

    For d >= 1:
        P(d) = sum_{r=1}^{min(len,d)}
               C(len,r) C(d-1,r-1) 2^(len-r).

    This function computes P(D), P(D+1), and sum_{d=0}^D P(d).
*/
ExtensionCounts countExtension(int len, int64 D, const vector<int>& inv) {
    int64 twoL = modPow(2, len);

    // Conventions: P(d) = 0 and A(d) = 0 for d < 0.
    if (D < 0) {
        return {0, D == -1 ? twoL : 0, 0};
    }

    if (len == 0) {
        return {D == 0 ? 1LL : 0LL, 0, 1};
    }

    if (D == 0) {
        int64 p1 = int64(len) * twoL % MOD * INV2 % MOD;
        return {twoL, p1, twoL};
    }

    /*
        V_r = C(len,r) C(D,r-1) 2^(len-r).

        P(D+1) = sum V_r.
        P(D)   = D^{-1} sum (D-r+1) V_r.
    */
    int limit = int(min<int64>(len, D + 1));
    int64 V = int64(len) * twoL % MOD * INV2 % MOD;

    int64 sumV = 0;
    int64 weighted = 0;

    for (int r = 1; r <= limit; ++r) {
        sumV += V;
        if (sumV >= MOD) sumV -= MOD;

        int64 factor = D - r + 1;
        weighted = (weighted + V * factor) % MOD;

        if (r < limit) {
            // 1/r - 1/(r+1) = 1/(r(r+1)).
            int64 delta = int64(inv[r]) - inv[r + 1];
            if (delta < 0) delta += MOD;

            V = V * (len - r) % MOD;
            V = V * factor % MOD;
            V = V * delta % MOD;
            V = V * INV2 % MOD;
        }
    }

    int64 pD1 = sumV;
    int64 pD = weighted * modPow(D, MOD - 2) % MOD;

    /*
        Let F(z) = sum_{d>=0} P(d) z^d = (2-z)^len / (1-z)^len.

        Then sum_{d=0}^D P(d) = [z^D] F(z)/(1-z), and
        F(z)/(1-z) = (2-z) F'(z) / len.

        Therefore:
        A(D) = (2(D+1)P(D+1) - D P(D)) / len.
    */
    int64 first = (2 * (D + 1)) % MOD * pD1 % MOD;
    int64 second = D % MOD * pD % MOD;
    int64 atMostD = (first - second + MOD) % MOD * inv[len] % MOD;

    return {pD, pD1, atMostD};
}

struct Kernel {
    int64 s;       // S_g(D)
    int64 exact;   // E_g(D)
    int64 exactNext; // E_g(D + 1)
};

// Computes S_g(D), E_g(D), and E_g(D + 1), where D >= 1.
//
// A_k = C(g, k) * C(D + k - 1, k - 1)
// z_k = A_k * (D + k)
//
// E_g(D)     = sum A_k
// E_g(D + 1) = (sum z_k) / (D + 1)
// S_g(D)     = 1 + sum z_k / k
//
// A_1 = g
// A_{k+1} = A_k * (g-k) * (D+k) / (k(k+1))
Kernel calculateKernel(int g, int64 D, const vector<int>& inv) {
    if (g == 0) return {1, 0, 0};

    int64 A = g;
    int64 sumA = 0;
    int64 sumZ = 0;
    int64 sumWeighted = 0;

    for (int k = 1; k <= g; ++k) {
        int64 z = A * (D + k) % MOD;

        sumA += A;
        if (sumA >= MOD) sumA -= MOD;

        sumZ += z;
        if (sumZ >= MOD) sumZ -= MOD;

        int64 weighted = z * inv[k] % MOD;
        sumWeighted += weighted;
        if (sumWeighted >= MOD) sumWeighted -= MOD;

        if (k < g) {
            A = z * (g - k) % MOD;
            A = A * inv[k] % MOD;
            A = A * inv[k + 1] % MOD;
        }
    }

    int64 exactNext = sumZ * modPow(D + 1, MOD - 2) % MOD;
    int64 s = (sumWeighted + 1) % MOD;

    return {s, sumA, exactNext};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    vector<int> position(Q), value(Q);
    for (int i = 0; i < Q; ++i) {
        cin >> position[i] >> value[i];
    }

    int maxGap = 0;
    for (int i = 1; i < Q; ++i) {
        maxGap = max(maxGap, position[i] - position[i - 1] - 1);
    }

    vector<int> inv(maxGap + 1, 0);
    if (maxGap >= 1) inv[1] = 1;
    for (int i = 2; i <= maxGap; ++i) {
        inv[i] = MOD - (int64)(MOD / i) * inv[MOD % i] % MOD;
    }

    // After the most recently processed remembered position:
    // dp0 = ways whose prefix maximum is value[i]
    // dp1 = ways whose prefix maximum is value[i] + 1
    int64 dp0 = 1;
    int64 dp1 = 0;

    for (int i = 1; i < Q; ++i) {
        int g = position[i] - position[i - 1] - 1;
        int64 D = (int64)value[i] - value[i - 1];

        int64 next0 = 0;
        int64 next1 = 0;

        if (D >= 1) {
            Kernel kernel = calculateKernel(g, D, inv);

            // S_g(D - 1) = S_g(D) - E_g(D).
            int64 previousS = (kernel.s - kernel.exact + MOD) % MOD;

            next0 = (dp0 * kernel.s + dp1 * previousS) % MOD;
            next1 = (dp0 * kernel.exactNext + dp1 * kernel.exact) % MOD;
        } else if (D == 0) {
            int64 powerOfTwo = modPow(2, g);
            int64 exactOne = 0;
            if (g > 0) {
                exactOne = (int64)g * modPow(2, g - 1) % MOD;
            }

            next0 = dp0 * powerOfTwo % MOD;
            next1 = (dp0 * exactOne + dp1 * powerOfTwo) % MOD;
        } else if (D == -1) {
            next1 = dp0 * modPow(2, g) % MOD;
        }

        dp0 = next0;
        dp1 = next1;
    }

    cout << (dp0 + dp1) % MOD << '\n';
    return 0;
}
