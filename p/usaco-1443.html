import sys


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    t = data[0]
    ns = data[1:1 + t]
    max_n = max(ns, default=1)

    # For P >= 2, the differing values form the interval
    # [444...445, 499...999], with P digits in each endpoint pattern.
    intervals = []
    lower = 45
    p = 2
    while lower <= max_n:
        upper = 5 * 10 ** (p - 1) - 1
        intervals.append((lower, upper))
        lower = 10 * lower - 5
        p += 1

    answers = []
    for n in ns:
        count = 0
        for lower, upper in intervals:
            if lower > n:
                break
            count += min(n, upper) - lower + 1
        answers.append(str(count))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    t = data[0]
    queries = data[1:1 + t]

    max_n = max(queries) if queries else 0

    intervals = []
    pow10 = 10  # 10^m for m >= 1

    while True:
        # Valid numbers for this m are:
        # from 4*10^m + t_m to 5*10^m - 1,
        # where t_m = (4*10^m + 5) // 9.
        t_m = (4 * pow10 + 5) // 9
        start = 4 * pow10 + t_m
        end = 5 * pow10 - 1

        if start > max_n:
            break

        intervals.append((start, end))
        pow10 *= 10

    answers = []
    for n in queries:
        ans = 0
        for start, end in intervals:
            if start > n:
                break
            ans += min(n, end) - start + 1
        answers.append(str(ans))

    sys.stdout.write("\n".join(answers))

if __name__ == "__main__":
    main()

import sys
from bisect import bisect_left


def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    t = int(data[0])

    MAX_P = 9  # 10^9 is the maximum possible value of N

    power10 = [1] * (MAX_P + 1)
    for p in range(1, MAX_P + 1):
        power10[p] = power10[p - 1] * 10

    threshold = [0] * (MAX_P + 1)
    upper = [0] * (MAX_P + 1)
    prefix = [0] * (MAX_P + 1)

    # prefix[p] counts all differing integers whose power is at most 10^p.
    for p in range(1, MAX_P + 1):
        if p == 1:
            threshold[p] = 5
        else:
            threshold[p] = 4 * power10[p - 1] + threshold[p - 1]

        upper[p] = 5 * power10[p - 1]
        prefix[p] = prefix[p - 1] + upper[p] - threshold[p]

    search_powers = power10[1:]
    answers = []

    for i in range(1, t + 1):
        n = int(data[i])

        # Smallest p such that 10^p >= n.
        # For n = 1 this gives p = 1, which correctly yields an answer of 0.
        p = bisect_left(search_powers, n) + 1

        answer = prefix[p - 1]

        # Add the qualifying integers from the current power-of-ten range.
        end = min(n, upper[p] - 1)
        if end >= threshold[p]:
            answer += end - threshold[p] + 1

        answers.append(str(answer))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()
