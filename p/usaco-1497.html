题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Bessie has a special function $f(x)$ that takes as input an integer in $[1, N]$ and returns an integer in $[1, N]$ ($1 \le N \le 2 \cdot 10^5$). Her function $f(x)$ is defined by $N$ integers $a_1 \ldots a_N$ where $f(x) = a_x$ ($1 \le a_i \le N$).

Bessie wants this function to be idempotent. In other words, it should satisfy $f(f(x)) = f(x)$ for all integers $x \in [1, N]$.

For a cost of $c_i$, Bessie can change the value of $a_i$ to any integer in $[1, N]$ ($1 \le c_i \le 10^9$). Determine the minimum total cost Bessie needs to make $f(x)$ idempotent.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $N$.

The second line contains $N$ space-separated integers $a_1,a_2,\dots,a_N$.

The third line contains $N$ space-separated integers $c_1,c_2,\dots,c_N$.

OUTPUT FORMAT (print output to the terminal / stdout):

Output the minimum total cost Bessie needs to make $f(x)$ idempotent.

SAMPLE INPUT:

5
2 4 4 5 3
1 1 1 1 1

SAMPLE OUTPUT:

3

We can change $a_1 = 4$, $a_4 = 4$, $a_5 = 4$. Since all $c_i$ equal one, the total cost is equal to $3$, the number of changes. It can be shown that there is no solution with only $2$ or fewer changes.

SAMPLE INPUT:

8
1 2 5 5 3 3 4 4
9 9 2 5 9 9 9 9

SAMPLE OUTPUT:

7

We change $a_3 = 3$ and $a_4 = 4$. The total cost is $2+5=7$.

SCORING:

Subtasks:

  • Input 3: $N\le 20$
  • Inputs 4-9: $a_i\ge i$
  • Inputs 10-15: All $a_i$ are distinct.
  • Inputs 16-21: No additional constraints.

Additionally, in each of the last three subtasks, the first half of tests will satisfy $c_i=1$ for all $i$.

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1);
    vector<int64> c(N + 1);

    for (int i = 1; i <= N; ++i) cin >> a[i];
    for (int i = 1; i <= N; ++i) cin >> c[i];

    vector<int> indegree(N + 1, 0);
    int64 totalCost = 0;

    for (int i = 1; i <= N; ++i) {
        ++indegree[a[i]];
        if (a[i] != i) totalCost += c[i];
    }

    // dp0[v]: maximum savings in v's tree subtree if v is not fixed.
    // dp1[v]: same, if v is fixed. The edge from v to its parent is excluded.
    vector<int64> dp0(N + 1, 0), dp1(N + 1, 0);
    vector<char> removed(N + 1, false);

    queue<int> q;
    for (int i = 1; i <= N; ++i) {
        if (indegree[i] == 0) q.push(i);
    }

    // Peel off all non-cycle vertices, accumulating tree DP values.
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        removed[v] = true;

        int p = a[v];
        dp0[p] += max(dp0[v], dp1[v]);
        dp1[p] += max(dp1[v], dp0[v] + c[v]);

        if (--indegree[p] == 0) q.push(p);
    }

    const int64 NEG = -(1LL << 60);
    vector<char> seen(N + 1, false);
    int64 totalSavings = 0;

    auto baseValue = [&](int v, int state) -> int64 {
        return state == 0 ? dp0[v] : dp1[v];
    };

    // Solve the remaining cycle of each component.
    for (int i = 1; i <= N; ++i) {
        if (removed[i] || seen[i]) continue;

        vector<int> cycle;
        int v = i;
        while (!seen[v]) {
            seen[v] = true;
            cycle.push_back(v);
            v = a[v];
        }

        int L = (int)cycle.size();

        if (L == 1) {
            int u = cycle[0];
            totalSavings += max(dp0[u], dp1[u]);
            continue;
        }

        // Vertices are in directed order: cycle[k] -> cycle[k+1].
        // An edge saves its cost exactly when its tail is not fixed and
        // its head is fixed.
        int64 best = NEG;

        for (int firstState = 0; firstState < 2; ++firstState) {
            int64 cur[2] = {NEG, NEG};
            cur[firstState] = baseValue(cycle[0], firstState);

            for (int k = 1; k < L; ++k) {
                int64 nxt[2] = {NEG, NEG};

                for (int prevState = 0; prevState < 2; ++prevState) {
                    if (cur[prevState] == NEG) continue;

                    for (int state = 0; state < 2; ++state) {
                        int64 value = cur[prevState]
                                    + baseValue(cycle[k], state);

                        if (prevState == 0 && state == 1) {
                            value += c[cycle[k - 1]];
                        }

                        nxt[state] = max(nxt[state], value);
                    }
                }

                cur[0] = nxt[0];
                cur[1] = nxt[1];
            }

            for (int lastState = 0; lastState < 2; ++lastState) {
                int64 value = cur[lastState];
                if (lastState == 0 && firstState == 1) {
                    value += c[cycle.back()];
                }
                best = max(best, value);
            }
        }

        totalSavings += best;
    }

    cout << totalCost - totalSavings << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1);
    vector<int64> c(N + 1);
    for (int i = 1; i <= N; ++i) cin >> a[i];
    for (int i = 1; i <= N; ++i) cin >> c[i];

    int64 totalCost = 0;
    for (int i = 1; i <= N; ++i) totalCost += c[i];

    vector<vector<int>> incoming(N + 1);
    vector<int> degree(N + 1, 0);
    for (int i = 1; i <= N; ++i) {
        incoming[a[i]].push_back(i);
        ++degree[a[i]];
    }

    // Remove all non-cycle nodes, from the leaves toward the cycles.
    vector<int> q;
    for (int i = 1; i <= N; ++i) {
        if (degree[i] == 0) q.push_back(i);
    }

    vector<int> order;
    order.reserve(N);
    for (size_t head = 0; head < q.size(); ++head) {
        int v = q[head];
        order.push_back(v);
        int p = a[v];
        if (--degree[p] == 0) q.push_back(p);
    }

    vector<char> isCycle(N + 1, false);
    for (int i = 1; i <= N; ++i) {
        if (degree[i] > 0) isCycle[i] = true;
    }

    // dp[v][s]: maximum savings in v's tree subtree, excluding the
    // saving on v's own outgoing edge, when v has state s.
    // State 1 means v is chosen as a fixed point.
    vector<array<int64, 2>> dp(N + 1, {0, 0});

    auto computeDP = [&](int v) {
        int64 d0 = 0, d1 = 0;
        for (int u : incoming[v]) {
            if (isCycle[u]) continue;
            d0 += max(dp[u][0], dp[u][1]);
            d1 += max(dp[u][0] + c[u], dp[u][1]);
        }
        dp[v][0] = d0;
        dp[v][1] = d1;
    };

    for (int v : order) computeDP(v);
    for (int v = 1; v <= N; ++v) {
        if (isCycle[v]) computeDP(v);
    }

    const int64 NEG = -(1LL << 60);
    vector<char> visited(N + 1, false);
    int64 totalSavings = 0;

    for (int start = 1; start <= N; ++start) {
        if (!isCycle[start] || visited[start]) continue;

        vector<int> cycle;
        int v = start;
        do {
            visited[v] = true;
            cycle.push_back(v);
            v = a[v];
        } while (v != start);

        int L = (int)cycle.size();

        if (L == 1) {
            int r = cycle[0];
            totalSavings += max(dp[r][0], dp[r][1] + c[r]);
            continue;
        }

        int64 best = 0;

        // Try both states for the first cycle node, then run linear DP.
        for (int firstState = 0; firstState <= 1; ++firstState) {
            int64 d[2] = {NEG, NEG};
            d[firstState] = dp[cycle[0]][firstState];

            for (int k = 1; k < L; ++k) {
                int prev = cycle[k - 1];
                int cur = cycle[k];
                int64 nd[2];

                nd[0] = dp[cur][0] + max(d[0], d[1]);
                nd[1] = dp[cur][1] + max(d[0] + c[prev], d[1]);

                d[0] = nd[0];
                d[1] = nd[1];
            }

            int last = cycle.back();
            int64 closingFromZero =
                (firstState == 1 ? c[last] : 0);

            best = max(best, d[0] + closingFromZero);
            best = max(best, d[1]);
        }

        totalSavings += best;
    }

    cout << totalCost - totalSavings << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const ll NEG = -(1LL << 60);

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1);
    vector<ll> cost(N + 1);

    for (int i = 1; i <= N; i++) cin >> a[i];

    ll totalCost = 0;
    for (int i = 1; i <= N; i++) {
        cin >> cost[i];
        totalCost += cost[i];
    }

    // weight[v] is the saving from selecting only v as a fixed point.
    vector<ll> weight(N + 1, 0);
    vector<int> indegree(N + 1, 0);

    for (int i = 1; i <= N; i++) {
        indegree[a[i]]++;
        if (a[i] == i)
            weight[i] += cost[i];
        else
            weight[a[i]] += cost[i];
    }

    // Remove all vertices outside the directed cycles.
    queue<int> q;
    for (int i = 1; i <= N; i++) {
        if (indegree[i] == 0) q.push(i);
    }

    vector<char> removed(N + 1, false);
    vector<int> order;
    order.reserve(N);

    while (!q.empty()) {
        int v = q.front();
        q.pop();

        removed[v] = true;
        order.push_back(v);

        int p = a[v];
        if (--indegree[p] == 0) q.push(p);
    }

    // dp[v][s]: best saving in v's off-cycle subtree when v has state s.
    // State 1 means v is chosen to become a fixed point.
    vector<array<ll, 2>> dp(N + 1);
    for (int i = 1; i <= N; i++) {
        dp[i] = {0, weight[i]};
    }

    // Process tree vertices from the leaves toward their cycle.
    for (int v : order) {
        int p = a[v];

        ll ifParentNotSelected = max(dp[v][0], dp[v][1]);
        ll ifParentSelected = max(dp[v][0], dp[v][1] - cost[v]);

        dp[p][0] += ifParentNotSelected;
        dp[p][1] += ifParentSelected;
    }

    vector<char> visited(N + 1, false);
    ll maximumSaving = 0;

    // Solve the remaining cycles, including self-loops.
    for (int start = 1; start <= N; start++) {
        if (removed[start] || visited[start]) continue;

        vector<int> cycle;
        int v = start;
        do {
            cycle.push_back(v);
            visited[v] = true;
            v = a[v];
        } while (v != start);

        if (cycle.size() == 1) {
            int u = cycle[0];
            maximumSaving += max(dp[u][0], dp[u][1]);
            continue;
        }

        ll best = 0;

        // Fix the state of the first cycle vertex to handle the closing edge.
        for (int firstState = 0; firstState <= 1; firstState++) {
            array<ll, 2> cur = {NEG, NEG};
            cur[firstState] = dp[cycle[0]][firstState];

            for (int i = 1; i < (int)cycle.size(); i++) {
                array<ll, 2> nxt = {NEG, NEG};
                int previous = cycle[i - 1];
                int current = cycle[i];

                for (int previousState = 0; previousState <= 1; previousState++) {
                    if (cur[previousState] == NEG) continue;

                    for (int currentState = 0; currentState <= 1; currentState++) {
                        ll value = cur[previousState] + dp[current][currentState];

                        // Both endpoints selected: the saving from the original
                        // edge previous -> current must be removed.
                        if (previousState && currentState)
                            value -= cost[previous];

                        nxt[currentState] = max(nxt[currentState], value);
                    }
                }

                cur = nxt;
            }

            int last = cycle.back();
            for (int lastState = 0; lastState <= 1; lastState++) {
                if (cur[lastState] == NEG) continue;

                ll value = cur[lastState];
                if (firstState && lastState)
                    value -= cost[last];

                best = max(best, value);
            }
        }

        maximumSaving += best;
    }

    cout << totalCost - maximumSaving << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1);
    for (int i = 1; i <= N; i++) cin >> a[i];

    vector<long long> c(N + 1);
    for (int i = 1; i <= N; i++) cin >> c[i];

    const long long INF = (1LL << 60);

    // dp[u][s][b]:
    // minimum cost inside u's off-cycle subtree,
    // where s = whether u is fixed,
    // b = whether at least one node in this subtree is fixed.
    vector<array<array<long long, 2>, 2>> dp(N + 1);

    for (int i = 1; i <= N; i++) {
        for (int s = 0; s < 2; s++) {
            for (int b = 0; b < 2; b++) {
                dp[i][s][b] = INF;
            }
        }
        dp[i][0][0] = 0;
        dp[i][1][1] = 0;
    }

    vector<int> indeg(N + 1, 0);
    for (int i = 1; i <= N; i++) {
        indeg[a[i]]++;
    }

    queue<int> q;
    for (int i = 1; i <= N; i++) {
        if (indeg[i] == 0) q.push(i);
    }

    // Process all non-cycle nodes from leaves toward cycles.
    while (!q.empty()) {
        int u = q.front();
        q.pop();

        // h[parent_state][subtree_flag]
        long long h[2][2];
        for (int t = 0; t < 2; t++) {
            for (int b = 0; b < 2; b++) {
                h[t][b] = INF;
            }
        }

        long long cu = c[u];

        for (int t = 0; t < 2; t++) {
            for (int b = 0; b < 2; b++) {
                // u is not fixed
                if (dp[u][0][b] < INF) {
                    long long g = (t == 1 ? 0LL : cu);
                    h[t][b] = min(h[t][b], dp[u][0][b] + g);
                }

                // u is fixed
                if (dp[u][1][b] < INF) {
                    h[t][b] = min(h[t][b], dp[u][1][b] + cu);
                }
            }
        }

        int p = a[u];

        long long ndp[2][2];
        for (int sp = 0; sp < 2; sp++) {
            for (int bp = 0; bp < 2; bp++) {
                ndp[sp][bp] = INF;
            }
        }

        for (int sp = 0; sp < 2; sp++) {
            for (int bp = 0; bp < 2; bp++) {
                long long base = dp[p][sp][bp];
                if (base >= INF) continue;

                for (int cf = 0; cf < 2; cf++) {
                    if (h[sp][cf] >= INF) continue;

                    long long val = base + h[sp][cf];
                    int nf = bp | cf;
                    ndp[sp][nf] = min(ndp[sp][nf], val);
                }
            }
        }

        for (int sp = 0; sp < 2; sp++) {
            for (int bp = 0; bp < 2; bp++) {
                dp[p][sp][bp] = ndp[sp][bp];
            }
        }

        indeg[p]--;
        if (indeg[p] == 0) q.push(p);
    }

    vector<char> isCycle(N + 1, 0);
    for (int i = 1; i <= N; i++) {
        if (indeg[i] > 0) isCycle[i] = 1;
    }

    auto solveCycle = [&](const vector<int>& cyc) -> pair<long long, long long> {
        int k = (int)cyc.size();

        if (k == 1) {
            int u = cyc[0];

            long long best0 = INF, best1 = INF;

            if (dp[u][0][0] < INF) {
                best0 = min(best0, dp[u][0][0] + c[u]);
            }
            if (dp[u][0][1] < INF) {
                best1 = min(best1, dp[u][0][1] + c[u]);
            }
            if (dp[u][1][1] < INF) {
                best1 = min(best1, dp[u][1][1]);
            }

            return {best0, best1};
        }

        long long ans[2] = {INF, INF};

        for (int start_s = 0; start_s < 2; start_s++) {
            long long cur[2][2];

            for (int i = 0; i < 2; i++) {
                for (int j = 0; j < 2; j++) {
                    cur[i][j] = INF;
                }
            }

            for (int b0 = 0; b0 < 2; b0++) {
                long long val = dp[cyc[0]][start_s][b0];
                if (val < INF) {
                    cur[start_s][b0] = min(cur[start_s][b0], val);
                }
            }

            for (int idx = 1; idx < k; idx++) {
                long long nxt[2][2];

                for (int i = 0; i < 2; i++) {
                    for (int j = 0; j < 2; j++) {
                        nxt[i][j] = INF;
                    }
                }

                int prevNode = cyc[idx - 1];
                int curNode = cyc[idx];
                long long cp = c[prevNode];

                for (int prev_s = 0; prev_s < 2; prev_s++) {
                    for (int flag = 0; flag < 2; flag++) {
                        long long base = cur[prev_s][flag];
                        if (base >= INF) continue;

                        for (int cur_s = 0; cur_s < 2; cur_s++) {
                            long long edgeCost =
                                (prev_s == 0 && cur_s == 1) ? 0LL : cp;

                            for (int cur_b = 0; cur_b < 2; cur_b++) {
                                long long local = dp[curNode][cur_s][cur_b];
                                if (local >= INF) continue;

                                long long val = base + local + edgeCost;
                                int nf = flag | cur_b;
                                nxt[cur_s][nf] = min(nxt[cur_s][nf], val);
                            }
                        }
                    }
                }

                for (int i = 0; i < 2; i++) {
                    for (int j = 0; j < 2; j++) {
                        cur[i][j] = nxt[i][j];
                    }
                }
            }

            int lastNode = cyc[k - 1];
            long long cp = c[lastNode];

            for (int last_s = 0; last_s < 2; last_s++) {
                for (int flag = 0; flag < 2; flag++) {
                    long long base = cur[last_s][flag];
                    if (base >= INF) continue;

                    long long edgeCost =
                        (last_s == 0 && start_s == 1) ? 0LL : cp;

                    long long total = base + edgeCost;
                    ans[flag] = min(ans[flag], total);
                }
            }
        }

        return {ans[0], ans[1]};
    };

    vector<char> visited(N + 1, 0);

    long long global0 = 0;      // no fixed node selected so far
    long long global1 = INF;    // at least one fixed node selected

    for (int i = 1; i <= N; i++) {
        if (isCycle[i] && !visited[i]) {
            vector<int> cyc;

            int u = i;
            while (!visited[u]) {
                visited[u] = 1;
                cyc.push_back(u);
                u = a[u];
            }

            auto [best0, best1] = solveCycle(cyc);

            long long ng0 = global0 + best0;
            long long ng1 = INF;

            if (global1 < INF) {
                ng1 = min(ng1, global1 + min(best0, best1));
            }

            if (global0 < INF) {
                ng1 = min(ng1, global0 + best1);
            }

            global0 = ng0;
            global1 = ng1;
        }
    }

    cout << global1 << '\n';

    return 0;
}

using namespace std;

using int64 = long long;
const int64 INF = (1LL << 62);

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1), indegree(N + 1, 0);
    for (int i = 1; i <= N; ++i) {
        cin >> a[i];
        ++indegree[a[i]];
    }

    vector<int64> c(N + 1);
    for (int i = 1; i <= N; ++i) {
        cin >> c[i];
    }

    // down0[v]: contribution of v's off-cycle descendants when v is not fixed.
    // down1[v]: contribution of v's off-cycle descendants when v is fixed.
    // The cost of editing v itself is not included.
    vector<int64> down0(N + 1, 0), down1(N + 1, 0);

    queue<int> q;
    for (int i = 1; i <= N; ++i) {
        if (indegree[i] == 0) {
            q.push(i);
        }
    }

    // Remove all non-cycle vertices, processing children before parents.
    while (!q.empty()) {
        int x = q.front();
        q.pop();

        // A removed vertex cannot be a self-loop.
        //
        // If the parent is not fixed:
        //   x fixed:     c[x] + down1[x]
        //   x not fixed: c[x] + down0[x]
        //
        // If the parent is fixed:
        //   x fixed:     c[x] + down1[x]
        //   x not fixed: down0[x]
        int64 whenParentNotFixed =
            c[x] + min(down0[x], down1[x]);
        int64 whenParentFixed =
            min(down0[x], c[x] + down1[x]);

        int p = a[x];
        down0[p] += whenParentNotFixed;
        down1[p] += whenParentFixed;

        if (--indegree[p] == 0) {
            q.push(p);
        }
    }

    vector<char> visited(N + 1, false);
    int64 answer = 0;

    for (int start = 1; start <= N; ++start) {
        if (indegree[start] == 0 || visited[start]) {
            continue;
        }

        vector<int> cycle;
        int v = start;
        do {
            visited[v] = true;
            cycle.push_back(v);
            v = a[v];
        } while (v != start);

        int k = static_cast<int>(cycle.size());

        if (k == 1) {
            // This vertex is already a fixed point.
            answer += down1[cycle[0]];
            continue;
        }

        int64 best = INF;

        // Fix the state of cycle[0], then run a two-state DP around the cycle.
        for (int firstState = 0; firstState <= 1; ++firstState) {
            array<int64, 2> dp = {INF, INF};
            dp[firstState] = (firstState == 1)
                                 ? down1[cycle[0]]
                                 : down0[cycle[0]];

            for (int i = 0; i + 1 < k; ++i) {
                array<int64, 2> nextDP = {INF, INF};
                int u = cycle[i];
                int w = cycle[i + 1];

                for (int currentState = 0; currentState <= 1; ++currentState) {
                    if (dp[currentState] == INF) {
                        continue;
                    }

                    for (int nextState = 0; nextState <= 1; ++nextState) {
                        // u's parent is w.
                        // If u is not fixed and its parent is fixed, u can
                        // remain unchanged. Otherwise u must be edited.
                        int64 localCost =
                            (currentState == 1 || nextState == 0) ? c[u] : 0LL;

                        int64 treeCost = (nextState == 1)
                                             ? down1[w]
                                             : down0[w];

                        nextDP[nextState] = min(
                            nextDP[nextState],
                            dp[currentState] + localCost + treeCost
                        );
                    }
                }

                dp = nextDP;
            }

            // Close the cycle: the parent of the last vertex is cycle[0].
            for (int lastState = 0; lastState <= 1; ++lastState) {
                if (dp[lastState] == INF) {
                    continue;
                }

                int64 localCost =
                    (lastState == 1 || firstState == 0)
                        ? c[cycle.back()]
                        : 0LL;

                best = min(best, dp[lastState] + localCost);
            }
        }

        answer += best;
    }

    cout << answer << '\n';
    return 0;
}

    int N;
    cin >> N;
    vector<int> a(N+1);
    vector<long long> c(N+1);

    vector<int> indeg(N+1,0);
    for(int i=1;i<=N;i++) indeg[a[i]]++;

    const long long NEG = -(1LL<<60);

    // For a tree node v (v is not on a cycle):
    //  X[v] = max savings inside subtree(v) (incl. v) assuming a[v] is NOT a fixed point
    //  Y[v] = max savings inside subtree(v) (incl. v) assuming a[v] IS a fixed point
    // sumX[v]/sumY[v] accumulate children contributions.
    vector<long long> X(N+1,0), Y(N+1,0), sumX(N+1,0), sumY(N+1,0);

    queue<int> q;
    for(int i=1;i<=N;i++) if(indeg[i]==0) q.push(i);

    // Peel off all non-cycle nodes, processing them bottom-up.
    while(!q.empty()){
        int v=q.front(); q.pop();
        X[v] = max(sumY[v], sumX[v]);
        Y[v] = max(sumY[v], sumX[v] + c[v]);
        int p=a[v];
        sumX[p] += X[v];
        sumY[p] += Y[v];
        if(--indeg[p]==0) q.push(p);
    }

    long long bestSavings = 0;
    vector<char> vis(N+1,0);

    for(int i=1;i<=N;i++){
        if(indeg[i]==0 || vis[i]) continue;   // not a cycle node, or already handled

        // Collect the cycle in order: cyc[i] -> cyc[i+1] -> ... -> cyc[0]
        vector<int> cyc;
        int v=i;
        do{ cyc.push_back(v); vis[v]=1; v=a[v]; }while(v!=i);
        int k=(int)cyc.size();

        if(k==1){
            // Self loop: a[u]==u, so u can be a fixed point with zero cost.
            int u=cyc[0];
            bestSavings += max(c[u] + sumY[u], sumX[u]);
