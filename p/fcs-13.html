#include <bits/stdc++.h>
using namespace std;

using P = pair<int, int>;

static const int T = 3000;
static const int N = 60;
static const int LOCAL_LIMIT = 180;
static const int EMERGENCY_LIMIT = 40;

set<P> blackCells;
vector<P> wallCells;
vector<P> fillCells;
int rx, ry, vx = 0, vy = 0;
int localUsed = 0, emergencyUsed = 0;

bool markable(P p) {
    return p.first >= 1 && p.first <= T && p.second >= 1 && p.second <= T;
}

bool isBlack(P p) {
    return blackCells.count(p);
}

vector<P> neighbors(P p) {
    vector<P> res;
    for (int dx = -1; dx <= 1; ++dx) {
        for (int dy = -1; dy <= 1; ++dy) {
            if (dx == 0 && dy == 0) continue;
            P q = {p.first + dx, p.second + dy};
            if (q.first > 0 && q.second > 0) res.push_back(q);
        }
    }
    return res;
}

int wallIndex(P p) {
    return p.first == N ? p.second : p.first;
}

bool baseWallCell(P p) {
    int i = wallIndex(p);
    return i % 3 != 0 || i <= 2 || i >= N - 2;
}

bool specialWallCell(P p) {
    int i = wallIndex(p);
    return i <= 2 || i >= N - 2;
}

bool adjacent(P a, P b) {
    return max(abs(a.first - b.first), abs(a.second - b.second)) <= 1;
}

vector<P> urgentWallCells(P p) {
    vector<P> res;
    for (P w : wallCells) {
        if (!isBlack(w) && adjacent(p, w)) res.push_back(w);
    }
    return res;
}

pair<int, int> deadlineInfo(P w, P p) {
    int best = 1000000;
    if (w.first == N) {
        int qx = N - 1;
        int lo = max(1, w.second - 1);
        int hi = min(N - 1, w.second + 1);
        for (int qy = lo; qy <= hi; ++qy) {
            best = min(best, max(abs(qx - p.first), abs(qy - p.second)));
        }
        return {best, abs(w.second - p.second)};
    } else {
        int qy = N - 1;
        int lo = max(1, w.first - 1);
        int hi = min(N - 1, w.first + 1);
        for (int qx = lo; qx <= hi; ++qx) {
            best = min(best, max(abs(qx - p.first), abs(qy - p.second)));
        }
        return {best, abs(w.first - p.first)};
    }
}

int dangerAfterMark(P candidate) {
    P p = {rx, ry};
    int danger = 0;
    for (P q : neighbors(p)) {
        if (q == candidate || isBlack(q)) continue;

        if (q.first >= N || q.second >= N) {
            danger += 1000;
            continue;
        }

        int whiteExits = 0;
        for (P w : wallCells) {
            if (w == candidate || isBlack(w)) continue;
            if (adjacent(q, w)) ++whiteExits;
        }
        if (whiteExits >= 2) danger += 10 * (whiteExits - 1);
    }
    return danger;
}

struct Plan {
    bool ok = false;
    int eta = 1000000;
    P target = {-1, -1};
    vector<P> offPath;
};

Plan makePlan(bool restrictToBox) {
    vector<P> dirs;
    if (vx != 0 || vy != 0) {
        dirs.push_back({vx, vy});
    } else {
        dirs = {
            {0, -1}, {-1, 0}, {1, -1}, {-1, -1},
            {1, 0}, {0, 1}, {1, 1}, {-1, 1}
        };
    }

    Plan best;
    int limit = restrictToBox ? N - 1 : T - 1;

    for (P d : dirs) {
        for (int k = 1; k <= 12; ++k) {
            P c = {rx + d.first * k, ry + d.second * k};
            if (c.first < 1 || c.second < 1 || c.first > limit || c.second > limit) continue;
            if (isBlack(c)) continue;

            bool bad = false;
            set<P> futurePath;
            for (int j = 1; j < k; ++j) {
                P q = {rx + d.first * j, ry + d.second * j};
                futurePath.insert(q);
                if (isBlack(q)) bad = true;
            }
            if (bad) continue;

            P previous = {c.first - d.first, c.second - d.second};
            vector<P> off;
            for (P q : neighbors(c)) {
                if (!markable(q) || q == previous || isBlack(q)) continue;
                if (futurePath.count(q)) {
                    bad = true;
                    break;
                }
                off.push_back(q);
            }
            if (bad) continue;
            if ((int)off.size() > k) continue;

            sort(off.begin(), off.end(), [](P a, P b) {
                if (a.second != b.second) return a.second < b.second;
                return a.first > b.first;
            });

            int eta = k + 1;
            if (eta < best.eta) {
                best.ok = true;
                best.eta = eta;
                best.target = c;
                best.offPath = off;
            }
        }
    }
    return best;
}

pair<P, bool> chooseWallMove() {
    vector<P> base, other;
    for (P w : wallCells) {
        if (isBlack(w)) continue;
        if (baseWallCell(w)) base.push_back(w);
        else other.push_back(w);
    }

    vector<P> candidates = !base.empty() ? base : other;
    if (candidates.empty()) return {{-1, -1}, false};

    int minimumDeadline = 1000000;
    for (P w : candidates) {
        minimumDeadline = min(minimumDeadline, deadlineInfo(w, {rx, ry}).first);
    }

    if (!base.empty() && minimumDeadline > 5) {
        vector<P> special;
        for (P w : candidates) {
            if (specialWallCell(w)) special.push_back(w);
        }
        if (!special.empty()) candidates = special;
    }

    P best = {-1, -1};
    int bestDanger = 1000000000;
    pair<int, int> bestInfo = {1000000000, 1000000000};

    for (P w : candidates) {
        int danger = dangerAfterMark(w);
        auto info = deadlineInfo(w, {rx, ry});
        if (best.first == -1 ||
            danger < bestDanger ||
            (danger == bestDanger && info < bestInfo)) {
            best = w;
            bestDanger = danger;
            bestInfo = info;
        }
    }

    if (bestDanger > 0 && emergencyUsed < EMERGENCY_LIMIT &&
        (rx >= N - 5 || ry >= N - 5)) {
        P emergency = {-1, -1};
        int emergencyDanger = bestDanger;
        for (P q : neighbors({rx, ry})) {
            if (q.first >= N || q.second >= N || !markable(q) || isBlack(q)) continue;
            int danger = dangerAfterMark(q);
            if (danger < emergencyDanger) {
                emergencyDanger = danger;
                emergency = q;
            }
        }
        if (emergency.first != -1) return {emergency, true};
    }

    return {best, false};
}

P chooseFillMove() {
    P best = {-1, -1};
    pair<int, pair<int, int>> bestScore = {1000000000, {1000000000, 1000000000}};

    for (P q : fillCells) {
        if (isBlack(q)) continue;
        int d = max(abs(q.first - rx), abs(q.second - ry));
        auto score = make_pair(d, make_pair(q.second, q.first));
        if (best.first == -1 || score < bestScore) {
            best = q;
            bestScore = score;
        }
    }
    return best;
}

P anyWhiteCell() {
    for (int x = 1; x <= 3000; ++x) {
        for (int y = 1; y <= 3000; ++y) {
            if (!isBlack({x, y})) return {x, y};
        }
    }
    return {1, 1};
}

#include <bits/stdc++.h>
using namespace std;

struct Cell {
    int x, y;
    bool operator==(const Cell& other) const {
        return x == other.x && y == other.y;
    }
};

struct Plan {
    Cell center{};
    Cell opening{};
    Cell approach{};
    vector<Cell> order;
    int age = 0;
};

static set<pair<int, int>> blackCells;

static bool markable(int x, int y) {
    return 1 <= x && x <= 3000 && 1 <= y && y <= 3000;
}

static bool isBlack(int x, int y) {
    return blackCells.count({x, y});
}

static int dist2(const Cell& a, const Cell& b) {
    int dx = a.x - b.x, dy = a.y - b.y;
    return dx * dx + dy * dy;
}

static int cheb(const Cell& a, const Cell& b) {
    return max(abs(a.x - b.x), abs(a.y - b.y));
}

static vector<Cell> cageNeighbors(Cell c) {
    vector<Cell> result;
    for (int dx = -1; dx <= 1; ++dx) {
        for (int dy = -1; dy <= 1; ++dy) {
            if (dx == 0 && dy == 0) continue;
            int x = c.x + dx, y = c.y + dy;
            if (markable(x, y)) result.push_back({x, y});
        }
    }
    return result;
}

static Plan makePlan(Cell r, const vector<Cell>& history) {
    Plan p;
    Cell d{0, 0};

    if (history.size() == 1) {
        if (r.y <= r.x) d = {0, -1};
        else d = {-1, 0};
    } else {
        int back = max(0, (int)history.size() - 7);
        int vx = r.x - history[back].x;
        int vy = r.y - history[back].y;

        Cell last = history[history.size() - 2];
        int lx = r.x - last.x;
        int ly = r.y - last.y;

        if (vx == 0 && vy == 0) {
            vx = lx;
            vy = ly;
        }

        int ax = abs(vx), ay = abs(vy);
        if (ax > 2 * ay) vy = 0;
        if (ay > 2 * ax) vx = 0;

        d.x = (vx > 0) - (vx < 0);
        d.y = (vy > 0) - (vy < 0);

        if (d.x == 0 && d.y == 0) {
            d.x = (lx > 0) - (lx < 0);
            d.y = (ly > 0) - (ly < 0);
        }
        if (d.x == 0 && d.y == 0) {
            if (r.y <= r.x) d = {0, -1};
            else d = {-1, 0};
        }
    }

    p.approach = d;

    int boundaryDistance = INT_MAX;
    if (d.x < 0) boundaryDistance = min(boundaryDistance, r.x - 1);
    if (d.y < 0) boundaryDistance = min(boundaryDistance, r.y - 1);

    if (history.size() == 1) {
        if (d.x < 0) p.center = {1, r.y};
        else p.center = {r.x, 1};
    } else if (boundaryDistance != INT_MAX && boundaryDistance > 0 &&
               boundaryDistance <= 16) {
        p.center = {r.x + boundaryDistance * d.x,
                    r.y + boundaryDistance * d.y};
    } else {
        p.center = {r.x + 8 * d.x, r.y + 8 * d.y};
    }

    p.center.x = max(1, min(3000, p.center.x));
    p.center.y = max(1, min(3000, p.center.y));

    vector<Cell> around = cageNeighbors(p.center);
    Cell candidate{p.center.x - d.x, p.center.y - d.y};
    bool found = false;
    for (Cell c : around) {
        if (c == candidate) {
            found = true;
            break;
        }
    }

    if (found) {
        p.opening = candidate;
    } else {
        p.opening = around[0];
        for (Cell c : around) {
            if (dist2(c, r) < dist2(p.opening, r)) p.opening = c;
        }
    }

    for (Cell c : around) {
        if (!(c == p.opening)) p.order.push_back(c);
    }

    stable_sort(p.order.begin(), p.order.end(), [&](const Cell& a, const Cell& b) {
        return dist2(a, r) > dist2(b, r);
    });

    return p;
}

#include <bits/stdc++.h>
using namespace std;

struct Position {
    int x, y;
};

struct Candidate {
    Position target;
    double weight = 0.0;
};

static constexpr int LIMIT = 3000;

bool operator<(const Position &a, const Position &b) {
    if (a.x != b.x) return a.x < b.x;
    return a.y < b.y;
}

Position clampPosition(int x, int y) {
    x = max(1, min(LIMIT, x));
    y = max(1, min(LIMIT, y));
    return {x, y};
}

Position projectToBoundary(const Position &p, int vx, int vy) {
    if (vx == 0 && vy == 0) return p;

    int steps = INT_MAX;

    if (vx < 0) steps = min(steps, p.x - 1);
    if (vx > 0) steps = min(steps, LIMIT - p.x);
    if (vy < 0) steps = min(steps, p.y - 1);
    if (vy > 0) steps = min(steps, LIMIT - p.y);

    if (steps == INT_MAX) return p;

    return clampPosition(p.x + vx * steps, p.y + vy * steps);
}

vector<Position> buildCage(Position target) {
    vector<Position> cage;

    if (target.y == 1) {
        cage.push_back({target.x + 1, target.y});
        cage.push_back({target.x - 1, target.y});
        cage.push_back({target.x + 1, target.y + 1});
        cage.push_back({target.x - 1, target.y + 1});
        cage.push_back({target.x, target.y + 1});
    } else if (target.y == LIMIT) {
        cage.push_back({target.x + 1, target.y});
        cage.push_back({target.x - 1, target.y});
        cage.push_back({target.x + 1, target.y - 1});
        cage.push_back({target.x - 1, target.y - 1});
        cage.push_back({target.x, target.y - 1});
    } else if (target.x == 1) {
        cage.push_back({target.x, target.y + 1});
        cage.push_back({target.x, target.y - 1});
        cage.push_back({target.x + 1, target.y + 1});
        cage.push_back({target.x + 1, target.y - 1});
        cage.push_back({target.x + 1, target.y});
    } else if (target.x == LIMIT) {
        cage.push_back({target.x, target.y + 1});
        cage.push_back({target.x, target.y - 1});
        cage.push_back({target.x - 1, target.y + 1});
        cage.push_back({target.x - 1, target.y - 1});
        cage.push_back({target.x - 1, target.y});
    } else {
        cage.push_back({target.x + 1, target.y});
        cage.push_back({target.x + 1, target.y + 1});
        cage.push_back({target.x, target.y + 1});
        cage.push_back({target.x - 1, target.y + 1});
        cage.push_back({target.x - 1, target.y});
        cage.push_back({target.x - 1, target.y - 1});
        cage.push_back({target.x, target.y - 1});
        cage.push_back({target.x + 1, target.y - 1});
    }

    for (int radius = 2; radius <= 3; ++radius) {
        for (int dx = -radius; dx <= radius; ++dx) {
            for (int dy = -radius; dy <= radius; ++dy) {
                if (max(abs(dx), abs(dy)) != radius) continue;
                int x = target.x + dx;
                int y = target.y + dy;
                if (1 <= x && x <= LIMIT && 1 <= y && y <= LIMIT) {
                    cage.push_back({x, y});
                }
            }
        }
    }

    cage.push_back(target);
    return cage;
}
