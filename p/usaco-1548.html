#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using i128 = __int128_t;

struct Vec {
    ll x, y;
};

static inline i128 cross(const Vec &a, const Vec &b) {
    return (i128)a.x * b.y - (i128)a.y * b.x;
}

static inline i128 dot(const Vec &a, const Vec &b) {
    return (i128)a.x * b.x + (i128)a.y * b.y;
}

static inline int half(const Vec &a) {
    return (a.y > 0 || (a.y == 0 && a.x > 0)) ? 0 : 1;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    while (T--) {
        int N;
        cin >> N;

        vector<Vec> v;
        v.reserve(N);

        for (int i = 0; i < N; i++) {
            ll h, p, s;
            cin >> h >> p >> s;
            Vec a{p - s, h - s};
            if (a.x != 0 || a.y != 0) v.push_back(a);
        }

        int n = (int)v.size();
        if (n < 3) {
            cout << 0 << '\n';
            continue;
        }

        sort(v.begin(), v.end(), [](const Vec &a, const Vec &b) {
            if (half(a) != half(b)) return half(a) < half(b);
            return cross(a, b) > 0;
        });

        vector<Vec> doubled(2 * n);
        for (int i = 0; i < 2 * n; i++) doubled[i] = v[i % n];

        // q[i]: first direction at least pi radians after v[i].
        // r[i]: first direction strictly more than pi radians after v[i].
        vector<int> q(n), r(n);
        int ptr = 1;

        for (int i = 0; i < n; i++) {
            ptr = max(ptr, i + 1);
            int lim = i + n;

            while (ptr < lim) {
                i128 cr = cross(v[i], doubled[ptr]);
                if (cr > 0 || (cr == 0 && dot(v[i], doubled[ptr]) > 0)) {
                    ptr++;
                } else {
                    break;
                }
            }

            q[i] = ptr;
            int z = ptr;
            while (z < lim && cross(v[i], doubled[z]) == 0) z++;
            r[i] = z;
        }

        vector<int> sameEnd(n);
        for (int l = 0; l < n; ) {
            int rr = l + 1;
            while (rr < n && cross(v[l], v[rr]) == 0) rr++;
            for (int i = l; i < rr; i++) sameEnd[i] = rr;
            l = rr;
        }

        vector<int> R(n);
        vector<ll> pref(n + 1, 0);
        for (int i = 0; i < n; i++) {
            R[i] = min(q[i], n);
            pref[i + 1] = pref[i] + R[i];
        }

        ll answer = 0;

        for (int i = 0; i < n; i++) {
            int L = min(r[i], n);
            if (L >= n) continue;

            int lo = sameEnd[i];
            int hi = min(q[i], n);
            if (lo >= hi) continue;

            int start = lower_bound(R.begin() + lo, R.begin() + hi, L + 1)
                        - R.begin();

            if (start < hi) {
                answer += pref[hi] - pref[start]
                          - (ll)(hi - start) * L;
            }
        }

        cout << answer << '\n';
    }

    return 0;
}

struct Point {

};

static inline ll cross(const Point& a, const Point& b) {
    return a.x * b.y - a.y * b.x;
}

static inline ll dot(const Point& a, const Point& b) {
    return a.x * b.x + a.y * b.y;
}

static inline int halfPlane(const Point& p) {
    return (p.y > 0 || (p.y == 0 && p.x >= 0)) ? 0 : 1;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        vector<Point> points;
        points.reserve(N);
        map<pair<ll, ll>, ll> rayCount;

        for (int i = 0; i < N; i++) {
            ll h, p, s;
            cin >> h >> p >> s;

            Point v{h - s, p - s};
            if (v.x == 0 && v.y == 0) continue;

            points.push_back(v);

            ll g = std::gcd(v.x < 0 ? -v.x : v.x,
                            v.y < 0 ? -v.y : v.y);
            rayCount[{v.x / g, v.y / g}]++;
        }

        int M = (int)points.size();
        if (M < 3) {
            cout << 0 << '\n';
            continue;
        }

        sort(points.begin(), points.end(), [](const Point& a, const Point& b) {
            int ha = halfPlane(a), hb = halfPlane(b);
            if (ha != hb) return ha < hb;
            return cross(a, b) > 0;
        });

        ll total = (ll)M * (M - 1) * (M - 2) / 6;

        // Count triples having a circular gap strictly greater than 180 degrees.
        ll bad = 0;
        int r = 0;

        for (int i = 0; i < M; i++) {
            if (r < i + 1) r = i + 1;

            while (r < i + M) {
                const Point& a = points[i];
                const Point& b = points[r % M];

                ll cr = cross(a, b);
                bool inside;

                if (cr > 0) {
                    inside = true; // Angular difference is in (0, 180).
                } else if (cr == 0 && dot(a, b) > 0 && r < M) {
                    inside = true; // Same direction, angular difference 0.
                } else {
                    inside = false;
                }

                if (!inside) break;
                r++;
            }

            ll c = r - i - 1;
            bad += c * (c - 1) / 2;
        }

        // Count triples containing an antipodal pair; these have a gap of
        // exactly 180 degrees and were not counted above.
        __int128 antipodal = 0;

        for (const auto& entry : rayCount) {
            auto key = entry.first;
            pair<ll, ll> opposite{-key.first, -key.second};

            if (!(key < opposite)) continue;

            auto it = rayCount.find(opposite);
            if (it == rayCount.end()) continue;

            ll a = entry.second;
            ll b = it->second;
            ll outside = M - a - b;

            antipodal += (__int128)a * b * outside;
            antipodal += (__int128)a * b * (a + b - 2) / 2;
        }

        ll answer = total - bad - (ll)antipodal;
        cout << answer << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using i128 = __int128_t;

struct Point {
    ll x, y;
};

struct Ray {
    Point p;
    ll weight;
};

static inline i128 cross(const Point& a, const Point& b) {
    return (i128)a.x * b.y - (i128)a.y * b.x;
}

static inline int halfPlane(const Point& a) {
    return (a.y > 0 || (a.y == 0 && a.x > 0)) ? 0 : 1;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        vector<Point> points;
        points.reserve(N);

        for (int i = 0; i < N; i++) {
            ll h, p, s;
            cin >> h >> p >> s;

            // A beats B exactly when cross(v_A, v_B) < 0,
            // where v = (p - h, s - h).
            Point v{p - h, s - h};
            if (v.x != 0 || v.y != 0) {
                points.push_back(v);
            }
        }

        const ll total = (ll)points.size();
        if (total < 3) {
            cout << 0 << '\n';
            continue;
        }

        sort(points.begin(), points.end(), [](const Point& a, const Point& b) {
            int ha = halfPlane(a), hb = halfPlane(b);
            if (ha != hb) return ha < hb;
            return cross(a, b) > 0;
        });

        // Compress points having the same direction into weighted rays.
        vector<Ray> rays;
        for (size_t i = 0; i < points.size();) {
            size_t j = i + 1;
            int hi = halfPlane(points[i]);

            while (j < points.size() &&
                   halfPlane(points[j]) == hi &&
                   cross(points[i], points[j]) == 0) {
                ++j;
            }

            rays.push_back({points[i], (ll)(j - i)});
            i = j;
        }

        const int M = (int)rays.size();

        // Cows tied with a given cow are precisely those on the same line
        // through the origin (either the same or the opposite ray).
        map<pair<ll, ll>, int> lineIds;
        vector<int> rayLine(M);
        vector<ll> lineWeight;

        for (int i = 0; i < M; i++) {
            ll x = rays[i].p.x;
            ll y = rays[i].p.y;
            ll ax = x < 0 ? -x : x;
            ll ay = y < 0 ? -y : y;
            ll g = std::gcd(ax, ay);

            x /= g;
            y /= g;

            if (x < 0 || (x == 0 && y < 0)) {
                x = -x;
                y = -y;
            }

            pair<ll, ll> key{x, y};
            auto it = lineIds.find(key);
            int id;

            if (it == lineIds.end()) {
                id = (int)lineWeight.size();
                lineIds[key] = id;
                lineWeight.push_back(0);
            } else {
                id = it->second;
            }

            rayLine[i] = id;
            lineWeight[id] += rays[i].weight;
        }

        // Number of triples containing no tied pair: the three cows must
        // come from three distinct lines.
        i128 sum1 = total, sum2 = 0, sum3 = 0;
        for (ll w : lineWeight) {
            sum2 += (i128)w * w;
            sum3 += (i128)w * w * w;
        }

        i128 answer =
            (sum1 * sum1 * sum1 - 3 * sum1 * sum2 + 2 * sum3) / 6;

        // in[i] = number of cows beating a cow on ray i.
        // These are the rays strictly inside the next CCW semicircle.
        const int D = 2 * M;
        vector<Point> dirs(D);
        vector<ll> pref(D + 1, 0);

        for (int i = 0; i < D; i++) {
            dirs[i] = rays[i % M].p;
            pref[i + 1] = pref[i] + rays[i % M].weight;
        }

        vector<ll> in(M, 0);
        int j = 0;

        for (int i = 0; i < M; i++) {
            if (j < i + 1) j = i + 1;
            int limit = i + M;

            while (j < limit && cross(dirs[i], dirs[j]) > 0) {
                ++j;
            }

            in[i] = pref[j] - pref[i + 1];
        }

        // Subtract transitive non-tied triples. The last term removes
        // pairs of cows on the same ray that were counted as a pair
        // both beaten by another cow.
        for (int i = 0; i < M; i++) {
            ll w = rays[i].weight;
            ll out = total - lineWeight[rayLine[i]] - in[i];

            answer -= (i128)w * out * (out - 1) / 2;
            answer += (i128)w * (w - 1) / 2 * in[i];
        }

        cout << (ll)answer << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using i128 = __int128_t;

struct Vec {
    int64 x, y;
};

static inline i128 cross(const Vec& a, const Vec& b) {
    return (i128)a.x * b.y - (i128)a.y * b.x;
}

static inline int halfPlane(const Vec& a) {
    return (a.y > 0 || (a.y == 0 && a.x > 0)) ? 0 : 1;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        vector<Vec> base;
        base.reserve(N);

        for (int i = 0; i < N; ++i) {
            int64 h, p, s;
            cin >> h >> p >> s;

            // The pairwise comparison sign is the sign of the cross product
            // of (h-s, p-s), up to a fixed reversal. Uniform strategies map
            // to the zero vector and cannot belong to a cycle.
            Vec v{h - s, p - s};
            if (v.x != 0 || v.y != 0) base.push_back(v);
        }

        int m = (int)base.size();
        if (m < 3) {
            cout << 0 << '\n';
            continue;
        }

        sort(base.begin(), base.end(), [](const Vec& a, const Vec& b) {
            int ha = halfPlane(a), hb = halfPlane(b);
            if (ha != hb) return ha < hb;
            return cross(a, b) > 0;
        });

        vector<Vec> a = base;
        a.insert(a.end(), base.begin(), base.end());
        int sz = 2 * m;

        // One past the end of the run of vectors having the same direction.
        vector<int> runEnd(sz);
        runEnd[sz - 1] = sz;
        for (int i = sz - 2; i >= 0; --i) {
            if (cross(a[i], a[i + 1]) == 0)
                runEnd[i] = runEnd[i + 1];
            else
                runEnd[i] = i + 1;
        }

        // up[j]: first direction at least pi radians after direction j.
        vector<int> up(sz);
        int k = 0;
        for (int j = 0; j < sz; ++j) {
            k = max(k, runEnd[j]);
            while (k < sz && cross(a[j], a[k]) > 0) ++k;
            up[j] = k;
        }

        vector<int64> pref(sz + 1, 0);
        for (int i = 0; i < sz; ++i) pref[i + 1] = pref[i] + up[i];

        int64 counted = 0;

        for (int i = 0; i < m; ++i) {
            int lo = runEnd[i]; // Skip directions equal to direction i.
            int hi = up[i];     // Directions strictly between 0 and pi after i.
            if (lo >= hi) continue;

            // First direction strictly more than pi after direction i.
            int left = up[i];

    int ha = half(a), hb = half(b);
    if (ha != hb) return ha < hb;
    return cross(a, b) > 0;
}

static int64 choose2(int64 x) {
    if (x < 2) return 0;
    return x * (x - 1) / 2;
}

static int64 choose3(int64 x) {
    if (x < 3) return 0;
    return x * (x - 1) * (x - 2) / 6;
}

        int64 oppositeTwice = 0;

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using i128 = __int128_t;

struct PairHash {
    static uint64_t splitmix64(uint64_t x) {
        x += 0x9e3779b97f4a7c15ULL;
        x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
        x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
        return x ^ (x >> 31);
    }

    size_t operator()(const pair<int64, int64>& p) const {
        static const uint64_t seed =
            chrono::steady_clock::now().time_since_epoch().count();

        uint64_t a = splitmix64(static_cast<uint64_t>(p.first) + seed);
        uint64_t b = splitmix64(static_cast<uint64_t>(p.second) + seed
                                + 0x9e3779b97f4a7c15ULL);
        return a ^ (b + 0x9e3779b97f4a7c15ULL
                    + (a << 6) + (a >> 2));
    }
};

struct Bucket {
    int64 x, y;
    int64 weight;
};

struct Direction {
    int64 x, y;
    int64 weight;
};

static int halfPlane(const Direction& d) {
    return (d.y > 0 || (d.y == 0 && d.x > 0)) ? 0 : 1;
}

static i128 cross(const Direction& a, const Direction& b) {
    return (i128)a.x * b.y - (i128)a.y * b.x;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        unordered_map<pair<int64, int64>, Bucket, PairHash> buckets;
        buckets.max_load_factor(0.7f);
        buckets.reserve(N);

        int64 active = 0;

        for (int i = 0; i < N; ++i) {
            int64 h, p, s;
            cin >> h >> p >> s;

            int64 x = 2 * h - p - s;
            int64 y = p - s;

            // The completely mixed strategy ties against every strategy.
            if (x == 0 && y == 0) continue;

            // Normalize so that equal rays have the same key.
            int64 g = std::gcd(std::llabs(x), std::llabs(y));
            x /= g;
            y /= g;

            pair<int64, int64> key{x, y};
            auto it = buckets.find(key);

            if (it == buckets.end()) {
                buckets.emplace(key, Bucket{x, y, 1});
            } else {
                ++it->second.weight;
            }

            ++active;
        }

        int64 seen = 0;
        int64 pairs = 0;
        int64 allTriples = 0;
        int64 boundaryTriples = 0;

        vector<Direction> directions;
        directions.reserve(buckets.size());

        for (const auto& entry : buckets) {
            const auto& key = entry.first;
            const Bucket& b = entry.second;

            // Number of choices using three distinct direction groups.
            allTriples += b.weight * pairs;
            pairs += b.weight * seen;
            seen += b.weight;

            directions.push_back({b.x, b.y, b.weight});

            // Exactly opposite rays tie. A triangle containing such a pair
            // has the origin on its boundary, rather than strictly inside.
            pair<int64, int64> opposite{-b.x, -b.y};
            if (key < opposite) {
                auto jt = buckets.find(opposite);
                if (jt != buckets.end()) {
                    int64 a = b.weight;
                    int64 c = jt->second.weight;
                    boundaryTriples += a * c * (active - a - c);
                }
            }
        }

        sort(directions.begin(), directions.end(),
             [](const Direction& a, const Direction& b) {
                 int ha = halfPlane(a);
                 int hb = halfPlane(b);
                 if (ha != hb) return ha < hb;
                 return cross(a, b) > 0;
             });

        int M = (int)directions.size();

        // Duplicate the circular ordering.
        vector<Direction> circular(2 * M);
        for (int i = 0; i < 2 * M; ++i) {
            circular[i] = directions[i % M];
        }

        vector<int64> prefixWeight(2 * M + 1, 0);
        vector<int64> prefixSquare(2 * M + 1, 0);

        for (int i = 0; i < 2 * M; ++i) {
            int64 w = circular[i].weight;
            prefixWeight[i + 1] = prefixWeight[i] + w;
            prefixSquare[i + 1] = prefixSquare[i] + w * w;
        }

        // Count triples lying in an open semicircle. Each such triple has
        // a unique direction immediately following its unique gap > 180°.
        int64 outsideTriples = 0;
        int r = 1;

        for (int i = 0; i < M; ++i) {
            r = max(r, i + 1);

            while (r < i + M && cross(circular[i], circular[r]) > 0) {
                ++r;
            }

            int64 sum = prefixWeight[r] - prefixWeight[i + 1];
            int64 sumSquares =
                prefixSquare[r] - prefixSquare[i + 1];

            int64 twoDistinctGroups = (sum * sum - sumSquares) / 2;
            outsideTriples += circular[i].weight * twoDistinctGroups;
        }

        int64 answer = allTriples - outsideTriples - boundaryTriples;
        cout << answer << '\n';
    }

    return 0;
}
