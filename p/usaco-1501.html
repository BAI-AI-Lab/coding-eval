#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using i128 = __int128_t;

static inline ll absll(ll x) { return x >= 0 ? x : -x; }

// Coefficients for abs(A), abs(B), then adjusted for their signs.
static pair<i128, i128> bezout(ll A, ll B) {
    ll r0 = absll(A), r1 = absll(B);
    i128 x0 = 1, x1 = 0, y0 = 0, y1 = 1;

    while (r1 != 0) {
        ll q = r0 / r1;
        ll nr = r0 % r1;
        r0 = r1;
        r1 = nr;

        i128 nx = x0 - (i128)q * x1;
        i128 ny = y0 - (i128)q * y1;
        x0 = x1; x1 = nx;
        y0 = y1; y1 = ny;
    }

    if (A < 0) x0 = -x0;
    if (B < 0) y0 = -y0;
    return {x0, y0};
}

// b is positive.
static i128 floorDiv(i128 a, i128 b) {
    i128 q = a / b;
    if (a % b < 0) --q;
    return q;
}

// b is positive.
static i128 ceilDiv(i128 a, i128 b) {
    i128 q = a / b;
    if (a % b > 0) ++q;
    return q;
}

static string toString(i128 x) {
    if (x == 0) return "0";
    bool neg = x < 0;
    if (neg) x = -x;
    string s;
    while (x > 0) {
        s.push_back(char('0' + int(x % 10)));
        x /= 10;
    }
    if (neg) s.push_back('-');
    reverse(s.begin(), s.end());
    return s;
}

    ios::sync_with_stdio(false);

    int Q;

    while (Q--) {
        ll a, b, c, d;
        cin >> a >> b >> c >> d;

        if (a == c && b == d) {
            cout << "0\n";
            continue;
        }
        if (a == 0 && b == 0) {
            cout << "-1\n";
            continue;
        }

        ll g = std::gcd(absll(a), absll(b));
        if (std::gcd(absll(c), absll(d)) != g) {
            cout << "-1\n";
            continue;
        }

        // Reduce to primitive vectors.
        ll A = a / g, B = b / g;
        ll C = c / g, D = d / g;

        auto [u, v] = bezout(A, B); // u*A + v*B = 1
        auto [z, t] = bezout(C, D); // z*C + t*D = 1

        /*
           Every integer determinant-one matrix taking (A,B) to (C,D)
           has the form

               M_k = Q * L^k * P^(-1),

           where L = [[1,1],[0,1]].  Its entries are affine in k.
           We need exactly those k for which all entries are nonnegative;
           such a matrix, and only such a matrix, is a product of the two
           operation matrices.
        */

        i128 p0 = (i128)C * u + (i128)t * B;
        i128 q0 = (i128)C * v - (i128)t * A;
        i128 r0 = (i128)D * u - (i128)z * B;
        i128 s0 = (i128)D * v + (i128)z * A;

        i128 cp = -(i128)C * B;
        i128 cq =  (i128)C * A;
        i128 cr = -(i128)D * B;
        i128 cs =  (i128)D * A;

        bool ok = true, hasLo = false, hasHi = false;
        i128 lo = 0, hi = 0;

        auto addConstraint = [&](i128 base, i128 coef) {
            if (!ok) return;
            if (coef == 0) {
                if (base < 0) ok = false;
            } else if (coef > 0) {
                i128 bound = ceilDiv(-base, coef);
                if (!hasLo || bound > lo) {
                    lo = bound;
                    hasLo = true;
                }
            } else {
                i128 bound = floorDiv(base, -coef);
                if (!hasHi || bound < hi) {
                    hi = bound;
                    hasHi = true;
                }
            }
        };

        addConstraint(p0, cp);
        addConstraint(q0, cq);
        addConstraint(r0, cr);
        addConstraint(s0, cs);

        if (!ok || (hasLo && hasHi && lo > hi)) {
            cout << "-1\n";
            continue;
        }

        // Factor a nonnegative SL(2,Z) matrix into the two operation
        // matrices, batching consecutive identical factors.
        auto wordLength = [&](i128 k) -> i128 {
            i128 p = p0 + cp * k;
            i128 q = q0 + cq * k;
            i128 r = r0 + cr * k;
            i128 s = s0 + cs * k;

            if (p < 0 || q < 0 || r < 0 || s < 0) return -1;

            i128 ans = 0;
            while (true) {
                if (q == 0) { // R^r
                    ans += r;
                    break;
                }
                if (r == 0) { // L^q
                    ans += q;
                    break;
                }

                i128 n;
                if (p >= r && q >= s) {
                    // The last factors are L: subtract the second row
                    // from the first row.
                    n = std::min(p / r, q / s);
                    if (n <= 0) return -1;
                    p -= n * r;
                    q -= n * s;
                } else {
                    // The last factors are R: subtract the first row
                    // from the second row.
                    n = std::min(r / p, s / q);
                    if (n <= 0) return -1;
                    r -= n * p;
                    s -= n * q;
                }
                ans += n;
            }
            return ans;
        };

        i128 best = -1;
        auto tryK = [&](i128 k) {
            i128 len = wordLength(k);
            if (len >= 0 && (best < 0 || len < best)) best = len;
        };

        // The shortest representation occurs at an endpoint of the
        // feasible interval.  Normally there is only one finite endpoint.
        if (hasLo) tryK(lo);
        if (hasHi && (!hasLo || hi != lo)) tryK(hi);
        if (!hasLo && !hasHi) tryK(0);

        if (best < 0) cout << "-1\n";
        else cout << toString(best) << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const ll INF = LLONG_MAX;

// a,b > 0 and c,d > 0. All intermediate states must remain positive.
ll positiveCore(ll a, ll b, ll c, ll d) {
    ll answer = 0;

    while (true) {
        if (c < a || d < b) return -1;
        if (a == c && b == d) return answer;
        if (c == d) return -1;

        if (c < d) {
            swap(a, b);
            swap(c, d);
            continue;
        }

        // Now c > d. The last reverse operations are forced to be c -= d.
        if (d == b && (c - a) % d == 0) {
            return answer + (c - a) / d;
        }

        // Reduce c to a value in [1,d], unless this would overshoot a.
        ll steps = (c - 1) / d;
        ll maximumSteps = (c - a) / d;
        if (steps > maximumSteps) return -1;

        c -= steps * d;
        answer += steps;
    }
}

// Nonnegative initial state, with a nonnegative target.
ll nonnegative(ll a, ll b, ll c, ll d) {
    if (c < a || d < b) return -1;
    if (a == c && b == d) return 0;
    if (a == 0 && b == 0) return -1;

    if (a == 0) {
        // To make the first coordinate nonzero, (0,b) -> (b,b) is forced.
        if (c == 0) return -1;
        ll result = positiveCore(b, b, c, d);
        return result < 0 ? -1 : result + 1;
    }

    if (b == 0) {
        if (d == 0) return -1;
        ll result = positiveCore(a, a, c, d);
        return result < 0 ? -1 : result + 1;
    }

    return positiveCore(a, b, c, d);
}

struct FSeg {
    ll x, y;       // Candidates are z = x-k*y, second coordinate z-y.
    ll maxK;
    ll prefix;
    ll minZ, maxZ;
};

struct BSeg {
    ll u, v;       // Candidates are (u-t*v,v).
    ll maxT;
    ll prefix;
    ll minZ, maxZ;
};

// Candidates for crossing from (x,-y), x>y, to the positive pair (x,x-y).
vector<FSeg> buildForwardSegments(ll A, ll B) {
    vector<FSeg> segments;
    segments.reserve(128);

    ll x = A, y = B, prefix = 0;
    while (x != y) {
        if (x > y) {
            ll q = x / y;
            ll r = x % y;
            ll maxK = r ? q - 1 : q - 2;

            if (maxK >= 0) {
                ll minZ = x - maxK * y;
                segments.push_back({x, y, maxK, prefix, minZ, x});
            }

            if (r) {
                prefix += q;
                x = r;
            } else {
                break;
            }
        } else {
            ll q = y / x;
            ll r = y % x;

            if (r) {
                prefix += q;
                y = r;
            } else {
                break;
            }
        }
    }
    return segments;
}

// Positive ancestors of the target which can serve as a crossing state.
vector<BSeg> buildBackwardSegments(ll C, ll D) {
    vector<BSeg> segments;
    segments.reserve(128);

    ll x = C, y = D, prefix = 0;
    while (x != y) {
        if (x > y) {
            ll q = x / y;
            ll r = x % y;
            ll maxT = r ? q - 1 : q - 2;

            if (maxT >= 0) {
                ll minZ = x - maxT * y;
                segments.push_back({x, y, maxT, prefix, minZ, x});
            }

            if (r) {
                prefix += q;
                x = r;
            } else {
                break;
            }
        } else {
            ll q = y / x;
            ll r = y % x;

            if (r) {
                prefix += q;
                y = r;
            } else {
                break;
            }
        }
    }
    return segments;
}

// Reach (g,-g), where g = gcd(A,B), using only sign-preserving operations.
pair<ll, ll> equalStateInfo(ll A, ll B) {
    ll x = A, y = B, distance = 0;

    while (x != y) {
        if (x > y) {
            ll q = x / y;
            ll r = x % y;
            if (r) {
                distance += q;
                x = r;
            } else {
                distance += q - 1;
                x = y;
            }
        } else {
            ll q = y / x;
            ll r = y % x;
            if (r) {
                distance += q;
                y = r;
            } else {
                distance += q - 1;
                y = x;
            }
        }
    }

    return {x, distance};
}

// Target is still in the quadrant (positive,negative).
ll oppositeDistance(ll A, ll B, ll C, ll D) {
    ll x = A, y = B, prefix = 0;

    while (true) {
        if (x == C && y == D) return prefix;
        if (x == y) return -1;

        if (x > y) {
            ll q = x / y;
            ll r = x % y;
            ll steps = r ? q : q - 1;
            ll endX = r ? r : y;

            if (D == y && C <= x && C >= endX && (x - C) % y == 0) {
                ll k = (x - C) / y;
                if (k <= steps) return prefix + k;
            }

            prefix += steps;
            x = endX;
        } else {
            ll q = y / x;
            ll r = y % x;
            ll steps = r ? q : q - 1;
            ll endY = r ? r : x;

            if (C == x && D <= y && D >= endY && (y - D) % x == 0) {
                ll k = (y - D) / x;
                if (k <= steps) return prefix + k;
            }

            prefix += steps;
            y = endY;
        }
    }
}

// Start is (A,-B), A,B>0; target is strictly positive.
ll crossToPositive(ll A, ll B, ll C, ll D) {
    ll best = INF;

    // Route through (g,-g), then (g,0), followed by nonnegative operations.
    auto [g, toEqual] = equalStateInfo(A, B);
    ll viaZero = nonnegative(g, 0, C, D);
    if (viaZero >= 0) {
        best = toEqual + 1 + viaZero;
    }

    vector<FSeg> forward = buildForwardSegments(A, B);
    vector<BSeg> backward = buildBackwardSegments(C, D);

    // Both lists consist of disjoint intervals, ordered by decreasing z.
    size_t i = 0, j = 0;
    while (i < forward.size() && j < backward.size()) {
        const FSeg &f = forward[i];
        const BSeg &b = backward[j];

        if (f.minZ > b.maxZ) {
            ++i;
            continue;
        }
        if (b.minZ > f.maxZ) {
            ++j;
            continue;
        }

        // Equality of second coordinates gives z-y = v, hence z = y+v.
        ll z = f.y + b.v;

        if (z >= f.minZ && z <= f.maxZ &&
            z >= b.minZ && z <= b.maxZ &&
            (f.x - z) % f.y == 0 &&
            (b.u - z) % b.v == 0) {

            ll k = (f.x - z) / f.y;
            ll t = (b.u - z) / b.v;
            ll total = f.prefix + k + 1 + b.prefix + t;
            best = min(best, total);
        }

        if (f.minZ > b.minZ) {
            ++i;
        } else if (f.minZ < b.minZ) {
            ++j;
        } else {
            ++i;
            ++j;
        }
    }

    return best == INF ? -1 : best;
}

ll solveQuery(ll a, ll b, ll c, ll d) {
    if (a == 0 && b == 0) {
        return (c == 0 && d == 0) ? 0 : -1;
    }

    if (a >= 0 && b >= 0) {
        if (c < 0 || d < 0) return -1;
        return nonnegative(a, b, c, d);
    }

    if (a <= 0 && b <= 0) {
        if (c > 0 || d > 0) return -1;
        return nonnegative(-a, -b, -c, -d);
    }

    // Normalize the opposite-sign start to a>0, b<0.
    if (a < 0) {
        swap(a, b);
        swap(c, d);
    }

    ll A = a;
    ll B = -b;

    if (c > 0 && d < 0) {
        return oppositeDistance(A, B, c, -d);
    }

    if (c < 0 && d > 0) {
        return -1;
    }

    if (c > 0 && d > 0) {
        return crossToPositive(A, B, c, d);
    }

    if (c < 0 && d < 0) {
        // Negate everything and swap coordinates.
        return crossToPositive(B, A, -d, -c);
    }

    // A target with a zero coordinate can only be reached immediately after
    // arriving at (g,-g).
    auto [g, toEqual] = equalStateInfo(A, B);
    if ((c == g && d == 0) || (c == 0 && d == -g)) {
        return toEqual + 1;
    }

    return -1;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using i128 = __int128_t;

struct Run {
    int64 x, y;       // First pair in this run
    int64 start;      // Global index of the first pair
    int64 n;          // Number of pairs in the run
    int64 lo, hi;     // Depth interval of these pairs
    bool horizontal;  // x > y
};

struct RunInfo {
    vector<Run> runs;
    int64 depth = 0;
    int64 root = 0;
};

static int64 abs64(int64 x) {
    return x < 0 ? -x : x;
}

// Compressed list of all non-root ancestors of (x,y), including (x,y).
// A run with x>y contains (x-t*y,y); one with y>x contains (x,y-t*x).
static RunInfo buildRuns(int64 x, int64 y) {
    RunInfo info;
    int64 index = 0;

    while (x != y) {
        Run r;
        r.x = x;
        r.y = y;
        r.start = index;
        r.horizontal = (x > y);

        if (x > y) {
            int64 q = x / y;
            r.n = (x % y == 0 ? q - 1 : q);
            x -= r.n * y;
        } else {
            int64 q = y / x;
            r.n = (y % x == 0 ? q - 1 : q);
            y -= r.n * x;
        }

        index += r.n;
        info.runs.push_back(r);
    }

    info.depth = index;
    info.root = x; // x == y == gcd of the original pair

    for (Run &r : info.runs) {
        r.hi = info.depth - r.start;
        r.lo = r.hi - r.n + 1;
    }
    return info;
}

// Number of parent steps from the pair used to build info to anc,
// or -1 if anc is not an ancestor.
static int64 ancestorDist(const RunInfo &info, pair<int64,int64> anc) {
    for (const Run &r : info.runs) {
        if (r.horizontal) {
            if (anc.second == r.y && anc.first > r.y && anc.first <= r.x) {
                int64 diff = r.x - anc.first;
                if (diff % r.y == 0) {
                    int64 t = diff / r.y;
                    if (t < r.n) return r.start + t;
                }
            }
        } else {
            if (anc.first == r.x && anc.second > r.x && anc.second <= r.y) {
                int64 diff = r.y - anc.second;
                if (diff % r.x == 0) {
                    int64 t = diff / r.x;
                    if (t < r.n) return r.start + t;
                }
            }
        }
    }

    if (anc.first == info.root && anc.second == info.root)
        return info.depth;
    return -1;
}

static int64 solveQuery(int64 a, int64 b, int64 c, int64 d) {
    // Zero initial states: one useful operation reaches (s,s), after which
    // the ordinary same-sign process applies.
    if (a == 0 || b == 0) {
        if (a == 0 && b == 0)
            return (c == 0 && d == 0) ? 0 : -1;

        if (a == c && b == d) return 0;

        int64 s = (b == 0 ? a : b);
        bool correctSign = (s > 0 && c > 0 && d > 0) ||
                           (s < 0 && c < 0 && d < 0);
        if (!correctSign) return -1;

        int64 x = abs64(c), y = abs64(d);
        if (std::gcd(x, y) != abs64(s)) return -1;

        RunInfo target = buildRuns(x, y);
        return target.depth + 1;
    }

    // Both initial components have the same sign.
    if ((a > 0 && b > 0) || (a < 0 && b < 0)) {
        bool correctSign = (a > 0 && c > 0 && d > 0) ||
                           (a < 0 && c < 0 && d < 0);
        if (!correctSign) return -1;

        pair<int64,int64> start = {abs64(a), abs64(b)};
        pair<int64,int64> target = {abs64(c), abs64(d)};
        RunInfo info = buildRuns(target.first, target.second);
        return ancestorDist(info, start);
    }

    // Canonicalize an opposite-sign start to a>0>b. Swapping both pairs
    // preserves operation counts.
    if (a < 0) {
        swap(a, b);
        swap(c, d);
    }

    int64 x = a, y = -b;
    RunInfo startInfo = buildRuns(x, y);
    int64 g = startInfo.root;

    // Zero targets are reachable only through (g,-g).
    if (c == 0 || d == 0) {
        if ((c == 0 && d == -g) || (c == g && d == 0))
            return startInfo.depth + 1;
        return -1;
    }

// Try to intersect one mixed-prefix segment and one candidate segment.
bool intersectSegments(const Segment& s, const Candidate& c,

    if (s.dir == 0) {
        // The second coordinate of the mixed-chain state is fixed.
        if (mode == 0) {
            // p2 = c.x - (t+1)*c.y = s.y
            i128 num = (i128)c.x - s.y;
            if (num < 0 || num % c.y != 0) ok = false;
            else t = num / c.y - 1;
        } else {
            // p2 = c.y - t*c.x = s.y
            i128 num = (i128)c.y - s.y;
            if (num < 0 || num % c.x != 0) ok = false;
            else t = num / c.x;
        }
    } else if (s.dir == 1) {
        // The first coordinate of the mixed-chain state is fixed.
        if (mode == 0) {
            // p1 = c.x - t*c.y = s.x
            i128 num = (i128)c.x - s.x;
            if (num < 0 || num % c.y != 0) ok = false;
            else t = num / c.y;
        } else {
            // p1 = c.y - (t+1)*c.x = s.x
            i128 num = (i128)c.y - s.x;
            if (num < 0 || num % c.x != 0) ok = false;
            else t = num / c.x - 1;
        }
    } else {
        ok = false;
    }

    if (!ok || t < 0 || t >= c.m) return false;
    int64 tt = (int64)t;

    i128 p1, p2;
    if (mode == 0) {
        p1 = (i128)c.x - (i128)tt * c.y;
        p2 = (i128)c.x - (i128)(tt + 1) * c.y;
    } else {
        p1 = (i128)c.y - (i128)(tt + 1) * c.x;
        p2 = (i128)c.y - (i128)tt * c.x;

    i128 prefixOffset;
    if (s.dir == 0) {
        if (p2 != s.y || p1 > s.x) return false;

    for (const Segment& s : target.seg) {
        if ((mode == 0 && s.dir == 0) || (mode == 1 && s.dir == 1)) {
            if (s.m == 0) continue;

            Candidate c;
            c.x = s.x;
            c.y = s.y;
            c.m = s.m; // Candidates are t=0,...,m-1.

                lo = hi - (i128)2 * (s.m - 1) * s.y;
            } else {
                hi = (i128)2 * s.y - s.x;
                lo = hi - (i128)2 * (s.m - 1) * s.x;
            }
            c.hi = (int64)hi;
            c.lo = (int64)lo;
            candidates.push_back(c);

    size_t i = 0, j = 0;

    // the first intersection also minimizes prefix length + suffix length.
    while (i < mixed.seg.size() && j < candidates.size()) {
        const Segment& s = mixed.seg[i];
        if (s.dir == -1) break;
        const Candidate& c = candidates[j];

        if (c.lo > s.hi) {
            ++j;
            continue;
        }
        if (s.lo > c.hi) {
            ++i;
            continue;
        }

        int64 cost;
        if (intersectSegments(s, c, mode, cost)) {
            best = cost;
            break;
        }

        if (s.lo > c.lo) ++i;
        else if (c.lo > s.lo) ++j;
        else {
            ++i;
            ++j;
        }
    }

    // At the terminal mixed state (g,-g), one can instead pass through an
    // axis state and start the monotone suffix from (g,g), or from (-g,-g).
    if (mixed.g == target.g) {
        i128 special = (i128)mixed.steps + 2 + target.steps;
        best = min(best, (int64)special);

ll positiveCore(ll a, ll b, ll c, ll d) {

            swap(a, b);
            swap(c, d);
            continue;

            return answer + (c - a) / d;

        // Reduce c to a value in [1,d], unless this would overshoot a.
        ll steps = (c - 1) / d;

        if (steps > maximumSteps) return -1;

        answer += steps;
    }
}

            int64 q = y / x;
            r.n = (y % x == 0 ? q - 1 : q);
            y -= r.n * x;
        }

        index += r.n;
        info.runs.push_back(r);
    }

    info.depth = index;
    info.root = x; // x == y == gcd of the original pair

    for (Run &r : info.runs) {
        r.hi = info.depth - r.start;
        r.lo = r.hi - r.n + 1;
    }
    return info;
}

// Number of parent steps from the pair used to build info to anc,
// or -1 if anc is not an ancestor.
static int64 ancestorDist(const RunInfo &info, pair<int64,int64> anc) {
    for (const Run &r : info.runs) {
        if (r.horizontal) {
            if (anc.second == r.y && anc.first > r.y && anc.first <= r.x) {
                int64 diff = r.x - anc.first;
                if (diff % r.y == 0) {
                    int64 t = diff / r.y;
                    if (t < r.n) return r.start + t;
                }
            }
        } else {
            if (anc.first == r.x && anc.second > r.x && anc.second <= r.y) {
                int64 diff = r.y - anc.second;
                if (diff % r.x == 0) {
                    int64 t = diff / r.x;
                    if (t < r.n) return r.start + t;
                }
            }
        }
    }

    if (anc.first == info.root && anc.second == info.root)
        return info.depth;
    return -1;
}

static int64 solveQuery(int64 a, int64 b, int64 c, int64 d) {
    // Zero initial states: one useful operation reaches (s,s), after which
    // the ordinary same-sign process applies.
    if (a == 0 || b == 0) {
        if (a == 0 && b == 0)
            return (c == 0 && d == 0) ? 0 : -1;

        if (a == c && b == d) return 0;

        int64 s = (b == 0 ? a : b);
        bool correctSign = (s > 0 && c > 0 && d > 0) ||
                           (s < 0 && c < 0 && d < 0);
        if (!correctSign) return -1;

        int64 x = abs64(c), y = abs64(d);
        if (std::gcd(x, y) != abs64(s)) return -1;

        RunInfo target = buildRuns(x, y);
        return target.depth + 1;
    }

    // Both initial components have the same sign.
    if ((a > 0 && b > 0) || (a < 0 && b < 0)) {
        bool correctSign = (a > 0 && c > 0 && d > 0) ||
                           (a < 0 && c < 0 && d < 0);
        if (!correctSign) return -1;

        pair<int64,int64> start = {abs64(a), abs64(b)};
        pair<int64,int64> target = {abs64(c), abs64(d)};
        RunInfo info = buildRuns(target.first, target.second);
        return ancestorDist(info, start);
    }

    // Canonicalize an opposite-sign start to a>0>b. Swapping both pairs
    // preserves operation counts.
    if (a < 0) {
        swap(a, b);
        swap(c, d);
    }

    int64 x = a, y = -b;
    RunInfo startInfo = buildRuns(x, y);
    int64 g = startInfo.root;

    // Zero targets are reachable only through (g,-g).
    if (c == 0 || d == 0) {
        if ((c == 0 && d == -g) || (c == g && d == 0))
            return startInfo.depth + 1;
        return -1;
    }

    // Still opposite signs: only the deterministic non-crossing chain applies.
    if (c > 0 && d < 0) {
        return ancestorDist(startInfo, {c, -d});
    }
    if (c < 0 && d > 0) return -1;

    // The target has equal signs. Normalize to positive absolute values.
    bool positiveTarget = (c > 0);
    int64 tx = abs64(c), ty = abs64(d);
    if (std::gcd(tx, ty) != g) return -1;

    RunInfo targetInfo = buildRuns(tx, ty);

    // Fallback: go to (g,-g), then through a zero state to (g,g) or (-g,-g),
    // and finally follow the same-sign path to the target.
    i128 best = (i128)startInfo.depth + targetInfo.depth + 2;

    // A crossing at a non-root ancestor (x,y), x>y, produces the positive
    // base (x,x-y). At x<y it produces the negative base with absolute
    // values (y-x,y). Match all such bases against ancestors of the target.
    const auto &S = startInfo.runs;
    const auto &T = targetInfo.runs;
    size_t i = 0, j = 0;

    while (i < S.size() && j < T.size()) {
        const Run &s = S[i];
        const Run &t = T[j];

        bool overlap = !(s.hi < t.lo || t.hi < s.lo);

        if (!overlap) {
            if (s.lo > t.hi) ++i;
            else ++j;
            continue;
        }

        // Positive bases come from horizontal runs, negative ones from
        // vertical runs. Equal pairs must also have the same orientation.
        if (s.horizontal == positiveTarget && s.horizontal == t.horizontal) {
            if (s.horizontal) {
                // Base: (s.x-k*s.y, s.x-(k+1)*s.y)
                // Target run: (t.x-l*t.y, t.y)
                int64 diff = s.x - t.y;
                if (diff > 0 && diff % s.y == 0) {
                    int64 h = diff / s.y; // h = k+1
                    if (h >= 1 && h <= s.n) {
                        int64 k = h - 1;
                        int64 baseX = s.x - k * s.y;
                        int64 diff2 = t.x - baseX;
                        if (diff2 >= 0 && diff2 % t.y == 0) {
                            int64 l = diff2 / t.y;
                            if (l < t.n) {
                                i128 cost = (i128)s.start + k + 1 + t.start + l;
                                if (cost < best) best = cost;
                            }
                        }
                    }
                }
            } else {
                // Base: (s.y-(k+1)*s.x, s.y-k*s.x)
                // Target run: (t.x, t.y-l*t.x)
                int64 diff = s.y - t.x;
                if (diff > 0 && diff % s.x == 0) {
                    int64 h = diff / s.x; // h = k+1
                    if (h >= 1 && h <= s.n) {
                        int64 k = h - 1;
                        int64 baseY = s.y - k * s.x;
                        int64 diff2 = t.y - baseY;
                        if (diff2 >= 0 && diff2 % t.x == 0) {
                            int64 l = diff2 / t.x;
                            if (l < t.n) {
                                i128 cost = (i128)s.start + k + 1 + t.start + l;
                                if (cost < best) best = cost;
                            }
                        }
                    }
                }
            }
        }

        // Enumerate only pairs of runs whose depth intervals overlap.
        if (s.lo > t.lo) ++i;
        else if (t.lo > s.lo) ++j;
        else {
            ++i;
            ++j;
        }
    }

    return (int64)best;
}

// Depth of a positive primitive pair (x, y) in the Stern-Brocot tree
// rooted at (1, 1), using reverse subtraction.
long long depth_positive(long long x, long long y) {
    long long d = 0;
    while (!(x == 1 && y == 1)) {
        if (x > y) {
            long long q = (x - 1) / y;
            x -= q * y;
            d += q;
        } else {
            long long q = (y - 1) / x;
            y -= q * x;
            d += q;
        }
    }
    return d;
}

// Move (x, y) up by 'steps' parent operations in the positive tree.
pair<long long, long long> climb(long long x, long long y, long long steps) {
    while (steps > 0 && !(x == 1 && y == 1)) {
        if (x > y) {
            long long q = (x - 1) / y;
            long long take = min(q, steps);
            x -= take * y;
            steps -= take;
        } else {
            long long q = (y - 1) / x;
            long long take = min(q, steps);
            y -= take * x;
            steps -= take;
        }
    }
    return {x, y};
}

// Minimum operations from positive primitive (x,y) to positive primitive (c,d).

    return -1;
}

// Build root-to-(X,Y) path in the positive primitive tree.
// X,Y > 0 and gcd(X,Y)=1.
Path build_path(long long X, long long Y) {
    vector<pair<char, long long>> rev;

    long long x = X, y = Y;
    while (!(x == 1 && y == 1)) {
        if (x > y) {
            long long q = (x - 1) / y;
            rev.push_back({'A', q});
            x -= q * y;
        } else {
            long long q = (y - 1) / x;
            rev.push_back({'B', q});
            y -= q * x;
        }
    }

    reverse(rev.begin(), rev.end());

    Path path;
    path.segs.push_back({0, 0, 1, 1, 0, 0});

    long long curx = 1, cury = 1, dep = 0;

    for (auto [op, q] : rev) {
        long long l = dep + 1;
        long long r = dep + q;

        if (op == 'A') {
            path.segs.push_back({l, r, curx + cury, cury, cury, 0});
            curx += q * cury;
        } else {
            path.segs.push_back({l, r, curx, cury + curx, 0, curx});
            cury += q * curx;
        }

        dep = r;
    }

    path.depth = dep;
    return path;
}

// Source is PN: (p, -q). Target is PP: (c, d).
// All p,q,c,d > 0.
long long calc(long long p, long long q, long long c, long long d) {
    Path ps = build_path(p, q);
    Path pt = build_path(c, d);

    long long DS = ps.depth;
    long long DC = pt.depth;
    long long limit = min(DS, DC);

    long long best = -1;

    size_t i = 0, j = 0;

    while (i < ps.segs.size() && j < pt.segs.size()) {
        long long L = max(ps.segs[i].l, pt.segs[j].l);
        long long R = min(ps.segs[i].r, pt.segs[j].r);

        if (L <= R) {
            long long LL = max(L, 1LL);
            long long RR = min(R, limit);

            if (LL <= RR) {
                const Seg &ss = ps.segs[i];
                const Seg &tt = pt.segs[j];

                __int128 Sx = (__int128)ss.x0 + (__int128)ss.sx * (LL - ss.l);
                __int128 Sy = (__int128)ss.y0 + (__int128)ss.sy * (LL - ss.l);
                __int128 Cx = (__int128)tt.x0 + (__int128)tt.sx * (LL - tt.l);
                __int128 Cy = (__int128)tt.y0 + (__int128)tt.sy * (LL - tt.l);

                __int128 sx = ss.sx, sy = ss.sy;
                __int128 tx = tt.sx, ty = tt.sy;

                // Need:
                // Sx(t) = Cx(t)
                // Sx(t) - Sy(t) = Cy(t)
                //
                // t = LL + k
                __int128 a1 = sx - tx;
                __int128 b1 = Cx - Sx;

                __int128 a2 = sx - sy - ty;
                __int128 b2 = Cy - (Sx - Sy);

                long long len = RR - LL;
                long long cand = -1;

                if (a1 == 0) {
                    if (b1 == 0) {
                        if (a2 == 0) {
                            if (b2 == 0) cand = RR;
                        } else {
                            if (b2 % a2 == 0) {
                                __int128 k = b2 / a2;
                                if (k >= 0 && k <= len) {
                                    cand = LL + (long long)k;
                                }
                            }
                        }
                    }
                } else {
                    if (b1 % a1 == 0) {
                        __int128 k = b1 / a1;
                        if (k >= 0 && k <= len) {
                            if (a2 * k == b2) {
                                cand = LL + (long long)k;
                            }
                        }
                    }
                }

                if (cand > best) best = cand;
            }
        }

        if (ps.segs[i].r < pt.segs[j].r) {
            ++i;
        } else if (ps.segs[i].r > pt.segs[j].r) {
            ++j;
        } else {
            ++i;
            ++j;
        }
    }

    if (best != -1) {
        return DS + DC + 1 - 2LL * best;
    } else {
        // Go through (1,0): mixed -> zero -> (1,1) -> target.
        return DS + DC + 2;
    }
}

long long solve_query(long long a, long long b, long long c, long long d) {
    long long g1 = gcdll(a, b);

    if (g1 == 0) {
        return (c == 0 && d == 0) ? 0 : -1;
    }

    long long g2 = gcdll(c, d);
    if (g1 != g2) return -1;

    a /= g1;
    b /= g1;
    c /= g1;
    d /= g1;

    // Source positive-positive.
    if (a > 0 && b > 0) {
        if (c > 0 && d > 0) return positive_distance(a, b, c, d);
        return -1;
    }

    // Source negative-negative.
    if (a < 0 && b < 0) {
        if (c < 0 && d < 0) return positive_distance(-a, -b, -c, -d);
        return -1;
    }

    // Source has a zero coordinate.
    if (a == 0 || b == 0) {
        if (a == 1 && b == 0) {
            if (c == 1 && d == 0) return 0;
            if (c > 0 && d > 0) return 1 + depth_positive(c, d);
            return -1;
        }

        if (a == -1 && b == 0) {
            if (c == -1 && d == 0) return 0;
            if (c < 0 && d < 0) return 1 + depth_positive(-c, -d);
            return -1;
        }

        if (a == 0 && b == 1) {
            if (c == 0 && d == 1) return 0;
            if (c > 0 && d > 0) return 1 + depth_positive(c, d);
            return -1;
        }

        if (a == 0 && b == -1) {
            if (c == 0 && d == -1) return 0;
            if (c < 0 && d < 0) return 1 + depth_positive(-c, -d);
            return -1;
        }

        return -1;
    }

    // Now source is mixed. If it is NP, swap coordinates to make it PN.
    if (a < 0 && b > 0) {
        swap(a, b);
        swap(c, d);
    }

    // Source is PN: (a,b) = (p, -q).
    long long p = a;
    long long q = -b;

    // Target PN.
    if (c > 0 && d < 0) {
        return mixed_same(p, q, c, -d);
    }

    // Target NP: impossible because mixed sign pattern cannot flip.
    if (c < 0 && d > 0) {
        return -1;
    }

    // Target PP.
    if (c > 0 && d > 0) {
        return calc(p, q, c, d);
    }

    // Target NN.
    if (c < 0 && d < 0) {
        return calc(q, p, -d, -c);
    }

    // Target has a zero coordinate.
    if ((c == 1 && d == 0) || (c == 0 && d == -1)) {
        return depth_positive(p, q) + 1;
    }

    return -1;
}

// Find k in [0, L] such that F(A(k)) = B(k), where A starts as (ax, ay),

bool solve_interval(ll ax, ll ay, ll bx, ll by, ll L, ll &kout) {

pair<i128, i128> bezout(long long a, long long b) {

    long long B = (b < 0 ? -b : b);

    i128 old_r = A, r = B;
    i128 old_s = 1, s = 0;
    i128 old_t = 0, t = 1;

    while (r != 0) {
        i128 q = old_r / r;

        i128 nr = old_r - q * r;
        i128 ns = old_s - q * s;
        i128 nt = old_t - q * t;

        old_r = r;
        r = nr;
        old_s = s;
        s = ns;
        old_t = t;
        t = nt;

    i128 y = old_t;

    if (b < 0) y = -y;

            i128 need = ceil_div(-val, coeff);

/*
For positive x and y, repeatedly subtract the smaller value from the larger
one until reaching (g,g), where g = gcd(x,y).

Reversing these subtractions gives a word over:
    A: (x,y) -> (x+y,y)
    B: (x,y) -> (x,y+x)

The word is stored in run-length encoded form.
*/
void buildPath(int64 x, int64 y, Path& path) {
    Run reversedRuns[MAX_RUNS];
    int count = 0;
    int64 length = 0;

    while (x != y) {
        if (x > y) {
            int64 k = (x - 1) / y;
            reversedRuns[count++] = {'A', k};
            x -= k * y;
            length += k;
        } else {
            int64 k = (y - 1) / x;
            reversedRuns[count++] = {'B', k};
            y -= k * x;
            length += k;
        }
    }

    path.count = count;
    path.length = length;
    for (int i = 0; i < count; ++i) {
        path.runs[i] = reversedRuns[count - 1 - i];
    }
}

char opposite(char ch) {
    return ch == 'A' ? 'B' : 'A';
}

/*
Longest common prefix of two paths.
If flipFirst is true, every A/B in the first path is interchanged.
*/
int64 commonPrefix(const Path& first, const Path& second, bool flipFirst) {
    int i = 0, j = 0;
    int64 remainingFirst = first.count ? first.runs[0].len : 0;
    int64 remainingSecond = second.count ? second.runs[0].len : 0;
    int64 result = 0;

    while (i < first.count && j < second.count) {
        char firstChar = flipFirst ? opposite(first.runs[i].ch) : first.runs[i].ch;
        char secondChar = second.runs[j].ch;

        if (firstChar != secondChar) break;

        int64 take = min(remainingFirst, remainingSecond);
        result += take;
        remainingFirst -= take;
        remainingSecond -= take;

        if (remainingFirst == 0) {
            ++i;
            if (i < first.count) remainingFirst = first.runs[i].len;
        }
        if (remainingSecond == 0) {
            ++j;
            if (j < second.count) remainingSecond = second.runs[j].len;
        }
    }

    return result;
}

/*
Largest position n <= cap such that both paths contain the requested
character at position n. Positions are zero-indexed.
*/
int64 lastCommonDesiredPosition(
    const Path& first,
    const Path& second,
    char desired,
    int64 cap
) {
    if (cap < 0) return -1;

    int i = 0, j = 0;
    int64 positionFirst = 0, positionSecond = 0;
    int64 remainingFirst = first.count ? first.runs[0].len : 0;
    int64 remainingSecond = second.count ? second.runs[0].len : 0;
    int64 answer = -1;

    while (i < first.count && j < second.count) {
        int64 endFirst = positionFirst + remainingFirst;
        int64 endSecond = positionSecond + remainingSecond;

        if (first.runs[i].ch == desired && second.runs[j].ch == desired) {
            int64 low = max(positionFirst, positionSecond);
            int64 high = min(endFirst, endSecond); // exclusive

            if (low < high) {
                int64 candidate = min(high - 1, cap);
                if (candidate >= low) {
                    answer = max(answer, candidate);
                }
            }
        }

        int64 take = min(remainingFirst, remainingSecond);
        positionFirst += take;
        positionSecond += take;
        remainingFirst -= take;
        remainingSecond -= take;

        if (remainingFirst == 0) {
            ++i;
            if (i < first.count) remainingFirst = first.runs[i].len;
        }
        if (remainingSecond == 0) {
            ++j;
            if (j < second.count) remainingSecond = second.runs[j].len;
        }
    }

    return answer;
}

bool isPrefix(const Path& prefix, const Path& path) {
    return commonPrefix(prefix, path, false) == prefix.length;
}

int signOf(int64 x) {
    return (x > 0) - (x < 0);
}

int64 absoluteValue(int64 x) {
    return x < 0 ? -x : x;
}

enum Kind {
    ZERO_ZERO,
    ZERO_ONE,
    SAME_SIGN,
    MIXED_SIGN
};

Kind classify(int64 x, int64 y) {
    if (x == 0 && y == 0) return ZERO_ZERO;
    if (x == 0 || y == 0) return ZERO_ONE;
    if ((x > 0) == (y > 0)) return SAME_SIGN;
    return MIXED_SIGN;
}

int64 solveQuery(int64 a, int64 b, int64 c, int64 d) {
    if (a == c && b == d) return 0;

    int64 sourceGcd = gcd(absoluteValue(a), absoluteValue(b));
    int64 targetGcd = gcd(absoluteValue(c), absoluteValue(d));

    // The gcd is invariant under both operations.
    if (sourceGcd != targetGcd) return -1;

    Kind sourceKind = classify(a, b);
    Kind targetKind = classify(c, d);

    if (sourceKind == ZERO_ZERO) {
        return -1;
    }

    if (sourceKind == ZERO_ONE) {
        // The only meaningful operation creates (x,x).
        if (targetKind != SAME_SIGN) return -1;

        int64 sourceSign = signOf(a != 0 ? a : b);
        if (signOf(c) != sourceSign) return -1;

        Path targetPath;
        buildPath(absoluteValue(c), absoluteValue(d), targetPath);
        return 1 + targetPath.length;
    }

    if (sourceKind == SAME_SIGN) {
        // The sign cannot change, and the target must be a descendant
        // of the source in the positive Euclidean tree.
        if (targetKind != SAME_SIGN) return -1;
        if (signOf(c) != signOf(a)) return -1;

        Path sourcePath, targetPath;
        buildPath(absoluteValue(a), absoluteValue(b), sourcePath);
        buildPath(absoluteValue(c), absoluteValue(d), targetPath);

        if (!isPrefix(sourcePath, targetPath)) return -1;
        return targetPath.length - sourcePath.length;
    }

    // The source has opposite signs.
    if (targetKind == ZERO_ZERO) {
        return -1;
    }

    if (targetKind == ZERO_ONE) {
        Path sourcePath;
        buildPath(absoluteValue(a), absoluteValue(b), sourcePath);

        // Equality must first be reached. Then either coordinate can be
        // changed into zero, retaining the sign of the other coordinate.
        bool valid =
            (c == 0 && d == signOf(b) * sourceGcd) ||
            (d == 0 && c == signOf(a) * sourceGcd);

        if (!valid) return -1;
        return sourcePath.length + 1;
    }

    if (targetKind == MIXED_SIGN) {
        // While the signs are opposite, staying mixed forces us to update
        // the coordinate with larger absolute value.
        if (signOf(c) != signOf(a) || signOf(d) != signOf(b)) {
            return -1;
        }

        Path sourcePath, targetPath;
        buildPath(absoluteValue(a), absoluteValue(b), sourcePath);
        buildPath(absoluteValue(c), absoluteValue(d), targetPath);

        if (!isPrefix(targetPath, sourcePath)) return -1;
        return sourcePath.length - targetPath.length;
    }

    // Mixed-sign source, same-sign target.
    Path sourcePath, targetPath;
    buildPath(absoluteValue(a), absoluteValue(b), sourcePath);
    buildPath(absoluteValue(c), absoluteValue(d), targetPath);

    // A is wanted when the first coordinate has the target sign.
    // Otherwise B is wanted.
    char wanted = signOf(c) == signOf(a) ? 'A' : 'B';

    /*
    For a prefix R L of the source path, exiting the mixed-sign region
    toward the desired sign produces a positive path equal to swap(R) L,
    where swap interchanges A and B.

    If n = |R|, this candidate is a prefix of the target exactly when:
      - the first n target characters equal the first n characters of swap(P),
      - P[n] = wanted,
      - T[n] = wanted.
    */
    int64 matched = commonPrefix(sourcePath, targetPath, true);
    int64 cap = min(matched, min(sourcePath.length, targetPath.length) - 1);

    int64 n = lastCommonDesiredPosition(
        sourcePath, targetPath, wanted, cap
    );

    if (n >= 0) {
        int64 k = n + 1;
        return sourcePath.length + targetPath.length - 2 * k + 1;
    }

    /*
    No earlier exit works. Continue to equality (g,-g), use two operations
    to reach either (g,g) or (-g,-g), and then follow the target's tree path.
    */
    return sourcePath.length + targetPath.length + 2;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int Q;
    cin >> Q;

    while (Q--) {
        int64 a, b, c, d;
        cin >> a >> b >> c >> d;
        cout << solveQuery(a, b, c, d) << '\n';
    }

    return 0;
}
