                    return plan;

                plan.cost += d;
                load -= d;

            if (rel < ev.steps) {
                int next = bestCycle[(idx + 1) % M];
                if (!adjacent(cur, next)) {
                    plan.valid = false;
                    return plan;
                }
                addMove(next);
            }
        }

        if (load != 0 || plan.ops.size() > 100000) plan.valid = false;
        return plan;
    };

        plan.cost = 0;
        plan.valid = true;
        vector<int> w = h;
        int remaining = 0;
        for (int x : w) if (x != 0) ++remaining;

        int cur = 0;
        ll load = 0;
        bool horizontalFirst = (variant & 1);
        int amountWeight = (variant >= 2) ? 15 : 0;

            plan.ops.push_back(op);
            plan.cost += 100 + load;
            cur = to;
        };

        while (remaining > 0) {
            if ((int)plan.ops.size() > 90000) {
                plan.valid = false;
                return plan;
            }

            if (load == 0 && w[cur] > 0) {
                int d = w[cur];
                plan.ops.push_back("+" + to_string(d));
                plan.cost += d;
                load += d;
                w[cur] = 0;
                --remaining;
                continue;
            }

            if (load > 0 && w[cur] < 0) {
                int d = (int)min<ll>(load, -w[cur]);
                plan.ops.push_back("-" + to_string(d));
                plan.cost += d;
                load -= d;
                w[cur] += d;
                if (w[cur] == 0) --remaining;
                continue;
            }

            int targetCell = -1;
            ll bestScore = INF;
            for (int cell = 0; cell < M; ++cell) {
                if (load == 0) {
                    if (w[cell] <= 0) continue;
                } else {
                    if (w[cell] >= 0) continue;
                }

                ll score = 100LL * dist[cur][cell];
                if (load == 0) {
                    score -= amountWeight * (ll)w[cell];
                } else if (variant >= 2 && -w[cell] < load) {
                    score += 40;
                }
                if (variant == 3) score += (ll)(rng() % 7);

                if (score < bestScore) {
                    bestScore = score;
                    targetCell = cell;
                }
            }

            if (targetCell < 0) {
                plan.valid = false;
                return plan;
            }

            int r = cur / N, c = cur % N;
            int tr = targetCell / N, tc = targetCell % N;
            bool canHorizontal = (c != tc);
            bool canVertical = (r != tr);
            bool goHorizontal = canHorizontal &&
                (!canVertical || horizontalFirst);

            if (goHorizontal) {
                if (tc > c) addMove(cur + 1, "R");
                else addMove(cur - 1, "L");
            } else {
                if (tr > r) addMove(cur + N, "D");
                else addMove(cur - N, "U");
            }
        }

        if (load != 0 || plan.ops.size() > 100000) plan.valid = false;
        return plan;
    };

    for (int variant = 0; variant < 4; ++variant) {
        Plan p = greedyPlan(variant);
        if (p.valid && (!answer.valid || p.cost < answer.cost)) {
            answer = move(p);
        }
    }

    for (const string& op : answer.ops) {
        cout << op << '\n';
    }
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

struct Op {
    int type; // 0: load, 1: unload, 2: U, 3: D, 4: L, 5: R
    int d;
};

struct Config {
    double emptyLook = 0.0;
    double nextLook = 0.0;
    double noise = 0.0;
    double amountBonus = 0.0;
    bool oppUnload = false;
    bool oppPick = false;
    bool oppLoad = false;
    int pathStyle = 0;
    int collectRadius = 0;
    int collectMaxLoad = 0;
    double collectRatio = 1.0;
};

class Solver {
    static constexpr int N = 20;
    static constexpr int CELLS = N * N;
    array<int, CELLS> initial{};
    mt19937 rng{123456789};
    vector<Op> bestOps;
    long long bestCost = (1LL << 62);

    static int row(int p) { return p / N; }
    static int col(int p) { return p % N; }

    static int distance(int a, int b) {
        return abs(row(a) - row(b)) + abs(col(a) - col(b));
    }

    pair<long long, vector<Op>> simulate(const Config &cfg) {
        array<int, CELLS> h = initial;
        vector<Op> ops;
        ops.reserve(12000);

        int pos = 0;
        int load = 0;
        int posCount = 0, negCount = 0;
        for (int i = 0; i < CELLS; ++i) {
            if (h[i] > 0) ++posCount;
            if (h[i] < 0) ++negCount;
        }

        long long cost = 0;

        auto doLoad = [&](int p) {
            int d = h[p];
            h[p] = 0;
            load += d;
            cost += d;
            --posCount;
            ops.push_back({0, d});
        };

        auto doUnload = [&](int p) {
            int d = min(load, -h[p]);
            h[p] += d;
            load -= d;
            cost += d;
            if (h[p] == 0) --negCount;
            ops.push_back({1, d});
        };

        auto handleCurrent = [&](bool intermediate) {
            if (h[pos] > 0) {
                if (load == 0) {
                    if (!intermediate || cfg.oppPick) {
                        doLoad(pos);
                        return true;
                    }
                } else if (cfg.oppLoad) {
                    doLoad(pos);
                    return true;
                }
            }
            if (h[pos] < 0 && load > 0 && (!intermediate || cfg.oppUnload)) {
                doUnload(pos);
                return true;
            }
            return false;
        };

        auto moveOne = [&](int type) {
            if (type == 2) pos -= N;
            else if (type == 3) pos += N;
            else if (type == 4) --pos;
            else ++pos;
            cost += 100 + load;
            ops.push_back({type, 0});
        };

        auto distanceField = [&](bool positive) {
            array<int, CELLS> dist;
            dist.fill(-1);
            queue<int> q;
            for (int i = 0; i < CELLS; ++i) {
                if ((positive && h[i] > 0) || (!positive && h[i] < 0)) {
                    dist[i] = 0;
                    q.push(i);
                }
            }
            const int dr[4] = {-1, 1, 0, 0};
            const int dc[4] = {0, 0, -1, 1};
            while (!q.empty()) {
                int p = q.front();
                q.pop();
                int r = row(p), c = col(p);
                for (int k = 0; k < 4; ++k) {
                    int nr = r + dr[k], nc = c + dc[k];
                    if (nr < 0 || nr >= N || nc < 0 || nc >= N) continue;
                    int np = nr * N + nc;
                    if (dist[np] != -1) continue;
                    dist[np] = dist[p] + 1;
                    q.push(np);
                }
            }
            return dist;
        };

        auto realNoise = [&]() {
            return uniform_real_distribution<double>(-1.0, 1.0)(rng);
        };

        while (true) {
            handleCurrent(false);
            if (posCount == 0 && negCount == 0 && load == 0) break;

            int target = -1;

            if (load == 0) {
                array<int, CELLS> negDist{};
                if (cfg.emptyLook != 0.0) negDist = distanceField(false);

                double best = 1e100;
                for (int s = 0; s < CELLS; ++s) {
                    if (h[s] <= 0) continue;
                    double score = distance(pos, s);
                    if (cfg.emptyLook != 0.0 && negDist[s] >= 0) {
                        score += cfg.emptyLook * negDist[s];
                    }
                    score -= cfg.amountBonus * (double)h[s] / 100.0;
                    score += cfg.noise * realNoise();
                    if (score < best) {
                        best = score;
                        target = s;
                    }
                }
            } else {
                if (cfg.collectRadius > 0 && load < cfg.collectMaxLoad) {
                    int nearestSource = -1;
                    int ds = 1000;
                    for (int s = 0; s < CELLS; ++s) {
                        if (h[s] <= 0) continue;
                        int d = distance(pos, s);
                        if (d < ds) {
                            ds = d;
                            nearestSource = s;
                        }
                    }
                    if (nearestSource != -1 && ds <= cfg.collectRadius) {
                        array<int, CELLS> negDist = distanceField(false);
                        int dn = negDist[pos] < 0 ? 1000 : negDist[pos];
                        if (ds < cfg.collectRatio * max(1, dn) + 1.0) {
                            target = nearestSource;
                        }
                    }
                }

                if (target == -1) {
                    array<int, CELLS> posDist{};
                    if (cfg.nextLook != 0.0) posDist = distanceField(true);

                    double best = 1e100;
                    for (int t = 0; t < CELLS; ++t) {
                        if (h[t] >= 0) continue;
                        double score = distance(pos, t);
                        if (cfg.nextLook != 0.0 && load <= -h[t] && posDist[t] >= 0) {
                            score += cfg.nextLook * posDist[t];
                        }
                        score -= cfg.amountBonus * (double)(-h[t]) / 100.0;
                        score += cfg.noise * realNoise();
                        if (score < best) {
                            best = score;
                            target = t;
                        }
                    }
                }
            }

            if (target == -1) break;

            while (pos != target) {
                int dr = row(target) - row(pos);
                int dc = col(target) - col(pos);
                vector<int> options;

                auto addHorizontal = [&]() {
                    if (dc > 0) options.push_back(5);
                    else if (dc < 0) options.push_back(4);
                };
                auto addVertical = [&]() {
                    if (dr > 0) options.push_back(3);
                    else if (dr < 0) options.push_back(2);
                };

                if (cfg.pathStyle == 0) {
                    addHorizontal();
                    addVertical();
                } else if (cfg.pathStyle == 1) {
                    addVertical();
                    addHorizontal();
                } else {
                    addHorizontal();
                    addVertical();
                    if (options.size() == 2) {
                        int chosen = 0;
                        if (cfg.pathStyle == 2) {
                            chosen = (rng() & 1);
                        } else {
                            double utility[2] = {0.0, 0.0};
                            for (int k = 0; k < 2; ++k) {
                                int np = pos;
                                if (options[k] == 2) np -= N;
                                else if (options[k] == 3) np += N;
                                else if (options[k] == 4) --np;
                                else ++np;

                                if (h[np] < 0 && load > 0 && cfg.oppUnload) {
                                    utility[k] += 2.0 + min(load, -h[np]) / 100.0;
                                }
                                if (h[np] > 0 &&
                                    ((load == 0 && cfg.oppPick) || (load > 0 && cfg.oppLoad))) {
                                    utility[k] += 2.0 + h[np] / 100.0;
                                }
                                utility[k] += uniform_real_distribution<double>(0.0, 0.2)(rng);
                            }
                            chosen = utility[1] > utility[0];
                        }
                        swap(options[0], options[chosen]);
                    }
                }

                moveOne(options[0]);
                if (pos != target && handleCurrent(true)) break;
            }
        }

        return {cost, move(ops)};
    }

public:
    void readInput() {
        int n;
        cin >> n;
        for (int i = 0; i < CELLS; ++i) cin >> initial[i];
    }

    void solve() {
        const int RUNS = 320;

        for (int run = 0; run < RUNS; ++run) {
            Config cfg;

            if (run == 1) {

                cfg.oppPick = true;

            } else if (run == 2) {

                cfg.oppUnload = true;
                cfg.oppPick = true;
                cfg.pathStyle = 3;
            } else if (run == 3) {
                cfg.emptyLook = 0.5;
                cfg.nextLook = 0.5;
                cfg.oppUnload = true;
                cfg.oppPick = true;
                cfg.pathStyle = 3;
            } else if (run == 4) {
                cfg.emptyLook = 0.5;
                cfg.nextLook = 0.25;
                cfg.oppUnload = true;
                cfg.oppPick = true;
                cfg.oppLoad = true;
                cfg.pathStyle = 3;
            } else if (run == 5) {
                cfg.emptyLook = 1.0;
                cfg.oppUnload = true;
                cfg.oppPick = true;
                cfg.oppLoad = true;
                cfg.pathStyle = 3;
                cfg.collectRadius = 2;
                cfg.collectMaxLoad = 400;
                cfg.collectRatio = 1.0;
            } else if (run >= 6) {

                const double noises[] = {0.0, 0.05, 0.15, 0.35, 0.7, 1.2, 2.0};

                cfg.emptyLook = looks[rng() % 7];
                cfg.nextLook = looks[rng() % 7];

            long long cost;
        };

        State current;
        current.order = initial;
        current.prefix.assign(m + 1, 0);
        current.cost = routeCost(current.order);

        auto rebuild = [&](State& state) {
            state.prefix.assign(m + 1, 0);
            for (int i = 0; i < m; ++i)
                state.prefix[i + 1] = state.prefix[i] + tasks[state.order[i]].delta;
            state.cost = routeCost(state.order);
        };

        rebuild(current);

        State runBest = current;
        if (current.cost < globalBestCost) {
            globalBestCost = current.cost;
            globalBest = current.order;
        }

        vector<int> segment;
        segment.reserve(m);
        uniform_real_distribution<double> uniform01(0.0, 1.0);

        for (int iteration = 0; iteration < iterations; ++iteration) {
            double progress = double(iteration) / double(iterations);
            double temperature = initialTemperature *
                pow(finalTemperature / initialTemperature, progress);

            int left = rng() % m;
            int right = left;
            int type = rng() % 100;

            auto randomOther = [&](int a) {
                if (rng() & 1) {
                    int span = min(30, m - 1);
                    int offset = 1 + rng() % span;
                    int b = a + ((rng() & 1) ? offset : -offset);
                    if (b < 0) b += m;
                    if (b >= m) b -= m;
                    return b;
                }
                return rng() % m;
            };

            if (type < 24) {
                int i = rng() % (m - 1);
                int j = i + 1;
                left = i;
                right = j;
                segment.assign(current.order.begin() + i,
                               current.order.begin() + j + 1);
                swap(segment.front(), segment.back());
            } else if (type < 44) {
                int i = randomOther(left);
                int j = randomOther(i);
                if (i == j) continue;
                left = min(i, j);
                right = max(i, j);
                segment.assign(current.order.begin() + left,
                               current.order.begin() + right + 1);
                swap(segment.front(), segment.back());
            } else if (type < 69) {
                int from = rng() % m;
                int to = randomOther(from);
                if (from == to) continue;

                left = min(from, to);
                right = max(from, to);
                segment.clear();

                if (from < to) {
                    for (int i = from + 1; i <= to; ++i)
                        segment.push_back(current.order[i]);
                    segment.push_back(current.order[from]);
                } else {
                    segment.push_back(current.order[from]);
                    for (int i = to; i < from; ++i)
                        segment.push_back(current.order[i]);
                }
            } else if (type < 84) {
                if (sourceTasks.empty() || sinkTasks.empty()) continue;
                int st = sourceTasks[rng() % sourceTasks.size()];
                int kt = sinkTasks[rng() % sinkTasks.size()];

                int si = -1, ki = -1;
                for (int i = 0; i < m; ++i) {
                    if (current.order[i] == st) si = i;
                    if (current.order[i] == kt) ki = i;
                }
                if (si < 0 || ki < 0 || si == ki) continue;

                left = min(si, ki);
                right = max(si, ki);
                segment.assign(current.order.begin() + left,
                               current.order.begin() + right + 1);
                swap(segment.front(), segment.back());
            } else if (type < 94) {
                int a = rng() % m;
                int b = a + 1 + rng() % (m - a - 1);
                int c = b + 1 + rng() % (m - b);
                left = a;
                right = c - 1;
                segment.clear();
                for (int i = b; i < c; ++i) segment.push_back(current.order[i]);
                for (int i = a; i < b; ++i) segment.push_back(current.order[i]);
            } else {
                int i = rng() % m;
                int j = rng() % m;
                if (i == j) continue;
                left = min(i, j);
                right = max(i, j);
                segment.assign(current.order.begin() + left,
                               current.order.begin() + right + 1);
                reverse(segment.begin(), segment.end());
            }

            int i = left, j = right;
            long long oldCost = 0;
            long long newCost = 0;

            int firstDistance;
            if (i == 0) firstDistance = startDist[current.order[i]];
            else firstDistance = distMatrix[current.order[i - 1]][current.order[i]];

            oldCost = 1LL * firstDistance * (100 + current.prefix[i]);

            for (int k = i; k < j; ++k) {
                oldCost += 1LL * distMatrix[current.order[k]][current.order[k + 1]]
                           * (100 + current.prefix[k]);
            }
            if (j + 1 < m) {
                oldCost += 1LL * distMatrix[current.order[j]][current.order[j + 1]]
                           * (100 + current.prefix[j]);
            }

            int load = current.prefix[i];
            for (int k = 0; k < (int)segment.size(); ++k) {
                int t = segment[k];
                int d;
                if (k == 0) {
                    d = (i == 0 ? startDist[t]
                                : distMatrix[current.order[i - 1]][t]);
                } else {
                    d = distMatrix[segment[k - 1]][t];
                }

                newCost += 1LL * d * (100 + load);
                load += tasks[t].delta;
                if (load < 0) {
                    newCost = (1LL << 60);
                    break;
                }
            }

            if (newCost == (1LL << 60)) continue;

            if (j + 1 < m) {
                newCost += 1LL * distMatrix[segment.back()][current.order[j + 1]]
                           * (100 + load);
            }

            long long delta = newCost - oldCost;
            bool accept = delta <= 0;
            if (!accept) {
                accept = uniform01(rng) <
                    exp(-double(delta) / max(0.0001, temperature));
            }
            if (!accept) continue;

            int newLoad = current.prefix[i];
            for (int k = 0; k < (int)segment.size(); ++k) {
                int pos = i + k;
                current.order[pos] = segment[k];
                newLoad += tasks[segment[k]].delta;
                current.prefix[pos] = newLoad;
            }
            current.cost += delta;

            if (current.cost < runBest.cost) {
                runBest = current;
                if (runBest.cost < globalBestCost) {
                    globalBestCost = runBest.cost;
                    globalBest = runBest.order;
                }
            }

            if ((iteration & 8191) == 8191 &&
                current.cost > runBest.cost + 5000) {
                current = runBest;
            }
        }

        if (runBest.cost < globalBestCost) {
            globalBestCost = runBest.cost;
            globalBest = runBest.order;
        }
        return {runBest.order, runBest.cost};
    }

    bool generatePartialPlan(
        double beta,
        int sourcePolicy,
        int automaticMode,
        bool verticalFirst,
        vector<Operation>& result,
        long long& resultCost
    ) const {
        vector<int> remaining(m);
        for (int i = 0; i < m; ++i) remaining[i] = tasks[i].delta;

        vector<Operation> operations;
        operations.reserve(20000);

        int x = 0, y = 0, load = 0;
        long long cost = 0;
        int remainingNonzero = m;

        auto nearestSinkDemand = [&](int source) {
            int best = INT_MAX;
            for (int t : sinkTasks) {
                if (remaining[t] >= 0) continue;
                int d = distMatrix[source][t];
                if (d < best) best = d;
            }
            if (best == INT_MAX) return INT_MAX;
            for (int t : sinkTasks) {
                if (remaining[t] < 0 &&
                    distMatrix[source][t] == best)
                    return -remaining[t];
            }
            return INT_MAX;
        };

        auto sourceAmount = [&](int t) {
            if (sourcePolicy == 0) return remaining[t];
            int demand = nearestSinkDemand(t);
            if (demand == INT_MAX) return remaining[t];
            return min(remaining[t], demand);
        };

        auto processCell = [&](bool forced) {
            int t = cellTask[x][y];
            if (t == -1 || remaining[t] == 0) return;

            if (remaining[t] > 0) {
                if (forced || (automaticMode & 1)) {
                    int amount = sourceAmount(t);
                    if (amount <= 0) return;
                    remaining[t] -= amount;
                    load += amount;
                    cost += amount;
                    operations.push_back({'+', amount});
                    --remainingNonzero;
                }
            } else {
                int amount = min(load, -remaining[t]);
                if (amount <= 0) return;
                if (forced || (automaticMode & 2)) {
                    remaining[t] += amount;
                    load -= amount;
                    cost += amount;
                    operations.push_back({'-', amount});
                    if (remaining[t] == 0) --remainingNonzero;
                }
            }
        };

        for (int iteration = 0; iteration < 2500 && remainingNonzero > 0; ++iteration) {
            int target = -1;
            double bestScore = 1e100;

            for (int t = 0; t < m; ++t) {
                if (remaining[t] == 0) continue;

                int amount, resultLoad;
                if (remaining[t] > 0) {
                    amount = sourceAmount(t);
                    if (amount <= 0) continue;
                    resultLoad = load + amount;
                } else {
                    if (load == 0) continue;
                    amount = min(load, -remaining[t]);
                    resultLoad = load - amount;
                }

                int d = abs(x - tasks[t].x) + abs(y - tasks[t].y);
                double score = d * (100.0 + load) + beta * resultLoad;
                if (target == -1 || score < bestScore ||
                    (score == bestScore && (rng() & 1))) {
                    bestScore = score;
                    target = t;
                }
            }

            if (target == -1) return false;

            int tx = tasks[target].x;
            int ty = tasks[target].y;

            auto moveVertical = [&]() {
                while (x != tx) {
                    if (x < tx) {
                        ++x;
                        operations.push_back({'D', 0});
                    } else {
                        --x;
                        operations.push_back({'U', 0});
                    }
                    cost += 100 + load;
                    processCell(false);
                }
            };

            auto moveHorizontal = [&]() {
                while (y != ty) {
                    if (y < ty) {
                        ++y;
                        operations.push_back({'R', 0});
                    } else {
                        --y;
                        operations.push_back({'L', 0});
                    }
                    cost += 100 + load;
                    processCell(false);
                }
            };

            if (verticalFirst) {
                moveVertical();
                moveHorizontal();
            } else {
                moveHorizontal();
                moveVertical();
            }
            processCell(true);

            if (operations.size() > 100000) return false;
        }

        if (remainingNonzero != 0 || load != 0) return false;

        result = std::move(operations);
        resultCost = cost;
        return true;
    }

    vector<Operation> makeFullPlan(const vector<int>& order) const {
        vector<Operation> operations;
        operations.reserve(order.size() * 8);

        int x = 0, y = 0;
        for (int t : order) {
            while (x < tasks[t].x) {
                ++x;
                operations.push_back({'D', 0});
            }
            while (x > tasks[t].x) {
                --x;
                operations.push_back({'U', 0});
            }
            while (y < tasks[t].y) {
                ++y;
                operations.push_back({'R', 0});
            }
            while (y > tasks[t].y) {
                --y;
                operations.push_back({'L', 0});
            }

            if (tasks[t].delta > 0)
                operations.push_back({'+', tasks[t].delta});
            else
                operations.push_back({'-', -tasks[t].delta});
        }
        return operations;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<vector<int>> height(N, vector<int>(N));
    for (int i = 0; i < N; ++i)
        for (int j = 0; j < N; ++j)
            cin >> height[i][j];

    Solver solver(N, std::move(height), 712367821ULL);

    if (solver.m == 0) return 0;

    vector<pair<vector<int>, long long>> candidates;

    auto addCandidate = [&](const vector<int>& order) {
        if ((int)order.size() != solver.m) return;
        long long cost = solver.routeCost(order);
        if (cost >= (1LL << 60)) return;
        candidates.push_back({order, cost});
    };

    const vector<double> betas = {0.0, 1.0, 2.0, 5.0, 10.0, 20.0};
    const vector<double> jitters = {0.0, 50.0};

    for (double beta : betas) {
        for (double jitter : jitters) {
            addCandidate(solver.greedyOrder(beta, jitter, false));
            addCandidate(solver.greedyOrder(beta, jitter, true));
        }
    }

    vector<int> allSources = solver.sourceTasks;
    vector<int> allSinks = solver.sinkTasks;
    vector<int> sourceFirst = solver.nearestOrder(allSources);
    vector<int> sinkPart = solver.nearestOrder(allSinks);
    sourceFirst.insert(sourceFirst.end(), sinkPart.begin(), sinkPart.end());
    addCandidate(sourceFirst);

    addCandidate(solver.repairFeasible(solver.snakeOrder(true)));
    addCandidate(solver.repairFeasible(solver.snakeOrder(false)));

    vector<int> beam = solver.beamOrder(80);
    addCandidate(beam);

    sort(candidates.begin(), candidates.end(),
         [](const auto& a, const auto& b) {
             return a.second < b.second;
         });

    for (int i = 0; i < min(4, (int)candidates.size()); ++i) {
        vector<int> extracted;
        for (int mode = 0; mode < 4; ++mode) {
            if (solver.realizeOrder(
                    candidates[i].first,
                    mode,
                    mode == 1 || mode == 3,
                    mode == 2 || mode == 3,
                    extracted)) {
                addCandidate(extracted);
            }
        }
    }

    sort(candidates.begin(), candidates.end(),
         [](const auto& a, const auto& b) {
             return a.second < b.second;
         });

    vector<int> globalBest = candidates.front().first;
    long long globalBestCost = candidates.front().second;

    vector<int> starts;
    for (int i = 0; i < min(4, (int)candidates.size()); ++i)
        starts.push_back(candidates[i].first);

    if (!beam.empty()) {
        bool exists = false;
        for (const auto& order : starts)
            if (order == beam) exists = true;
        if (!exists && starts.size() < 5) starts.push_back(beam);
    }

    for (const auto& order : starts) {
        solver.anneal(order, 80000, 1400.0, 2.0,
                      globalBest, globalBestCost);
    }

    solver.anneal(globalBest, 180000, 600.0, 1.0,
                  globalBest, globalBestCost);

    vector<int> extracted;
    for (int mode = 0; mode < 4; ++mode) {
        if (solver.realizeOrder(
                globalBest,
                mode,
                mode == 1 || mode == 3,
                mode == 2 || mode == 3,
                extracted)) {
            long long cost = solver.routeCost(extracted);
            if (cost < globalBestCost) {
                globalBestCost = cost;
                globalBest = extracted;
            }
        }
    }

    if (globalBestCost < (1LL << 60)) {
        solver.anneal(globalBest, 60000, 350.0, 1.0,
                      globalBest, globalBestCost);
    }

    vector<Operation> fullOperations = solver.makeFullPlan(globalBest);
    long long fullCost = globalBestCost + solver.baseCost;

    vector<Operation> partialOperations;
    long long partialCost = (1LL << 60);

    const vector<double> partialBetas = {0.0, 0.7, 1.5, 3.0, 6.0, 10.0};

    for (double beta : partialBetas) {
        for (int policy = 0; policy < 2; ++policy) {
            for (int automatic = 0; automatic < 4; ++automatic) {
                vector<Operation> operations;
                long long cost;
                if (solver.generatePartialPlan(
                        beta, policy, automatic, (rng() & 1) != 0,
                        operations, cost) &&
                    cost < partialCost) {
                    partialCost = cost;
                    partialOperations = std::move(operations);
                }
            }
        }
    }

    const vector<Operation>* answer = &fullOperations;
    if (partialCost < fullCost) answer = &partialOperations;

    string output;
    output.reserve(answer->size() * 10);

    for (const Operation& op : *answer) {
        if (op.type == '+' || op.type == '-') {
            output.push_back(op.type);
            output += to_string(op.amount);
        } else {
            output.push_back(op.type);
        }
        output.push_back('\n');
    }

    cout << output;
    return 0;
}
