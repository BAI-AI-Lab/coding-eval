#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

struct Clue {
    int r, c, x; // x = 0 for '+', 1 for '-'
};

static int parity(int64 x) {
    int p = int(x % 2);
    return p < 0 ? p + 2 : p;
}

static int64 floorDiv2(int64 x) {
    if (x >= 0) return x / 2;
    return -((-x + 1) / 2);
}

static int64 ceilDiv2(int64 x) {
    return floorDiv2(x + 1);
}

// Number of integers in [lo, hi] with the requested parity.
static int64 countWithParity(int64 lo, int64 hi, int p) {
    if (lo > hi) return 0;
    int64 first = lo + (p - parity(lo) + 2) % 2;
    if (first > hi) return 0;
    return (hi - first) / 2 + 1;
}

// Count L in [l0,l1], H in [h0,h1], both of parity p,
// subject to H-L >= separation.
static int64 countPairs(int64 l0, int64 l1,
                        int64 h0, int64 h1,
                        int p, int64 separation) {
    if (l0 > l1 || h0 > h1) return 0;

    int64 il = ceilDiv2(l0 - p);
    int64 ir = floorDiv2(l1 - p);
    int64 jl = ceilDiv2(h0 - p);
    int64 jr = floorDiv2(h1 - p);
    if (il > ir || jl > jr) return 0;

    // L = p+2i, H = p+2j. We need j-i >= need.
    int64 need = (separation + 1) / 2;
    int64 result = 0;

    // i+need <= jl: every permissible j works.
    int64 fullEnd = min(ir, jl - need);
    if (il <= fullEnd) {
        int64 n = fullEnd - il + 1;
        result += n * (jr - jl + 1);
    }

    // jl < i+need <= jr: the number of j values decreases linearly.
    int64 st = max(il, jl - need + 1);
    int64 en = min(ir, jr - need);
    if (st <= en) {
        int64 n = en - st + 1;
        int64 A = jr - need + 1;
        result += A * n - (st + en) * n / 2;
    }

    return result;
}

static int checkerboardCount(const vector<Clue>& clues) {
    int answer = 0;
    for (int base = 0; base < 2; ++base) {
        bool ok = true;
        for (const auto& cl : clues) {
            int expected = base ^ parity(cl.r + cl.c);
            if (expected != cl.x) {
                ok = false;
                break;
            }
        }
        answer += ok;
    }
    return answer;
}

// orient = 0: q = r+c; orient = 1: q = r-c.
static int64 singleWallCount(int R, int C, int orient,
                             const vector<Clue>& clues) {
    int64 qmin = orient == 0 ? 2 : 1 - (int64)C;
    int64 qmax = orient == 0 ? R + (int64)C : R - 1;

    int64 answer = 0;
    for (int b = 0; b < 2; ++b) {
        int base = 1 - b;
        int64 lo = qmin, hi = qmax - 1;

        for (const auto& cl : clues) {
            int64 q = orient == 0 ? cl.r + (int64)cl.c
                                  : cl.r - (int64)cl.c;
            int requiredZ = cl.x ^ parity(q) ^ base;
            if (requiredZ) {
                lo = max(lo, q);       // q <= K
            } else {
                hi = min(hi, q - 1);   // q > K
            }
        }
        answer += countWithParity(lo, hi, b);
    }
    return answer;
}

static int64 doubleWallCount(int R, int C, int orient,
                             const vector<Clue>& clues) {
    int64 qmin = orient == 0 ? 2 : 1 - (int64)C;
    int64 qmax = orient == 0 ? R + (int64)C : R - 1;
    int64 separation = max(R, C);

    int64 answer = 0;

    for (int b = 0; b < 2; ++b) {
        int base = 1 - b;
        vector<int64> ones;
        ones.reserve(clues.size());

        bool hasZero = false;
        int64 minZero = 0, maxZero = 0;

        for (const auto& cl : clues) {
            int64 q = orient == 0 ? cl.r + (int64)cl.c
                                  : cl.r - (int64)cl.c;
            int requiredZ = cl.x ^ parity(q) ^ base;
            if (requiredZ) {
                ones.push_back(q);
            } else {
                if (!hasZero) {
                    hasZero = true;
                    minZero = maxZero = q;
                } else {
                    minZero = min(minZero, q);
                    maxZero = max(maxZero, q);
                }
            }
        }

        sort(ones.begin(), ones.end());

        if (hasZero) {
            // All zeroes lie strictly between L and H. No one can lie in
            // their bounding interval; lower and upper ones force bounds.
            int64 l0 = qmin, l1 = minZero - 1;
            int64 h0 = maxZero + 1, h1 = qmax;

            auto it = lower_bound(ones.begin(), ones.end(), minZero);
            if (it != ones.begin()) {
                l0 = max(l0, *prev(it));
            }

            bool bad = (it != ones.end() && *it <= maxZero);
            if (bad) continue;

            auto jt = upper_bound(ones.begin(), ones.end(), maxZero);
            if (jt != ones.end()) {
                h1 = min(h1, *jt);
            }

            answer += countPairs(l0, l1, h0, h1, b, separation);
        } else {
            // The open interval (L,H) must contain no coordinate of a one.
            // Thus it lies in one of the gaps between consecutive such
            // coordinates. Different gaps count disjoint sets of pairs.
            ones.erase(unique(ones.begin(), ones.end()), ones.end());
            int k = int(ones.size());

            for (int t = 0; t <= k; ++t) {
                int64 l0 = qmin, l1 = qmax;
                int64 h0 = qmin, h1 = qmax;

                if (t > 0) {
                    int64 a = ones[t - 1];
                    l0 = max(l0, a);
                    h0 = max(h0, a + 1);
                }
                if (t < k) {
                    int64 bnd = ones[t];
                    l1 = min(l1, bnd - 1);
                    h1 = min(h1, bnd);
                }

                answer += countPairs(l0, l1, h0, h1, b, separation);
            }
        }
    }

    return answer;
}

static bool validLine(const vector<int>& a) {
    int plusRuns = 0;
    for (int i = 0; i < int(a.size()); ++i) {
        if (i > 0 && a[i - 1] == 1 && a[i] == 1) return false;
        if (i > 0 && a[i - 1] == 0 && a[i] == 0) ++plusRuns;
    }
    return plusRuns <= 1;
}

static int solveTwoByTwo(const vector<Clue>& clues) {
    int answer = 0;
    for (int mask = 0; mask < 16; ++mask) {
        vector<int> x(4);
        for (int i = 0; i < 4; ++i) x[i] = (mask >> i) & 1;

        bool ok = true;
        for (const auto& cl : clues) {
            int idx = (cl.r - 1) * 2 + (cl.c - 1);
            if (x[idx] != cl.x) {
                ok = false;
                break;
            }
        }
        if (!ok) continue;

        ok &= validLine({x[0], x[1]});
        ok &= validLine({x[2], x[3]});
        ok &= validLine({x[0], x[2]});
        ok &= validLine({x[1], x[3]});
        answer += ok;
    }
    return answer;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    while (T--) {
        int R, C, X;
        cin >> R >> C >> X;

        vector<Clue> clues(X);
        for (auto& cl : clues) {
            char ch;
            cin >> ch >> cl.r >> cl.c;
            cl.x = (ch == '-');
        }

        if (R == 2 && C == 2) {
            cout << solveTwoByTwo(clues) << '\n';
            continue;
        }

        int64 answer = checkerboardCount(clues);

        if (min(R, C) == 1) {
            // The two diagonal orientations coincide on a one-dimensional
            // grid, so only one is counted.
            answer += singleWallCount(R, C, 0, clues);
        } else {
            for (int orient = 0; orient < 2; ++orient) {
                answer += singleWallCount(R, C, orient, clues);
                answer += doubleWallCount(R, C, orient, clues);
            }
        }

        cout << answer << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

struct Clue {
    int r, c, p;
};

struct FamilyData {
    bool hasO[2] = {false, false};
    bool hasI[2] = {false, false};
    int64 maxO[2] = {0, 0};
    int64 minI[2] = {0, 0};
    int64 maxI[2] = {0, 0};
    vector<int64> valuesO[2];
};

static int parity(int64 x) {
    x %= 2;
    if (x < 0) x += 2;
    return static_cast<int>(x);
}

static int64 countWithParity(int64 lo, int64 hi, int p) {
    if (lo > hi) return 0;
    if (parity(lo) != p) ++lo;
    if (lo > hi) return 0;
    return (hi - lo) / 2 + 1;
}

static FamilyData buildData(const vector<Clue>& clues, bool useSum) {
    FamilyData d;

    for (const Clue& clue : clues) {
        int64 t = useSum ? int64(clue.r) + clue.c
                         : int64(clue.c) - clue.r;
        int q = parity(t + clue.p);

        int o = q;
        if (!d.hasO[o]) {
            d.hasO[o] = true;
            d.maxO[o] = t;
        } else {
            d.maxO[o] = max(d.maxO[o], t);
        }
        d.valuesO[o].push_back(t);

        int in = 1 - q;
        if (!d.hasI[in]) {
            d.hasI[in] = true;
            d.minI[in] = d.maxI[in] = t;
        } else {
            d.minI[in] = min(d.minI[in], t);
            d.maxI[in] = max(d.maxI[in], t);
        }
    }

    for (int p = 0; p < 2; ++p) {
        sort(d.valuesO[p].begin(), d.valuesO[p].end());
    }
    return d;
}

// One-transition grids: parameter k lies in [A, B-1].
static int64 countHalfplanes(int64 A, int64 B, const FamilyData& d) {
    int64 result = 0;

    for (int p = 0; p < 2; ++p) {
        int64 lo = A;
        int64 hi = B - 1;

        if (d.hasO[p]) lo = max(lo, d.maxO[p]);
        if (d.hasI[p]) hi = min(hi, d.minI[p] - 1);

        result += countWithParity(lo, hi, p);
    }
    return result;
}

// Two-transition grids: parameters l,u satisfy A <= l < u <= B,
// u-l >= M, and l,u have the same parity.
static int64 countBands(int64 A, int64 B, int64 M, const FamilyData& d) {
    int64 result = 0;
    int64 lastL = B - M;
    if (lastL < A) return 0;

    for (int p = 0; p < 2; ++p) {
        int64 hiL = lastL;
        if (d.hasI[p]) hiL = min(hiL, d.minI[p] - 1);

        int64 l = A;
        if (parity(l) != p) ++l;

        const vector<int64>& v = d.valuesO[p];
        size_t ptr = 0;

        for (; l <= hiL; l += 2) {
            while (ptr < v.size() && v[ptr] <= l) ++ptr;

            int64 loU = l + M;
            if (d.hasI[p]) loU = max(loU, d.maxI[p] + 1);

            int64 hiU = B;
            if (ptr < v.size()) hiU = min(hiU, v[ptr]);

            result += countWithParity(loU, hiU, p);
        }
    }
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int R, C, X;
        cin >> R >> C >> X;

        vector<Clue> clues(X);
        for (Clue& clue : clues) {
            char v;
            cin >> v >> clue.r >> clue.c;
            clue.p = (v == '-');
        }

        int pureCount = 0;
        for (int p = 0; p < 2; ++p) {
            bool ok = true;
            for (const Clue& clue : clues) {
                if (parity(int64(p) + clue.r + clue.c) != clue.p) {
                    ok = false;
                    break;
                }
            }
            pureCount += ok;
        }

        FamilyData differenceData = buildData(clues, false);
        FamilyData sumData = buildData(clues, true);

        const int64 diffA = 1LL - R;
        const int64 diffB = int64(C) - 1;
        const int64 sumA = 2;
        const int64 sumB = int64(R) + C;
        const int64 M = max(R, C);

        int64 answer = pureCount;
        answer += countHalfplanes(diffA, diffB, differenceData);

        if (min(R, C) > 1) {
            answer += countHalfplanes(sumA, sumB, sumData);

            // In the 2x2 case these two bands describe the same all-plus grid.
            if (!(R == 2 && C == 2)) {
                answer += countBands(diffA, diffB, M, differenceData);
            }
            answer += countBands(sumA, sumB, M, sumData);
        }

        cout << answer << '\n';
    }

    return 0;
}

using int64 = long long;

struct Memory {
    int r, c;
    int value;
    int parity;
    array<int, 4> dist;
};

static int64 solveOneDimensional(int n, const vector<Memory>& memories, bool byRow) {
    vector<int> fixed(n + 1, 0);
    for (const auto& m : memories) {
        int pos = byRow ? m.r : m.c;
        fixed[pos] = m.value;
    }

    // State: last value, and whether the unique possible "++" has been used.
    int64 dp[2][2] = {};
    auto index = [](int x) { return x == 1 ? 1 : 0; };

    for (int v : {-1, 1}) {
        if (fixed[1] == 0 || fixed[1] == v) {
            dp[index(v)][0] = 1;
        }
    }

    for (int pos = 2; pos <= n; ++pos) {
        int64 ndp[2][2] = {};
        for (int last = 0; last < 2; ++last) {
            int lastValue = last ? 1 : -1;
            for (int used = 0; used < 2; ++used) {
                int64 ways = dp[last][used];
                if (!ways) continue;
                for (int v : {-1, 1}) {
                    if (fixed[pos] != 0 && fixed[pos] != v) continue;
                    if (v != lastValue) {
                        ndp[index(v)][used] += ways;
                    } else if (v == 1 && used == 0) {
                        ndp[1][1] += ways;
                    }
                }
            }
        }
        memcpy(dp, ndp, sizeof(dp));
    }

    int64 answer = 0;
    for (int last = 0; last < 2; ++last) {
        for (int used = 0; used < 2; ++used) {
            answer += dp[last][used];
        }
    }
    return answer;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int64 R, C;
        int X;
        cin >> R >> C >> X;

        vector<Memory> memories(X);
        for (int i = 0; i < X; ++i) {
            char ch;
            int r, c;
            cin >> ch >> r >> c;
            memories[i].r = r;
            memories[i].c = c;
            memories[i].value = (ch == '+' ? 1 : -1);
            memories[i].parity = ((r + c) % 2 == 0 ? 1 : -1);
        }

        if (R == 1 || C == 1) {
            int n = (int)max(R, C);
            bool byRow = (C == 1);
            cout << solveOneDimensional(n, memories, byRow) << '\n';
            continue;
        }

        const int64 cells = R * C;

        // A[k] = number of cells at Manhattan distance < k from a corner.
        // Only areas at most cells/2 are needed for the canonical representation.
        vector<int64> area(1, 0);
        const int maxPossibleK = (int)(R + C - 1);
        for (int k = 1; k <= maxPossibleK; ++k) {
            int d = k - 1;
            int64 lo = max<int64>(0, (int64)d - (C - 1));
            int64 hi = min<int64>(R - 1, d);
            int64 diagonal = (hi >= lo ? hi - lo + 1 : 0);
            int64 nextArea = area.back() + diagonal;
            if (2 * nextArea > cells) break;
            area.push_back(nextArea);
        }

        const int Kmax = (int)area.size() - 1;

        // Corner order: top-left, top-right, bottom-left, bottom-right.
        vector<vector<pair<int, int>>> ordered(4);
        for (int i = 0; i < X; ++i) {
            auto& m = memories[i];
            int64 r = m.r, c = m.c;
            m.dist[0] = (int)((r - 1) + (c - 1));
            m.dist[1] = (int)((r - 1) + (C - c));
            m.dist[2] = (int)((R - r) + (c - 1));
            m.dist[3] = (int)((R - r) + (C - c));

            for (int corner = 0; corner < 4; ++corner) {
                if (m.dist[corner] < Kmax) {
                    ordered[corner].push_back({m.dist[corner], i});
                }
            }
        }
        for (int corner = 0; corner < 4; ++corner) {
            sort(ordered[corner].begin(), ordered[corner].end());
        }

        // A triangle of size k at a corner requires a particular checkerboard phase.
        int factor[4];
        factor[0] = 1;
        factor[1] = (C % 2 == 1 ? 1 : -1);
        factor[2] = (R % 2 == 1 ? 1 : -1);
        factor[3] = ((R + C) % 2 == 0 ? 1 : -1);

        int64 answer = 0;
        const int minDim = (int)min(R, C);
        const int doubleLimit = min(Kmax, minDim - 1);

        for (int base : {1, -1}) {
            int mismatch = 0;
            vector<int> weight(X);
            for (int i = 0; i < X; ++i) {
                int baseValue = base * memories[i].parity;
                if (baseValue != memories[i].value) {
                    ++mismatch;
                    weight[i] = -1;
                } else {
                    weight[i] = 1;
                }
            }

            // weights[corner][k]: total adjustment caused by flipped remembered cells.
            vector<vector<int>> weights(4, vector<int>(Kmax + 1, 0));
            for (int corner = 0; corner < 4; ++corner) {
                int ptr = 0;
                const auto& v = ordered[corner];
                for (int k = 1; k <= Kmax; ++k) {
                    int sum = weights[corner][k - 1];
                    while (ptr < (int)v.size() && v[ptr].first < k) {
                        sum += weight[v[ptr].second];
                        ++ptr;
                    }
                    weights[corner][k] = sum;
                }
            }

            // Unmodified checkerboard.
            if (mismatch == 0) ++answer;

            // One flipped corner triangle.
            for (int corner = 0; corner < 4; ++corner) {
                for (int k = 1; k <= Kmax; ++k) {
                    int sizeSign = (k % 2 == 0 ? 1 : -1);
                    if (factor[corner] * sizeSign != base) continue;
                    if (2 * area[k] == cells && base == -1) continue;
                    if (mismatch + weights[corner][k] == 0) {
                        ++answer;
                    }
                }
            }

            // Two triangles at opposite corners. Their sizes must have disjoint supports.
            const pair<int, int> oppositePairs[2] = {{0, 3}, {1, 2}};
            for (auto [first, second] : oppositePairs) {
                if (doubleLimit < 1) continue;

                int firstParity = (base * factor[first] == -1 ? 1 : 0);
                int secondParity = (base * factor[second] == -1 ? 1 : 0);

                unordered_map<int, vector<int>> positions;
                positions.reserve((size_t)min<int64>(doubleLimit + 1, 2LL * X + 16));

                int secondStart = (secondParity == 1 ? 1 : 2);
                for (int l = secondStart; l <= doubleLimit; l += 2) {
                    positions[weights[second][l]].push_back(l);
                }

                int firstStart = (firstParity == 1 ? 1 : 2);
                for (int k = firstStart; k <= doubleLimit; k += 2) {
                    int maxL = min(doubleLimit, minDim - k);
                    if (maxL < 1) continue;

                    // Binary search the largest size satisfying the canonical area rule.
                    int lo = 1, hi = maxL, best = 0;
                    while (lo <= hi) {
                        int mid = (lo + hi) / 2;
                        int64 twiceArea = 2 * (area[k] + area[mid]);
                        bool ok = (base == 1 ? twiceArea <= cells : twiceArea < cells);
                        if (ok) {
                            best = mid;
                            lo = mid + 1;
                        } else {
                            hi = mid - 1;
                        }
                    }
                    if (best == 0) continue;

                    int wanted = -mismatch - weights[first][k];
                    auto it = positions.find(wanted);
                    if (it == positions.end()) continue;

                    const auto& vec = it->second;
                    answer += (int64)(upper_bound(vec.begin(), vec.end(), best) - vec.begin());
                }
            }
        }

// subject to H-L >= separation.
static int64 countPairs(int64 l0, int64 l1,
                        int64 h0, int64 h1,
                        int p, int64 separation) {

    int64 il = ceilDiv2(l0 - p);
    int64 ir = floorDiv2(l1 - p);
    int64 jl = ceilDiv2(h0 - p);
    int64 jr = floorDiv2(h1 - p);

    int64 need = (separation + 1) / 2;

        result += n * (jr - jl + 1);

/*
For each parameter value, records whether it is invalid for at least one
constraint. Parameter parities are stored separately.
*/
struct ParityBad {
    int n;
    vector<int> diff[2];

    explicit ParityBad(int n = 0) : n(n), diff[2](vector<int>(n + 3, 0)) {}

    void add(int l, int r, int parity) {
        l = max(l, 1);
        r = min(r, n);
        if (l > r) return;

        if ((l & 1) != parity) ++l;
        if ((r & 1) != parity) --r;
        if (l > r) return;

        ++diff[parity][l];
        --diff[parity][r + 2];
    }

    vector<int> build() const {
        vector<int> bad(n + 1, 0);
        int count[2] = {0, 0};

        for (int i = 1; i <= n; ++i) {
            count[i & 1] += diff[i & 1][i];
            bad[i] = count[0] + count[1];
        }
        return bad;
    }
};

static int parityOf(int x) {
    return (x % 2 + 2) % 2;
}

/*
For a defective row H_h, the cell in column c is '+' exactly when

  h and c have the same parity and h >= c, or
  h and c have different parity and h < c.

This routine marks parameter values for which h = coef * parameter + constant
gives the wrong remembered sign.
*/
static void markAffineWrong(
    ParityBad &bad,
    int coef,
    int constant,
    int c,
    bool wantedPlus,
    int L,
    int R
) {
    if (L > R) return;

    int constantParity = parityOf(constant);

    for (int p = 0; p < 2; ++p) {
        int hParity = (p + constantParity) & 1;
        bool sameParity = (hParity == (c & 1));

        // Actual sign is '+' iff:
        //   same parity: h >= c
        //   other parity: h < c
        bool wrongMeansAtLeastC;
        if (wantedPlus) {
            wrongMeansAtLeastC = !sameParity;
        } else {
            wrongMeansAtLeastC = sameParity;
        }

        if (wrongMeansAtLeastC) {
            // h >= c
            if (coef == 1) {
                int low = c - constant;
                bad.add(max(L, low), R, p);
            } else {
                int high = constant - c;
                bad.add(L, min(R, high), p);
            }
        } else {
            // h < c
            if (coef == 1) {
                int high = c - constant - 1;
                bad.add(L, min(R, high), p);
            } else {
                int low = constant - c + 1;
                bad.add(max(L, low), R, p);
            }
        }
    }
}

        long long answer = 0;
        for (int a = 1; a <= K; ++a) {
            answer += (badValues[a] == 0);
        }
        return answer;
    }

    // A defective block touching the bottom boundary, preceded by pure rows.
    long long countBottomSingle(bool startsAtN) const {
        int K = min(n, R - 1);
        if (K <= 0) return 0;

        ParityBad bad(K);

        for (const Cell &cell : cells) {
            int r = cell.r;
            int firstBottomRow = R - r + 1;

            if (startsAtN) {
                // h = n - b - r + R + 1
                markAffineWrong(
                    bad, -1, n - r + R + 1, cell.c, cell.plus,
                    firstBottomRow, K
                );
            } else {
                // h = b + r - R
                markAffineWrong(
                    bad, 1, r - R, cell.c, cell.plus,
                    firstBottomRow, K
                );
            }

            // If b < R-r+1, this cell is in the pure prefix.
            int pureRight = min(K, firstBottomRow - 1);
            if (pureRight >= 1) {
                int baseType = bottomBaseType(startsAtN);
                for (int p = 0; p < 2; ++p) {
                    int distanceParity = (R & 1) ^ p ^ (r & 1);
                    bool expected = patternPlus(baseType, cell.c, distanceParity);
                    if (expected != cell.plus) {
                        bad.add(1, pureRight, p);
                    }
                }
            }
        }

        vector<int> badValues = bad.build();
        long long answer = 0;
        for (int b = 1; b <= K; ++b) {
            answer += (badValues[b] == 0);
        }
        return answer;
    }

    /*
    Two boundary blocks:
      top ends at H_1 and bottom starts at H_n, or
      top ends at H_n and bottom starts at H_1.
    */
    long long countTwoBoundaryBlocks(bool topEndsAtN) const {
        bool bottomStartsAtN = !topEndsAtN;

        ParityBad topBad(n), bottomBad(n);

        for (const Cell &cell : cells) {
            int r = cell.r;

            if (topEndsAtN) {
                markAffineWrong(topBad, -1, n + r, cell.c, cell.plus, r, n);
            } else {
                markAffineWrong(topBad, 1, 1 - r, cell.c, cell.plus, r, n);
            }

            int firstBottomRow = R - r + 1;
            if (bottomStartsAtN) {
                markAffineWrong(
                    bottomBad, -1, n - r + R + 1, cell.c, cell.plus,
                    firstBottomRow, n
                );
            } else {
                markBadBottomStart:
                markAffineWrong(
                    bottomBad, 1, r - R, cell.c, cell.plus,
                    firstBottomRow, n
                );
            }
        }

        vector<int> topBadValues = topBad.build();
        vector<int> bottomBadValues = bottomBad.build();

        // Prefix counts of valid bottom lengths, separated by parity.
        vector<array<int, 2>> pref(n + 1);
        pref[0] = {0, 0};
        for (int b = 1; b <= n; ++b) {
            pref[b] = pref[b - 1];
            if (bottomBadValues[b] == 0) ++pref[b][b & 1];
        }

        /*
        For each parity of a, find the first remembered row in the pure middle
        whose sign would be wrong there.
        */
        const int INF = R + 1;
        vector<vector<int>> firstWrong(2, vector<int>(R + 2, INF));

        for (const Cell &cell : cells) {
            int baseType = topBaseType(topEndsAtN);
            for (int p = 0; p < 2; ++p) {
                int distanceParity = (cell.r & 1) ^ p ^ 1;
                bool expected = patternPlus(baseType, cell.c, distanceParity);
                if (expected != cell.plus) {
                    firstWrong[p][cell.r] = cell.r;
                }
            }
        }

        for (int p = 0; p < 2; ++p) {
            for (int r = R; r >= 1; --r) {
                firstWrong[p][r] = min(firstWrong[p][r], firstWrong[p][r + 1]);
            }
        }

        int maxSum = min(R - 1, C);
        long long answer = 0;

        for (int a = 1; a <= n; ++a) {
            if (topBadValues[a] != 0) continue;

            int high = min(n, maxSum - a);
            if (high < 1) continue;

            int low = 1;
            int wrongRow = firstWrong[a & 1][a + 1];
            if (wrongRow != INF) {
                low = max(low, R - wrongRow + 1);
            }
            if (low > high) continue;

            // R-a-b must have the same parity as C.
            int requiredParity = (R & 1) ^ (C & 1) ^ (a & 1);
            answer += pref[high][requiredParity] - pref[low - 1][requiredParity];
        }

        return answer;
    }

    // An interior defective block spanning H_1, ..., H_n.
    long long countInterior(bool increasing) const {
        int maxL = R - n;
        if (maxL < 2) return 0;

        ParityBad bad(maxL);

        for (const Cell &cell : cells) {
            int r = cell.r;

            // Cell lies in the defective block.
            int blockLeft = max(2, r - n + 1);
            int blockRight = min(maxL, r);
            if (increasing) {
                // h = -l + r + 1
                markAffineWrong(
                    bad, -1, r + 1, cell.c, cell.plus,
                    blockLeft, blockRight
                );
            } else {
                // h = l + n - r
                markAffineWrong(
                    bad, 1, n - r, cell.c, cell.plus,
                    blockLeft, blockRight
                );
            }

            // Cell lies in the pure prefix.
            int leftLeft = r + 1;
            int leftRight = maxL;
            if (leftLeft <= leftRight) {
                int baseType = increasing ? 0 : ((C & 1) ? 0 : 1);
                for (int p = 0; p < 2; ++p) {
                    int distanceParity = p ^ 1 ^ (r & 1);
                    bool expected = patternPlus(baseType, cell.c, distanceParity);
                    if (expected != cell.plus) {
                        bad.add(leftLeft, leftRight, p);
                    }
                }
            }

            // Cell lies in the pure suffix.
            int rightLeft = 2;
            int rightRight = min(maxL, r - n);
            if (rightLeft <= rightRight) {
                int baseType = increasing ? ((C & 1) ? 0 : 1) : 0;
                for (int p = 0; p < 2; ++p) {
                    int distanceParity = (r & 1) ^ p ^ (n & 1);
                    bool expected = patternPlus(baseType, cell.c, distanceParity);
                    if (expected != cell.plus) {
                        bad.add(rightLeft, rightRight, p);
                    }
                }
            }
        }

        vector<int> badValues = bad.build();
        long long answer = 0;
        for (int l = 2; l <= maxL; ++l) {
            answer += (badValues[l] == 0);
        }
        return answer;
    }

    // All rows are defective, with a monotone sequence of defect positions.
    long long countAllDefectiveMonotone(bool increasing) const {
        int low, high;
        if (increasing) {
            low = 1;
            high = n - R + 1;
        } else {
            low = R;
            high = n;
        }
        if (low > high) return 0;

        ParityBad bad(high - low + 1);

        for (const Cell &cell : cells) {
            if (increasing) {
                // h = a + r - 1
                markAffineWrong(
                    bad, 1, cell.r - 1, cell.c, cell.plus,
                    low, high
                );
            } else {
                // h = a - r + 1
                markAffineWrong(
                    bad, 1, 1 - cell.r, cell.c, cell.plus,
                    low, high
                );
            }
        }

        vector<int> badValues = bad.build();
        long long answer = 0;
        for (int a = low; a <= high; ++a) {
            answer += (badValues[a - low] == 0);
        }
        return answer;
    }

    /*
    All rows are defective and the special jump H_1 <-> H_n is used.
    C must be even.
    */
    long long countSpecialAllDefective(bool jumpOneToN) const {
        int low, high;

        if (jumpOneToN) {
            low = max(1, R - n);
            high = min(n, R - 1);
        } else {
            low = max(1, n + 2 - R);
            high = min(n, 2 * n + 1 - R);
        }
        if (low > high) return 0;

        ParityBad bad(high - low + 1);

        for (const Cell &cell : cells) {
            int r = cell.r;

            if (jumpOneToN) {
                // Before the jump: h = k-r+1.
                markAffineWrong(
                    bad, 1, 1 - r, cell.c, cell.plus,
                    max(low, r), high
                );

                // The row containing H_n.
                int k = r - 1;
                if (k >= low && k <= high && !hCellIsPlus(n, cell.c)) {
                    bad.add(k - low, k - low, k & 1);
                }

                // After the jump: h = n+k+1-r.
                markAffineWrong(
                    bad, 1, n + 1 - r, cell.c, cell.plus,
                    low, min(high, r - 2)
                );
            } else {
                // Before the jump: h = a+r-1.
                markAffineWrong(
                    bad, 1, r - 1, cell.c, cell.plus,
                    low, min(high, n - r + 1)
                );

                // The row containing H_1.
                int a = n - r;
                if (a >= low && a <= high && !hCellIsPlus(1, cell.c)) {
                    bad.add(a - low, a - low, a & 1);
                }

                // After the jump: h = a+r-n-1.
                markAffineWrong(
                    bad, 1, r - n - 1, cell.c, cell.plus,
                    low, min(high, n - r - 1)
                );
            }
        }

        vector<int> badValues = bad.build();
        long long answer = 0;
        for (int p = low; p <= high; ++p) {
            answer += (badValues[p - low] == 0);
        }
        return answer;
    }

public:
    GridCounter(int R, int C, vector<Cell> cells)
        : R(R), C(C), n(C - 1), cells(std::move(cells)) {}

    long long solve() const {
        long long answer = 0;

        answer += countCheckerboard(0);
        answer += countCheckerboard(1);

        answer += countTopSingle(false);
        answer += countTopSingle(true);
        answer += countBottomSingle(false);
        answer += countBottomSingle(true);

        answer += countTwoBoundaryBlocks(false);
        answer += countTwoBoundaryBlocks(true);

        answer += countInterior(true);
        answer += countInterior(false);

        answer += countAllDefectiveMonotone(true);
        answer += countAllDefectiveMonotone(false);

        if ((C & 1) == 0) {
            answer += countSpecialAllDefective(true);
            answer += countSpecialAllDefective(false);
        }

        return answer;
    }
};

#include <bits/stdc++.h>
#include <boost/multiprecision/cpp_int.hpp>
using namespace std;
using boost::multiprecision::cpp_int;

/*
    A row of length C is valid exactly when:
      - it contains no "--";
      - it contains at most one "++".

    There are C+1 possible valid rows. Number them from 0 through C:
      - state 0 is the alternating row beginning with '+';
      - state C is the alternating row beginning with '-';
      - state p, 1 <= p < C, is the alternating row whose only equal
        adjacent pair is "++" between positions p and p+1.

    For two consecutive rows, their equality positions must all contain '+'.
    Moreover, each column may contain at most one such vertical "++".
    Consequently, walking through the row states uses labeled edges, with
    each label used at most once.

    The compatibility graph of row states is a theta graph:
      - most edges have one label;
      - one edge has no label and may be used repeatedly;
      - at most one additional edge has two labels.

    The memoized search below stores the used labels as a union of intervals.
*/

struct Edge {
    int to;
    vector<int> labels;
};

struct Interval {
    int l, r;

    bool operator<(const Interval& other) const {
        return tie(l, r) < tie(other.l, other.r);
    }
};

static bool contains(const vector<Interval>& used, int x) {
    for (const auto& in : used) {
        if (in.l <= x && x <= in.r) return true;
    }
    return false;
}

static vector<Interval> addLabels(const vector<Interval>& used,
                                  const vector<int>& labels) {
    vector<Interval> result = used;

    for (int x : labels) {
        bool inserted = false;

        for (auto& in : result) {
            if (in.l <= x && x <= in.r) {
                inserted = true;
                break;
            }
        }
        if (inserted) continue;

        result.push_back({x, x});
        sort(result.begin(), result.end());

        vector<Interval> merged;
        for (const auto& in : result) {
            if (!merged.empty() && in.l <= merged.back().r + 1) {
                merged.back().r = max(merged.back().r, in.r);
            } else {
                merged.push_back(in);
            }
        }
        result.swap(merged);
    }

    return result;
}

static bool cellIsPlus(int state, int col, int C) {
    if (state == 0) {
        return (col & 1) != 0;
    }
    if (state == C) {
        return (col & 1) == 0;
    }

    if (col <= state) {
        return ((col + state) & 1) == 0;
    } else {
        return ((col + state) & 1) != 0;
    }
}

static string makeKey(int row, int state, const vector<Interval>& used) {
    string key;
    key.reserve(24 + 16 * used.size());

    auto appendInt = [&](int x) {
        for (int i = 0; i < 4; ++i) {
            key.push_back(static_cast<char>((x >> (8 * i)) & 255));
        }
    };

    appendInt(row);
    appendInt(state);
    appendInt(static_cast<int>(used.size()));
    for (const auto& in : used) {
        appendInt(in.l);
        appendInt(in.r);
    }
    return key;
}

    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int R, C, X;
        cin >> R >> C >> X;

        vector<tuple<char, int, int>> clues(X);
        for (auto& [ch, r, c] : clues) {
            cin >> ch >> r >> c;
            --r;
            --c;
        }

        // Use the smaller dimension as the number of rows.
        if (R > C) {
            swap(R, C);
            for (auto& [ch, r, c] : clues) {
                swap(r, c);
            }
        }

        vector<vector<pair<int, int>>> rowClues(R);
        for (const auto& [ch, r, c] : clues) {
            rowClues[r].push_back({c, ch == '+'});
        }

        vector<vector<Edge>> graph(C + 1);

        auto addEdge = [&](int u, int v, vector<int> labels) {
            graph[u].push_back({v, labels});
            graph[v].push_back({u, move(labels)});
        };

        if (C == 1) {
            addEdge(0, 1, {});
            addEdge(0, 0, {1});
        } else if (C == 2) {
            addEdge(0, 1, {1});
            addEdge(1, 2, {2});
            addEdge(0, 2, {});
            addEdge(1, 1, {1, 2});
        } else if (C & 1) {
            for (int i = 0; i + 1 < C; ++i) {
                addEdge(i, i + 1, {i + 1});
            }
            addEdge(0, C - 1, {C});
            addEdge(0, C, {});
        } else {
            for (int i = 0; i + 1 < C; ++i) {
                addEdge(i, i + 1, {i + 1});
            }
            addEdge(C - 1, C, {C});
            addEdge(0, C, {});
            addEdge(1, C - 1, {1, C});
        }

        unordered_map<string, cpp_int> memo;
        memo.reserve(1 << 18);
        memo.max_load_factor(0.7f);

        auto dfs = [&](auto&& self, int row, int state,
                       const vector<Interval>& used) -> cpp_int {
            for (const auto& [col, wantPlus] : rowClues[row]) {
                if (cellIsPlus(state, col, C) != wantPlus) {
                    return 0;
                }
            }

            if (row + 1 == R) {
                return 1;
            }

            string key = makeKey(row, state, used);
            auto it = memo.find(key);
            if (it != memo.end()) return it->second;

            cpp_int answer = 0;

            for (const Edge& e : graph[state]) {
                bool usable = true;
                for (int label : e.labels) {
                    if (contains(used, label)) {
                        usable = false;
                        break;
                    }
                }
                if (!usable) continue;

                vector<Interval> nextUsed = addLabels(used, e.labels);
                answer += self(self, row + 1, e.to, nextUsed);
            }

            memo.emplace(move(key), answer);
            return answer;
        };

        vector<Interval> empty;
        cpp_int answer = dfs(dfs, 0, 0, empty);
        cout << answer << '\n';
    }
