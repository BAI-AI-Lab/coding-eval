import sys
import math


def main():
    n = int(sys.stdin.read())

    # Let d = x - y. Then:
    # n = d^3 + 3*d^2*y + 3*d*y^2, so d^3 < n.
    d = 1
    while d * d * d < n:
        if n % d == 0:
            q = n // d

            # q = 3*y^2 + 3*d*y + d^2
            # Its discriminant is 12*q - 3*d^2.
            discriminant = 12 * q - 3 * d * d
            if discriminant >= 0:
                root = math.isqrt(discriminant)
                if root * root == discriminant:
                    numerator = root - 3 * d
                    if numerator > 0 and numerator % 6 == 0:
                        y = numerator // 6
                        x = y + d
                        if x * x * x - y * y * y == n:
                            print(x, y)
                            return
        d += 1

    print(-1)


if __name__ == "__main__":
    main()

import sys
from math import isqrt

def main():
    data = sys.stdin.read().strip()
    if not data:
        return
    N = int(data)

    # d = x - y
    for d in range(1, 10**6 + 1):
        # Since y >= 1, N > d^3
        if d * d * d >= N:
            break
        if N % d != 0:
            continue

        S = N // d
        # S = 3*y^2 + 3*d*y + d^2
        # Solve for y: 3*y^2 + 3*d*y + (d^2 - S) = 0
        D = 12 * S - 3 * d * d
        if D < 0:
            continue

        r = isqrt(D)
        if r * r != D:
            continue

        num = r - 3 * d
        if num > 0 and num % 6 == 0:
            y = num // 6
            x = y + d
            if x * x * x - y * y * y == N:
                print(x, y)
                return

    print(-1)

if __name__ == "__main__":
    main()
