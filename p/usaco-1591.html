Milk Buckets

USACO 2026 Third · 银组 · 2026-02 · L2 中等 · Python 3.11 · 时限 4 秒 · 测试点等分

题面

来源：USACO 官网。模型收到的是同一份题面的文字。

There are $N$ ($1\le N\le 2\cdot 10^5$) buckets in a stack where the $i$-th bucket from the top has capacity $a_i$ gallons ($1\le a_i\le 10^9$). A tap above the top bucket sends one gallon of milk per second into the first bucket per second. There is also a pool below bucket $N$.

When a bucket reaches its capacity after $t$ seconds, at the start of the $t+1$th second it flips to dump its contents into either the bucket below if it is not the last bucket or the pool otherwise (it reverts back to filling at the end of the $t+1$th second). A bucket cannot collect milk while it is flipped; any milk arriving at the bucket above during this second is lost. Additionally, any amount of milk exceeding the capacity of the bucket below is lost.

Handle $Q$ ($1\le Q\le 3\cdot 10^5$) queries, each specified by three integers $i$, $v$, and $t$:

  1. First, set $a_i=v$ ($1\le i\le N, 1\le v \le 10^9$).
  2. Then answer the following question: Suppose that at time $0$, all buckets as well as the pool are empty. Determine the number of gallons of milk in the pool after $t$ seconds ($1\le t\le 10^{18}$).

The $a_i=v$ updates persist through later queries.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $N$.

The second line contains $a_1\dots a_N$.

The next line contains $Q$.

Then the following $Q$ lines contain three integers $i$, $v$, and $t$. This means that you should set $a_i=v$ and then answer the question for $t$.

OUTPUT FORMAT (print output to the terminal / stdout):

Output the answer to each question on a separate line.

import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    a = data[1:n + 1]
    q = data[n + 1]

    # P = (a[0] + 1) * product(ceil(a[i] / a[i-1]), i >= 1).
    # Products larger than 10^18 can be saturated, since they always give 0.
    LIMIT = 10**18 + 1

    size = 1
    while size < n:
        size <<= 1

    seg = [1] * (2 * size)
    seg[size] = a[0] + 1
    for i in range(1, n):
        seg[size + i] = (a[i] + a[i - 1] - 1) // a[i - 1]

    for i in range(size - 1, 0, -1):
        product = seg[2 * i] * seg[2 * i + 1]
        seg[i] = product if product <= LIMIT else LIMIT

    def update(pos, value, seg=seg, size=size, limit=LIMIT):
        node = size + pos
        if seg[node] == value:
            return
        seg[node] = value
        node >>= 1
        while node:
            product = seg[2 * node] * seg[2 * node + 1]
            if product > limit:
                product = limit
            if seg[node] == product:
                break
            seg[node] = product
            node >>= 1

    answers = []
    ptr = n + 2

    for _ in range(q):
        pos = data[ptr] - 1
        v = data[ptr + 1]
        t = data[ptr + 2]
        ptr += 3

        if a[pos] != v:
            a[pos] = v

            if pos == 0:
                update(0, v + 1)
            else:
                update(pos, (v + a[pos - 1] - 1) // a[pos - 1])

            if pos + 1 < n:
                update(pos + 1, (a[pos + 1] + v - 1) // v)

        x = t - n + 1
        if x > 0:
            dumps = x // seg[1]
            if dumps:
                answers.append(str(dumps * a[-1]))
                continue
        answers.append("0")

    sys.stdout.write("\n".join(answers))

if __name__ == "__main__":
    main()

import sys


def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    N = next(it)
    a = [0] + [next(it) for _ in range(N)]
    Q = next(it)

    CAP = 10**18 + 1

    # Segment tree over factors:
    # factor[i] = ceil(a_i / a_{i-1}) for i = 2..N
    size = 1
    while size < N - 1:
        size <<= 1

    tree = [1] * (2 * size)

    for i in range(2, N + 1):
        pos = i - 2
        tree[size + pos] = (a[i] + a[i - 1] - 1) // a[i - 1]

    for p in range(size - 1, 0, -1):
        val = tree[p << 1] * tree[p << 1 | 1]
        tree[p] = CAP if val > CAP else val

    def update(pos: int, value: int) -> None:
        idx = size + pos
        tree[idx] = value
        idx >>= 1
        while idx:
            val = tree[idx << 1] * tree[idx << 1 | 1]
            tree[idx] = CAP if val > CAP else val
            idx >>= 1

    base = a[1] + 1
    answers = []

    for _ in range(Q):
        i = next(it)
        v = next(it)
        t = next(it)

        a[i] = v

        if i == 1:
            base = a[1] + 1
            if N >= 2:
                update(0, (a[2] + a[1] - 1) // a[1])
        else:
            # Update factor i: ceil(a_i / a_{i-1})
            update(i - 2, (a[i] + a[i - 1] - 1) // a[i - 1])

            # Update factor i+1 if it exists
            if i + 1 <= N:
                update(i - 1, (a[i + 1] + a[i] - 1) // a[i])

        period = base * tree[1]
        if period > CAP:
            period = CAP

        x = t - N + 1
        if x <= 0:
            answers.append("0")
        else:
            answers.append(str(a[N] * (x // period)))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()

import sys
from math import prod


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    a = data[1:n + 1]
    q = data[n + 1]
    ptr = n + 2

    LIMIT = 10**18

    # active[j] stores ceil(a[j + 1] / a[j]) when that factor is > 1.
    # Factors equal to 1 are omitted because they do not affect the product.
    active = {}
    for j in range(n - 1):
        factor = (a[j + 1] + a[j] - 1) // a[j]
        if factor > 1:
            active[j] = factor

    def change_factor(j, new_factor):
        old_factor = active.get(j, 1)
        if old_factor == new_factor:
            return

        if new_factor == 1:
            active.pop(j, None)
        else:
            active[j] = new_factor

    answers = []

    for _ in range(q):
        idx = data[ptr] - 1
        value = data[ptr + 1]
        t = data[ptr + 2]
        ptr += 3

        if a[idx] != value:
            a[idx] = value

            if idx == 0:
                # Changing a[0] affects the factor from bucket 1 to bucket 2.
                if n > 1:
                    change_factor(0, (a[1] + value - 1) // value)
            else:
                # Factor from bucket idx to bucket idx + 1.
                change_factor(
                    idx - 1,
                    (value + a[idx - 1] - 1) // a[idx - 1]
                )

                # Factor from bucket idx + 1 to bucket idx + 2.
                if idx + 1 < n:
                    change_factor(
                        idx,
                        (a[idx + 1] + value - 1) // value
                    )

        if len(active) >= 60:
            answers.append("0")
            continue

        period = (a[0] + 1) * prod(active.values())

        if period > LIMIT:
            answers.append("0")
            continue

        dumps = (t - n + 1) // period
        if dumps < 0:
            dumps = 0

        answers.append(str(dumps * a[-1]))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()
