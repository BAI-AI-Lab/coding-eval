题面

来源：USACO 官网。模型收到的是同一份题面的文字。

**Note: The time limit for this problem is 4s, twice the default.**

Farmer John has a binary tree with $N$ nodes where the nodes are numbered from $1$ to $N$ ($1 \leq N < 2\cdot 10^5$ and $N$ is odd). For $i>1$, the parent of node $i$ is $\lfloor i/2\rfloor$. Each node has an initial integer value $a_i$, and a cost $c_i$ to change the initial value to any other integer value ($0\le a_i,c_i\le 10^9$).

He has been tasked by the Federal Bovine Intermediary (FBI) with finding an approximate median value within this tree, and has devised a clever algorithm to do so.

He starts at the last node $N$ and works his way backward. At every step of the algorithm, if a node would not be the median of it and its two children, he swaps the values of the current node and the child value that would be the median. At the end of this algorithm, the value at node $1$ (the root) is the median approximation.

The FBI has also given Farmer John a list of $Q$ $(1 \leq Q \leq 2\cdot 10^5)$ independent queries each specified by a target value $m$ ($0\le m\le 10^9$). For each query, FJ will first change some of the node's initial values, and then execute the median approximation algorithm. For each query, determine the minimum possible total cost for FJ to make the output of the algorithm equal to $m$.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line of input contains $N$.

The next $N$ lines each contain two integers $a_i$ and $c_i$.

The next line contains $Q$.

The next $Q$ lines each contain a target value $m$.

OUTPUT FORMAT (print output to the terminal / stdout):

Output $Q$ lines, the minimum possible total cost for each target value $m$.

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

static constexpr int LIM = 1000000000;

struct Entry {
    int start;
    array<int64, 3> dp; // minimum costs for output < m, = m, > m
};

struct Function {
    vector<Entry> seg;
};

int N;
vector<int> a;
vector<int64> c;

inline int64 min3(int64 x, int64 y, int64 z) {
    return min(x, min(y, z));
}

void addSegment(Function &f, int start, const array<int64, 3> &dp) {
    if (!f.seg.empty()) {
        const auto &last = f.seg.back().dp;
        if (last[0] == dp[0] && last[1] == dp[1] && last[2] == dp[2]) {
            return;
        }
    }
    f.seg.push_back({start, dp});
}

Function makeLeaf(int value, int64 cost) {
    Function f;
    f.seg.reserve(3);

    if (value > 0) {
        addSegment(f, 0, {cost, cost, 0});
    }
    addSegment(f, value, {cost, 0, cost});
    if (value < LIM) {
        addSegment(f, value + 1, {0, cost, cost});
    }
    return f;
}

Function combine(const Function &left, const Function &right,
                 int value, int64 cost) {
    Function result;
    result.seg.reserve(left.seg.size() + right.seg.size() + 2);

    size_t p = 0, q = 0;
    int pos = 0;

    while (pos <= LIM) {
        while (p + 1 < left.seg.size() &&
               left.seg[p + 1].start <= pos) {
            ++p;
        }
        while (q + 1 < right.seg.size() &&
               right.seg[q + 1].start <= pos) {
            ++q;
        }

        const auto &y = left.seg[p].dp;
        const auto &z = right.seg[q].dp;

        int next = LIM + 1;
        if (p + 1 < left.seg.size()) {
            next = min(next, left.seg[p + 1].start);
        }
        if (q + 1 < right.seg.size()) {
            next = min(next, right.seg[q + 1].start);
        }

        array<int64, 3> x;
        if (pos < value) {
            x = {cost, cost, 0};
            next = min(next, value);
        } else if (pos == value) {
            x = {cost, 0, cost};
            next = min(next, value + 1);
        } else {
            x = {0, cost, cost};
        }

        int64 minY = min3(y[0], y[1], y[2]);
        int64 minZ = min3(z[0], z[1], z[2]);

        // The median is below m exactly when at least two inputs are below m.
        int64 below = min3(
            y[0] + z[0],
            x[0] + y[0] + minZ,
            x[0] + z[0] + minY
        );

        // Similarly for the median being above m.
        int64 above = min3(
            y[2] + z[2],
            x[2] + y[2] + minZ,
            x[2] + z[2] + minY
        );

        // Either at least two inputs equal m, or the inputs are
        // below m, equal to m, and above m in some order.
        int64 equalPair = min3(
            y[1] + z[1],
            x[1] + y[1] + minZ,
            x[1] + z[1] + minY
        );

        int64 mixed = min3(
            x[0] + min(y[1] + z[2], y[2] + z[1]),
            x[1] + min(y[0] + z[2], y[2] + z[0]),
            x[2] + min(y[0] + z[1], y[1] + z[0])
        );

        int64 equal = min(equalPair, mixed);
        addSegment(result, pos, {below, equal, above});

        pos = next;
    }

    return result;
}

Function build(int u) {
    if (2 * u > N) {
        return makeLeaf(a[u], c[u]);
    }

    Function left = build(2 * u);
    Function right = build(2 * u + 1);
    return combine(left, right, a[u], c[u]);
}

int64 query(const Function &f, int m) {
    int lo = 0, hi = static_cast<int>(f.seg.size()) - 1;
    while (lo < hi) {
        int mid = (lo + hi + 1) / 2;
        if (f.seg[mid].start <= m) {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    return f.seg[lo].dp[1];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> N;
    a.resize(N + 1);
    c.resize(N + 1);

    for (int i = 1; i <= N; ++i) {
        cin >> a[i] >> c[i];
    }

    Function root = build(1);

    int Q;
    cin >> Q;
    while (Q--) {
        int m;
        cin >> m;
        cout << query(root, m) << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<ll> a(N + 1), c(N + 1);
    for (int i = 1; i <= N; i++) {
        cin >> a[i] >> c[i];
    }

    int Q;
    cin >> Q;

    vector<pair<ll, int>> queries(Q);
    for (int i = 0; i < Q; i++) {
        cin >> queries[i].first;
        queries[i].second = i;
    }
    sort(queries.begin(), queries.end());

    // For the current target m:
    // low[i], eq[i], high[i] are the minimum costs to make subtree i
    // output a value < m, = m, or > m, respectively.
    vector<ll> low(N + 1), eq(N + 1), high(N + 1);

    // Costs of making node i's own value belong to each category.
    vector<ll> selfLow(N + 1), selfEq(N + 1), selfHigh(N + 1);

    // Initial state corresponds to m < a_i for every node.
    for (int i = 1; i <= N; i++) {
        selfLow[i] = c[i];
        selfEq[i] = c[i];
        selfHigh[i] = 0;
    }

    auto recalc = [&](int i) {
        if (2 * i > N) {
            low[i] = selfLow[i];
            eq[i] = selfEq[i];
            high[i] = selfHigh[i];
            return;
        }

        int l = 2 * i;
        int r = 2 * i + 1;

        // The median is below m iff at least two inputs are below m.
        low[i] = min({
            selfLow[i] + low[l],
            selfLow[i] + low[r],
            low[l] + low[r]
        });

        // The median is above m iff at least two inputs are above m.
        high[i] = min({
            selfHigh[i] + high[l],
            selfHigh[i] + high[r],
            high[l] + high[r]
        });

        // Either at least two inputs equal m, or exactly one equals m
        // and the other two lie on opposite sides of m.
        ll best = min({
            selfEq[i] + eq[l],
            selfEq[i] + eq[r],
            eq[l] + eq[r]
        });

        best = min(best, selfEq[i] +
            min(low[l] + high[r], high[l] + low[r]));

        best = min(best, eq[l] +
            min(selfLow[i] + high[r], selfHigh[i] + low[r]));

        best = min(best, eq[r] +
            min(selfLow[i] + high[l], selfHigh[i] + low[l]));

        eq[i] = best;
    };

    for (int i = N; i >= 1; i--) {
        recalc(i);
    }

    vector<int> order(N);
    iota(order.begin(), order.end(), 1);
    sort(order.begin(), order.end(), [&](int x, int y) {
        return a[x] < a[y];
    });

    // Changing a node's category only affects its ancestors.
    auto updateNode = [&](int i, bool makeEqual) {
        if (makeEqual) {
            // Current target satisfies m == a_i.
            selfLow[i] = c[i];
            selfEq[i] = 0;
            selfHigh[i] = c[i];
        } else {
            // Current and all future targets satisfy m > a_i.
            selfLow[i] = 0;
            selfEq[i] = c[i];
            selfHigh[i] = c[i];
        }

        for (int v = i; ; v /= 2) {
            recalc(v);
            if (v == 1) break;
        }
    };

    vector<ll> answer(Q);
    size_t p = 0;
    size_t qi = 0;

    while (qi < queries.size()) {
        ll m = queries[qi].first;
        size_t qEnd = qi;
        while (qEnd < queries.size() && queries[qEnd].first == m) {
            qEnd++;
        }

        // Nodes with a_i < m permanently enter the "below target" state.
        while (p < order.size() && a[order[p]] < m) {
            updateNode(order[p], false);
            p++;
        }

        // Nodes with a_i == m temporarily enter the equality state.
        size_t equalBegin = p;
        while (p < order.size() && a[order[p]] == m) {
            updateNode(order[p], true);
            p++;
        }
        size_t equalEnd = p;

        for (size_t j = qi; j < qEnd; j++) {
            answer[queries[j].second] = eq[1];
        }

        // For subsequent, larger targets these nodes are below the target.
        for (size_t j = equalBegin; j < equalEnd; j++) {
            updateNode(order[j], false);
        }

        qi = qEnd;
    }

    for (ll x : answer) {
        cout << x << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct State {
    ll less, exact, greater;
    bool operator==(const State& other) const {
        return less == other.less && exact == other.exact &&
               greater == other.greater;
    }
};

struct Segment {
    int start;
    State state;
};

using Function = vector<Segment>;

int N;
vector<int> a;
vector<ll> c;

static inline State combineStates(const State& x, const State& y,
                                  ll m, int value, ll cost) {
    ll rootLess, rootExact, rootGreater;
    if (m < value) {
        rootLess = cost;
        rootExact = cost;
        rootGreater = 0;
    } else if (m == value) {
        rootLess = cost;
        rootExact = 0;
        rootGreater = cost;
    } else {
        rootLess = 0;
        rootExact = cost;
        rootGreater = cost;
    }

    // The median is below m iff at least two of the three values are below m.
    ll less = min(rootLess + min(x.less, y.less), x.less + y.less);

    // Similarly for the median being above m.
    ll greater = min(rootGreater + min(x.greater, y.greater),
                     x.greater + y.greater);

    // The current node's value equals m.
    ll fromEqual = min(
        min(x.exact, y.exact),
        min(x.less + y.greater, x.greater + y.less)
    );

    // The current node's value is below m.
    ll fromLess = min(
        x.exact + min(y.exact, y.greater),
        y.exact + min(x.exact, x.greater)
    );

    // The current node's value is above m.
    ll fromGreater = min(
        x.exact + min(y.less, y.exact),
        y.exact + min(x.less, x.exact)
    );

    ll exact = min(rootExact + fromEqual,
                   min(rootLess + fromLess, rootGreater + fromGreater));

    return {less, exact, greater};
}

Function mergeFunctions(const Function& left, const Function& right,
                        int value, ll cost) {
    Function result;
    result.reserve(left.size() + right.size() + 2);

    size_t i = 0, j = 0;
    ll m = 0;
    const ll INF = numeric_limits<ll>::max();

    while (true) {
        while (i + 1 < left.size() && left[i + 1].start <= m) ++i;
        while (j + 1 < right.size() && right[j + 1].start <= m) ++j;

        State current = combineStates(left[i].state, right[j].state,
                                      m, value, cost);

        if (result.empty() || !(result.back().state == current)) {
            result.push_back({(int)m, current});
        }

        ll next = INF;
        if (i + 1 < left.size()) {
            next = min(next, (ll)left[i + 1].start);
        }
        if (j + 1 < right.size()) {
            next = min(next, (ll)right[j + 1].start);
        }

        // The current node's three possible cost states change at value
        // and value + 1.
        if ((ll)value > m) next = min(next, (ll)value);
        if ((ll)value + 1 > m) next = min(next, (ll)value + 1);

        if (next == INF) break;
        m = next;
    }

    return result;
}

Function build(int node) {
    if (node > N / 2) {
        Function f;
        f.reserve(3);

        auto addSegment = [&](int start, State state) {
            if (f.empty() || !(f.back().state == state)) {
                f.push_back({start, state});
            }
        };

        if (a[node] == 0) {
            addSegment(0, {c[node], 0, c[node]});
            addSegment(1, {0, c[node], c[node]});
        } else {
            addSegment(0, {c[node], c[node], 0});
            addSegment(a[node], {c[node], 0, c[node]});
            addSegment(a[node] + 1, {0, c[node], c[node]});
        }
        return f;
    }

    Function left = build(node * 2);
    Function right = build(node * 2 + 1);
    return mergeFunctions(left, right, a[node], c[node]);
}

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 200005;
const long long INF = (long long)4e18;

int N;
int a[MAXN], c[MAXN];
unsigned char pref[MAXN]; // 0 = less than m, 1 = equal to m, 2 = greater than m
long long dp[MAXN][3];
int med3[3][3][3];

inline void pull(int v) {
    if ((v << 1) > N) {
        long long cv = c[v];
        int pv = pref[v];
        dp[v][0] = (pv == 0 ? 0 : cv);
        dp[v][1] = (pv == 1 ? 0 : cv);
        dp[v][2] = (pv == 2 ? 0 : cv);
        return;
    }

    int l = v << 1;
    int r = l | 1;

    long long cv = c[v];
    int pv = pref[v];

    long long w[3];
    w[0] = (pv == 0 ? 0 : cv);
    w[1] = (pv == 1 ? 0 : cv);
    w[2] = (pv == 2 ? 0 : cv);

    long long best[3] = {INF, INF, INF};

    for (int x = 0; x < 3; ++x) {
        long long wx = w[x];
        for (int y = 0; y < 3; ++y) {
            long long base = wx + dp[l][y];
            for (int z = 0; z < 3; ++z) {
                int out = med3[x][y][z];
                long long val = base + dp[r][z];
                if (val < best[out]) best[out] = val;
            }
        }
    }

    dp[v][0] = best[0];
    dp[v][1] = best[1];
    dp[v][2] = best[2];
}

inline void changePref(int v, int np) {
    if ((int)pref[v] == np) return;
    pref[v] = (unsigned char)np;

    int u = v;
    while (u >= 1) {
        pull(u);
        u >>= 1;
    }
}

struct Query {
    int m;
    int idx;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> N;

    for (int i = 1; i <= N; ++i) {
        cin >> a[i] >> c[i];
        pref[i] = 2; // initially all values are greater than m
    }

    for (int x = 0; x < 3; ++x) {
        for (int y = 0; y < 3; ++y) {
            for (int z = 0; z < 3; ++z) {
                int mn = min(x, min(y, z));
                int mx = max(x, max(y, z));
                med3[x][y][z] = x + y + z - mn - mx;
            }
        }
    }

    for (int i = N; i >= 1; --i) {
        pull(i);
    }

    int Q;
    cin >> Q;

    vector<Query> queries(Q);
    for (int i = 0; i < Q; ++i) {
        cin >> queries[i].m;
        queries[i].idx = i;
    }

    sort(queries.begin(), queries.end(), [](const Query &p, const Query &q) {
        return p.m < q.m;
    });

    vector<int> ord(N);
    iota(ord.begin(), ord.end(), 1);

    sort(ord.begin(), ord.end(), [&](int i, int j) {
        return a[i] < a[j];
    });

    vector<long long> ans(Q);

    int ptr = 0;

    for (int qi = 0; qi < Q; ) {
        int m = queries[qi].m;

        // All nodes with a[i] < m should now have free category "less".
        while (ptr < N && a[ord[ptr]] < m) {
            changePref(ord[ptr], 0);
            ++ptr;
        }

        // All nodes with a[i] == m should now have free category "equal".
        int q = ptr;
        while (q < N && a[ord[q]] == m) {
            changePref(ord[q], 1);
            ++q;
        }

        long long res = dp[1][1];

        do {
            ans[queries[qi].idx] = res;
            ++qi;
        } while (qi < Q && queries[qi].m == m);
    }

    for (int i = 0; i < Q; ++i) {
        cout << ans[i] << '\n';
    }

    return 0;
}

        queries[i] = {m, i};
    }
    sort(queries.begin(), queries.end());

    const int firstTarget = queries.front().first;

    // Category of each original value relative to the current target:
    // LT: a[i] < target, EQ: a[i] == target, GT: a[i] > target.
    vector<int> state(N + 1);
    for (int i = 1; i <= N; ++i) {
        if (a[i] < firstTarget) state[i] = LT;
        else if (a[i] == firstTarget) state[i] = EQ;
        else state[i] = GT;
    }

    // dp[i][category] is the minimum cost to make the value produced
    // by node i have the specified category relative to the target.
    vector<array<int64, 3>> dp(N + 1);

    auto pull = [&](int v) {
        // Cost of making node v's own initial value have each category.
        array<int64, 3> own{cost[v], cost[v], cost[v]};
        own[state[v]] = 0;

        const int left = 2 * v;
        if (left > N) {
            dp[v] = own;
            return;
        }

        // N is odd, so every internal node has exactly two children.
        const auto &L = dp[left];
        const auto &R = dp[left + 1];

        const int64 minOwn = min(own[LT], min(own[EQ], own[GT]));
        const int64 minL = min(L[LT], min(L[EQ], L[GT]));
        const int64 minR = min(R[LT], min(R[EQ], R[GT]));

        array<int64, 3> result;

        // The median is LT exactly when at least two inputs are LT.
        result[LT] = min(
            own[LT] + L[LT] + minR,
            min(own[LT] + R[LT] + minL,
                L[LT] + R[LT] + minOwn)
        );

        // The median is GT exactly when at least two inputs are GT.
        result[GT] = min(
            own[GT] + L[GT] + minR,
            min(own[GT] + R[GT] + minL,
                L[GT] + R[GT] + minOwn)
        );

        int64 twoEqual = min(
            own[EQ] + L[EQ] + minR,
            min(own[EQ] + R[EQ] + minL,
                L[EQ] + R[EQ] + minOwn)

        // Exactly one input equals the target, while the other two
        // are on opposite sides of the target.
        int64 oneEqual = min(

            min(L[EQ] + min(own[LT] + R[GT], own[GT] + R[LT]),
                R[EQ] + min(own[LT] + L[GT], own[GT] + L[LT]))
        );

        result[EQ] = min(twoEqual, oneEqual);
        dp[v] = result;
    };

    // Children have larger indices, so this is a valid bottom-up order.
    for (int i = N; i >= 1; --i) {
        pull(i);

    vector<Event> events;

        state[v] = newType;

        // Only this node and its ancestors are affected.
        for (int u = v; u > 0; u >>= 1) {
