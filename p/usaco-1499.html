#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, M;
    cin >> N >> M;

    vector<int> adj(N, 0);
    for (int i = 0; i < M; ++i) {
        int a, b;
        cin >> a >> b;
        --a; --b;
        adj[a] |= 1 << b;
        adj[b] |= 1 << a;
    }

    int size = 1 << N;
    vector<int> edges(size, 0);

    for (int mask = 1; mask < size; ++mask) {
        int bit = mask & -mask;
        int v = __builtin_ctz(bit);
        int rest = mask ^ bit;
        edges[mask] = edges[rest] + __builtin_popcount(adj[v] & rest);
    }

    const int INF = 1e9;
    vector<int> dp(size, INF);
    dp[0] = 0;

    for (int mask = 1; mask < size; ++mask) {
        int required = mask & -mask;
        int totalPairs = __builtin_popcount(mask) * (__builtin_popcount(mask) - 1) / 2;

        for (int part = mask; part; part = (part - 1) & mask) {
            if (!(part & required)) continue;
            int rest = mask ^ part;
            int crossPairs = __builtin_popcount(part) * __builtin_popcount(rest);
            int crossEdges = edges[mask] - edges[part] - edges[rest];
            int cost = edges[part] + (crossPairs - crossEdges) + dp[rest];
            dp[mask] = min(dp[mask], cost);
        }
    }

    cout << dp[size - 1] << '\n';
    return 0;
}

        --a;
        --b;
        g[a][b] = g[b][a] = true;
    }

    vector<int> adj(N, 0);

    // Build complement graph H = complement of G.
    for (int i = 0; i < N; i++) {
        for (int j = i + 1; j < N; j++) {
            if (!g[i][j]) {
                adj[i] |= (1 << j);
                adj[j] |= (1 << i);
            }
        }
    }

    int totalMasks = 1 << N;
    int all = totalMasks - 1;

    vector<int> edgeCount(totalMasks, 0);

    for (int mask = 1; mask < totalMasks; mask++) {
        int lsb = mask & -mask;
        int v = __builtin_ctz(lsb);
        int rest = mask ^ lsb;

        edgeCount[mask] = edgeCount[rest] +
                          __builtin_popcount((unsigned)(adj[v] & rest));
    }

    vector<int> weight(totalMasks, 0);

    for (int mask = 1; mask < totalMasks; mask++) {
        int sz = __builtin_popcount((unsigned)mask);
        int completeEdges = sz * (sz - 1) / 2;

        weight[mask] = completeEdges - 2 * edgeCount[mask];
    }

    const int INF = 1e9;
    vector<int> dp(totalMasks, INF);
    dp[0] = 0;

    for (int mask = 1; mask < totalMasks; mask++) {
        int best = INF;

        for (int sub = mask; sub; sub = (sub - 1) & mask) {
            best = min(best, dp[mask ^ sub] + weight[sub]);
        }

        dp[mask] = best;
    }

    int answer = edgeCount[all] + dp[all];
    cout << answer << '\n';

    return 0;

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, M;
    cin >> N >> M;

    vector<int> adj(N, 0);
    for (int i = 0; i < M; ++i) {
        int a, b;
        cin >> a >> b;
        --a;
        --b;
        adj[a] |= 1 << b;
        adj[b] |= 1 << a;
    }

    const int LIMIT = 1 << N;
    vector<int> popcount(LIMIT, 0);
    vector<int> inducedEdges(LIMIT, 0);
    vector<int> weight(LIMIT, 0);

    // weight[mask] = C(|mask|, 2) - 2 * inducedEdges[mask].
    for (int mask = 1; mask < LIMIT; ++mask) {
        int v = __builtin_ctz(mask);
        int rest = mask & (mask - 1);

        popcount[mask] = popcount[rest] + 1;
        inducedEdges[mask] =
            inducedEdges[rest] + __builtin_popcount(adj[v] & rest);

        weight[mask] =
            popcount[mask] * (popcount[mask] - 1) / 2
            - 2 * inducedEdges[mask];
    }

    // dp[mask] is the maximum sum of weights over all partitions of mask.
    vector<int> dp(LIMIT, 0);

    for (int mask = 1; mask < LIMIT; ++mask) {
        int anchor = mask & -mask;
        int others = mask ^ anchor;
        int best = INT_MIN;

        // Enumerate the part containing anchor.
        for (int sub = others;; sub = (sub - 1) & others) {
            int part = sub | anchor;
            int remainder = others ^ sub;

            best = max(best, weight[part] + dp[remainder]);

            if (sub == 0) break;
        }

        dp[mask] = best;
    }

    int totalPairs = N * (N - 1) / 2;
    cout << totalPairs - M - dp[LIMIT - 1] << '\n';

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, M;
    cin >> N >> M;

    vector<int> adj(N, 0);
    for (int i = 0; i < M; ++i) {
        int a, b;
        cin >> a >> b;
        --a;
        --b;
        adj[a] |= 1 << b;
        adj[b] |= 1 << a;
    }

    const int LIMIT = 1 << N;

    // edgeCount[mask] = number of original edges entirely inside mask.
    // vertexCount[mask] = number of vertices in mask.
    vector<int> edgeCount(LIMIT, 0);
    vector<int> vertexCount(LIMIT, 0);

    for (int mask = 1; mask < LIMIT; ++mask) {
        int v = __builtin_ctz(mask);
        int rest = mask & (mask - 1);

        vertexCount[mask] = vertexCount[rest] + 1;
        edgeCount[mask] =
            edgeCount[rest] + __builtin_popcount(adj[v] & rest);
    }

    const int INF = 1e9;
    vector<int> dp(LIMIT, INF);
    dp[0] = 0;

    for (int mask = 1; mask < LIMIT; ++mask) {
        int pivot = mask & -mask;
        int rest = mask ^ pivot;
        int best = INF;

        // Choose the part containing pivot.
        for (int sub = rest;; sub = (sub - 1) & rest) {
            int part = sub | pivot;
            int remaining = mask ^ part;

            // All original edges crossing between part and remaining.
            int originalCross =
                edgeCount[mask] - edgeCount[part] - edgeCount[remaining];

            // Every crossing pair must be an edge in the final graph.
            int additions =
                vertexCount[part] * vertexCount[remaining] - originalCross;

            // Original edges inside one part must be removed.
            int removals = edgeCount[part];

            best = min(best, removals + additions + dp[remaining]);

            if (sub == 0) break;
        }

        dp[mask] = best;
    }

    cout << dp[LIMIT - 1] << '\n';
    return 0;
}
