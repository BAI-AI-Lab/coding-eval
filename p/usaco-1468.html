import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:]))
    del data

    remaining = [0] * (n + 1)
    for x in a:
        remaining[x] += 1

    seen = bytearray(n + 1)
    distinct = 0
    answer = 0

    for x in a:
        # This is the penultimate occurrence of x. Every different value
        # before it can form a moo whose last two values are x.
        if remaining[x] == 2:
            answer += distinct - seen[x]

        if not seen[x]:
            seen[x] = 1
            distinct += 1

        remaining[x] -= 1

    sys.stdout.write(str(answer))

main()

import sys
from array import array

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    N = int(data[0])
    total = array('I', [0]) * (N + 1)
    a = array('I', [0]) * N

    for i in range(N):
        v = int(data[i + 1])
        a[i] = v
        total[v] += 1

    del data

    present = bytearray(N + 1)
    ans = 0
    distinct = 0

    for v in a:
        if total[v] == 2:
            ans += distinct
            if present[v]:
                ans -= 1

        if present[v] == 0:
            present[v] = 1
            distinct += 1

        total[v] -= 1

    sys.stdout.write(str(ans))

if __name__ == "__main__":
    main()

import sys


def solve():
    values = map(int, sys.stdin.buffer.read().split())
    n = next(values)

    # 0: unseen, 1: seen once, 2: seen at least twice
    state = bytearray(n + 1)

    distinct_seen = 0
    answer = 0

    for value in values:
        if state[value] == 0:
            state[value] = 1
            distinct_seen += 1
        elif state[value] == 1:
            # Every previously seen distinct value except this one
            # can form a moo (x, value, value).
            answer += distinct_seen - 1
            state[value] = 2

    print(answer)


if __name__ == "__main__":
    solve()
