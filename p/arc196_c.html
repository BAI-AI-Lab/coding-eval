import sys

MOD = 998244353

# Optional fast exact convolution using double FFT with 15-bit splitting.
# A pure-modular NTT fallback is provided below.
try:
    import numpy as _np
except Exception:
    _np = None


_REV = {}
_ROOTS = {}


def _ntt(a, inverse):
    n = len(a)

    rev = _REV.get(n)
    if rev is None:
        rev = [0] * n
        half = n >> 1
        for i in range(1, n):
            rev[i] = (rev[i >> 1] >> 1) | ((i & 1) * half)
        _REV[n] = rev

    for i in range(n):
        j = rev[i]
        if i < j:
            a[i], a[j] = a[j], a[i]

    length = 2
    while length <= n:
        key = (length, inverse)
        roots = _ROOTS.get(key)
        if roots is None:
            wlen = pow(3, (MOD - 1) // length, MOD)
            if inverse:
                wlen = pow(wlen, MOD - 2, MOD)
            roots = [1] * (length >> 1)
            for i in range(1, len(roots)):
                roots[i] = roots[i - 1] * wlen % MOD
            _ROOTS[key] = roots

        half = length >> 1
        for start in range(0, n, length):
            for j in range(half):
                x = start + j
                y = x + half
                u = a[x]
                v = a[y] * roots[j] % MOD
                s = u + v
                if s >= MOD:
                    s -= MOD
                d = u - v
                if d < 0:
                    d += MOD
                a[x] = s
                a[y] = d
        length <<= 1

    if inverse:
        inv_n = pow(n, MOD - 2, MOD)
        for i in range(n):
            a[i] = a[i] * inv_n % MOD


def _ntt_convolution(a, b):
    need = len(a) + len(b) - 1
    n = 1 << (need - 1).bit_length()
    fa = a + [0] * (n - len(a))
    fb = b + [0] * (n - len(b))
    _ntt(fa, False)
    _ntt(fb, False)
    for i in range(n):
        fa[i] = fa[i] * fb[i] % MOD
    _ntt(fa, True)
    return fa[:need]


def _fft_convolution(a, b):
    need = len(a) + len(b) - 1
    n = 1 << (need - 1).bit_length()

    aa = _np.asarray(a, dtype=_np.int64)
    bb = _np.asarray(b, dtype=_np.int64)

    split = 15
    base = 1 << split
    mask = base - 1

    a0 = aa & mask
    a1 = aa >> split
    b0 = bb & mask
    b1 = bb >> split

    fa0 = _np.fft.rfft(a0, n)
    fa1 = _np.fft.rfft(a1, n)
    fb0 = _np.fft.rfft(b0, n)
    fb1 = _np.fft.rfft(b1, n)

    c00 = _np.rint(
        _np.fft.irfft(fa0 * fb0, n)[:need]
    ).astype(_np.int64)
    c01 = _np.rint(
        _np.fft.irfft(fa0 * fb1 + fa1 * fb0, n)[:need]
    ).astype(_np.int64)
    c11 = _np.rint(
        _np.fft.irfft(fa1 * fb1, n)[:need]
    ).astype(_np.int64)

    base2 = base * base % MOD
    return (
        c00 % MOD
        + (c01 % MOD) * base
        + (c11 % MOD) * base2
    ) % MOD


def _convolution(a, b):
    if _np is not None:
        return _fft_convolution(a, b)
    return _ntt_convolution(a, b)


def main():
    data = sys.stdin.read().split()
    if not data:
        return

    n = int(data[0])
    s = data[1]

    # The first and last added backward edges can exist only in this case.
    if s[0] != "B" or s[-1] != "W":
        print(0)
        return

    fact = [1] * (n + 1)
    for i in range(1, n + 1):
        fact[i] = fact[i - 1] * i % MOD

    invfact = [1] * (n + 1)
    invfact[n] = pow(fact[n], MOD - 2, MOD)
    for i in range(n, 0, -1):
        invfact[i - 1] = invfact[i] * i % MOD

    # Compress the input to relevant ends of W-runs.
    # An entry (b,w) means: after b black vertices and w white vertices,
    # a nonempty W-run ends, and w >= b.
    bs = []
    ws = []

    b = 0
    w = 0
    i = 0
    length = 2 * n

    while i < length:
        # s[i] is B here.
        b += 1
        i += 1
        old_w = w
        while i < length and s[i] == "W":
            w += 1
            i += 1
        if w > old_w and w >= b:
            bs.append(b)
            ws.append(w)

    m = len(bs)

    # If x[b] is the compressed value for group b, then, for a relevant
    # group (b,w),
    #
    # x[b] = (w! - sum_{previous groups k} x[k] * (w-k)!) / (w-b)!.
    #
    # Compute this online convolution by CDQ divide and conquer.
    values = [0] * m
    incoming = [0] * m

    small_limit = 30000 if _np is not None else 80000
    fft_factor = 6 if _np is not None else 20

    def cross(left, mid, right):
        ids = [u for u in range(left, mid) if values[u]]
        if not ids:
            return

        pairs = [(bs[u], values[u]) for u in ids]
        naive_cost = len(ids) * (right - mid)

        bmin = bs[ids[0]]
        bmax = bs[ids[-1]]
        wmin = ws[mid]
        wmax = ws[right - 1]

        plen = bmax - bmin + 1
        tmin = wmin - bmax
        tmax = wmax - bmin
        zlen = tmax - tmin + 1
        conv_size = plen + zlen - 1
        fft_size = 1 << (conv_size - 1).bit_length()

        use_naive = (
            naive_cost <= small_limit
            or fft_size * fft_factor > naive_cost
        )

        if use_naive:
            for v in range(mid, right):
                wv = ws[v]
                total = incoming[v]
                for bu, au in pairs:
                    total += au * fact[wv - bu]
                incoming[v] = total % MOD
            return

        x = [0] * plen
        for u in ids:
            x[bs[u] - bmin] = values[u]
        z = fact[tmin:tmax + 1]

        conv = _convolution(x, z)
        offset = bmin + tmin

        for v in range(mid, right):
            add = int(conv[ws[v] - offset])
            val = incoming[v] + add
            if val >= MOD:
                val -= MOD
            incoming[v] = val

    sys.setrecursionlimit(1000000)

    def cdq(left, right):
        if right - left == 1:
            t = left
            values[t] = (
                (fact[ws[t]] - incoming[t]) % MOD
                * invfact[ws[t] - bs[t]] % MOD
            )
            return

        mid = (left + right) >> 1
        cdq(left, mid)
        cross(left, mid, right)
        cdq(mid, right)

    cdq(0, m)
    print(values[-1] % MOD)


if __name__ == "__main__":
    main()

MOD = 998244353
PRIMITIVE_ROOT = 3


def main() -> None:
    sys.setrecursionlimit(1 << 20)
    input = sys.stdin.readline

    N = int(input())
    S = input().strip()

    # For strong connectivity, the first vertex must be black and the last white.
    if S[0] != 'B' or S[-1] != 'W':
        print(0)
        return

    fact = [1] * (N + 1)
    for i in range(1, N + 1):
        fact[i] = fact[i - 1] * i % MOD

    invfact = [1] * (N + 1)
    invfact[N] = pow(fact[N], MOD - 2, MOD)
    for i in range(N, 0, -1):
        invfact[i - 1] = invfact[i] * i % MOD

    # Important cuts: ends of W-runs immediately followed by B.
    # Store (number of B before the cut, number of W before the cut).
    C = [0]
    W = [0]

    b = w = 0
    prev = ''
    for ch in S:
        if ch == 'B':
            if prev == 'W' and w - b >= 0:
                C.append(b)
                W.append(w)
            b += 1
        else:
            w += 1
        prev = ch

    M = len(C) - 1
    h = [0] * (M + 1)
    for i in range(1, M + 1):
        h[i] = W[i] - C[i]  # >= 0 for every stored cut

    # For small M, the quadratic DP is faster.
    if M <= 3000:
        F = [0] * (M + 1)
        F[0] = 1

        for i in range(1, M + 1):
            s = 0
            wi = W[i]
            for j in range(i):
                s += F[j] * fact[wi - C[j]]
            s %= MOD
            F[i] = (-s * invfact[h[i]]) % MOD

        ans = 0
        for j in range(M + 1):
            ans += F[j] * fact[N - C[j]]
            ans %= MOD
        print(ans)
        return

    # ---------- NTT preparation ----------
    max_size = 1
    while max_size < 2 * N + 5:
        max_size <<= 1
    max_log = max_size.bit_length() - 1

    roots = [0] * (max_log + 1)
    inv_roots = [0] * (max_log + 1)
    for k in range(1, max_log + 1):
        roots[k] = pow(PRIMITIVE_ROOT, (MOD - 1) // (1 << k), MOD)
        inv_roots[k] = pow(roots[k], MOD - 2, MOD)

    inv_size = [1] * (max_log + 1)
    inv2 = (MOD + 1) // 2
    for k in range(1, max_log + 1):
        inv_size[k] = inv_size[k - 1] * inv2 % MOD

    rev_cache = {}

    def ntt(a, invert):
        n = len(a)

        rev = rev_cache.get(n)
        if rev is None:
            bits = n.bit_length() - 1
            rev = [0] * n
            for i in range(1, n):
                rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (bits - 1))
            rev_cache[n] = rev

        for i in range(n):
            j = rev[i]
            if i < j:
                a[i], a[j] = a[j], a[i]

        length = 2
        level = 1
        used_roots = inv_roots if invert else roots

        while length <= n:
            wlen = used_roots[level]
            half = length >> 1

            for start in range(0, n, length):
                w = 1
                end = start + half
                j = start
                while j < end:
                    u = a[j]
                    v = a[j + half] * w % MOD

                    x = u + v
                    if x >= MOD:
                        x -= MOD
                    y = u - v
                    if y < 0:
                        y += MOD

                    a[j] = x
                    a[j + half] = y

                    w = w * wlen % MOD
                    j += 1

            length <<= 1
            level += 1

        if invert:
            inv_n = inv_size[n.bit_length() - 1]
            for i in range(n):
                a[i] = a[i] * inv_n % MOD

    def convolution(a, b):
        n = len(a)
        m = len(b)
        if n == 0 or m == 0:
            return []

        if n * m <= 4096:
            res = [0] * (n + m - 1)
            for i, ai in enumerate(a):
                if ai == 0:
                    continue
                for j, bj in enumerate(b):
                    res[i + j] = (res[i + j] + ai * bj) % MOD
            return res

        need = n + m - 1
        size = 1
        while size < need:
            size <<= 1

        fa = a + [0] * (size - n)
        fb = b + [0] * (size - m)

        ntt(fa, False)
        ntt(fb, False)

        for i in range(size):
            fa[i] = fa[i] * fb[i] % MOD

        ntt(fa, True)
        return fa[:need]

    F = [0] * (M + 1)
    F[0] = 1
    acc = [0] * (M + 1)

    def cdq(l, r):
        if l == r:
            if l > 0:
                F[l] = (-acc[l] * invfact[h[l]]) % MOD
            return

        mid = (l + r) >> 1
        cdq(l, mid)

        left_count = mid - l + 1
        right_count = r - mid

        if left_count * right_count <= 12000:
            for i in range(mid + 1, r + 1):
                wi = W[i]
                total = 0
                for j in range(l, mid + 1):
                    total += F[j] * fact[wi - C[j]]
                acc[i] = (acc[i] + total) % MOD
        else:
            cmax = C[mid]
            wmin = W[mid + 1]
            base = wmin - cmax

            len_c = cmax - C[l] + 1
            len_w = W[r] - wmin + 1

            a = [0] * len_c
            for j in range(l, mid + 1):
                u = cmax - C[j]
                a[len_c - 1 - u] = F[j]

            blen = len_c + len_w - 1
            bpoly = [0] * blen
            for t in range(blen):
                bpoly[t] = fact[base + t]

            conv = convolution(a, bpoly)

            offset = len_c - 1
            for i in range(mid + 1, r + 1):
                v = W[i] - wmin
                acc[i] = (acc[i] + conv[offset + v]) % MOD

        cdq(mid + 1, r)

    cdq(0, M)

    ans = 0
    for j in range(M + 1):
        ans += F[j] * fact[N - C[j]]
        ans %= MOD

    print(ans)


if __name__ == "__main__":
    main()

import sys

MOD = 998244353
G = 3

def ntt(a, invert):
    n = len(a)
    j = 0
    for i in range(1, n):
        bit = n >> 1
        while j & bit:
            j ^= bit
            bit >>= 1
        j ^= bit
        if i < j:
            a[i], a[j] = a[j], a[i]

    length = 2
    while length <= n:
        wlen = pow(G, (MOD - 1) // length, MOD)
        if invert:
            wlen = pow(wlen, MOD - 2, MOD)
        half = length >> 1
        for i in range(0, n, length):
            w = 1
            for j in range(i, i + half):
                u = a[j]
                v = a[j + half] * w % MOD
                a[j] = (u + v) % MOD
                a[j + half] = (u - v) % MOD
                w = w * wlen % MOD
        length <<= 1

    if invert:
        inv_n = pow(n, MOD - 2, MOD)
        for i in range(n):
            a[i] = a[i] * inv_n % MOD


def main():
    input = sys.stdin.readline
    N = int(input())
    S = input().strip()

    if S[0] != 'B' or S[-1] != 'W':
        print(0)
        return

    fact = [1] * (N + 1)
    for i in range(1, N + 1):
        fact[i] = fact[i - 1] * i % MOD

    invfact = [1] * (N + 1)
    invfact[N] = pow(fact[N], MOD - 2, MOD)
    for i in range(N, 0, -1):
        invfact[i - 1] = invfact[i] * i % MOD

    # c[i] for i=1..N-1: number of whites before the (i+1)-th black.
    # c[0] = 0 is used for the empty prefix in the inclusion-exclusion.
    c = [0] * N
    bcnt = 0
    wcnt = 0
    for ch in S:
        if ch == 'B':
            bcnt += 1
            if bcnt >= 2:
                c[bcnt - 1] = wcnt
        else:
            wcnt += 1

    dp = [0] * N
    res = [0] * N

    sys.setrecursionlimit(1 << 20)

    valid = [False] * N
    for i in range(1, N):
        valid[i] = (c[i] >= i)

    NAIVE_LIMIT = 60

    def solve(l, r):
        if l > r:
            return
        if l == r:
            if l == 0:
                dp[0] = 1
            else:
                if valid[l]:
                    dp[l] = (-invfact[c[l] - l] * res[l]) % MOD
                else:
                    dp[l] = 0
            return

        mid = (l + r) >> 1
        solve(l, mid)

        rv = [i for i in range(mid + 1, r + 1) if valid[i]]
        if rv:
            n_min = c[rv[0]]
            n_max = c[rv[-1]]
            k_min = n_min - mid
            k_max = n_max - l
            if k_min < 0:
                k_min = 0

            if k_max >= k_min:
                L = mid - l + 1
                A = dp[l:mid + 1]
                K_len = k_max - k_min + 1
                F = fact[k_min:k_min + K_len]
                need = L + K_len - 1

                if need <= NAIVE_LIMIT:
                    C = [0] * need
                    for ai, av in enumerate(A):
                        if av:
                            base = ai
                            for bi, bv in enumerate(F):
                                C[base + bi] = (C[base + bi] + av * bv) % MOD
                else:
                    n = 1
                    while n < need:
                        n <<= 1
                    arr1 = A + [0] * (n - L)
                    arr2 = F + [0] * (n - K_len)
                    ntt(arr1, False)
                    ntt(arr2, False)
                    for i in range(n):
                        arr1[i] = arr1[i] * arr2[i] % MOD
                    ntt(arr1, True)
                    C = arr1

                base = l + k_min
                for i in rv:
                    idx = c[i] - base
                    if 0 <= idx < need:
                        res[i] = (res[i] + C[idx]) % MOD

        solve(mid + 1, r)

    if N >= 1:
        solve(0, N - 1)

    ans = 0
    for j in range(N):
        ans = (ans + dp[j] * fact[N - j]) % MOD

    print(ans)


if __name__ == "__main__":
    main()
