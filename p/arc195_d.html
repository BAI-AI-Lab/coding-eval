import sys


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    pos = 0
    t = data[pos]
    pos += 1
    answers = []

    for _ in range(t):
        n = data[pos]
        pos += 1

        # Run-length encode the sequence.
        values = []
        counts = []
        for _ in range(n):
            x = data[pos]
            pos += 1
            if values and values[-1] == x:
                counts[-1] += 1
            else:
                values.append(x)
                counts.append(1)

        m = len(values)

        # dp[i]: answer for the contiguous suffix of runs i..m-1.
        # aux[i]: answer for run i followed by runs i+2..m-1
        #         (run i+1 is absent).
        dp = [0] * (m + 1)
        aux = [0] * m

        for i in range(m - 1, -1, -1):
            j = i + 2

            if j >= m:
                aux[i] = 1
            elif values[i] == values[j]:
                # The two equal-valued parts form one leading run. Its length
                # does not affect the answer.
                aux[i] = dp[j]
            else:
                best = 1 + dp[j]
                k = j + 1
                if (
                    k < m
                    and counts[j] == 1
                    and counts[k] == 1
                    and values[k] == values[i]
                ):
                    best = min(best, 2 + aux[j])
                aux[i] = best

            best = 1 + dp[i + 1]
            if (
                i + 2 < m
                and counts[i + 1] == 1
                and counts[i + 2] == 1
                and values[i + 2] == values[i]
            ):
                best = min(best, 2 + aux[i + 1])
            dp[i] = best

        answers.append(str(dp[0]))

    sys.stdout.write("\n".join(answers) + "\n")


if __name__ == "__main__":
    main()

import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
    T = next(it)
    answers = []

    for _ in range(T):
        N = next(it)

        values = []
        lengths = []

        for _ in range(N):
            x = next(it)
            if not values or values[-1] != x:
                values.append(x)
                lengths.append(1)
            else:
                lengths[-1] += 1

        m = len(values)
        dp = [0] * (m + 1)

        for i in range(1, m + 1):
            best = dp[i - 1]

            # Delete runs i-1 and i by one beneficial swap.
            # Need left neighbor i-2 and right neighbor i+1.
            if 3 <= i <= m - 1:
                if lengths[i - 2] == 1 and lengths[i - 1] == 1:
                    if values[i - 3] == values[i - 1] and values[i - 2] == values[i]:
                        cand = 1 + dp[i - 3]
                        if cand > best:
                            best = cand

            dp[i] = best

        answers.append(str(m - dp[m]))

    sys.stdout.write("\n".join(answers))

if __name__ == "__main__":
    solve()

import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    idx = 1
    out = []

    for _ in range(t):
        n = data[idx]
        idx += 1
        A = data[idx:idx + n]
        idx += n

        colors = []
        lengths = []
        ops = 0

        # initial number of runs in A
        runs = 0
        prev = None

        for x in A:
            # append x to the current sequence
            if colors and colors[-1] == x:
                lengths[-1] += 1
            else:
                colors.append(x)
                lengths.append(1)

            if prev is None or x != prev:
                runs += 1
            prev = x

            # greedily apply beneficial swaps (XYXY -> XXYY)
            while len(colors) >= 4:
                c1 = colors[-4]
                c2 = colors[-3]
                c3 = colors[-2]
                c4 = colors[-1]

                if (c1 == c3 and c2 == c4 and c1 != c2 and
                        lengths[-3] == 1 and lengths[-2] == 1):
                    ops += 1

                    l1 = lengths[-4] + 1
                    l4 = lengths[-1] + 1

                    # remove the middle two runs and keep the outer two
                    del colors[-4:]
                    del lengths[-4:]

                    if colors and colors[-1] == c1:
                        lengths[-1] += l1
                    else:
                        colors.append(c1)
                        lengths.append(l1)

                    if colors and colors[-1] == c2:
                        lengths[-1] += l4
                    else:
                        colors.append(c2)
                        lengths.append(l4)
                else:
                    break

        out.append(str(runs - ops))

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
