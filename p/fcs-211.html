#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
const int64 INF = (int64)4e18;

struct Node {
    string id;
    int64 x, y;
    char type;
};

struct Edge {
    int u, v;
};

struct Tree {
    vector<Edge> edges;
    int64 cost = 0;
};

struct DSU {
    vector<int> p, sz;
    explicit DSU(int n = 0) { init(n); }
    void init(int n) {
        p.resize(n);
        sz.assign(n, 1);
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) {
        while (p[x] != x) {
            p[x] = p[p[x]];
            x = p[x];
        }
        return x;
    }
    bool unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b);
        p[b] = a;
        sz[a] += sz[b];
        return true;
    }
};

class Solver {
public:
    int N, K, M;
    vector<Node> nodes;
    vector<int64> w;
    vector<vector<int>> starOrder;
    vector<int> relayOrder;
    chrono::steady_clock::time_point startTime;
    Tree best;

    explicit Solver(vector<Node> inputNodes, int n, int k) : N(n), K(k) {
        for (const auto &node : inputNodes)
            if (node.type != 'C') nodes.push_back(node);
        for (const auto &node : inputNodes)
            if (node.type == 'C') nodes.push_back(node);
        M = N + K;
        startTime = chrono::steady_clock::now();
    }

    bool timeUp() const {
        return chrono::duration<double>(chrono::steady_clock::now() - startTime).count() > 8.5;
    }

    inline int64 weight(int u, int v) const {
        return w[(size_t)u * M + v];
    }

    void buildWeights() {
        w.assign((size_t)M * M, 0);
        for (int i = 0; i < M; ++i) {
            for (int j = i + 1; j < M; ++j) {
                int64 value = INF;
                if (!(i >= N && j >= N)) {
                    int64 dx = nodes[i].x - nodes[j].x;
                    int64 dy = nodes[i].y - nodes[j].y;
                    int64 d2 = dx * dx + dy * dy;
                    int factor = 10;
                    if (i < N && j < N &&
                        (nodes[i].type == 'S' || nodes[j].type == 'S'))
                        factor = 8;
                    value = d2 * factor;
                }
                w[(size_t)i * M + j] = value;
                w[(size_t)j * M + i] = value;
            }
        }
    }

    vector<Edge> mst(const vector<int> &vertices, int relayPercent) const {
        int n = (int)vertices.size();
        vector<Edge> result;
        if (n <= 1) return result;

        vector<int64> key(n, INF);
        vector<int> parent(n, -1);
        vector<char> used(n, false);
        key[0] = 0;
        result.reserve(n - 1);

        for (int it = 0; it < n; ++it) {
            int u = -1;
            for (int i = 0; i < n; ++i) {
                if (!used[i] && (u == -1 || key[i] < key[u])) u = i;
            }
            if (u == -1 || key[u] == INF) break;
            used[u] = true;
            if (parent[u] != -1)
                result.push_back({vertices[u], vertices[parent[u]]});

            int gu = vertices[u];
            for (int v = 0; v < n; ++v) {
                if (used[v]) continue;
                int gv = vertices[v];
                int64 value = weight(gu, gv);
                if (value == INF) continue;
                if (relayPercent != 100 && (gu >= N || gv >= N))
                    value = value * relayPercent / 100;
                if (value < key[v]) {
                    key[v] = value;
                    parent[v] = u;
                }
            }
        }
        return result;
    }

    Tree pruneRelayLeaves(const vector<Edge> &edges) const {
        Tree result;
        int E = (int)edges.size();
        if (E == 0) return result;

        struct Arc {
            int to, id, next;
        };
        vector<int> degree(M, 0), head(M, -1), queueVertices;
        vector<Arc> arcs;
        arcs.reserve(2 * E);
        vector<char> removed(E, false);

        for (int i = 0; i < E; ++i) {
            int u = edges[i].u, v = edges[i].v;
            ++degree[u];
            ++degree[v];
            arcs.push_back({v, i, head[u]});
            head[u] = (int)arcs.size() - 1;
            arcs.push_back({u, i, head[v]});
            head[v] = (int)arcs.size() - 1;
        }

        for (int i = N; i < M; ++i)
            if (degree[i] == 1) queueVertices.push_back(i);

        for (size_t q = 0; q < queueVertices.size(); ++q) {
            int u = queueVertices[q];
            if (degree[u] != 1) continue;
            for (int a = head[u]; a != -1; a = arcs[a].next) {
                int id = arcs[a].id;
                if (removed[id]) continue;
                removed[id] = true;
                int v = arcs[a].to;
                --degree[u];
                --degree[v];
                if (v >= N && degree[v] == 1) queueVertices.push_back(v);
                break;
            }
        }

        result.edges.reserve(E);
        for (int i = 0; i < E; ++i) {
            if (!removed[i]) {
                result.edges.push_back(edges[i]);
                result.cost += weight(edges[i].u, edges[i].v);
            }
        }
        return result;
    }

    int activeRelayCount(const Tree &t) const {
        vector<char> active(K, false);
        int count = 0;
        for (const auto &e : t.edges) {
            if (e.u >= N && !active[e.u - N]) {
                active[e.u - N] = true;
                ++count;
            }
            if (e.v >= N && !active[e.v - N]) {
                active[e.v - N] = true;
                ++count;
            }
        }
        return count;
    }

    Tree normalize(Tree t) const {
        for (int iteration = 0; iteration < 4; ++iteration) {
            vector<char> active(K, false);
            for (const auto &e : t.edges) {
                if (e.u >= N) active[e.u - N] = true;
                if (e.v >= N) active[e.v - N] = true;
            }

            vector<int> vertices(N);
            iota(vertices.begin(), vertices.end(), 0);
            int oldActive = 0;
            for (int c = 0; c < K; ++c) {
                if (active[c]) {
                    vertices.push_back(N + c);
                    ++oldActive;
                }
            }

            Tree next = pruneRelayLeaves(mst(vertices, 100));
            if (next.cost > t.cost) break;
            int newActive = activeRelayCount(next);
            t = std::move(next);
            if (newActive >= oldActive) break;
        }
        return t;
    }

    void buildRelayOrders() {
        starOrder.assign(K, {});
        vector<int64> score(K, 0);

        for (int c = 0; c < K; ++c) {
            int g = N + c;
            starOrder[c].resize(N);
            iota(starOrder[c].begin(), starOrder[c].end(), 0);
            sort(starOrder[c].begin(), starOrder[c].end(), [&](int a, int b) {
                int64 wa = weight(g, a), wb = weight(g, b);
                if (wa != wb) return wa < wb;
                return a < b;
            });
            if (N >= 2)
                score[c] = weight(g, starOrder[c][0]) + weight(g, starOrder[c][1]);
            else if (N == 1)
                score[c] = weight(g, starOrder[c][0]);
        }

        relayOrder.resize(K);
        iota(relayOrder.begin(), relayOrder.end(), 0);
        sort(relayOrder.begin(), relayOrder.end(), [&](int a, int b) {
            if (score[a] != score[b]) return score[a] < score[b];
            return a < b;
        });
    }

    bool insertionPass(Tree &current, const vector<int> &order) {
        bool improved = false;
        vector<char> active(M, false);
        for (const auto &e : current.edges) {
            if (e.u >= N) active[e.u] = true;
            if (e.v >= N) active[e.v] = true;
        }

        vector<Edge> sortedTree = current.edges;
        auto sortTree = [&]() {
            sort(sortedTree.begin(), sortedTree.end(), [&](const Edge &a, const Edge &b) {
                int64 wa = weight(a.u, a.v), wb = weight(b.u, b.v);
                if (wa != wb) return wa < wb;
                if (a.u != b.u) return a.u < b.u;
                return a.v < b.v;
            });
        };
        sortTree();

        for (int c : order) {
            if (timeUp() || current.cost == 0) break;
            int g = N + c;
            if (active[g]) continue;

            DSU dsu(M);
            int target = (int)current.edges.size() + 1;
            int selected = 0;
            size_t i = 0;
            int j = 0;
            vector<Edge> candidateEdges;
            candidateEdges.reserve(target);

            while (selected < target &&
                   (i < sortedTree.size() || j < N)) {
                Edge e;
                if (i >= sortedTree.size()) {
                    e = {g, starOrder[c][j++]};
                } else if (j >= N) {
                    e = sortedTree[i++];
                } else {
                    int64 treeW = weight(sortedTree[i].u, sortedTree[i].v);
                    int64 starW = weight(g, starOrder[c][j]);
                    if (treeW <= starW) e = sortedTree[i++];
                    else e = {g, starOrder[c][j++]};
                }

                if (dsu.unite(e.u, e.v)) {
                    candidateEdges.push_back(e);
                    ++selected;
                }
            }

            if (selected != target) continue;
            Tree candidate = pruneRelayLeaves(candidateEdges);
            if (candidate.cost < current.cost) {
                current = std::move(candidate);
                improved = true;

                fill(active.begin(), active.end(), false);
                for (const auto &e : current.edges) {
                    if (e.u >= N) active[e.u] = true;
                    if (e.v >= N) active[e.v] = true;
                }
                sortedTree = current.edges;
                sortTree();
            }
        }
        return improved;
    }

    vector<Edge> connectNeighbors(const vector<int> &neighbors) const {
        int d = (int)neighbors.size();
        vector<Edge> added;
        if (d <= 1) return added;

        vector<int64> key(d, INF);
        vector<int> parent(d, -1);
        vector<char> used(d, false);
        key[0] = 0;
        added.reserve(d - 1);

        for (int it = 0; it < d; ++it) {
            int u = -1;
            for (int i = 0; i < d; ++i)
                if (!used[i] && (u == -1 || key[i] < key[u])) u = i;
            used[u] = true;
            if (parent[u] != -1)
                added.push_back({neighbors[u], neighbors[parent[u]]});

            for (int v = 0; v < d; ++v) {
                if (used[v]) continue;
                int64 value = weight(neighbors[u], neighbors[v]);
                if (value < key[v]) {
                    key[v] = value;
                    parent[v] = u;
                }
            }
        }
        return added;
    }

    bool deletionPass(Tree &current, const vector<int> &order) {
        bool improved = false;
        vector<char> active(M, false);
        for (const auto &e : current.edges) {
            if (e.u >= N) active[e.u] = true;
            if (e.v >= N) active[e.v] = true;
        }

        for (int c : order) {
            if (timeUp() || current.cost == 0) break;
            int g = N + c;
            if (!active[g]) continue;

            vector<Edge> remaining;
            vector<int> neighbors;
            remaining.reserve(current.edges.size());

            for (const auto &e : current.edges) {
                if (e.u == g || e.v == g) {
                    neighbors.push_back(e.u == g ? e.v : e.u);
                } else {
                    remaining.push_back(e);
                }
            }
            if (neighbors.empty()) continue;

            vector<Edge> added = connectNeighbors(neighbors);
            remaining.insert(remaining.end(), added.begin(), added.end());
            Tree candidate = pruneRelayLeaves(remaining);

            if (candidate.cost < current.cost) {
                current = std::move(candidate);
                improved = true;
                fill(active.begin(), active.end(), false);
                for (const auto &e : current.edges) {
                    if (e.u >= N) active[e.u] = true;
                    if (e.v >= N) active[e.v] = true;
                }
            }
        }
        return improved;
    }

    Tree localSearch(Tree current, const vector<int> &order) {
        current = normalize(current);
        for (int round = 0; round < 5 && !timeUp() && current.cost > 0; ++round) {
            int64 before = current.cost;
            insertionPass(current, order);
            if (!timeUp()) current = normalize(current);
            if (!timeUp()) {
                deletionPass(current, order);
                current = normalize(current);
            }
            if (current.cost >= before) break;
        }
        return current;
    }

    void updateBest(const Tree &t) {
        if (t.cost < best.cost ||
            (t.cost == best.cost && t.edges.size() < best.edges.size()))
            best = t;
    }

    void solve() {
        buildWeights();

        vector<int> robots(N);
        iota(robots.begin(), robots.end(), 0);
        Tree base = pruneRelayLeaves(mst(robots, 100));
        best = base;

        if (base.cost == 0 || K == 0) return;

        buildRelayOrders();

        Tree initial = localSearch(base, relayOrder);
        updateBest(initial);
        if (best.cost == 0) return;

        vector<Tree> seeds;
        vector<int> allVertices(M);
        iota(allVertices.begin(), allVertices.end(), 0);
        const int percentages[] = {65, 80, 100, 115, 135, 165, 220};

        for (int percent : percentages) {
            if (timeUp()) break;
            Tree seed = pruneRelayLeaves(mst(allVertices, percent));
            seed = normalize(seed);
            updateBest(seed);
            seeds.push_back(std::move(seed));
        }

        if (!timeUp() && K >= 3) {
            int choices[2] = {K / 3, 2 * K / 3};
            for (int q : choices) {
                if (q <= 0 || timeUp()) continue;
                vector<int> vertices = robots;
                for (int i = 0; i < q; ++i)
                    vertices.push_back(N + relayOrder[i]);
                Tree seed = pruneRelayLeaves(mst(vertices, 100));
                seed = normalize(seed);
                updateBest(seed);
                seeds.push_back(std::move(seed));
            }
        }

        sort(seeds.begin(), seeds.end(), [](const Tree &a, const Tree &b) {
            if (a.cost != b.cost) return a.cost < b.cost;
            return a.edges.size() < b.edges.size();
        });

        int usedSeeds = 0;
        for (const auto &seed : seeds) {
            if (usedSeeds >= 4 || timeUp() || best.cost == 0) break;
            Tree candidate = localSearch(seed, relayOrder);
            updateBest(candidate);
            ++usedSeeds;
        }

        mt19937 rng(123456789);
        for (int attempt = 0; attempt < 3 && !timeUp() && best.cost > 0; ++attempt) {
            vector<int> randomOrder = relayOrder;
            shuffle(randomOrder.begin(), randomOrder.end(), rng);
            Tree candidate = localSearch(base, randomOrder);
            updateBest(candidate);
        }
    }

    void output() const {
        vector<char> selected(K, false);
        for (const auto &e : best.edges) {
            if (e.u >= N) selected[e.u - N] = true;
            if (e.v >= N) selected[e.v - N] = true;
        }

        bool any = false, first = true;
        for (int c = 0; c < K; ++c) {
            if (!selected[c]) continue;
            any = true;
            if (!first) cout << '#';
            first = false;
            cout << nodes[N + c].id;
        }
        if (!any) cout << '#';
        cout << '\n';

        for (size_t i = 0; i < best.edges.size(); ++i) {
            if (i) cout << '#';
            cout << nodes[best.edges[i].u].id << '-'
                 << nodes[best.edges[i].v].id;
        }
        cout << '\n';
    }
};

struct DSU {
    vector<int> p;
    explicit DSU(int n) : p(n, -1) {}

    int find(int x) {
        while (p[x] >= 0) {
            if (p[p[x]] >= 0) p[x] = p[p[x]];
            x = p[x];
        }
        return x;
    }

    bool unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return false;
        if (p[a] > p[b]) swap(a, b);
        p[a] += p[b];
        p[b] = a;
        return true;
    }
};

int64 edgeWeight(int u, int v) {
    int64 dx = nodes[u].x - nodes[v].x;
    int64 dy = nodes[u].y - nodes[v].y;
    int64 d2 = dx * dx + dy * dy;

    char a = nodes[u].type;
    char b = nodes[v].type;
    if (a == 'C' && b == 'C') return INF;
    if (a == 'C' || b == 'C') return 5 * d2;

Solution pruneRelayLeaves(Solution s) {
    int E = (int)s.edges.size();
    vector<vector<int>> adj(M);
    vector<int> degree(M, 0);
    vector<char> removed(E, 0);

    s.active.assign(M, 0);
    for (int i = 0; i < N; ++i) s.active[i] = 1;

    for (int i = 0; i < E; ++i) {
        int u = s.edges[i].u;
        int v = s.edges[i].v;
        adj[u].push_back(i);
        adj[v].push_back(i);
        ++degree[u];
        ++degree[v];
        s.active[u] = 1;
        s.active[v] = 1;
    }

    queue<int> q;
    for (int r = N; r < M; ++r) {
        if (s.active[r] && degree[r] <= 1) q.push(r);

    if (D[a].tp == 'C' || D[b].tp == 'C') return 20 * d;
    if (D[a].tp == 'S' || D[b].tp == 'S') return 12 * d;
    return 20 * d;
}

// ---------------- kd-tree ----------------
struct KDTree {
    struct Nd { ll x, y; int id; int l = -1, r = -1; int ax = 0; };
    vector<Nd> t;
    int root = -1;
    const vector<char>* al = nullptr;

    int build(vector<Nd>& v, int l, int r, int dep) {
        if (l >= r) return -1;
        int mid = (l + r) >> 1, ax = dep & 1;
        nth_element(v.begin() + l, v.begin() + mid, v.begin() + r,
                    [ax](const Nd& a, const Nd& b) { return ax ? (a.y < b.y) : (a.x < b.x); });
        int id = (int)t.size();
        t.push_back(v[mid]);
        t[id].ax = ax;
        t[id].l = build(v, l, mid, dep + 1);
        t[id].r = build(v, mid + 1, r, dep + 1);
        return id;
    }
    void init(vector<Nd>& v, const vector<char>* alive) {
        al = alive;
        t.clear();
        t.reserve(v.size() + 1);
        root = build(v, 0, (int)v.size(), 0);
    }
    void srch(int n, ll qx, ll qy, ll& best, int& bid) const {
        const Nd& nd = t[n];
        ll dx = qx - nd.x, dy = qy - nd.y, d = dx * dx + dy * dy;
        if ((*al)[nd.id] && d < best) { best = d; bid = nd.id; }
        ll delta = nd.ax ? (qy - nd.y) : (qx - nd.x);
        int n1 = delta < 0 ? nd.l : nd.r;
        int n2 = delta < 0 ? nd.r : nd.l;
        if (n1 >= 0) srch(n1, qx, qy, best, bid);
        if (n2 >= 0) { ll bd = delta * delta; if (bd < best) srch(n2, qx, qy, best, bid); }
    }
    int nearest(ll qx, ll qy, ll& bd) const {
        ll b = INF; int id = -1;
        if (root >= 0) srch(root, qx, qy, b, id);
        bd = b; return id;
    }
    void ksrch(int n, ll qx, ll qy, priority_queue<pair<ll,int>>& h, int k) const {
        const Nd& nd = t[n];
        ll dx = qx - nd.x, dy = qy - nd.y, d = dx * dx + dy * dy;
        if ((*al)[nd.id]) {
            if ((int)h.size() < k) h.push({d, nd.id});
            else if (d < h.top().first) { h.pop(); h.push({d, nd.id}); }
        }
        ll delta = nd.ax ? (qy - nd.y) : (qx - nd.x);
        int n1 = delta < 0 ? nd.l : nd.r;
        int n2 = delta < 0 ? nd.r : nd.l;
        if (n1 >= 0) ksrch(n1, qx, qy, h, k);
        if (n2 >= 0) { ll bd = delta * delta; if ((int)h.size() < k || bd < h.top().first) ksrch(n2, qx, qy, h, k); }
    }
    void knearest(ll qx, ll qy, int k, vector<int>& out) const {
        priority_queue<pair<ll,int>> h;
        if (root >= 0) ksrch(root, qx, qy, h, k);
        out.clear();
        while (!h.empty()) { out.push_back(h.top().second); h.pop(); }
        reverse(out.begin(), out.end());
    }
};

KDTree kdRob, kdRel, kdAct;

void rebuildRelayKD() {
    vector<KDTree::Nd> pts;
    for (int r : relays) if (relayAlive[r])
        pts.push_back({2 * D[r].x, 2 * D[r].y, r, -1, -1, 0});
    kdRel.init(pts, &relayAlive);
}

// ---------------- union find / mst ----------------
struct UFD {
    vector<int> p;
    void init(int n) { p.resize(n); iota(p.begin(), p.end(), 0); }
    int find(int x) { while (p[x] != x) { p[x] = p[p[x]]; x = p[x]; } return x; }
    bool uni(int a, int b) { a = find(a); b = find(b); if (a == b) return false; p[a] = b; return true; }
};

ll primMST(const vector<int>& vs, vector<TE>& out) {
    out.clear();
    int m = (int)vs.size();
    if (m <= 1) return 0;
    vector<ll> best(m, INF); vector<int> par(m, -1); vector<char> done(m, 0);
    best[0] = 0; ll tot = 0;
    for (int it = 0; it < m; it++) {
        int bi = -1; ll bv = INF;
        for (int i = 0; i < m; i++) if (!done[i] && best[i] < bv) { bv = best[i]; bi = i; }
        if (bi < 0) break;
        done[bi] = 1; tot += bv;
        if (par[bi] >= 0) out.push_back({vs[bi], vs[par[bi]], bv});
        for (int i = 0; i < m; i++) if (!done[i]) {
            int a = vs[bi], b = vs[i];
            if (D[a].tp == 'C' && D[b].tp == 'C') continue;
            ll w = Wf(a, b);
            if (w < best[i]) { best[i] = w; par[i] = bi; }
        }
    }
    return tot;
}

// merged kruskal of two edge lists already sorted by weight
ll kruskalMerge(const vector<TE>& A, const vector<TE>& B, vector<TE>& out) {
    out.clear();
    UFD uf; uf.init(M);
    size_t i = 0, j = 0; ll tot = 0;
    while (i < A.size() || j < B.size()) {
        TE e;
        if (j >= B.size() || (i < A.size() && A[i].w <= B[j].w)) e = A[i++];
        else e = B[j++];
        if (D[e.u].tp == 'C' && D[e.v].tp == 'C') continue;
        if (uf.uni(e.u, e.v)) { out.push_back(e); tot += e.w; }
    }
    return tot;
}

void setTree(const vector<TE>& es) {
    gEdges = es; gCost = 0;
    for (auto& e : gEdges) gCost += e.w;
}

// exact MST over the currently active node set (robots + used relays)
void exactMST() {
    vector<int> act;
    act.reserve(robots.size() + relays.size());
    for (int r : robots) act.push_back(r);
    for (int r : relays) if (!relayAlive[r]) act.push_back(r);
    vector<TE> es;
    ll c = primMST(act, es);
    if (c <= gCost) setTree(es);
}

// remove relays that ended up as useless leaves
void pruneLeaves() {
    bool changed = true;
    while (changed) {
        changed = false;
        vector<int> deg(M, 0);
        for (auto& e : gEdges) { deg[e.u]++; deg[e.v]++; }
        for (int r : relays) {
            if (relayAlive[r]) continue;
            if (deg[r] == 1) {
                for (size_t i = 0; i < gEdges.size(); i++) {
                    if (gEdges[i].u == r || gEdges[i].v == r) {
                        gCost -= gEdges[i].w;
                        gEdges.erase(gEdges.begin() + i);
                        break;
                    }
                }
                relayAlive[r] = 1;
                changed = true;
                break;
            }
        }
    }
}

// ---------------- robot level structure (sparse, relay aware) ----------------
vector<TE> buildRobotLevel() {
    vector<TE> cand;
    // plain robot MST edges (guarantees connectivity + good baseline)
    {
        vector<TE> es; primMST(robots, es);
        for (auto& e : es) cand.push_back(e);
    }
    // near robot-robot direct edges
    for (int u : robots) {
        vector<int> nr; kdRob.knearest(D[u].x, D[u].y, 9, nr);
        for (int v : nr) if (v != u) cand.push_back({u, v, Wf(u, v)});
    }
    // relay mediated robot pairs
    for (int c : relays) {
        vector<int> nr; kdRob.knearest(D[c].x, D[c].y, 8, nr);
        int k = (int)nr.size();
        for (int i = 0; i < k; i++)
            for (int j = i + 1; j < k; j++) {
                int u = nr[i], v = nr[j];
                ll w = 40 * (d2f(u, c) + d2f(c, v));
                cand.push_back({u, v, w});
            }
    }
    sort(cand.begin(), cand.end(), [](const TE& a, const TE& b) { return a.w < b.w; });
    UFD uf; uf.init(M);
    vector<TE> tree;
    for (auto& e : cand) if (uf.uni(e.u, e.v)) tree.push_back(e);
    if ((int)tree.size() != (int)robots.size() - 1) {
        vector<TE> es; primMST(robots, es);
        for (auto& e : es) if (uf.uni(e.u, e.v)) tree.push_back(e);
    }
    return tree;
}

// ---------------- greedy relay subdivision ----------------
void phaseA() {
    vector<int> rl;
    for (int i = 0; i < (int)gEdges.size(); i++) {
        int u = gEdges[i].u, v = gEdges[i].v;
        if (D[u].tp != 'C' && D[v].tp != 'C') rl.push_back(i);
    }
    sort(rl.begin(), rl.end(), [](int a, int b) {
        return Af(gEdges[a].u, gEdges[a].v) > Af(gEdges[b].u, gEdges[b].v);
    });
    int since = 0;
    while (!rl.empty() && !timeUp()) {
        ll bestGain = 0; int bestPos = -1, bestC = -1;
        for (int pos = 0; pos < (int)rl.size(); pos++) {
            TE e = gEdges[rl[pos]];
            ll A = Af(e.u, e.v);
            if (A <= bestGain) break;
            ll qx = D[e.u].x + D[e.v].x, qy = D[e.u].y + D[e.v].y;
            ll dd; int c = kdRel.nearest(qx, qy, dd);
            if (c < 0) continue;
            ll gain = A - 20 * dd;
            if (gain > bestGain) { bestGain = gain; bestPos = pos; bestC = c; }
        }
        if (bestPos < 0) break;
        int ei = rl[bestPos]; TE e = gEdges[ei];
        ll w1 = Wf(e.u, bestC), w2 = Wf(bestC, e.v);
        gEdges[ei] = {e.u, bestC, w1};
        gEdges.push_back({bestC, e.v, w2});
        gCost += w1 + w2 - e.w;
        rl.erase(rl.begin() + bestPos);
        relayAlive[bestC] = 0;
        if (++since >= 40) { rebuildRelayKD(); since = 0; }
    }
}

// ---------------- exact insertion of a free relay (hub / restructure aware) ----
bool b1Step() {
    vector<int> act;
    act.reserve(robots.size() + relays.size());
    for (int r : robots) act.push_back(r);
    for (int r : relays) if (!relayAlive[r]) act.push_back(r);
    if (act.empty()) return false;

    for (int i = 0; i < M; i++) nodeIn[i] = 0;
    for (int a : act) nodeIn[a] = 1;
    {
        vector<KDTree::Nd> pts; pts.reserve(act.size());
        for (int a : act) pts.push_back({D[a].x, D[a].y, a, -1, -1, 0});
        kdAct.init(pts, &nodeIn);
    }
    vector<TE> st = gEdges;
    sort(st.begin(), st.end(), [](const TE& a, const TE& b) { return a.w < b.w; });
    ll maxW = st.empty() ? 0 : st.back().w;

    ll bestDelta = 0; int bestC = -1; vector<TE> bestTree;
    vector<int> nr; vector<TE> star; vector<TE> out;
    for (int c : relays) {
        if (!relayAlive[c]) continue;
        kdAct.knearest(D[c].x, D[c].y, 12, nr);
        if (nr.empty()) continue;
        ll mn = INF;
        for (int x : nr) mn = min(mn, Wf(c, x));
        if (mn >= maxW) continue;          // cannot possibly improve
        star.clear();
        for (int x : nr) star.push_back({c, x, Wf(c, x)});
        sort(star.begin(), star.end(), [](const TE& a, const TE& b) { return a.w < b.w; });
        ll cost = kruskalMerge(st, star, out);
        ll delta = cost - gCost;
        if (delta < bestDelta) { bestDelta = delta; bestC = c; bestTree = out; }
    }
    if (bestC < 0) return false;
    setTree(bestTree);
    relayAlive[bestC] = 0;
    return true;
}

    if (!(cin >> N >> K)) return 0;
    M = N + K;
    D.resize(M);
    for (int i = 0; i < M; i++) {
        int id; ll x, y; char t;
        cin >> id >> x >> y >> t;
        D[i] = {x, y, id, t};

    for (int i = 0; i < M; i++) {
        if (D[i].tp == 'C') relays.push_back(i); else robots.push_back(i);

    for (int r : relays) relayAlive[r] = 1;
    for (int r : robots) robotAlive[r] = 1;

    {
        vector<KDTree::Nd> pts; pts.reserve(robots.size());
        for (int r : robots) pts.push_back({D[r].x, D[r].y, r, -1, -1, 0});
        kdRob.init(pts, &robotAlive);
    }
    rebuildRelayKD();

    setTree(buildRobotLevel());

    phaseA();

        ll c0 = gCost;

        if (a > b) swap(a, b);

    if (links.empty()) cout << '\n';

    }
