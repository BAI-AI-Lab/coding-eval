import sys

MOD = 1_000_000_007
MOD2 = 2 * MOD


def main():
    data = sys.stdin.buffer.read().split()
    t = int(data[0])
    answers = []

    for i in range(1, t + 1):
        s = data[i]
        n = 0
        transformed = False

        # Interpreting the parity digits as a binary number gives n.
        # If no transformation is needed, these digits equal the original ones.
        for c in s:
            if c > 49:  # A digit other than '0' or '1'
                transformed = True
            n = ((n << 1) + (c & 1)) % MOD2

        # The number of operations after obtaining n is floor(3*n/2).
        # Using n modulo 2*MOD preserves both the result modulo MOD and parity.
        ans = (3 * n - (n & 1)) // 2
        if transformed:
            ans += 1

        answers.append(str(ans % MOD))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

INV2 = (MOD + 1) // 2


def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    t = int(data[0])
    out = []
    idx = 1

    for _ in range(t):
        s = data[idx]
        idx += 1

        b = 0
        has_non_binary_digit = False

        for ch in s:
            d = ch - 48
            parity = d & 1
            b = (b * 2 + parity) % MOD
            if d > 1:
                has_non_binary_digit = True

        last_parity = (s[-1] - 48) & 1

        if last_parity:
            h = ((3 * b - 1) % MOD) * INV2 % MOD
        else:
            h = (3 * b % MOD) * INV2 % MOD

        if has_non_binary_digit:
            h += 1

        out.append(str(h % MOD))

    sys.stdout.write("\n".join(out))

    solve()

import sys

MOD = 1_000_000_007
INV2 = 500_000_004


def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    t = int(data[0])
    answers = []

    for i in range(1, t + 1):
        s = data[i]

        b_mod = 0
        has_invalid_digit = False

        for ch in s:
            digit = ch - 48  # ASCII digit to integer

            if digit > 1:
                has_invalid_digit = True

            # Parity of the original digit.
            bit = digit & 1

            # Interpret the parity digits as a binary number.
            b_mod = (2 * b_mod + bit) % MOD

        # The parity of b is the parity of its last binary digit.
        parity = (s[-1] - 48) & 1

        # floor(3b / 2) = (3b - (b mod 2)) / 2
        answer = ((3 * b_mod - parity) * INV2) % MOD

        if has_invalid_digit:
            answer = (answer + 1) % MOD

        answers.append(str(answer))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()
