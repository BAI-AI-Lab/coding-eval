import sys


def min_operations(a, b, c, d):
    if c < a or d < b:
        return -1

    x, y = c, d
    operations = 0

    # Reverse the operations. Outside the initial state, the larger
    # coordinate must have been the one most recently increased.
    while x != a or y != b:
        if x < a or y < b or x == y:
            return -1

        if x == a:
            difference = y - b
            if difference % a:
                return -1
            return operations + difference // a

        if y == b:
            difference = x - a
            if difference % b:
                return -1
            return operations + difference // b

        if x > y:
            count = (x - 1) // y
            x -= count * y
        else:
            count = (y - 1) // x
            y -= count * x

        operations += count

    return operations


def main():
    data = sys.stdin.buffer.read().split()
    t = int(data[0])
    answers = []
    index = 1

    for _ in range(t):
        a = int(data[index])
        b = int(data[index + 1])
        c = int(data[index + 2])
        d = int(data[index + 3])
        index += 4
        answers.append(str(min_operations(a, b, c, d)))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

def min_operations(a, b, c, d):

        return -1

    x, y = c, d
    operations = 0

    while True:
        if x == a and y == b:
            return operations

            return -1

import sys


def minimum_operations(a: int, b: int, c: int, d: int) -> int:
    x, y = c, d
    operations = 0

    while True:
        # We have reached the initial pair.
        if x == a and y == b:
            return operations

        # Reverse operations never increase a coordinate.
        # Equal piles have no valid predecessor.
        if x < a or y < b or x == y:
            return -1

        if x > y:
            # The initial pair may occur before the end of this
            # batch of subtractions.
            if y == b and x >= a and (x - a) % y == 0:
                return operations + (x - a) // y

            count = (x - 1) // y
            x -= count * y
            operations += count
        else:
            # Symmetric case: repeatedly subtract x from y.
            if x == a and y >= b and (y - b) % x == 0:
                return operations + (y - b) // x

            count = (y - 1) // x
            y -= count * x
            operations += count


def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    t = data[0]
    index = 1
    answers = []

    for _ in range(t):
        a, b, c, d = data[index:index + 4]
        index += 4
        answers.append(str(minimum_operations(a, b, c, d)))

    sys.stdout.write("\n".join(answers) + "\n")


if __name__ == "__main__":
    solve()
