import sys
from collections import deque

sys.setrecursionlimit(1000000)


class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, fr, to, cap):
        fwd = [to, cap, len(self.g[to])]
        rev = [fr, 0, len(self.g[fr])]
        self.g[fr].append(fwd)
        self.g[to].append(rev)

    def bfs(self, s, t):
        self.level = [-1] * self.n
        self.level[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            for to, cap, _ in self.g[v]:
                if cap > 0 and self.level[to] == -1:
                    self.level[to] = self.level[v] + 1
                    q.append(to)
        return self.level[t] != -1

    def dfs(self, v, t, flow):
        if v == t:
            return flow
        while self.it[v] < len(self.g[v]):
            e = self.g[v][self.it[v]]
            if e[1] > 0 and self.level[e[0]] == self.level[v] + 1:
                pushed = self.dfs(e[0], t, min(flow, e[1]))
                if pushed:
                    e[1] -= pushed
                    self.g[e[0]][e[2]][1] += pushed
                    return pushed
            self.it[v] += 1
        return 0

    def max_flow(self, s, t):
        result = 0
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                pushed = self.dfs(s, t, 10**18)
                if not pushed:
                    break
                result += pushed
        return result


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[:3]
    edges = []
    adj = [[] for _ in range(n)]
    pos = 3
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        edges.append((u, v))
        adj[u].append(v)

    # No attainable distance can exceed the unweighted shortest-path length.
    dist = [-1] * n
    dist[0] = 0
    q = deque([0])
    while q:
        v = q.popleft()
        for to in adj[v]:
            if dist[to] == -1:
                dist[to] = dist[v] + 1
                q.append(to)
    max_distance = dist[n - 1]

    inf = m + 1

    # Minimum number of edges that must have weight 1 to force distance >= d.
    def required(d):
        size = n * d
        source = size
        sink = size + 1
        flow = Dinic(size + 2)

        # Binary variables x[v][i] indicate whether label[v] >= i+1.
        # Enforce x[v][i] >= x[v][i+1].
        for v in range(n):
            base = v * d
            for i in range(d - 1):
                flow.add_edge(base + i, base + i + 1, inf)

        for u, v in edges:
            ub = u * d
            vb = v * d
            # Cost max(0, label[v] - label[u]).
            for i in range(d):
                flow.add_edge(ub + i, vb + i, 1)
            # Forbid label[v] - label[u] > 1.
            for i in range(d - 1):
                flow.add_edge(ub + i, vb + i + 1, inf)

        # label[1] = 0 and label[N] = d.
        sink_base = (n - 1) * d
        for i in range(d):
            flow.add_edge(source, i, inf)
            flow.add_edge(sink_base + i, sink, inf)

        return flow.max_flow(source, sink)

    answer = 0
    for d in range(1, max_distance + 1):
        if required(d) <= k:
            answer = d
        else:
            break

    print(answer)


if __name__ == "__main__":
    main()

import sys
from collections import deque

sys.setrecursionlimit(1 << 25)


class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, fr, to, cap):
        self.g[fr].append([to, cap, len(self.g[to])])
        self.g[to].append([fr, 0, len(self.g[fr]) - 1])

    def max_flow(self, s, t):
        flow = 0
        INF_FLOW = 10**18
        n = self.n

        while True:
            level = [-1] * n
            level[s] = 0
            q = deque([s])

            while q:
                v = q.popleft()
                for to, cap, rev in self.g[v]:
                    if cap > 0 and level[to] < 0:
                        level[to] = level[v] + 1
                        q.append(to)

            if level[t] < 0:
                break

            it = [0] * n

            def dfs(v, f):
                if v == t:
                    return f

                while it[v] < len(self.g[v]):
                    e = self.g[v][it[v]]
                    to, cap, rev = e

                    if cap > 0 and level[v] + 1 == level[to]:
                        ret = dfs(to, min(f, cap))
                        if ret > 0:
                            e[1] -= ret
                            self.g[to][rev][1] += ret
                            return ret

                    it[v] += 1

                return 0

            while True:
                f = dfs(s, INF_FLOW)
                if f == 0:
                    break
                flow += f

        return flow


def solve():
    input = sys.stdin.readline

    N, M, K = map(int, input().split())
    edges = []
    for _ in range(M):
        u, v = map(int, input().split())
        edges.append((u, v))

    def feasible(L):
        if L == 0:
            return True

        source = 0
        sink = 1
        node_count = 2 + N * L
        dinic = Dinic(node_count)

        INF = 10**7

        def idx(v, k):
            # v: 1-indexed, k: 1..L
            return 2 + (v - 1) * L + (k - 1)

        # Monotonicity: x[v][k+1] <= x[v][k]
        for v in range(1, N + 1):
            for k in range(1, L):
                dinic.add_edge(idx(v, k + 1), idx(v, k), INF)

        # h[1] = 0, h[N] = L
        for k in range(1, L + 1):
            dinic.add_edge(idx(1, k), sink, INF)
            dinic.add_edge(source, idx(N, k), INF)

        for u, v in edges:
            # Cost: max(0, h[v] - h[u]), one unit per level
            for k in range(1, L + 1):
                dinic.add_edge(idx(v, k), idx(u, k), 1)

            # Constraint: h[v] <= h[u] + 1
            for k in range(1, L):
                dinic.add_edge(idx(v, k + 1), idx(u, k), INF)

        return dinic.max_flow(source, sink) <= K

    low, high = 0, N - 1

    while low < high:
        mid = (low + high + 1) // 2
        if feasible(mid):
            low = mid
        else:
            high = mid - 1

    print(low)


if __name__ == "__main__":
    solve()

import sys
from collections import deque

sys.setrecursionlimit(1000000)


class Dinic:
    def __init__(self, n):
        self.n = n
        self.ad = [[] for _ in range(n)]

    def add_edge(self, fr, to, cap):
        fwd = [to, cap, len(self.ad[to])]
        rev = [fr, 0, len(self.ad[fr])]
        self.ad[fr].append(fwd)
        self.ad[to].append(rev)

    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, rev in self.ad[u]:
                if cap > 0 and self.level[v] < 0:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] >= 0

    def dfs(self, u, t, f):
        if u == t:
            return f
        while self.it[u] < len(self.ad[u]):
            e = self.ad[u][self.it[u]]
            v, cap, rev = e
            if cap > 0 and self.level[v] == self.level[u] + 1:
                ret = self.dfs(v, t, min(f, cap))
                if ret > 0:
                    e[1] -= ret
                    self.ad[v][rev][1] += ret
                    return ret
            self.it[u] += 1
        return 0

    def max_flow(self, s, t):
        flow = 0
        INF_FLOW = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF_FLOW)
                if f == 0:
                    break
                flow += f
        return flow


def main():
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return

    N, M, K = data[0], data[1], data[2]
    edges = []
    idx = 3
    for _ in range(M):
        u = data[idx]
        v = data[idx + 1]
        idx += 2
        edges.append((u, v))

    def feasible(D):
        if D == 0:
            return True

        node_count = N * D + 2
        SRC = node_count - 2
        SNK = node_count - 1
        dinic = Dinic(node_count)

        # INF is larger than any possible finite cut.
        INF = M * D + 1

        def vid(v, i):
            return (v - 1) * D + (i - 1)

        # p_1 = 0, p_N = D
        for i in range(1, D + 1):
            dinic.add_edge(vid(1, i), SNK, INF)   # X_{1,i} = 0
            dinic.add_edge(SRC, vid(N, i), INF)   # X_{N,i} = 1

        # X_{v,i+1} => X_{v,i}
        for v in range(1, N + 1):
            for i in range(1, D):
                dinic.add_edge(vid(v, i + 1), vid(v, i), INF)

        for u, v in edges:
            # p_v <= p_u + 1
            for i in range(1, D):
                dinic.add_edge(vid(v, i + 1), vid(u, i), INF)

            # An edge is selected iff p_v = p_u + 1.
            for i in range(1, D + 1):
                dinic.add_edge(vid(v, i), vid(u, i), 1)

        return dinic.max_flow(SRC, SNK) <= K

    for D in range(min(N - 1, K), -1, -1):
        if feasible(D):
            print(D)
            return


if __name__ == "__main__":
    main()
