import sys


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N, K = data[0], data[1]
    n = N * K

    graph = [[] for _ in range(n + 1)]
    pos = 2
    for _ in range(n - 1):
        u, v = data[pos], data[pos + 1]
        pos += 2
        graph[u].append(v)
        graph[v].append(u)

    parent = [0] * (n + 1)
    parent[1] = -1
    order = []
    stack = [1]

    while stack:
        v = stack.pop()
        order.append(v)
        for to in graph[v]:
            if to == parent[v]:
                continue
            parent[to] = v
            stack.append(to)

    # length[v] is the size of the unfinished path ending at v.
    # Zero means that the whole subtree of v has been decomposed.
    length = [0] * (n + 1)
    child_count = [0] * (n + 1)
    child_sum = [0] * (n + 1)

    possible = True

    for v in reversed(order):
        count = child_count[v]
        total_length = child_sum[v]

        if count > 2 or (count == 2 and total_length + 1 != K):
            possible = False
            break

        length[v] = (total_length + 1) % K

        if v != 1 and length[v] != 0:
            p = parent[v]
            child_count[p] += 1
            child_sum[p] += length[v]

    if possible and length[1] == 0:
        sys.stdout.write("Yes\n")
    else:
        sys.stdout.write("No\n")


if __name__ == "__main__":
    main()

def main() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N, K = data[0], data[1]
    M = N * K

    if K == 1:
        print("Yes")
        return

    adj = [[] for _ in range(M)]

    idx = 2
    for _ in range(M - 1):
        u = data[idx] - 1
        v = data[idx + 1] - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)

    parent = [-1] * M
    parent[0] = -2
    order = [0]

    for v in order:
        for to in adj[v]:
            if to == parent[v]:
                continue
            parent[to] = v
            order.append(to)

    state = [0] * M

    for v in reversed(order):
        open_lengths = []
        invalid = False

        for to in adj[v]:
            if parent[to] == v:
                st = state[to]
                if st == -1:
                    invalid = True
                    break
                if st > 0:
                    open_lengths.append(st)
                    if len(open_lengths) > 2:
                        invalid = True
                        break

        if invalid:
            state[v] = -1
            continue

        cnt = len(open_lengths)

        if cnt == 0:
            state[v] = 1
        elif cnt == 1:
            length = open_lengths[0] + 1
            if length == K:
                state[v] = 0
            elif length < K:
                state[v] = length
            else:
                state[v] = -1
        elif cnt == 2:
            length = open_lengths[0] + open_lengths[1] + 1
            if length == K:
                state[v] = 0
            else:
                state[v] = -1
        else:
            state[v] = -1

    print("Yes" if state[0] == 0 else "No")


if __name__ == "__main__":
    main()

import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    it = iter(data)
    N = int(next(it))
    K = int(next(it))
    n = N * K

    adj = [[] for _ in range(n)]
    for _ in range(n - 1):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # Root the tree at 0
    parent = [-1] * n
    visited = [False] * n
    order = [0]
    visited[0] = True

    idx = 0
    while idx < len(order):
        v = order[idx]
        idx += 1
        for to in adj[v]:
            if not visited[to]:
                visited[to] = True
                parent[to] = v
                order.append(to)

    # Subtree sizes
    sz = [1] * n
    for v in reversed(order):
        for to in adj[v]:
            if parent[to] == v:
                sz[v] += sz[to]

    # In any valid decomposition, an edge is used iff the subtree side
    # size is not divisible by K.
    selected_parent = [False] * n
    deg = [0] * n

    for v in range(1, n):
        if sz[v] % K != 0:
            selected_parent[v] = True
            p = parent[v]
            deg[v] += 1
            deg[p] += 1

    # Used edges must form paths, so maximum degree is at most 2.
    if any(d > 2 for d in deg):
        print("No")
        return

    # Check that every connected component of the used edges has K vertices.
    comp_visited = [False] * n
    for start in range(n):
        if comp_visited[start]:
            continue

        stack = [start]
        comp_visited[start] = True
        cnt = 0

        while stack:
            v = stack.pop()
            cnt += 1

            p = parent[v]
            if p != -1 and selected_parent[v]:
                if not comp_visited[p]:
                    comp_visited[p] = True
                    stack.append(p)

            for to in adj[v]:
                if parent[to] == v and selected_parent[to]:
                    if not comp_visited[to]:
                        comp_visited[to] = True
                        stack.append(to)

        if cnt != K:
            print("No")
            return

    print("Yes")


if __name__ == "__main__":
    solve()
