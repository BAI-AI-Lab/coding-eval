#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        vector<pair<int64, int64>> jobs;
        jobs.reserve(N);

        for (int i = 0; i < N; ++i) {
            int64 s, t;
            cin >> s >> t;
            // Starting by s is equivalent to completing by s + t.
            jobs.push_back({s + t, t});
        }

        sort(jobs.begin(), jobs.end(),
             [](const auto& a, const auto& b) {
                 return a.first < b.first;
             });

        priority_queue<int64> durations;
        int64 elapsed = 0;

        for (const auto& [deadline, duration] : jobs) {
            elapsed += duration;
            durations.push(duration);

            while (elapsed > deadline) {
                elapsed -= durations.top();
                durations.pop();
            }
        }

        cout << durations.size() << '\n';
    }

    return 0;
}
