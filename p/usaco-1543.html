题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Bessie is designing a nuclear reactor to power Farmer John's lucrative new AI data center business, CowWeave!

The reactor core consists of $N$ ($1\le N \le 2\cdot 10^5$) fuel rods, numbered $1$ through $N$. The $i$-th rod has a "stable operating range" $[l_i, r_i]$ ($-10^9 \leq l_i \leq r_i \leq 10^9$), meaning it can only generate power if its energy $a_i$ (chosen by Bessie) satisfies $l_i \le a_i \le r_i$; otherwise, it sits idle and does not generate power. Moreover, $a_i$ must always be an integer. Note that $a_i$ can be any integer, not limited to $[-10^9, 10^9]$.

However, quantum interactions between the rods mean that there are $M$ constraints of the form $(x, y, z)$ where Bessie must satisfy $a_x + a_y = z$ ($1 \leq x,y \leq N$ and $-10^9\le z\le 10^9$) to prevent the reactor from melting down.

Help Bessie find the maximum number of power-generating rods she can achieve in her design without it melting down!

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $T$ ($1\le T\le 10$), the number of independent tests. Each test is specified in the following format:

  • The first line contains the two integers $N$ and $M$.
  • The second line contains the $N$ integers $l_1, \dots, l_N$.
  • The third line contains the $N$ integers $r_1, \dots, r_N$.
  • The next $M$ lines each contain three integers $x$, $y$, and $z$, each representing a constraint.

It is guaranteed that neither the sum of $N$ nor the sum of $M$ over all tests exceeds $4\cdot 10^5$.

OUTPUT FORMAT (print output to the terminal / stdout):

If no choice of rod energies exists that satisfies all constraints, output $-1$. Otherwise, output the maximum number of power-generating rods Bessie can achieve.

In the second test, the constraints require that:

  1. $a_1 + a_1 = 2$
  2. $a_2 + a_2 = 10$

Choosing energies $a=[1, 5, 3]$ results in $2$ power-generating rods because:

  • $l_1 = 1 \leq a_1 \leq 1 = r_1$
  • $l_3 = 3 \leq a_3 \leq 3 = r_3$

and $a$ satisfies all required constraints.

import sys
from array import array


def read_integers(data):
    i = 0
    n = len(data)
    while i < n:
        while i < n and data[i] <= 32:
            i += 1
        if i >= n:
            break
        sign = 1
        if data[i] == 45:
            sign = -1
            i += 1
        value = 0
        while i < n and data[i] > 32:
            value = value * 10 + data[i] - 48
            i += 1
        yield sign * value


def solve_test(n, left, right, head, to, nxt, edge_z):
    # sign: 0 = unvisited, 1 = a_i = b_i + t, 2 = a_i = b_i - t
    sign = bytearray(n)
    b = [0] * n
    answer = 0

    for root in range(n):
        if sign[root]:
            continue

        sign[root] = 1
        b[root] = 0
        component = [root]
        q = 0
        fixed_t = None

        while q < len(component):
            u = component[q]
            q += 1
            su = sign[u]
            su_value = 1 if su == 1 else -1
            bu = b[u]

            e = head[u]
            while e != -1:
                v = to[e]
                z = edge_z[e]
                sv = sign[v]

                if sv == 0:
                    sign[v] = 3 - su
                    b[v] = z - bu
                    component.append(v)
                elif sv != su:
                    # The parameter cancels, leaving a constant equation.
                    if bu + b[v] != z:
                        return -1
                else:
                    # 2 * su_value * t + bu + b[v] = z.
                    numerator = z - bu - b[v]
                    if numerator & 1:
                        return -1
                    t_value = (numerator // 2) * su_value
                    if fixed_t is None:
                        fixed_t = t_value
                    elif fixed_t != t_value:
                        return -1

                e = nxt[e]

        if fixed_t is not None:
            # This component has exactly one integer solution.
            for u in component:
                if sign[u] == 1:
                    value = b[u] + fixed_t
                else:
                    value = b[u] - fixed_t
                if left[u] <= value <= right[u]:
                    answer += 1
            continue

        if len(component) == 1:
            # Its operating range is nonempty and t is unrestricted.
            answer += 1
            continue

        # Each rod is active exactly when t belongs to one integer interval.
        # Maximize the number of intervals containing the chosen t.
        events = []
        for u in component:
            bu = b[u]
            if sign[u] == 1:
                lo = left[u] - bu
                hi = right[u] - bu
            else:
                lo = bu - right[u]
                hi = bu - left[u]
            events.append((lo, 1))
            events.append((hi + 1, -1))

        events.sort()
        current = 0
        best = 0
        i = 0
        total_events = len(events)

        while i < total_events:
            position = events[i][0]
            j = i
            # Apply all changes together because intervals are closed.
            while j < total_events and events[j][0] == position:
                current += events[j][1]
                j += 1
            if current > best:
                best = current
            i = j

        answer += best

    return answer


def main():
    data = sys.stdin.buffer.read()
    if not data:
        return

    integers = read_integers(data)
    tests = next(integers)
    output = []

    for _ in range(tests):
        n = next(integers)
        m = next(integers)

        left = array('i')
        right = array('i')
        for _ in range(n):
            left.append(next(integers))
        for _ in range(n):
            right.append(next(integers))

        head = array('i', [-1]) * n
        to = array('i')
        nxt = array('i')
        edge_z = array('i')

        for _ in range(m):
            x = next(integers) - 1
            y = next(integers) - 1
            z = next(integers)

            edge_id = len(to)
            to.append(y)
            edge_z.append(z)
            nxt.append(head[x])
            head[x] = edge_id

            if x != y:
                edge_id = len(to)
                to.append(x)
                edge_z.append(z)
                nxt.append(head[y])
                head[y] = edge_id

        result = solve_test(n, left, right, head, to, nxt, edge_z)
        output.append(str(result))

    sys.stdout.write('\n'.join(output))


if __name__ == '__main__':
    main()

import sys


def solve():
    data = sys.stdin.buffer.read()
    idx = 0
    ndata = len(data)

    def next_int():
        nonlocal idx
        while idx < ndata and data[idx] <= 32:
            idx += 1
        sign = 1
        if data[idx] == 45:  # '-'
            sign = -1
            idx += 1
        num = 0
        while idx < ndata and data[idx] > 32:
            num = num * 10 + (data[idx] - 48)
            idx += 1
        return sign * num

    T = next_int()
    answers = []

    for _ in range(T):
        N = next_int()
        M = next_int()

        # Release large objects from previous test case.
        adj = None
        c = None
        sign = None
        l = None
        r = None
        events = None
        queue = None
        comp = None

        l = [0] * N
        r = [0] * N

        for i in range(N):
            l[i] = next_int()
        for i in range(N):
            r[i] = next_int()

        adj = [[] for _ in range(N)]

        for _ in range(M):
            x = next_int() - 1
            y = next_int() - 1
            z = next_int()
            adj[x].append((y, z))
            if x != y:
                adj[y].append((x, z))

        sign = bytearray(N)  # 0 = unvisited, 1 = +, 2 = -
        c = [0] * N

        answer = 0
        impossible = False

        for start in range(N):
            if sign[start] != 0:
                continue

            sign[start] = 1
            c[start] = 0

            queue = [start]
            qidx = 0
            comp = []

            fixed_t = None
            bad = False

            while qidx < len(queue):
                u = queue[qidx]
                qidx += 1
                comp.append(u)

                su = sign[u]
                cu = c[u]
                su_val = 1 if su == 1 else -1

                for v, z in adj[u]:
                    sv = sign[v]

                    if sv == 0:
                        sign[v] = 2 if su == 1 else 1
                        c[v] = z - cu
                        queue.append(v)
                    else:
                        sv_val = 1 if sv == 1 else -1
                        coeff = su_val + sv_val
                        rhs = z - cu - c[v]

                        if coeff == 0:
                            if rhs != 0:
                                bad = True
                                break
                        else:
                            if rhs % coeff != 0:
                                bad = True
                                break

                            val = rhs // coeff
                            if fixed_t is None:
                                fixed_t = val
                            elif fixed_t != val:
                                bad = True
                                break

                if bad:
                    break

            if bad:
                impossible = True
                break

            if fixed_t is None:
                events = []

                for u in comp:
                    if sign[u] == 1:
                        L = l[u] - c[u]
                        R = r[u] - c[u]
                    else:
                        L = c[u] - r[u]
                        R = c[u] - l[u]

                    events.append((L, 1))
                    events.append((R + 1, -1))

                events.sort()

                cur = 0
                best = 0
                i = 0
                E = len(events)

                while i < E:
                    x = events[i][0]
                    while i < E and events[i][0] == x:
                        cur += events[i][1]
                        i += 1
                    if cur > best:
                        best = cur

                answer += best

            else:
                cnt = 0

                for u in comp:
                    su_val = 1 if sign[u] == 1 else -1
                    value = su_val * fixed_t + c[u]

                    if l[u] <= value <= r[u]:
                        cnt += 1

                answer += cnt

        if impossible:
            answers.append("-1")
        else:
            answers.append(str(answer))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()

import sys


def integer_stream():
    for line in sys.stdin.buffer:
        yield from map(int, line.split())


def solve_case(n, lower, upper, graph):
    # For each vertex v:
    #   a[v] = sign[v] * t + offset[v]
    # sign[v] == 0 means the vertex has not been visited.
    sign = [0] * (n + 1)
    offset = [0] * (n + 1)
    answer = 0

    for root in range(1, n + 1):
        if sign[root] != 0:
            continue

        # An isolated rod can always choose an energy in its range.
        if not graph[root]:
            answer += 1
            sign[root] = 1
            continue

        sign[root] = 1
        offset[root] = 0
        component = [root]
        fixed_t = None

        # The component list also acts as a BFS queue. Appended vertices
        # are subsequently processed by this loop.
        for u in component:
            su = sign[u]
            bu = offset[u]

            for v, z in graph[u]:
                sv = sign[v]

                if sv == 0:
                    # a[v] = z - a[u]
                    sign[v] = -su
                    offset[v] = z - bu
                    component.append(v)

                elif sv == -su:
                    # The coefficient of t cancels, so the offsets must agree.
                    if bu + offset[v] != z:
                        return -1

                else:
                    # Both signs are equal:
                    # 2 * su * t = z - bu - offset[v].
                    numerator = su * (z - bu - offset[v])

                    # t is the root's energy and therefore must be an integer.
                    if numerator & 1:
                        return -1

                    candidate = numerator // 2
                    if fixed_t is None:
                        fixed_t = candidate
                    elif fixed_t != candidate:
                        return -1

        if fixed_t is None:
            # This component has one free integer parameter t.
            # Each rod is active for an interval of possible values of t.
            events = []

            for u in component:
                b = offset[u]

                if sign[u] == 1:
                    # lower[u] <= t + b <= upper[u]
                    left = lower[u] - b
                    right = upper[u] - b
                else:
                    # lower[u] <= -t + b <= upper[u]
                    left = b - upper[u]
                    right = b - lower[u]

                # Add at left, remove at right + 1.
                # At equal coordinates, removals are processed first.
                events.append(2 * left + 1)
                events.append(2 * (right + 1))

            events.sort()

            active = 0
            best = 0
            for event in events:
                if event & 1:
                    active += 1
                    if active > best:
                        best = active
                else:
                    active -= 1

            answer += best

        else:
            # This component has a unique solution.
            for u in component:
                value = sign[u] * fixed_t + offset[u]
                if lower[u] <= value <= upper[u]:
                    answer += 1

    return answer


def main():
    data = integer_stream()
    get = data.__next__

    try:
        tests = get()
    except StopIteration:
        return

    answers = []

    for _ in range(tests):
        n = get()
        m = get()

        lower = [0] + [get() for _ in range(n)]
        upper = [0] + [get() for _ in range(n)]

        graph = [[] for _ in range(n + 1)]

        for _ in range(m):
            x = get()
            y = get()
            z = get()

            if x == y:
                graph[x].append((x, z))
            else:
                graph[x].append((y, z))
                graph[y].append((x, z))

        answers.append(str(solve_case(n, lower, upper, graph)))

        del graph, lower, upper

    sys.stdout.write("\n".join(answers) + "\n")


if __name__ == "__main__":
    main()
