#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using i128 = __int128_t;

const int INF = 1000000000;

struct Cost {
    ll K, S; // K + sqrt(S)
};

bool costLess(const Cost& A, const Cost& B) {
    ll c = B.K - A.K;
    if (c == 0) return A.S < B.S;

    if (c > 0) {
        if (A.S <= B.S) return true;
        i128 C = c;
        i128 D = (i128)A.S - B.S - C * C;
        if (D < 0) return true;
        return D * D < (i128)4 * C * C * B.S;
    } else {
        i128 D0 = -(i128)c;
        i128 D = (i128)B.S - A.S - D0 * D0;
        if (D <= 0) return false;
        return (i128)4 * D0 * D0 * A.S < D * D;
    }
}

struct Entry {
    Cost c;
    int j, b;
};

struct EntryCmp {
    bool operator()(const Entry& a, const Entry& b) const {
        if (costLess(b.c, a.c)) return true;
        if (costLess(a.c, b.c)) return false;
        if (a.j != b.j) return a.j > b.j;
        return a.b > b.b;
    }
};

struct ActiveSegTree {
    int n;
    vector<int> mn, mx;

    ActiveSegTree(int n_) : n(n_), mn(4 * n_ + 5, INF), mx(4 * n_ + 5, 0) {}

    void update(int node, int l, int r, int p, int v) {
        if (l == r) {
            mn[node] = mx[node] = v;
            return;
        }
        int m = (l + r) >> 1;
        if (p <= m) update(node << 1, l, m, p, v);
        else update(node << 1 | 1, m + 1, r, p, v);
        mn[node] = min(mn[node << 1], mn[node << 1 | 1]);
        mx[node] = max(mx[node << 1], mx[node << 1 | 1]);
    }

    void update(int p, int v) {
        update(1, 1, n, p, v);
    }

    int firstGreater(int node, int l, int r, int ql, int qr, int v) {
        if (r < ql || l > qr || mx[node] <= v) return n + 1;
        if (l == r) return l;
        int m = (l + r) >> 1;
        int res = firstGreater(node << 1, l, m, ql, qr, v);
        if (res != n + 1) return res;
        return firstGreater(node << 1 | 1, m + 1, r, ql, qr, v);
    }

    int firstGreater(int ql, int qr, int v) {
        if (ql > qr) return n + 1;
        return firstGreater(1, 1, n, ql, qr, v);
    }

    int lastLess(int node, int l, int r, int ql, int qr, int v) {
        if (r < ql || l > qr || mn[node] >= v) return 0;
        if (l == r) return l;
        int m = (l + r) >> 1;
        int res = lastLess(node << 1 | 1, m + 1, r, ql, qr, v);
        if (res != 0) return res;
        return lastLess(node << 1, l, m, ql, qr, v);
    }

    int lastLess(int ql, int qr, int v) {
        if (ql > qr) return 0;
        return lastLess(1, 1, n, ql, qr, v);
    }
};

struct RunMap {
    struct Info {
        int s, e, v;
    };

    int n;
    map<int, int> mp;

    RunMap(int n_, int initial) : n(n_) {
        mp[0] = initial;
    }

    map<int, int>::iterator split(int p) {
        if (p == 0) return mp.begin();
        auto it = mp.lower_bound(p);
        if (it != mp.end() && it->first == p) return it;
        auto prv = prev(it);
        return mp.emplace(p, prv->second).first;
    }

    void assignRange(int l, int r, int v) {
        split(l);
        if (r < n) split(r + 1);
        auto it = mp.lower_bound(l);
        auto jt = (r < n ? mp.lower_bound(r + 1) : mp.end());
        mp.erase(it, jt);
        mp[l] = v;
    }

    int get(int p) const {
        return prev(mp.upper_bound(p))->second;
    }

    Info info(int p) const {
        auto it = prev(mp.upper_bound(p));
        auto nx = next(it);
        return {it->first, nx == mp.end() ? n : nx->first - 1, it->second};
    }
};

struct ValidSet {
    int n;
    vector<int> nx, pr;
    vector<char> alive;

    ValidSet(int n_) : n(n_), nx(n_ + 2), pr(n_ + 1), alive(n_ + 1, 1) {
        iota(nx.begin(), nx.end(), 0);
        iota(pr.begin(), pr.end(), 0);
    }

    int findNext(int x) {
        if (x > n) return n + 1;
        return nx[x] == x ? x : nx[x] = findNext(nx[x]);
    }

    int findPrev(int x) {
        if (x < 0) return -1;
        return pr[x] == x ? x : pr[x] = findPrev(pr[x]);
    }

    int successor(int x) {
        return findNext(x);
    }

    int predecessor(int x) {
        return findPrev(x);
    }

    void eraseRange(int l, int r) {
        l = max(l, 0);
        r = min(r, n);
        if (l > r) return;
        int j = findNext(l);
        while (j <= r) {
            alive[j] = 0;
            int nj = findNext(j + 1);
            nx[j] = nj;
            pr[j] = findPrev(j - 1);
            j = nj;
        }
    }
};

struct Cow {
    int s, x, y, idx;
};

struct Solver {
    int n, X, Y, kx;
    vector<ll> a, axisCost;

    ActiveSegTree active;
    RunMap L, R;
    ValidSet valid;

    priority_queue<Entry, vector<Entry>, EntryCmp> lowPQ, highPQ;

    int firstHigh, lastLow;

    Solver(int n_, int X_, int Y_, const vector<int>& xs)
        : n(n_), X(X_), Y(Y_), a(n_ + 1), axisCost(n_ + 1),
          active(n_), L(n_, 0), R(n_, INF), valid(n_),
          firstHigh(n_ + 1), lastLow(0) {

        kx = 0;
        for (int i = 1; i <= n; i++) {
            if (xs[i] < X) kx++;
        }

        for (int j = 0; j <= n; j++) {
            if (j < kx) a[j] = xs[j + 1];
            else if (j == kx) a[j] = X;
            else a[j] = xs[j];
            axisCost[j] = llabs((ll)X - a[j]);
        }
    }

    Cost makeCost(int j, int b) const {
        return {
            axisCost[j] + llabs((ll)Y - b),
            a[j] * a[j] + (ll)b * b
        };
    }

    void addLowCandidate(int g) {
        if (g < 0 || g > n) return;
        auto run = R.info(g);
        if (run.v >= Y) return;

        int st = max(run.s, kx + 1);
        if (st > run.e || !valid.alive[st]) return;

        lowPQ.push({makeCost(st, run.v), st, run.v});
    }

    void addHighCandidate(int g) {
        if (g < 0 || g > n) return;
        auto run = L.info(g);
        if (run.v <= Y) return;

        int en = min(run.e, kx - 1);
        if (en < run.s || en < 0 || !valid.alive[en]) return;

        highPQ.push({makeCost(en, run.v), en, run.v});
    }

    void activate(int k, int y) {
        if (k > 1) {
            int i = active.firstGreater(1, k - 1, y);
            if (i <= k - 1) valid.eraseRange(i, k - 1);
        }
        if (k < n) {
            int i = active.lastLess(k + 1, n, y);
            if (i >= k + 1) valid.eraseRange(k, i - 1);
        }

        if (y > Y) firstHigh = min(firstHigh, k);
        else lastLow = max(lastLow, k);

        bool changedL = false, changedR = false;
        int leftR = 0, rightR = 0;

        auto it = prev(L.mp.upper_bound(k));
        if (it->second < y) {
            auto jt = it;
            while (jt != L.mp.end() && jt->second < y) ++jt;
            int r = (jt == L.mp.end() ? n : jt->first - 1);
            L.assignRange(k, r, y);
            changedL = true;
        }

        int r = k - 1;
        if (r >= 0) {
            it = prev(R.mp.upper_bound(r));
            if (it->second > y) {
                auto first = it;
                while (first != R.mp.begin()) {
                    auto prv = prev(first);
                    if (prv->second <= y) break;
                    first = prv;
                }
                leftR = first->first;
                rightR = r;
                R.assignRange(leftR, rightR, y);
                changedR = true;
            }
        }

        active.update(k, y);

        if (changedL) {
            addHighCandidate(k - 1);
            addHighCandidate(k);
        }
        if (changedR) {
            addLowCandidate(leftR);
            addLowCandidate(rightR + 1);
        }
    }

    static ll floorSqrt(ll s) {
        ll r = sqrtl((long double)s);
        while ((i128)(r + 1) * (r + 1) <= s) r++;
        while ((i128)r * r > s) r--;
        return r;
    }

    ll answer() {
        Cost best{(ll)X + Y, 0};
        bool have = true;

        auto take = [&](const Cost& c) {
            if (!have || costLess(c, best)) {
                best = c;
                have = true;
            }
        };

        if (lastLow < firstHigh) {
            int l = lastLow;
            int r = firstHigh - 1;
            int j = min(max(kx, l), r);
            take(makeCost(j, Y));
        }

        if (lastLow > 0) {
            int lim = min(kx, lastLow - 1);
            if (lim >= 0) {
                int j = valid.predecessor(lim);
                if (j >= 0) take(makeCost(j, R.get(j)));
            }
        }

        if (firstHigh <= n) {
            int lim = max(kx, firstHigh);
            int j = valid.successor(lim);
            if (j <= n) take(makeCost(j, L.get(j)));
        }

        while (!lowPQ.empty()) {
            const Entry& e = lowPQ.top();
            bool ok = e.j > kx && valid.alive[e.j] && R.get(e.j) == e.b;
            if (!ok) lowPQ.pop();
            else break;
        }
        if (!lowPQ.empty()) take(lowPQ.top().c);

        while (!highPQ.empty()) {
            const Entry& e = highPQ.top();
            bool ok = e.j < kx && valid.alive[e.j] && L.get(e.j) == e.b;
            if (!ok) highPQ.pop();
            else break;
        }
        if (!highPQ.empty()) take(highPQ.top().c);

        return best.K + floorSqrt(best.S);
    }
};

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct Comp {
    int mnx, mxx, mny, mxy;
};

struct Cow {
    int s, x, y;
};

static ll floorSqrt(ll n) {
    ll r = (ll)sqrt((double)n);
    while ((r + 1) * (r + 1) <= n) ++r;
    while (r * r > n) --r;
    return r;
}

class Solver {
public:
    int X, Y;
    map<int, Comp*> comps; // Ordered by minimum x, key = mnx.
    multiset<ll> gapValues;

    Solver(int X_, int Y_) : X(X_), Y(Y_) {
        gapValues.insert(gapCost(nullptr, nullptr));
    }

    // The valid rectangle between two consecutive components is
    // [left.maxX, right.minX] x [left.maxY, right.minY].
    // A null left/right denotes the unbounded initial/final side.
    ll gapCost(const Comp* left, const Comp* right) const {
        ll a = X, b = Y;

        if (left) {
            a = max(a, (ll)left->mxx);
            b = max(b, (ll)left->mxy);
        }
        if (right) {
            a = min(a, (ll)right->mnx);
            b = min(b, (ll)right->mny);
        }

        ll axisDistance = llabs(a - X) + llabs(b - Y);
        return axisDistance + floorSqrt(a * a + b * b);
    }

    void removeGap(const Comp* left, const Comp* right) {
        ll value = gapCost(left, right);
        auto it = gapValues.find(value);
        gapValues.erase(it);
    }

    void addGap(const Comp* left, const Comp* right) {
        gapValues.insert(gapCost(left, right));
    }

    void addPoint(int x, int y) {
        auto right = comps.lower_bound(x);

        bool inside = false;
        map<int, Comp*>::iterator center;

        if (right != comps.begin()) {
            auto previous = prev(right);
            if (previous->second->mxx > x) {
                inside = true;
                center = previous;
            }
        }

        // Existing components that will merge with the new segment:
        // the half-open range [first, last).
        map<int, Comp*>::iterator first, last;

        if (inside) {
            first = center;
            while (first != comps.begin()) {
                auto previous = prev(first);
                if (previous->second->mxy > y) first = previous;
                else break;
            }

            last = next(center);
            while (last != comps.end() && last->second->mny < y) {
                ++last;
            }
        } else {
            first = right;
            while (first != comps.begin()) {
                auto previous = prev(first);
                if (previous->second->mxy > y) first = previous;
                else break;
            }

            last = right;
            while (last != comps.end() && last->second->mny < y) {
                ++last;
            }
        }

        Comp* leftOutside = nullptr;
        if (first != comps.begin()) {
            leftOutside = prev(first)->second;
        }

        Comp* rightOutside = nullptr;
        if (last != comps.end()) {
            rightOutside = last->second;
        }

        if (first == last) {
            // A new singleton component splits one existing gap.
            removeGap(leftOutside, rightOutside);
        } else {
            // Remove every gap incident to the merged range.
            removeGap(leftOutside, first->second);

            for (auto it = first; it != last; ++it) {
                auto nx = next(it);
                Comp* nextComp = (nx == comps.end() ? nullptr : nx->second);
                removeGap(it->second, nextComp);
            }
        }

        Comp* merged = new Comp{x, x, y, y};

        for (auto it = first; it != last; ++it) {
            Comp* c = it->second;
            merged->mnx = min(merged->mnx, c->mnx);
            merged->mxx = max(merged->mxx, c->mxx);
            merged->mny = min(merged->mny, c->mny);
            merged->mxy = max(merged->mxy, c->mxy);
        }

        if (first != last) {
            comps.erase(first, last);
        }

        auto inserted = comps.emplace(merged->mnx, merged).first;

        Comp* newLeft = nullptr;
        if (inserted != comps.begin()) {
            newLeft = prev(inserted)->second;
        }

        Comp* newRight = nullptr;
        auto after = next(inserted);
        if (after != comps.end()) {
            newRight = after->second;
        }

        addGap(newLeft, merged);
        addGap(merged, newRight);
    }

    ll answer() const {
        return *gapValues.begin();
    }
};

int evaluate(const Rect &re) {
    ll a, b;

    if (SX < re.xl) a = re.xl;
    else if (SX > re.xr) a = re.xr;

    else if (SY > re.yr) b = re.yr;

class BoundTree {
    int n;
    vector<int> mx, lazy;

    void apply(int v, int value) {
        mx[v] = value;
        lazy[v] = value;
    }

    void push(int v) {
        if (lazy[v] == -1) return;
        apply(v * 2, lazy[v]);
        apply(v * 2 + 1, lazy[v]);
        lazy[v] = -1;
    }

    void assignRec(int v, int l, int r, int ql, int qr, int value) {
        if (ql <= l && r <= qr) {
            apply(v, value);
            return;
        }

        push(v);
        int m = (l + r) / 2;
        if (ql <= m) assignRec(v * 2, l, m, ql, qr, value);
        if (qr > m) assignRec(v * 2 + 1, m + 1, r, ql, qr, value);
        mx[v] = max(mx[v * 2], mx[v * 2 + 1]);

For each possible x-axis turning point a, maintains the best path using a.

States:
  middle: L <= Y <= U, so the best y-axis turning point is Y
  lower:  L > Y, so the best y-axis turning point is L
  upper:  U < Y, so the best y-axis turning point is U

For a fixed lower/upper bound z, the cost
  |X-a| + sqrt(a^2+z^2) + constant

        double middle = INF_COST;
        double lower = INF_COST;
        double upper = INF_COST;

        int lowerRight = -1;
        int upperLeft = -1;
        int upperRight = -1;

        int lowerTag = -1;
        int upperTag = -1;

        if (a == -1) return b;
        if (b == -1) return a;
        return min(a, b);
    }

    double middleCost(int index) const {
        double a = coord[index];
        return fabs(a - X) + sqrt(a * a + 1.0 * Y * Y);
    }

    double lowerCost(int index, int bound) const {
        double a = coord[index];
        return fabs(a - X) + sqrt(a * a + 1.0 * bound * bound)
             + (bound - Y);
    }

    double upperCost(int index, int bound) const {
        double a = coord[index];
        return fabs(a - X) + sqrt(a * a + 1.0 * bound * bound)
             + (Y - bound);

            cur.upper = INF_COST;

            applyLower(v * 2, cur.lowerTag);
            applyLower(v * 2 + 1, cur.lowerTag);
            cur.lowerTag = -1;

            cur.middle = INF_COST;

            cur.lowerLeft = cur.lowerRight = -1;
            cur.lowerTag = -1;

class PointSegmentTree {
    int base;
    vector<int> maxId, minId;
    const vector<int>& y;

    int betterMax(int a, int b) const {
        if (a == -1) return b;
        if (b == -1) return a;
        return y[a] > y[b] ? a : b;
    }

    int betterMin(int a, int b) const {
        if (a == -1) return b;
        if (b == -1) return a;
        return y[a] < y[b] ? a : b;
    }

public:
    PointSegmentTree(int n, const vector<int>& values) : y(values) {
        base = 1;
        while (base < n) base <<= 1;
        maxId.assign(2 * base, -1);
        minId.assign(2 * base, -1);
    }

        int p = base + position;
        maxId[p] = minId[p] = id;

        for (p >>= 1; p > 0; p >>= 1) {
            maxId[p] = betterMax(maxId[2 * p], maxId[2 * p + 1]);
            minId[p] = betterMin(minId[2 * p], minId[2 * p + 1]);
        }
    }

    // Query [left, right).
    int queryMax(int left, int right) const {
        int answer = -1;
        for (left += base, right += base; left < right;
             left >>= 1, right >>= 1) {
            if (left & 1) answer = betterMax(answer, maxId[left++]);
            if (right & 1) answer = betterMax(answer, maxId[--right]);
        }
        return answer;
    }

    int queryMin(int left, int right) const {
        int answer = -1;
        for (left += base, right += base; left < right;
             left >>= 1, right >>= 1) {
            if (left & 1) answer = betterMin(answer, minId[left++]);
            if (right & 1) answer = betterMin(answer, minId[--right]);

    void refreshOuterBoundaries() {
        if (componentByLx.empty()) {
            boundaryTree.setPoint(0, MinSegmentTree::INF);
            boundaryTree.setPoint(N + 1, MinSegmentTree::INF);
            return;
        }

        int first = componentByLx.begin()->second;
        int last = prev(componentByLx.end())->second;

        boundaryTree.setPoint(0, leftBoundaryCost(first));
        boundaryTree.setPoint(N + 1, rightBoundaryCost(last));
    }

    void insertSingleton(int id, int x, int y) {
        auto it = componentByLx.lower_bound(x);

        int after = (it == componentByLx.end() ? -1 : it->second);
        int before = -1;
        if (it != componentByLx.begin()) {
            before = prev(it)->second;
        }

        // Remove the old before -> after boundary.
        if (before != -1) {
            boundaryTree.setPoint(before, MinSegmentTree::INF);
        }

        Component current;
        current.lx = current.rx = x;
        current.ly = current.ry = y;
        current.lxId = current.rxId = current.lyId = current.ryId = id;
        current.prv = before;
        current.nxt = after;

        component[id] = current;
        active[id] = true;

        if (before != -1) component[before].nxt = id;
        if (after != -1) component[after].prv = id;

        componentByLx[x] = id;

        if (before != -1) {
            boundaryTree.setPoint(before, middleBoundaryCost(before, id));
        }
        if (after != -1) {
            boundaryTree.setPoint(id, middleBoundaryCost(id, after));
        }

        refreshOuterBoundaries();
    }

    void mergeRange(int left, int right, int id, int x, int y) {
        mergeBuffer.clear();

        int current = left;
        while (true) {
            mergeBuffer.push_back(current);
            if (current == right) break;
            current = component[current].nxt;
        }

        int before = component[left].prv;
        int after = component[right].nxt;

        // Remove all boundaries touching the merged range.
        if (before != -1) {
            boundaryTree.setPoint(before, MinSegmentTree::INF);
        }
        for (int c : mergeBuffer) {
            boundaryTree.setPoint(c, MinSegmentTree::INF);
        }

        Component merged;
        merged.lx = merged.rx = x;
        merged.ly = merged.ry = y;
        merged.lxId = merged.rxId = merged.lyId = merged.ryId = id;

        int totalSize = 1;
        for (int c : mergeBuffer) {
            absorb(merged, component[c]);
            totalSize += dsuSize[c];

            componentByLx.erase(component[c].lx);
            parent[c] = id;
            active[c] = false;
        }

        dsuSize[id] = totalSize;

        merged.prv = before;
        merged.nxt = after;
        component[id] = merged;
        active[id] = true;

        if (before != -1) component[before].nxt = id;
        if (after != -1) component[after].prv = id;

        componentByLx[merged.lx] = id;

        if (before != -1) {
            boundaryTree.setPoint(before, middleBoundaryCost(before, id));
        }
        if (after != -1) {
            boundaryTree.setPoint(id, middleBoundaryCost(id, after));
        }

        refreshOuterBoundaries();
    }

public:
    DynamicSolver(int n, int startX, int targetY,
                  const vector<int>& yValues,
                  const vector<int>& positions,
                  int coordinateCount)
        : N(n), X(startX), Y(targetY), M(coordinateCount),
          y(yValues), position(positions),
          parent(n + 1), dsuSize(n + 1, 1),
          component(n + 1), active(n + 1, false),
          pointTree(coordinateCount, yValues),
          boundaryTree(n + 2) {
        iota(parent.begin(), parent.end(), 0);
        mergeBuffer.reserve(n);
    }

    void insertPoint(int id) {
        int x = position[id];
        int px = componentCoordinates(id);

        /*
        A new point (px, py) crosses an old point in either of two ways:
          old.x <  px and old.y > py
          old.x >  px and old.y < py
        */

        int northWestWitness = pointTree.queryMax(0, x);
        int southEastWitness = pointTree.queryMin(x + 1, M);

        int left = -1, right = -1;

        // Components containing a north-west crossing point.
        if (northWestWitness != -1 &&
            y[northWestWitness] > componentCoordinates(id)) {
            int root = findRoot(northWestWitness);
            int first = root;

            while (component[first].prv != -1 &&
                   component[component[first].prv].ry >
                       componentCoordinates(id)) {
                first = component[first].prv;
            }

            left = first;
            right = root;
        }

        // Components containing a south-east crossing point.
        if (southEastWitness != -1 &&
            y[southEastWitness] < componentCoordinates(id)) {
            int root = findRoot(southEastWitness);
            int last = root;

            while (component[last].nxt != -1 &&
                   component[component[last].nxt].ly <
                       componentCoordinates(id)) {
                last = component[last].nxt;
            }

            if (left == -1) {
                left = root;
                right = last;
            } else {
                if (component[root].lx < component[left].lx) {
                    left = root;
                }
                if (component[last].lx > component[right].lx) {
                    right = last;
                }
            }
        }

        if (left == -1) {
            insertSingleton(id, componentCoordinates(id),
                            componentCoordinates(id));
        } else {
            mergeRange(left, right, id, componentCoordinates(id),
                       componentCoordinates(id));
        }

        pointTree.insert(x, id);
    }

    int answer() const {
        if (componentByLx.empty()) {
            return (int)floorSqrt(1LL * X * X + 1LL * Y * Y);
        }
        return boundaryTree.minimum();
    }

private:
    // Kept separate so the insertion code above remains readable.
    int componentCoordinates(int id) const {
        return originalY[id];
    }

    const vector<int>& originalY;
};

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct Event {
    int s, x, y;
};

struct Block {
    int firstY = 0, lastY = 0;
    int minX = 0, maxX = 0;
    int size = 0;

    ll cost = 0;
    bool inCostSet = false;
};

class Solver {
    static constexpr ll INF = (1LL << 60);

    int X, Y;
    vector<Block> blocks;

    // Ordered by the smallest y-coordinate in each block.
    map<int, int> order;

    // One entry per block: (answer contributed by this block, block ID).
    set<pair<ll, int>> costSet;

    // Active obstacles with y < Y and y > Y, respectively.
    priority_queue<int> belowMax;
    priority_queue<int, vector<int>, greater<int>> aboveMin;

    static ll integerSqrt(ll n) {
        ll r = static_cast<ll>(sqrt(static_cast<long double>(n)));
        while ((r + 1) * (r + 1) <= n) ++r;
        while (r * r > n) --r;
        return r;
    }

    ll pathCost(ll a, ll b) const {
        return llabs(static_cast<ll>(X) - a)
             + llabs(static_cast<ll>(Y) - b)
             + integerSqrt(a * a + b * b);
    }

    // For a feasible interval [L, U], an optimal a is X clamped to it.
    ll optimalA(ll L, ll U) const {
        if (L > U) return -1;
        if (static_cast<ll>(X) < L) return L;
        if (static_cast<ll>(X) > U) return U;
        return X;
    }

    ll computeBlockCost(int id) const {
        const Block &b = blocks[id];
        auto it = order.find(b.firstY);

        int prevId = -1, nextId = -1;
        if (it != order.begin()) prevId = prev(it)->second;
        auto nit = next(it);
        if (nit != order.end()) nextId = nit->second;

        ll answer = INF;

        if (b.size == 1) {
            // The block's own obstacle is excluded because its y-coordinate
            // equals the candidate b.
            ll L = (prevId == -1 ? 0 : blocks[prevId].maxX);
            ll U = (nextId == -1 ? INF : blocks[nextId].minX);
            ll a = optimalA(L, U);
            if (a >= 0) answer = min(answer, pathCost(a, b.firstY));
        } else {
            // Candidate using the first obstacle in the block.
            {
                ll L = (prevId == -1 ? 0 : blocks[prevId].maxX);
                ll U = b.minX;
                ll a = optimalA(L, U);
                if (a >= 0) answer = min(answer, pathCost(a, b.firstY));
            }

            // Candidate using the last obstacle in the block.
            {
                ll L = b.maxX;
                ll U = (nextId == -1 ? INF : blocks[nextId].minX);
                ll a = optimalA(L, U);
                if (a >= 0) answer = min(answer, pathCost(a, b.lastY));
            }
        }

        return answer;
    }

    void refreshBlock(int id) {
        Block &b = blocks[id];
        if (b.inCostSet) {
            costSet.erase({b.cost, id});
        }

        b.cost = computeBlockCost(id);
        costSet.insert({b.cost, id});
        b.inCostSet = true;
    }

public:
    Solver(int startX, int targetY, int maxN) : X(startX), Y(targetY) {
        blocks.reserve(maxN + 1);
    }

    void insertObstacle(int x, int y) {
        if (y < Y) {
            belowMax.push(x);
        } else {
            aboveMin.push(x);
        }

        const int newId = static_cast<int>(blocks.size());
        blocks.emplace_back();

        Block merged;
        merged.firstY = merged.lastY = y;
        merged.minX = merged.maxX = x;
        merged.size = 1;

        auto lower = order.lower_bound(y);

        // Does the new y-coordinate lie inside an existing block's y-range?
        int containingId = -1;
        if (lower != order.begin()) {
            auto p = prev(lower);
            if (blocks[p->second].lastY > y) {
                containingId = p->second;
            }
        }

        vector<int> removed;
        int outsidePrev = -1, outsideNext = -1;
        bool hasRange = false;

        map<int, int>::iterator leftMost, rightMost;

        if (containingId != -1) {
            auto anchor = order.find(blocks[containingId].firstY);

            leftMost = anchor;
            while (leftMost != order.begin()) {
                auto p = prev(leftMost);
                if (blocks[p->second].maxX <= x) break;
                leftMost = p;
            }

            auto after = next(anchor);
            while (after != order.end() &&
                   blocks[after->second].minX < x) {
                ++after;
            }
            rightMost = prev(after);
            hasRange = true;
        } else {
            bool leftTouched = false, rightTouched = false;

            auto lit = lower;
            while (lit != order.begin()) {
                auto p = prev(lit);
                if (blocks[p->second].maxX <= x) break;
                lit = p;
                leftTouched = true;
            }

            auto rit = lower;
            while (rit != order.end() &&
                   blocks[rit->second].minX < x) {
                ++rit;
                rightTouched = true;
            }

            if (leftTouched) {
                leftMost = lit;
                rightMost = lit;
                hasRange = true;
            }

            if (rightTouched) {
                rightMost = prev(rit);
                if (!leftTouched) leftMost = rightMost;
                hasRange = true;
            }
        }

        if (!hasRange) {
            if (lower != order.begin()) {
                outsidePrev = prev(lower)->second;
            }
            if (lower != order.end()) {
                outsideNext = lower->second;
            }

            blocks[newId] = merged;
            order.emplace(y, newId);

            refreshBlock(newId);
            if (outsidePrev != -1) refreshBlock(outsidePrev);
            if (outsideNext != -1) refreshBlock(outsideNext);
            return;
        }

        if (leftMost != order.begin()) {
            outsidePrev = prev(leftMost)->second;
        }
        auto afterRight = next(rightMost);
        if (afterRight != order.end()) {
            outsideNext = afterRight->second;
        }

        for (auto it = leftMost;; ++it) {
            removed.push_back(it->second);
            if (it == rightMost) break;
        }

        for (int id : removed) {
            const Block &b = blocks[id];
            merged.minX = min(merged.minX, b.minX);
            merged.maxX = max(merged.maxX, b.maxX);
            merged.firstY = min(merged.firstY, b.firstY);
            merged.lastY = max(merged.lastY, b.lastY);
            merged.size += b.size;
        }

        for (int id : removed) {
            if (blocks[id].inCostSet) {
                costSet.erase({blocks[id].cost, id});
                blocks[id].inCostSet = false;
            }
            order.erase(blocks[id].firstY);
        }

        blocks[newId] = merged;
        order.emplace(merged.firstY, newId);

        refreshBlock(newId);
        if (outsidePrev != -1) refreshBlock(outsidePrev);
        if (outsideNext != -1) refreshBlock(outsideNext);
    }

    ll answer() const {
        // Going through the origin is always possible.
        ll result = static_cast<ll>(X) + Y;

        if (!costSet.empty()) {
            result = min(result, costSet.begin()->first);
        }

        // Candidate whose y-axis endpoint is exactly Y.
        ll L = belowMax.empty() ? 0 : belowMax.top();
        ll U = aboveMin.empty() ? INF : aboveMin.top();
        ll a = optimalA(L, U);
        if (a >= 0) {
            result = min(result, pathCost(a, Y));
        }

        return result;
    }
};
