#pragma GCC optimize("O3")
#include <bits/stdc++.h>
using namespace std;

static constexpr uint32_t MOD = 1000000007;
static constexpr int MAXK = 20;

struct Poly {
    uint32_t a[MAXK + 1];
    uint32_t b[MAXK + 1];
};

static inline uint32_t addmod(uint32_t a, uint32_t b) {
    uint32_t s = a + b;
    return s - (MOD & (uint32_t)-(int)(s >= MOD));
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, K;
    cin >> N >> K;
    const int M = 1 << K;

    vector<Poly> dp(M);
    vector<int> cows(N);

    for (int i = 0; i < N; ++i) {
        cin >> cows[i];
        ++dp[cows[i]].a[0];
    }

    vector<uint8_t> pc(M);
    for (int i = 1; i < M; ++i)
        pc[i] = pc[i >> 1] + (i & 1);

    // dp[x].a + delta * dp[x].b represents
    // sum_y f[y] * (1+delta)^|x&y| * r^|y\x|, modulo delta^2.
    for (int lev = 1, bit = 1; lev <= K; ++lev, bit <<= 1) {
        for (int base = 0; base < M; base += bit << 1) {
            for (int off = 0; off < bit; ++off) {
                Poly &p = dp[base + off];
                Poly &q = dp[base + off + bit];

                int maxd = lev - pc[off];
                uint32_t prevA = 0, prevB = 0;

                for (int d = 0; d <= maxd; ++d) {
                    uint32_t a0 = p.a[d], b0 = p.b[d];
                    uint32_t a1 = q.a[d], b1 = q.b[d];

                    q.a[d] = addmod(a0, a1);
                    q.b[d] = addmod(addmod(b0, b1), a1);

                    p.a[d] = addmod(a0, prevA);
                    p.b[d] = addmod(b0, prevB);

                    prevA = a1;
                    prevB = b1;
                }
            }
        }
    }

    vector<uint32_t> inv(K + 1);
    inv[1] = 1;
    for (int i = 2; i <= K; ++i)
        inv[i] = MOD - (uint64_t)(MOD / i) * inv[MOD % i] % MOD;

    vector<int> answer(M, -1);

    for (int x : cows) {
        if (answer[x] == -1) {
            int s = pc[x];
            uint64_t res = 0;
            for (int v = 0; v <= K - s; ++v) {
                res += (uint64_t)dp[x].b[v] * inv[s + v];
                res %= MOD;
            }
            answer[x] = (int)res;
        }
        cout << answer[x] << '\n';
    }

    return 0;
}

static constexpr int MOD = 1000000007;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, K;
    cin >> N >> K;

    const int S = 1 << K;
    const int stride = K + 1;

    vector<int> masks(N);
    // Low 32 bits: unweighted sum. High 32 bits: intersection-weighted sum.
    vector<uint64_t> dp((size_t)S * stride, 0);

    for (int i = 0; i < N; ++i) {
        cin >> masks[i];
        ++dp[(size_t)masks[i] * stride];
    }

    // After processing all bits, entry (A,r) describes bitstrings B with
    // |B\A|=r. The high part is sum_B count[B]*|A∩B|.
    for (int j = 0, bit = 1; j < K; ++j, bit <<= 1) {
        const int maxR = j + 1;
        for (int base = 0; base < S; base += 2 * bit) {
            for (int off = 0; off < bit; ++off) {
                size_t i0 = (size_t)(base + off) * stride;
                size_t i1 = i0 + (size_t)bit * stride;
                uint64_t previous = 0;

                for (int r = 0; r <= maxR; ++r) {
                    uint64_t x = dp[i0 + r];
                    uint64_t y = dp[i1 + r];

                    dp[i0 + r] = x + previous;
                    // Adding y<<32 increments the weighted sum by the
                    // unweighted sum in y, accounting for one common bit.
                    dp[i1 + r] = x + y + (y << 32);
                    previous = y;
                }
            }
        }
    }

    vector<long long> inv(2 * K + 1, 0);
    inv[1] = 1;
    for (int i = 2; i <= 2 * K; ++i) {
        inv[i] = MOD - (long long)(MOD / i) * inv[MOD % i] % MOD;
    }

    vector<int> answer(S, 0);
    for (int a = 1; a < S; ++a) {
        const size_t row = (size_t)a * stride;
        const int pc = __builtin_popcount((unsigned)a);
        long long sum = 0;

        for (int r = 0; r <= K; ++r) {
            uint32_t weighted = dp[row + r] >> 32;
            sum += (long long)weighted * inv[pc + r];
        }
        answer[a] = sum % MOD;
    }

    for (int x : masks) {
        cout << answer[x] << '\n';
    }

static const int MOD = 1000000007;

long long modpow(long long a, long long e) {
    long long r = 1;
    while (e) {
        if (e & 1) r = r * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return r;
}

    a += b;
    if (a >= MOD) a -= MOD;
}

    cin >> N >> K;

    vector<int> masks(N), freq(M, 0);

        cin >> x;
        ++freq[x];

    for (int i = 1; i < M; ++i)
        pc[i] = pc[i >> 1] + (i & 1);

    vector<int> f0(M), f1(M);
    for (int i = 1; i < M; ++i) {
        f0[i] = freq[i] % MOD;
        f1[i] = (long long)freq[i] * pc[i] % MOD;
    }

    // Subset zeta transforms:
    // f[mask] = sum of the original values over submasks of mask.
    for (int bit = 1; bit < M; bit <<= 1) {
        for (int base = 0; base < M; base += bit << 1) {
            for (int j = 0; j < bit; ++j) {
                addmod(f0[base + bit + j], f0[base + j]);
                addmod(f1[base + bit + j], f1[base + j]);
            }
        }

    for (int i = 1; i <= K; ++i)

        f0[i] = (long long)f0[i] * w % MOD;
        f1[i] = (long long)f1[i] * w % MOD;

    }

#include <bits/stdc++.h>
using namespace std;

static const int MOD = 1000000007;

long long modpow(long long a, long long e) {
    long long r = 1;
    while (e) {
        if (e & 1) r = r * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return r;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, K;
    cin >> N >> K;

    int M = 1 << K;
    vector<int> cnt(M, 0);
    vector<int> cows(N);

    for (int i = 0; i < N; i++) {
        int x;
        cin >> x;
        cows[i] = x;
        cnt[x]++;
    }

    vector<unsigned char> pc(M, 0);
    for (int i = 1; i < M; i++) {
        pc[i] = pc[i >> 1] + (i & 1);
    }

    vector<int> A(M), B(M);

    for (int mask = 0; mask < M; mask++) {
        A[mask] = cnt[mask];
        B[mask] = (long long)cnt[mask] * pc[mask] % MOD;
    }

    // Superset zeta transform:
    // A[u] = sum_{t superset u} cnt[t]
    // B[u] = sum_{t superset u} cnt[t] * popcount(t)
    for (int bit = 0; bit < K; bit++) {
        int step = 1 << bit;
        int block = step << 1;

        for (int base = 0; base < M; base += block) {
            for (int off = 0; off < step; off++) {
                int low = base + off;
                int high = low + step;

                A[low] += A[high];
                if (A[low] >= MOD) A[low] -= MOD;

                B[low] += B[high];
                if (B[low] >= MOD) B[low] -= MOD;
            }
        }
    }

    int Nmod = N % MOD;

    vector<int> fact(K + 1), invfact(K + 1);
    fact[0] = 1;
    for (int i = 1; i <= K; i++) {
        fact[i] = (long long)fact[i - 1] * i % MOD;
    }

    invfact[K] = modpow(fact[K], MOD - 2);
    for (int i = K; i >= 1; i--) {
        invfact[i - 1] = (long long)invfact[i] * i % MOD;
    }

    // f[a][p] = integral_0^1 x^(a-1) (x-1)^p dx
    //         = (-1)^p (a-1)! p! / (a+p)!
    vector<vector<int>> f(K + 1, vector<int>(K + 1, 0));

    for (int a = 1; a <= K; a++) {
        for (int p = 0; p <= K - a; p++) {
            long long val = (long long)fact[a - 1] * fact[p] % MOD;
            val = val * invfact[a + p] % MOD;
            if (p & 1) val = (MOD - val) % MOD;
            f[a][p] = (int)val;
        }
    }

    vector<vector<int>> masksByPop(K + 1);
    for (int mask = 0; mask < M; mask++) {
        masksByPop[pc[mask]].push_back(mask);
    }

    vector<int> ans(M, 0);

    // p = 0 contribution.
    // For r = complement(s), q = popcount(r), a = popcount(s) = K - q:
    // p0 = f[a][0] * (B[0] + a * A[0]) - N
    for (int q = 0; q <= K - 1; q++) {
        int a = K - q;

        int term = (B[0] + (long long)a * A[0]) % MOD;
        int baseVal = (long long)f[a][0] * term % MOD;

        baseVal -= Nmod;
        if (baseVal < 0) baseVal += MOD;

        for (int r : masksByPop[q]) {
            ans[r] = baseVal;
        }
    }

    if (K > 1) {
        int R = K - 1;
        size_t total = (size_t)R * M;

        vector<uint64_t> dpA(total, 0);
        vector<uint64_t> dpB(total, 0);

        for (int p = 1; p <= K - 1; p++) {
            size_t base = (size_t)(p - 1) * M;

            for (int mask : masksByPop[p]) {
                dpA[base + mask] = A[mask];

                addmod(z[low], z[high]);
                addmod(w[low], w[high]);
            }
        }
    }

    for (int mask : cows) {
        long long answer =
            (1LL * popcount[mask] * z[mask] + w[mask] - N) % MOD;

        if (answer < 0) answer += MOD;
        cout << answer << '\n';
    }

    return 0;
}
