import sys
from math import comb

MOD = 998244353


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, k = data[0], data[1]
    a = data[2:2 + n]

    binomial = [comb(k, t) for t in range(k + 1)]

    # moments[m] = sum of P_j^m for already processed prefix sums P_j.
    # Initially only P_0 = 0 is present.
    moments = [0] * (k + 1)
    moments[0] = 1

    prefix = 0
    answer = 0

    for i in range(n):
        prefix = (prefix + a[i]) % MOD

        powers = [1] * (k + 1)
        for m in range(1, k + 1):
            powers[m] = powers[m - 1] * prefix % MOD

        # sum_j (prefix - P_j)^K, expanded by the binomial theorem.
        current = 0
        for t in range(k + 1):
            m = k - t
            term = binomial[t] * powers[t] % MOD * moments[m] % MOD
            if m & 1:
                current -= term
            else:
                current += term

        answer = (answer + current) % MOD

        # Add the current prefix sum for future right endpoints.
        for m in range(k + 1):
            moments[m] += powers[m]
            if moments[m] >= MOD:
                moments[m] -= MOD

    print(answer)


if __name__ == "__main__":
    main()

import sys
import math

MOD = 998244353

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N, K = data[0], data[1]
    A = data[2:]

    # binomial coefficients C(K, j)
    C = [math.comb(K, j) for j in range(K + 1)]

    # prev_sum[p] = sum_{l=0}^{r-1} (S_l)^p mod MOD
    prev_sum = [0] * (K + 1)
    prev_sum[0] = 1  # S_0^0 = 1, S_0^p = 0 for p > 0

    powS = [1] * (K + 1)
    S = 0
    ans = 0

    for a in A:
        S = (S + a) % MOD

        # powers of current prefix sum S_r
        for p in range(1, K + 1):
            powS[p] = powS[p - 1] * S % MOD

        # add contribution for all l < r
        for j in range(K + 1):
            term = C[j] * powS[j] % MOD * prev_sum[K - j] % MOD
            if (K - j) & 1:
                ans -= term
            else:
                ans += term

        ans %= MOD

        # include current prefix sum for future r
        for p in range(K + 1):
            prev_sum[p] += powS[p]
            if prev_sum[p] >= MOD:
                prev_sum[p] -= MOD

    print(ans % MOD)

if __name__ == "__main__":
    main()
