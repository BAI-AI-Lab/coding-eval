struct Endgame {

    vector<int> dist, move;
    vector<vector<int>> patterns;

    static int nextConfig(int key, int qmask, int reply) {
        int m = key >> 8;
        int states = key & 255;
        int newMask = 0, newM = 0;

        for (int i = 0; i < m; i++) {
            int s = (states >> (2 * i)) & 3;
            int b = (qmask >> i) & 1;
            int h = reply ^ b;
            int ns = NXT[s][h];
            if (ns != -1) {
                newMask |= ns << (2 * newM);
                newM++;
            }
        }

        if (newM <= 2) return -1;
        return (newM << 8) | newMask;
    }

struct Candidate {

    int h = 0;
    int run = 0;

static vector<Candidate> applyAnswer(const vector<Candidate>& cs,
                                     const vector<int>& z, int y) {
    vector<Candidate> nxt;
    for (int i = 0; i < (int)cs.size(); ++i) {
        int required = y ^ z[i];
        Candidate d = cs[i];

        if (d.run == 0) {
            d.h = required;
            d.run = 1;
        } else {
            if (d.run == 2 && required == d.h) continue;
            if (required == d.h) {
                d.run = 2;
            } else {
                d.h = required;
                d.run = 1;
            }
        }
        nxt.push_back(d);
    }
    return nxt;
}

static const vector<pair<int, int>> smallBounds = {
    {0, 0}, {0, 1}, {0, 2}, {1, 1}, {1, 2}, {2, 2}
};

static vector<vector<int>> smallPatterns() {
    vector<vector<int>> p;
    for (auto [a, b] : smallBounds) {
        vector<int> z(3, 0);
        for (int i = a; i <= b; ++i) z[i] = 1;
        p.push_back(z);
    }
    return p;
}

static const vector<vector<int>> patterns3 = smallPatterns();
static unordered_map<string, int> memoSmall;
static unordered_map<string, int> choiceSmall;

static string smallKey(const vector<Candidate>& cs, int depth) {
    string s = to_string(depth) + ":";
    for (const auto& c : cs) {
        s.push_back(char('0' + c.h * 2 + (c.run - 1)));
        s.push_back(',');
    }
    return s;
}

static bool solveSmall(const vector<Candidate>& cs, int depth) {
    if ((int)cs.size() <= 2) return true;
    if (depth == 0) return false;

    string key = smallKey(cs, depth);
    auto it = memoSmall.find(key);
    if (it != memoSmall.end()) return it->second;

    for (int p = 0; p < 6; ++p) {
        bool ok = true;
        for (int y = 0; y <= 1; ++y) {
            vector<Candidate> nxt = applyAnswer(cs, patterns3[p], y);
            if (!nxt.empty() && !solveSmall(nxt, depth - 1)) {
                ok = false;
                break;
            }
        }
        if (ok) {
            memoSmall[key] = 1;
            choiceSmall[key] = p;
            return true;
        }
    }

    memoSmall[key] = 0;
    return false;
}

#include <bits/stdc++.h>
using namespace std;

struct Candidate {
    int pos;
    int group;
    int older = -1;
    int last = -1;
    bool alive = true;
};

class ThreeCandidateSolver {
    static constexpr int STATES = 125;
    static constexpr int INF = 1e9;

    int pw[3] = {1, 5, 25};
    const array<int, 7> masks = {0, 1, 2, 3, 4, 6, 7};

    int aliveCount(int state) const {
        int result = 0;
        for (int i = 0; i < 3; ++i) {
            int digit = (state / pw[i]) % 5;
            if (digit != 4) ++result;
        }
        return result;
    }

    int transition(int state, int mask, int response) const {
        int nextState = 0;

        for (int i = 0; i < 3; ++i) {
            int digit = (state / pw[i]) % 5;

            if (digit == 4) {
                nextState += 4 * pw[i];
                continue;
            }

            int older = digit >> 1;
            int last = digit & 1;
            int inside = (mask >> i) & 1;
            int honest = response ^ inside;

            if (older == last && last == honest) {
                digit = 4;
            } else {
                digit = (last << 1) | honest;
            }

            nextState += digit * pw[i];
        }

        return nextState;
    }

public:
    array<int, STATES> distance{};
    array<int, STATES> action{};

    ThreeCandidateSolver() {
        distance.fill(INF);
        action.fill(-1);

        for (int state = 0; state < STATES; ++state) {
            if (aliveCount(state) <= 2) {
                distance[state] = 0;
            }
        }

        bool changed = true;
        while (changed) {
            changed = false;

            for (int state = 0; state < STATES; ++state) {
                if (distance[state] != INF) continue;

                int bestDistance = INF;
                int bestAction = -1;

                for (int mask : masks) {
                    bool possible = true;
                    int maximumNextDistance = 0;

                    for (int response = 0; response <= 1; ++response) {
                        int nextState = transition(state, mask, response);

                        if (distance[nextState] == INF) {
                            possible = false;
                            break;
                        }

                        maximumNextDistance =
                            max(maximumNextDistance, distance[nextState]);
                    }

                    if (possible && maximumNextDistance + 1 < bestDistance) {
                        bestDistance = maximumNextDistance + 1;
                        bestAction = mask;
                    }
                }

                if (bestAction != -1) {
                    distance[state] = bestDistance;
                    action[state] = bestAction;
                    changed = true;
                }
            }
        }
    }
};

static void applyResponse(vector<Candidate> &candidates, int response,
                          int selectedGroup) {
    for (Candidate &candidate : candidates) {
        if (!candidate.alive) continue;

        int inside = (candidate.group == selectedGroup);
        int honest = response ^ inside;

        if (candidate.older == candidate.last &&
            candidate.last == honest) {
            candidate.alive = false;
        } else {
            candidate.older = candidate.last;
            candidate.last = honest;
        }
    }
}

static void removeDead(vector<Candidate> &candidates) {
    vector<Candidate> remaining;
    remaining.reserve(candidates.size());

    for (const Candidate &candidate : candidates) {
        if (candidate.alive) remaining.push_back(candidate);
    }

    candidates.swap(remaining);
}

static pair<int, int> intervalForMask(const vector<Candidate> &candidates,

                freePosition == candidates[1].pos ||
                freePosition == candidates[2].pos)) {
            ++freePosition;

        return {freePosition, freePosition};

    if (mask == 1) return {candidates[2].pos, candidates[2].pos};
    if (mask == 2) return {candidates[1].pos, candidates[1].pos};
    if (mask == 3) return {candidates[1].pos, candidates[2].pos};
    if (mask == 4) return {candidates[0].pos, candidates[0].pos};
    if (mask == 6) return {candidates[0].pos, candidates[1].pos};
    return {candidates[0].pos, candidates[2].pos};
}

                int l = candidates[starts[selectedGroup]].pos;
                int r = candidates[starts[selectedGroup] +
                                   groupSize[selectedGroup] - 1].pos;

                int response = ask(l, r);
                applyResponse(candidates, response, selectedGroup);
            }

            removeDead(candidates);
        }

        if (candidates.size() == 3) {
            while (candidates.size() == 3) {
                int state = 0;

                for (int i = 0; i < 3; ++i) {
                    int pairCode = (candidates[i].older << 1) |
                                   candidates[i].last;
                    state += pairCode * (i == 0 ? 1 : (i == 1 ? 5 : 25));
                }

                int mask = solver.action[state];
                if (mask == -1) {
                    mask = 7;
                }

                auto [l, r] = intervalForMask(candidates, mask, n);
                int response = ask(l, r);

                for (int i = 0; i < 3; ++i) {
                    if (!candidates[i].alive) continue;

                    int inside = (mask >> i) & 1;
                    int honest = response ^ inside;

                    if (candidates[i].older == candidates[i].last &&
                        candidates[i].last == honest) {
                        candidates[i].alive = false;
                    } else {
                        candidates[i].older = candidates[i].last;
                        candidates[i].last = honest;
                    }
                }

                removeDead(candidates);
            }
        }

        if (!candidates.empty()) {
            cout << "! " << candidates[0].pos << endl;

            int result;
            cin >> result;

            if (result == 0 && candidates.size() > 1) {
                cout << "! " << candidates[1].pos << endl;
                cin >> result;
            }
        }

        cout << "#" << endl;
    }

    return 0;
}
