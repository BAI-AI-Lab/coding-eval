import sys


def main():
    s = sys.stdin.buffer.readline().strip()
    n = len(s)
    r = s[::-1]

    # KMP failure function for r.
    failure = [0] * n
    j = 0
    for i in range(1, n):
        c = r[i]
        while j and c != r[j]:
            j = failure[j - 1]
        if c == r[j]:
            j += 1
        failure[i] = j

    # Longest suffix of s that is also a prefix of r.
    # Such a suffix is exactly a palindromic suffix of s.
    j = 0
    for c in s:
        while j and c != r[j]:
            j = failure[j - 1]
        if c == r[j]:
            j += 1

    # Keep the longest palindromic suffix and append the minimum necessary part.
    answer = s + s[:n - j][::-1]
    sys.stdout.buffer.write(answer)


if __name__ == "__main__":
    main()

import sys

def main():
    S = sys.stdin.readline().strip()
    n = len(S)
    R = S[::-1]

    # Build R + '$' + S and compute its prefix function.
    # The last value is the length of the longest palindromic suffix of S.
    T = R + '$' + S
    L = len(T)
    pi = [0] * L
    j = 0

    for i in range(1, L):
        while j > 0 and T[i] != T[j]:
            j = pi[j - 1]
        if T[i] == T[j]:
            j += 1
        pi[i] = j

    p = j  # longest palindromic suffix length
    sys.stdout.write(S + R[p:] + '\n')

if __name__ == "__main__":
    main()
