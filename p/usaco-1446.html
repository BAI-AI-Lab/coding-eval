import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
    t = next(it)
    answers = []

    for _ in range(t):
        n = next(it)
        a = [next(it) for _ in range(n)]

        total = sum(a)
        window_size = n // 2 + 1

        current = sum(a[:window_size])
        bessie = current

        for left in range(1, n - window_size + 1):
            current += a[left + window_size - 1] - a[left - 1]
            if current < bessie:
                bessie = current

        answers.append(f"{bessie} {total - bessie}")

    sys.stdout.write("\n".join(answers))

if __name__ == "__main__":
    main()

import sys


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    t = data[0]
    pos = 1
    answers = []

    for _ in range(t):
        n = data[pos]
        pos += 1

        end = pos + n
        block_len = n // 2 + 1

        window_sum = sum(data[pos:pos + block_len])
        bessie = window_sum

        for i in range(pos + block_len, end):
            window_sum += data[i] - data[i - block_len]
            if window_sum < bessie:
                bessie = window_sum

        total = sum(data[pos:end])
        elsie = total - bessie
        answers.append(f"{bessie} {elsie}")

        pos = end

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()
