#include <bits/stdc++.h>
using namespace std;

using Board = array<uint8_t, 100>;

static const char DIR_CHAR[4] = {'F', 'B', 'L', 'R'};

Board tiltBoard(const Board &b, int dir) {
    Board out{};
    if (dir == 0) {
        for (int c = 0; c < 10; ++c) {
            int w = 0;
            for (int r = 0; r < 10; ++r) {
                uint8_t v = b[r * 10 + c];
                if (v) out[w++ * 10 + c] = v;
            }
        }
    } else if (dir == 1) {
        for (int c = 0; c < 10; ++c) {
            int w = 9;
            for (int r = 9; r >= 0; --r) {
                uint8_t v = b[r * 10 + c];
                if (v) out[w-- * 10 + c] = v;
            }
        }
    } else if (dir == 2) {
        for (int r = 0; r < 10; ++r) {
            int w = 0;
            for (int c = 0; c < 10; ++c) {
                uint8_t v = b[r * 10 + c];
                if (v) out[r * 10 + w++] = v;
            }
        }
    } else {
        for (int r = 0; r < 10; ++r) {
            int w = 9;
            for (int c = 9; c >= 0; --c) {
                uint8_t v = b[r * 10 + c];
                if (v) out[r * 10 + w--] = v;
            }
        }
    }
    return out;
}

void putCandy(Board &b, int flavor, int rank) {
    for (int i = 0; i < 100; ++i) {
        if (b[i] == 0) {
            if (rank == 0) {
                b[i] = static_cast<uint8_t>(flavor);
                return;
            }
            --rank;
        }
    }
}

int componentScore(const Board &b) {
    array<uint8_t, 100> used{};
    int st[100];
    int result = 0;

    for (int s = 0; s < 100; ++s) {
        if (b[s] == 0 || used[s]) continue;
        uint8_t color = b[s];
        int top = 0;
        int sz = 0;
        st[top++] = s;
        used[s] = 1;

        while (top) {
            int x = st[--top];
            ++sz;
            int r = x / 10;
            int c = x % 10;

            if (r > 0) {
                int y = x - 10;
                if (!used[y] && b[y] == color) {
                    used[y] = 1;
                    st[top++] = y;
                }
            }
            if (r < 9) {
                int y = x + 10;
                if (!used[y] && b[y] == color) {
                    used[y] = 1;
                    st[top++] = y;
                }
            }
            if (c > 0) {
                int y = x - 1;
                if (!used[y] && b[y] == color) {
                    used[y] = 1;
                    st[top++] = y;
                }
            }
            if (c < 9) {
                int y = x + 1;
                if (!used[y] && b[y] == color) {
                    used[y] = 1;
                    st[top++] = y;
                }
            }
        }
        result += sz * sz;
    }
    return result;
}

int heuristic(const Board &b) {
    int score = componentScore(b);
    int proximity = 0;

    for (int r = 0; r < 10; ++r) {
        for (int c = 0; c < 10; ++c) {
            uint8_t v = b[r * 10 + c];
            if (!v) continue;

            if (r < 9) {
                if (c > 0 && b[(r + 1) * 10 + c - 1] == v) ++proximity;
                if (c < 9 && b[(r + 1) * 10 + c + 1] == v) ++proximity;
            }
            if (r < 8 && b[(r + 2) * 10 + c] == v) ++proximity;
            if (c < 8 && b[r * 10 + c + 2] == v) ++proximity;
        }
    }
    return score * 16 + proximity;
}

uint64_t splitmix64(uint64_t x) {
    x += 0x9e3779b97f4a7c15ULL;
    x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
    x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
    return x ^ (x >> 31);
}

struct RNG {
    uint64_t s;
    explicit RNG(uint64_t seed) : s(seed ? seed : 0x123456789abcdefULL) {}

    uint64_t next() {
        uint64_t x = s;
        x ^= x >> 12;
        x ^= x << 25;
        x ^= x >> 27;
        s = x;
        return x * 2685821657736338717ULL;
    }

    int integer(int n) {
        return static_cast<int>(next() % static_cast<uint64_t>(n));
    }
};

double exactValue(const Board &b, int turn, const array<int, 100> &flavors) {
    if (turn >= 100) return componentScore(b);

    int emptyCount = 100 - turn;
    double total = 0.0;

    for (int rank = 0; rank < emptyCount; ++rank) {
        Board placed = b;
        putCandy(placed, flavors[turn], rank);

        if (turn == 99) {
            total += componentScore(placed);
        } else {
            double best = -1.0;
            for (int d = 0; d < 4; ++d) {
                Board nb = tiltBoard(placed, d);
                best = max(best, exactValue(nb, turn + 1, flavors));
            }
            total += best;
        }
    }
    return total / emptyCount;
}

double greedyRollout(const Board &start, int nextTurn,
                     const array<int, 100> &flavors, uint64_t seed) {
    if (100 - nextTurn <= 2) return exactValue(start, nextTurn, flavors);

    RNG placeRng(splitmix64(seed));
    RNG policyRng(splitmix64(seed ^ 0xd1b54a32d192ed03ULL));
    Board b = start;

    for (int turn = nextTurn; turn <= 97; ++turn) {
        int rank = placeRng.integer(100 - turn);
        Board placed = b;
        putCandy(placed, flavors[turn], rank);

        if (turn == 97) {
            double best = -1.0;
            for (int d = 0; d < 4; ++d) {
                Board nb = tiltBoard(placed, d);
                best = max(best, exactValue(nb, 98, flavors));
            }
            return best;
        }

        int bestValue = -1;
        int ties = 0;
        Board bestBoard{};

        for (int d = 0; d < 4; ++d) {
            Board nb = tiltBoard(placed, d);
            int value = heuristic(nb);

            if (value > bestValue) {
                bestValue = value;
                bestBoard = nb;
                ties = 1;
            } else if (value == bestValue) {
                ++ties;
                if (policyRng.integer(ties) == 0) bestBoard = nb;
            }
        }
        b = bestBoard;
    }

    return exactValue(b, 98, flavors);
}

double beamRollout(const Board &start, int nextTurn,
                   const array<int, 100> &flavors, uint64_t seed) {
    struct Node {
        Board b;
        int value;
    };

    int horizon = 100 - nextTurn;
    int width = (horizon <= 5 ? 8 : 4);

    RNG rng(splitmix64(seed));
    array<int, 100> ranks{};
    for (int turn = nextTurn; turn <= 97; ++turn) {
        ranks[turn] = rng.integer(100 - turn);
    }

    vector<Node> current;
    current.push_back({start, 0});

    for (int turn = nextTurn; turn <= 97; ++turn) {
        if (turn == 97) {
            double best = -1.0;
            for (const Node &node : current) {
                Board placed = node.b;
                putCandy(placed, flavors[turn], ranks[turn]);
                for (int d = 0; d < 4; ++d) {
                    Board nb = tiltBoard(placed, d);
                    best = max(best, exactValue(nb, 98, flavors));
                }
            }
            return best;
        }

        vector<Node> next;
        next.reserve(current.size() * 4);

        for (const Node &node : current) {
            Board placed = node.b;
            putCandy(placed, flavors[turn], ranks[turn]);

            for (int d = 0; d < 4; ++d) {
                Board nb = tiltBoard(placed, d);
                next.push_back({nb, heuristic(nb)});
            }
        }

        sort(next.begin(), next.end(), [](const Node &a, const Node &b) {
            return a.value > b.value;
        });
        if (static_cast<int>(next.size()) > width) next.resize(width);
        current = move(next);
    }

    return exactValue(current.front().b, 98, flavors);
}

double rollout(const Board &start, int nextTurn,
               const array<int, 100> &flavors, uint64_t seed) {
    int horizon = 100 - nextTurn;
    if (horizon <= 4) return exactValue(start, nextTurn, flavors);
    if (horizon <= 6) return beamRollout(start, nextTurn, flavors, seed);
    return greedyRollout(start, nextTurn, flavors, seed);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    array<int, 100> flavors{};
    for (int i = 0; i < 100; ++i) {
        if (!(cin >> flavors[i])) return 0;
    }

    const auto startTime = chrono::steady_clock::now();
    const double TIME_BUDGET = 1.80;

    Board board{};

    for (int turn = 0; turn < 100; ++turn) {
        int p;
        if (!(cin >> p)) return 0;
        putCandy(board, flavors[turn], p - 1);

        if (turn == 99) return 0;

        Board candidates[4];
        int candidateHeuristic[4];

        for (int d = 0; d < 4; ++d) {
            candidates[d] = tiltBoard(board, d);
            candidateHeuristic[d] = heuristic(candidates[d]);
        }

        int nextTurn = turn + 1;
        int horizon = 100 - nextTurn;
        double values[4] = {0, 0, 0, 0};

        if (horizon <= 4) {
            for (int d = 0; d < 4; ++d) {
                values[d] = exactValue(candidates[d], nextTurn, flavors);
            }
        } else {
            double sums[4] = {0, 0, 0, 0};
            int iterations = 0;

            double elapsed = chrono::duration<double>(
                chrono::steady_clock::now() - startTime).count();
            double remaining = TIME_BUDGET - elapsed;
            int decisionsLeft = 99 - turn;

            if (remaining > 0.01) {
                double localEnd = elapsed + remaining / decisionsLeft;
                uint64_t baseSeed =
                    0x6a09e667f3bcc909ULL ^
                    (static_cast<uint64_t>(turn + 1) *
                     0x9e3779b97f4a7c15ULL);

                while (true) {
                    if (iterations > 0) {
                        double now = chrono::duration<double>(
                            chrono::steady_clock::now() - startTime).count();
                        if (now >= localEnd) break;
                    }

                    uint64_t seed = splitmix64(
                        baseSeed +
                        static_cast<uint64_t>(iterations + 1) *
                            0xbf58476d1ce4e5b9ULL);

                    for (int d = 0; d < 4; ++d) {
                        sums[d] += rollout(
                            candidates[d], nextTurn, flavors, seed);
                    }
                    ++iterations;
                }
            }

            if (iterations > 0) {
                for (int d = 0; d < 4; ++d) {
                    values[d] = sums[d] / iterations;
                }
            } else {
                for (int d = 0; d < 4; ++d) {
                    values[d] = candidateHeuristic[d];
                }
            }
        }

        int bestDir = 0;
        for (int d = 1; d < 4; ++d) {
            if (values[d] > values[bestDir] + 1e-9 ||
                (abs(values[d] - values[bestDir]) <= 1e-9 &&
                 candidateHeuristic[d] > candidateHeuristic[bestDir])) {
                bestDir = d;
            }
        }

        board = candidates[bestDir];
        cout << DIR_CHAR[bestDir] << '\n' << flush;
    }

    return 0;
}

struct RNG {
    uint32_t x;
    explicit RNG(uint32_t seed = 2463534242u) : x(seed ? seed : 1u) {}
    inline uint32_t next() {
        x ^= x << 13;
        x ^= x >> 17;
        x ^= x << 5;
        return x;
    }
};

struct State {
    array<uint8_t, 100> a{};
};

class Solver {
    static constexpr char DIR_CHAR[4] = {'F', 'B', 'L', 'R'};
    int flavor[100];
    RNG rng;

    static void place(State &s, int f, int p) {
        for (int i = 0; i < 100; ++i) {
            if (s.a[i] == 0 && --p == 0) {
                s.a[i] = static_cast<uint8_t>(f);
                return;
            }
        }
    }

    static void tilt(State &s, int dir) {
        uint8_t tmp[10];
        if (dir == 0) {
            for (int c = 0; c < 10; ++c) {
                int n = 0;
                for (int r = 0; r < 10; ++r) {
                    uint8_t v = s.a[r * 10 + c];
                    if (v) tmp[n++] = v;
                }
                for (int r = 0; r < 10; ++r)
                    s.a[r * 10 + c] = r < n ? tmp[r] : 0;
            }
        } else if (dir == 1) {
            for (int c = 0; c < 10; ++c) {
                int n = 0;
                for (int r = 9; r >= 0; --r) {
                    uint8_t v = s.a[r * 10 + c];
                    if (v) tmp[n++] = v;
                }
                for (int r = 9, k = 0; r >= 0; --r, ++k)
                    s.a[r * 10 + c] = k < n ? tmp[k] : 0;
            }
        } else if (dir == 2) {
            for (int r = 0; r < 10; ++r) {
                int n = 0;
                for (int c = 0; c < 10; ++c) {
                    uint8_t v = s.a[r * 10 + c];
                    if (v) tmp[n++] = v;
                }
                for (int c = 0; c < 10; ++c)
                    s.a[r * 10 + c] = c < n ? tmp[c] : 0;
            }
        } else {
            for (int r = 0; r < 10; ++r) {
                int n = 0;
                for (int c = 9; c >= 0; --c) {
                    uint8_t v = s.a[r * 10 + c];
                    if (v) tmp[n++] = v;
                }
                for (int c = 9, k = 0; c >= 0; --c, ++k)
                    s.a[r * 10 + c] = k < n ? tmp[k] : 0;
            }
        }
    }

    static int score(const State &s) {
        uint8_t seen[100] = {};

int componentScore(const Board& board) {

    constexpr int dc[4] = {0, 0, -1, 1};

    for (int start = 0; start < CELLS; ++start) {
        if (board[start] == 0 || visited[start]) continue;

        int top = 0;
        int size = 0;
        stack[top++] = start;
        visited[start] = true;
        unsigned char color = board[start];

        while (top > 0) {
            int pos = stack[--top];
            ++size;

            int r = pos / N;
            int c = pos % N;

            for (int k = 0; k < 4; ++k) {

                if (board[next] == color && !visited[next]) {

                    stack[top++] = next;
                }

Evaluation evaluateBoard(const Board& board) {
    Evaluation result;
    bool visited[CELLS] = {};
    int stack[CELLS];

    int rowCount[3][N] = {};
    int columnCount[3][N] = {};

    for (int pos = 0; pos < CELLS; ++pos) {
        if (board[pos] == 0) continue;
        int color = board[pos] - 1;
        ++rowCount[color][pos / N];
        ++columnCount[color][pos % N];
    }

    for (int color = 0; color < 3; ++color) {
        for (int i = 0; i < N; ++i) {
            result.projection += rowCount[color][i] * rowCount[color][i];
            result.projection += columnCount[color][i] * columnCount[color][i];
        }
    }

    constexpr int dr[4] = {-1, 1, 0, 0};
    constexpr int dc[4] = {0, 0, -1, 1};

    for (int start = 0; start < CELLS; ++start) {
        if (board[start] == 0 || visited[start]) continue;

        int top = 0;
        int size = 0;
        stack[top++] = start;
        visited[start] = true;
        unsigned char color = board[start];

        while (top > 0) {

            ++size;

            int r = pos / N;
            int c = pos % N;

            for (int k = 0; k < 4; ++k) {
                int nr = r + dr[k];
                int nc = c + dc[k];
                if (nr < 0 || nr >= N || nc < 0 || nc >= N) continue;

                int next = nr * N + nc;

                    ++result.allEdges;
                    if (board[next] == color) ++result.sameEdges;
