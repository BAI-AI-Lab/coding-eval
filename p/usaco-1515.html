import sys


def main():
    input = sys.stdin.buffer.readline
    n, m = map(int, input().split())

    # wins[j] is a bitmask of all symbols that beat symbol j.
    wins = [0] * n
    bits = [1 << i for i in range(n)]

    for i in range(n):
        row = input().strip()
        for j, result in enumerate(row):
            if result == 87:  # 'W': symbol i+1 beats symbol j+1
                wins[j] |= bits[i]
            elif result == 76:  # 'L': symbol j+1 beats symbol i+1
                wins[i] |= bits[j]

    answers = []
    total = n * n

    for _ in range(m):
        a, b = map(int, input().split())
        both = wins[a - 1] & wins[b - 1]
        k = both.bit_count()

        # A pair guarantees a win iff at least one of its two symbols
        # beats both of Elsie's possible symbols.
        bad = (n - k) * (n - k)
        answers.append(str(total - bad))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

import sys


def main() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # winmask[x] has bit y set iff symbol y beats symbol x.
    winmask = [0] * (n + 1)
    bits = [1 << i for i in range(n + 1)]

    for i in range(1, n + 1):
        row = next(it)
        wi = winmask[i]

        for j, ch in enumerate(row, start=1):
            if ch == ord('W'):      # symbol i beats symbol j
                winmask[j] |= bits[i]
            elif ch == ord('L'):    # symbol i loses to symbol j
                wi |= bits[j]

        winmask[i] = wi

    answers = []
    for _ in range(m):
        s1 = int(next(it))
        s2 = int(next(it))

        # Number of symbols that beat both of Elsie's symbols.
        common_winners = (winmask[s1] & winmask[s2]).bit_count()

        # Count ordered pairs (L, R) containing at least one common winner.
        ans = 2 * n * common_winners - common_winners * common_winners
        answers.append(str(ans))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

import sys


def solve():
    read = sys.stdin.buffer.readline
    first = read().split()
    if not first:
        return

    n, m = map(int, first)

    # incoming[s] will have bit x set exactly when symbol x beats symbol s.
    byte_count = (n + 7) // 8
    incoming_bytes = [bytearray(byte_count) for _ in range(n)]
    bit_in_byte = (1, 2, 4, 8, 16, 32, 64, 128)

    WIN = ord('W')
    LOSE = ord('L')

    for i in range(n):
        row = read().strip()
        i_byte = i >> 3
        i_bit = bit_in_byte[i & 7]
        incoming_i = incoming_bytes[i]

        for j, outcome in enumerate(row):
            if outcome == WIN:
                # Symbol i beats symbol j, so i beats incoming target j.
                incoming_bytes[j][i_byte] |= i_bit
            elif outcome == LOSE:
                # Symbol j beats symbol i, so j beats incoming target i.
                incoming_i[j >> 3] |= bit_in_byte[j & 7]

    incoming = [
        int.from_bytes(mask, "little")
        for mask in incoming_bytes
    ]

    try:
        popcount = int.bit_count
    except AttributeError:
        def popcount(value):
            return bin(value).count("1")

    total = n * n
    answers = []

    for _ in range(m):
        s1, s2 = map(int, read().split())
        s1 -= 1
        s2 -= 1

        # Symbols that beat both of Elsie's possible choices.
        common_winners = popcount(incoming[s1] & incoming[s2])

        # A Bessie pair fails exactly when neither hoof holds a common winner.
        failing_choices = n - common_winners
        answer = total - failing_choices * failing_choices
        answers.append(str(answer))

    sys.stdout.write("\n".join(answers) + "\n")


if __name__ == "__main__":
    solve()
