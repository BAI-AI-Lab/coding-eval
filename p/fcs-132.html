#include <bits/stdc++.h>
using namespace std;

static inline uint64_t mix64(uint64_t x) {
    x += 0x9e3779b97f4a7c15ULL;
    x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
    x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
    return x ^ (x >> 31);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int R, H;
    if (!(cin >> R >> H)) return 0;

    const auto startTime = chrono::steady_clock::now();

    vector<uint64_t> code;
    code.reserve(1000);
    vector<vector<int>> positions(49);

    for (int id = 0; id < 2401 && (int)code.size() < 1000; ++id) {
        int c[4], t = id;
        for (int j = 0; j < 4; ++j) {
            c[j] = t % 7;
            t /= 7;
        }

        int val[7];
        bool ok = true;
        for (int x = 0; x < 7; ++x) {
            int v = c[3];
            v = (v * x + c[2]) % 7;
            v = (v * x + c[1]) % 7;
            v = (v * x + c[0]) % 7;
            val[x] = v;
            if (x < 5 && v == 0) ok = false;
        }
        if (!ok) continue;

        int pos = (int)code.size() + 1;
        uint64_t mask = 0;
        for (int x = 0; x < 7; ++x) {
            int bit = x * 7 + val[x];
            mask |= 1ULL << bit;
            positions[bit].push_back(pos);
        }
        code.push_back(mask);
    }

    const int N = (int)code.size();
    vector<uint64_t> unions;
    unions.reserve(N * (N + 1) / 2);
    for (int i = 0; i < N; ++i)
        for (int j = i; j < N; ++j)
            unions.push_back(code[i] | code[j]);

    uint64_t currentMask = 0;
    for (uint64_t c : code) currentMask |= c;

    vector<int> order;
    for (int s = 0; s < 7; ++s)
        for (int b = 0; b < 7; ++b) {
            int bit = b * 7 + s;
            if ((currentMask >> bit) & 1ULL) order.push_back(bit);
        }

    const int HS = 1 << 20;
    vector<uint64_t> keys(HS);
    vector<uint32_t> tags(HS, 0);
    uint32_t stamp = 0;

    auto uniqueForMask = [&](uint64_t mask) {
        ++stamp;
        for (uint64_t full : unions) {
            uint64_t key = full & mask;
            int h = (int)(mix64(key) & (HS - 1));
            while (tags[h] == stamp) {
                if (keys[h] == key) return false;
                h = (h + 1) & (HS - 1);
            }
            tags[h] = stamp;
            keys[h] = key;
        }
        return true;
    };

    for (int bit : order) {
        if (__builtin_popcountll(currentMask) <= 26) break;
        double elapsed = chrono::duration<double>(
            chrono::steady_clock::now() - startTime).count();
        if (elapsed > 0.10) break;

        uint64_t candidate = currentMask & ~(1ULL << bit);
        if (uniqueForMask(candidate)) currentMask = candidate;
    }

    vector<int> sentBits;
    for (int bit = 0; bit < 49; ++bit)
        if ((currentMask >> bit) & 1ULL)
            sentBits.push_back(bit);

    for (int bit : sentBits) {
        const auto &v = positions[bit];
        cout << "? " << v.size();
        for (int p : v) cout << ' ' << p;
        cout << '\n';
        cout.flush();
    }

    cout << "@\n";
    cout.flush();

    int L;
    cin >> L;
    uint64_t answer = 0;
    for (int i = 0; i < L; ++i) {
        int x;
        cin >> x;
        if (x && i < (int)sentBits.size())
            answer |= 1ULL << sentBits[i];
    }

    for (int i = 0; i < N; ++i) {
        for (int j = i; j < N; ++j) {
            if (((code[i] | code[j]) & currentMask) == answer) {
                cout << "! " << i + 1 << ' ' << j + 1 << '\n';
                cout.flush();
                return 0;
            }
        }
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

static constexpr int N = 1000;
static constexpr int BITS = 10;
static constexpr int LIMIT = (1 << BITS) - 1;

/*
For two labels a and b, let d = a XOR b and D = support(d).

After querying both the zero-set and the one-set of every bit, we know
the unordered pair of bits at each position. The only remaining ambiguity
is that, independently at every bit in D, the two labels may be swapped.

For a parity mask t, querying positions for which parity(label & t) is 1
distinguishes such a swap at a nonempty proper subset S of D exactly when

    parity(t & D) = 0 and parity(t & S) = 1.

The following routine constructs a small family of parity masks hitting
every possible pair (D, S).
*/
static vector<int> buildHittingSet() {
    constexpr int MASKS = 1 << BITS;
    constexpr int WORDS = MASKS / 64;

    int constraintCount = 0;
    for (int d = 1; d <= LIMIT; ++d) {
        constraintCount += (1 << __builtin_popcount((unsigned)d)) - 2;
    }

    vector<uint64_t> rows((size_t)constraintCount * WORDS, 0);
    vector<int> gain(MASKS, 0);

    int index = 0;
    for (int d = 1; d <= LIMIT; ++d) {
        for (int s = (d - 1) & d; s; s = (s - 1) & d) {
            uint64_t *row = rows.data() + (size_t)index * WORDS;

            int p = s & -s;
            int difference = d ^ s;
            int q = difference & -difference;
            int freeMask = LIMIT ^ (p | q);

            int u = freeMask;
            while (true) {
                int parityS = __builtin_parity(
                    (unsigned)(u & (s ^ p))
                );
                int parityDifference = __builtin_parity(
                    (unsigned)(u & (difference ^ q))
                );

                int valueP = 1 ^ parityS;
                int valueQ = 1 ^ parityDifference;

                int t = u;
                if (valueP) t |= p;
                if (valueQ) t |= q;

                row[t >> 6] |= 1ULL << (t & 63);
                ++gain[t];

                if (u == 0) break;
                u = (u - 1) & freeMask;
            }
            ++index;
        }
    }

    vector<unsigned char> covered(constraintCount, 0);
    vector<int> selected;
    int remaining = constraintCount;

    while (remaining > 0 && selected.size() < 45) {
        int best = -1;
        for (int t = 0; t < MASKS; ++t) {
            if (best == -1 || gain[t] > gain[best]) best = t;
        }

        if (best == -1 || gain[best] == 0) break;
        selected.push_back(best);

        for (int i = 0; i < constraintCount; ++i) {
            if (covered[i]) continue;

            const uint64_t *row = rows.data() + (size_t)i * WORDS;
            if (!(row[best >> 6] & (1ULL << (best & 63)))) continue;

            covered[i] = 1;
            --remaining;

            for (int w = 0; w < WORDS; ++w) {
                uint64_t values = row[w];
                while (values) {
                    int bit = __builtin_ctzll(values);
                    values &= values - 1;
                    --gain[(w << 6) | bit];
                }
            }
        }
    }

    if (remaining == 0 && selected.size() <= 45) {
        sort(selected.begin(), selected.end());
        return selected;
    }

    // Every pair (D, S) is hit by a mask {i,j} with i in S and j in D\S.
    vector<int> fallback;
    for (int i = 0; i < BITS; ++i) {
        for (int j = i + 1; j < BITS; ++j) {
            fallback.push_back((1 << i) | (1 << j));
        }
    }
    return fallback;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int R, H;
    if (!(cin >> R >> H)) return 0;

    vector<int> parityMasks = buildHittingSet();

    vector<uint16_t> valueMask(N), zeroMask(N);
    vector<uint64_t> checksumMask(N, 0);

    for (int label = 0; label < N; ++label) {
        uint16_t ones = 0;
        for (int bit = 0; bit < BITS; ++bit) {
            if ((label >> bit) & 1) ones |= uint16_t(1 << bit);
        }
        valueMask[label] = ones;
        zeroMask[label] = uint16_t(LIMIT ^ ones);

        for (int j = 0; j < (int)parityMasks.size(); ++j) {
            if (__builtin_parity((unsigned)(label & parityMasks[j]))) {
                checksumMask[label] |= 1ULL << j;
            }
        }
    }

    vector<vector<int>> queries;
    queries.reserve(20 + parityMasks.size());

    for (int bit = 0; bit < BITS; ++bit) {
        vector<int> zeros, ones;
        for (int position = 0; position < N; ++position) {
            if ((position >> bit) & 1) {
                ones.push_back(position + 1);
            } else {
                zeros.push_back(position + 1);
            }
        }
        queries.push_back(move(zeros));
        queries.push_back(move(ones));
    }

    for (int t : parityMasks) {
        vector<int> positions;
        for (int position = 0; position < N; ++position) {
            if (__builtin_parity((unsigned)(position & t))) {
                positions.push_back(position + 1);
            }
        }
        queries.push_back(move(positions));
    }

    for (const auto &query : queries) {
        cout << "? " << query.size();
        for (int position : query) cout << ' ' << position;
        cout << endl;
    }

    cout << "@" << endl;

    int returned;
    if (!(cin >> returned)) return 0;

    vector<int> response(returned);
    for (int &answer : response) {
        if (!(cin >> answer)) return 0;
    }

    int required = 20 + (int)parityMasks.size();
    if (returned < required) return 0;

    uint16_t observedOnes = 0;
    uint16_t observedZeros = 0;

    for (int bit = 0; bit < BITS; ++bit) {
        if (response[2 * bit]) observedOnes |= uint16_t(1 << bit);
        if (response[2 * bit + 1]) observedZeros |= uint16_t(1 << bit);
    }

    uint64_t observedChecksum = 0;
    for (int j = 0; j < (int)parityMasks.size(); ++j) {
        if (response[20 + j]) observedChecksum |= 1ULL << j;
    }

    int answerA = -1, answerB = -1;

    for (int a = 0; a < N && answerA == -1; ++a) {
        for (int b = a; b < N; ++b) {
            if ((valueMask[a] | valueMask[b]) != observedOnes) continue;
            if ((zeroMask[a] | zeroMask[b]) != observedZeros) continue;
            if ((checksumMask[a] | checksumMask[b]) != observedChecksum) continue;

            answerA = a + 1;
            answerB = b + 1;
            break;
        }
    }

    if (answerA == -1) return 0;

    cout << "! " << answerA << ' ' << answerB << endl;
    return 0;
}
