import sys
from itertools import product

def solve():
    data = sys.stdin.read().split()
    if not data:
        return

    N = int(data[0])
    S = data[1]
    T = data[2]

    f = [-1] * 26

    for s, t in zip(S, T):
        a = ord(s) - ord('a')
        b = ord(t) - ord('a')

        if f[a] == -1:
            f[a] = b
        elif f[a] != b:
            print(-1)
            return

    present = [i for i in range(26) if f[i] != -1]
    targets = set(f[i] for i in present)

    # If all 26 letters must remain present, no temporary letter is available.
    # The only possible transformation is the identity.
    if len(targets) == 26:
        possible = all(f[i] == i for i in present)
        print(0 if possible else -1)
        return

    groups_by_target = {d: [] for d in targets}
    for a in present:
        groups_by_target[f[a]].append(a)

    groups = list(groups_by_target.values())
    merge_count = len(present) - len(targets)

    answer = 10**9

    for reps in product(*groups):
        selected = set(reps)

        non_fixed = sum(1 for a in reps if f[a] != a)

        visited = set()
        cycles = 0

        for start in reps:
            if start in visited:
                continue

            cur = start
            path_pos = {}
            path = []

            while cur in selected and cur not in visited:
                visited.add(cur)
                path_pos[cur] = len(path)
                path.append(cur)
                cur = f[cur]

            if cur in selected and cur in path_pos:
                cycle_len = len(path) - path_pos[cur]
                if cycle_len >= 2:
                    cycles += 1

        cost = merge_count + non_fixed + cycles
        answer = min(answer, cost)

    print(answer)


if __name__ == "__main__":
    solve()

import sys

def solve():
    input = sys.stdin.readline
    N = int(input().strip())
    S = input().strip()
    T = input().strip()

    f = [-1] * 26
    domain = [False] * 26
    target_used = [False] * 26

    for sc, tc in zip(S, T):
        s = ord(sc) - 97
        t = ord(tc) - 97

        if f[s] == -1:
            f[s] = t
        elif f[s] != t:
            print(-1)
            return

        domain[s] = True
        target_used[t] = True

    # If S uses all 26 letters and T uses all 26 letters but S != T,
    # no operation can be performed without reducing the number of distinct letters.
    if S != T and all(domain) and all(target_used):
        print(-1)
        return

    # Count characters that must be changed at least once.
    changed = 0
    for c in range(26):
        if domain[c] and f[c] != c:
            changed += 1

    # In-degree from source letters.
    indeg = [0] * 26
    for c in range(26):
        if domain[c]:
            indeg[f[c]] += 1

    # Find cycles of length >= 2 that have no extra incoming edges from outside the cycle.
    state = [0] * 26
    extra = 0

    for v in range(26):
        if domain[v] and state[v] == 0:
            path = []
            cur = v

            while domain[cur] and state[cur] == 0:
                state[cur] = 1
                path.append(cur)
                cur = f[cur]

            if domain[cur] and state[cur] == 1:
                idx = path.index(cur)
                cycle = path[idx:]

                if len(cycle) >= 2:
                    pure = True
                    for node in cycle:
                        if indeg[node] != 1:
                            pure = False
                            break
                    if pure:
                        extra += 1

            for node in path:
                state[node] = 2

    print(changed + extra)

if __name__ == "__main__":
    solve()
