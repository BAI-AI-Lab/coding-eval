import sys


def can_make(a, k):
    n = len(a)

    if k == 1:
        return all(x == a[0] for x in a)
    if n <= k:
        return True

    inf = k + 1

    # lcp[i][j] = length of the longest common prefix of a[i:] and a[j:].
    lcp = [[0] * (n + 1) for _ in range(n + 1)]
    for i in range(n - 1, -1, -1):
        for j in range(n - 1, -1, -1):
            if a[i] == a[j]:
                lcp[i][j] = lcp[i + 1][j + 1] + 1

    # Proper base lengths that can be repeated to obtain each length.
    divisors = [[] for _ in range(n + 1)]
    for length in range(2, n + 1):
        divisors[length] = [
            d for d in range(1, length // 2 + 1) if length % d == 0
        ]

    # dp[l][r] = minimum PRINT statements needed to output a[l:r],
    # capped at k+1 when more than k are needed.
    dp = [[inf] * (n + 1) for _ in range(n)]
    for i in range(n):
        dp[i][i + 1] = 1

    for length in range(2, n + 1):
        for left in range(n - length + 1):
            right = left + length
            best = inf

            # Wrap a program for a shorter substring in REP, repeating it.
            for base_len in divisors[length]:
                if lcp[left][left + base_len] >= length - base_len:
                    best = min(best, dp[left][left + base_len])
                    if best == 1:
                        break

            # Concatenate programs for two consecutive substrings.
            if best != 1:
                for mid in range(left + 1, right):
                    cost = dp[left][mid] + dp[mid][right]
                    if cost < best:
                        best = cost
                        if best == 2:
                            break

            dp[left][right] = best

    return dp[0][n] <= k


def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    it = iter(data)
    t = int(next(it))
    answers = []

    for _ in range(t):
        n = int(next(it))
        k = int(next(it))
        a = [int(next(it)) for _ in range(n)]
        answers.append("YES" if can_make(a, k) else "NO")

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

import sys


def can_make(seq, K):
    n = len(seq)
    INF = 10**9

    # dp[l][r] = minimum number of PRINT statements needed
    # to output seq[l:r].
    dp = [[0] * (n + 1) for _ in range(n + 1)]

    for i in range(n):
        dp[i][i + 1] = 1

    for length in range(2, n + 1):
        for l in range(n - length + 1):
            r = l + length
            best = INF

            # Concatenate two programs.
            for mid in range(l + 1, r):
                val = dp[l][mid] + dp[mid][r]
                if val < best:
                    best = val

            # Repeat a smaller program.
            for p in range(1, length):
                if length % p != 0:
                    continue

                if dp[l][l + p] >= best:
                    continue

                ok = True
                for i in range(l + p, r):
                    if seq[i] != seq[i - p]:
                        ok = False
                        break

                if ok:
                    best = dp[l][l + p]

            dp[l][r] = best

    return dp[0][n] <= K


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    idx = 1
    answers = []

    for _ in range(t):
        n = data[idx]
        k = data[idx + 1]
        idx += 2

        seq = data[idx:idx + n]
        idx += n

        answers.append("YES" if can_make(seq, k) else "NO")

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":

import sys


def build_proper_divisors(limit):
    """divisors[length] contains every proper divisor of length."""
    divisors = [[] for _ in range(limit + 1)]
    for d in range(1, limit // 2 + 1):
        for length in range(2 * d, limit + 1, d):
            divisors[length].append(d)
    return divisors


def can_generate(sequence, k, proper_divisors):
    n = len(sequence)

    # With one PRINT statement, the output must be constant.
    if k == 1:
        first = sequence[0]
        return all(value == first for value in sequence)

    INF = k + 1

    # run_end[i] is the exclusive end of the constant run starting at i.
    run_end = [n] * n
    for i in range(n - 2, -1, -1):
        if sequence[i] == sequence[i + 1]:
            run_end[i] = run_end[i + 1]
        else:
            run_end[i] = i + 1

    # lcp[i][j] = longest common prefix of sequence[i:] and sequence[j:].
    # Only entries with i < j are needed.
    lcp = [[0] * (n + 1) for _ in range(n + 1)]
    for i in range(n - 1, -1, -1):
        current = lcp[i]
        following = lcp[i + 1]
        value = sequence[i]
        for j in range(n - 1, i, -1):
            if value == sequence[j]:
                current[j] = following[j + 1] + 1

    # dp[left][right] is the minimum number of PRINT statements needed
    # for sequence[left:right], capped at INF.
    dp = [[INF] * (n + 1) for _ in range(n + 1)]

    for length in range(1, n + 1):
        divisors = proper_divisors[length]

        for left in range(n - length + 1):
            right = left + length
            row = dp[left]

            # REP length PRINT value generates a constant sequence.
            if right <= run_end[left]:
                row[right] = 1
                continue

            best = INF
            lcp_row = lcp[left]

            # The sequence may be repetitions of a shorter program.
            for d in divisors:
                prefix_cost = row[left + d]
                if prefix_cost < best and lcp_row[left + d] >= length - d:
                    best = prefix_cost
                    # A nonconstant sequence requires at least two PRINTs.
                    if best == 2:
                        break

            # The sequence may be the concatenation of two programs.
            if best > 2:
                for middle in range(left + 1, right):
                    cost = row[middle] + dp[middle][right]
                    if cost < best:
                        best = cost
                        if best == 2:
                            break

            row[right] = best

    return dp[0][n] <= k


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    t = data[0]
    position = 1
    tests = []
    max_n = 0

    for _ in range(t):
        n = data[position]
        k = data[position + 1]
        position += 2

        sequence = data[position:position + n]
        position += n

        tests.append((sequence, k))
        max_n = max(max_n, n)

    proper_divisors = build_proper_divisors(max_n)

    answers = []
    for sequence, k in tests:
        if can_generate(sequence, k, proper_divisors):
            answers.append("YES")
        else:
            answers.append("NO")

    sys.stdout.write("\n".join(answers) + "\n")


if __name__ == "__main__":
    main()
