import sys


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, k = data[0], data[1]
    q = data[2]
    pos = 3

    beauty = [[0] * n for _ in range(n)]

    # sums[i][j] is the sum of the K x K square with upper-left corner (i, j).
    m = n - k + 1
    sums = [[0] * m for _ in range(m)]

    best = 0
    answers = []

    for _ in range(q):
        r = data[pos] - 1
        c = data[pos + 1] - 1
        value = data[pos + 2]
        pos += 3

        delta = value - beauty[r][c]
        beauty[r][c] = value

        # Only squares containing (r, c) change. Since every update is
        # positive, the global maximum can never decrease.
        i_start = max(0, r - k + 1)
        i_end = min(r, m - 1)
        j_start = max(0, c - k + 1)
        j_end = min(c, m - 1)

        for i in range(i_start, i_end + 1):
            row = sums[i]
            for j in range(j_start, j_end + 1):
                new_sum = row[j] + delta
                row[j] = new_sum
                if new_sum > best:
                    best = new_sum

        answers.append(str(best))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

import sys


def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    N = next(it)
    K = next(it)
    Q = next(it)

    W = N - K + 1
    B = 24
    BR = (W + B - 1) // B

    # val[i][j] is the window sum excluding its block's lazy value.
    val = [[0] * W for _ in range(W)]
    lazy = [[0] * BR for _ in range(BR)]
    bmax = [[0] * BR for _ in range(BR)]

    cows = [[0] * N for _ in range(N)]

    global_max = 0
    answers = []

    for _ in range(Q):
        r = next(it) - 1
        c = next(it) - 1
        v = next(it)

        old = cows[r][c]
        delta = v - old
        cows[r][c] = v

        if delta:
            r1 = r - K + 1
            if r1 < 0:
                r1 = 0
            r2 = r
            if r2 >= W:
                r2 = W - 1

            c1 = c - K + 1
            if c1 < 0:
                c1 = 0
            c2 = c
            if c2 >= W:
                c2 = W - 1

            br1 = r1 // B
            br2 = r2 // B
            bc1 = c1 // B
            bc2 = c2 // B

            for bi in range(br1, br2 + 1):
                i_start = bi * B
                i_end = i_start + B - 1
                if i_end >= W:
                    i_end = W - 1

                ir1 = r1 if r1 > i_start else i_start
                ir2 = r2 if r2 < i_end else i_end
                if ir1 > ir2:
                    continue

                lazy_row = lazy[bi]
                bmax_row = bmax[bi]

                for bj in range(bc1, bc2 + 1):
                    j_start = bj * B
                    j_end = j_start + B - 1
                    if j_end >= W:
                        j_end = W - 1

                    jc1 = c1 if c1 > j_start else j_start
                    jc2 = c2 if c2 < j_end else j_end
                    if jc1 > jc2:
                        continue

                    if (
                        i_start >= r1
                        and i_end <= r2
                        and j_start >= c1
                        and j_end <= c2
                    ):
                        # Whole block is covered.
                        lazy_row[bj] += delta
                        bm = bmax_row[bj] + delta
                        bmax_row[bj] = bm
                        if bm > global_max:
                            global_max = bm
                    else:
                        # Only part of the block is covered.
                        lz = lazy_row[bj]
                        max_new = -1

                        for i in range(ir1, ir2 + 1):
                            row = val[i]
                            for j in range(jc1, jc2 + 1):
                                nv = row[j] + delta
                                row[j] = nv
                                actual = nv + lz
                                if actual > max_new:
                                    max_new = actual

                        bm = bmax_row[bj]
                        if max_new > bm:
                            bm = max_new
                            bmax_row[bj] = bm

                        if bm > global_max:
                            global_max = bm

        answers.append(str(global_max))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()

import sys


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, k, q = data[0], data[1], data[2]
    width = n - k + 1

    beauty = [[0] * n for _ in range(n)]

    # picture_sums[rr][cc] is the sum of the K x K picture whose
    # upper-left corner is (rr, cc).
    picture_sums = [[0] * width for _ in range(width)]

    # For each row/column, the possible upper-left coordinates of
    # pictures containing that row/column.
    valid_starts = [
        tuple(range(max(0, x - k + 1), min(x, width - 1) + 1))
        for x in range(n)
    ]

    best = 0
    answers = []
    pos = 3

    for _ in range(q):
        r = data[pos] - 1
        c = data[pos + 1] - 1
        v = data[pos + 2]
        pos += 3

        delta = v - beauty[r][c]
        beauty[r][c] = v

        columns = valid_starts[c]

        for rr in valid_starts[r]:
            picture_row = picture_sums[rr]
            for cc in columns:
                new_sum = picture_row[cc] + delta
                picture_row[cc] = new_sum
                if new_sum > best:
                    best = new_sum

        answers.append(str(best))

    sys.stdout.write("\n".join(answers) + "\n")


if __name__ == "__main__":
    solve()
