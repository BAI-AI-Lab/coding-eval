#include <bits/stdc++.h>
using namespace std;

struct Cell {
    int x, y;
};

struct Orientation {
    int w = 0, h = 0, r = 0, f = 0;
    long long mx = 0, my = 0;
    vector<Cell> cells;
    vector<pair<int, int>> rowReq;
    array<uint16_t, 10> rowMask{};
};

struct Piece {
    int k = 0;
    int bboxArea = 0;
    int maxDim = 0;
    vector<Orientation> orientations;
};

struct Placement {
    int orient = 0;
    int x = 0, y = 0;
};

struct Result {
    int W = 0, H = 0;
    vector<Placement> placements;
};

static uint64_t mixHash(uint64_t x) {
    x += 0x9e3779b97f4a7c15ULL;
    x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
    x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
    return x ^ (x >> 31);
}

static Result packStrip(const vector<Piece>& pieces, const vector<int>& order, int W) {
    const int n = (int)pieces.size();
    const int words = (W + 63) / 64;
    constexpr int MAX_WORDS = 8;

    vector<uint64_t> occupied;
    vector<int> freeCells;
    int H = 0;
    int firstNonFull = 0;

    Result result;
    result.placements.resize(n);

    const int dx[4] = {1, -1, 0, 0};
    const int dy[4] = {0, 0, 1, -1};

    for (int id : order) {
        const Piece& piece = pieces[id];
        bool placed = false;

        for (int y = firstNonFull; y <= H && !placed; ++y) {
            if (y < H && freeCells[y] == 0) continue;

            int bestOrient = -1;
            int bestX = INT_MAX;
            int bestTop = INT_MAX;
            int bestContact = -1;

            for (int oi = 0; oi < (int)piece.orientations.size(); ++oi) {
                const Orientation& o = piece.orientations[oi];
                if (o.w > W) continue;

                int top = max(H, y + o.h);
                if (bestOrient != -1 && top > bestTop) continue;

                bool possible = true;
                for (auto [ry, need] : o.rowReq) {
                    int row = y + ry;
                    if (row < H && freeCells[row] < need) {
                        possible = false;
                        break;
                    }
                }
                if (!possible) continue;

                int numX = W - o.w + 1;
                int usedWords = (numX + 63) / 64;
                uint64_t valid[MAX_WORDS];

                for (int i = 0; i < words; ++i) valid[i] = ~0ULL;
                for (int i = usedWords; i < words; ++i) valid[i] = 0;

                int rem = numX & 63;
                if (rem) valid[usedWords - 1] = (1ULL << rem) - 1;
                else valid[usedWords - 1] = ~0ULL;

                bool any = true;
                for (const Cell& c : o.cells) {
                    int row = y + c.y;
                    if (row >= H) continue;

                    const uint64_t* bits = &occupied[(size_t)row * words];
                    int sh = c.x;
                    any = false;

                    if (sh == 0) {
                        for (int i = 0; i < usedWords; ++i) {
                            valid[i] &= ~bits[i];
                            any |= (valid[i] != 0);
                        }
                    } else {
                        for (int i = 0; i < usedWords; ++i) {
                            uint64_t shifted = bits[i] >> sh;
                            if (i + 1 < words) {
                                shifted |= bits[i + 1] << (64 - sh);
                            }
                            valid[i] &= ~shifted;
                            any |= (valid[i] != 0);
                        }
                    }
                    if (!any) break;
                }
                if (!any) continue;

                for (int wi = 0; wi < usedWords; ++wi) {
                    uint64_t mask = valid[wi];
                    while (mask) {
                        int b = __builtin_ctzll(mask);
                        mask &= mask - 1;
                        int x = wi * 64 + b;
                        if (x >= numX) continue;

                        int contact = 0;
                        for (const Cell& c : o.cells) {
                            int ax = x + c.x;
                            int ay = y + c.y;

                            for (int d = 0; d < 4; ++d) {
                                int sx = c.x + dx[d];
                                int sy = c.y + dy[d];

                                if (sx >= 0 && sx < o.w && sy >= 0 && sy < o.h &&
                                    ((o.rowMask[sy] >> sx) & 1U)) {
                                    continue;
                                }

                                int nx = ax + dx[d];
                                int ny = ay + dy[d];

                                if (nx < 0 || nx >= W || ny < 0) {
                                    ++contact;
                                } else if (ny < H) {
                                    if ((occupied[(size_t)ny * words + (nx >> 6)] >> (nx & 63)) & 1ULL) {
                                        ++contact;
                                    }
                                }
                            }
                        }

                        bool better = false;
                        if (bestOrient == -1) better = true;
                        else if (top != bestTop) better = top < bestTop;
                        else if (contact != bestContact) better = contact > bestContact;
                        else if (x != bestX) better = x < bestX;

                        if (better) {
                            bestOrient = oi;
                            bestX = x;
                            bestTop = top;
                            bestContact = contact;
                        }
                    }
                }
            }

            if (bestOrient != -1) {
                const Orientation& o = piece.orientations[bestOrient];
                int newH = max(H, y + o.h);

                if (newH > H) {
                    occupied.resize((size_t)newH * words, 0ULL);
                    freeCells.resize(newH, W);
                    H = newH;
                }

                for (const Cell& c : o.cells) {
                    int ax = bestX + c.x;
                    int ay = y + c.y;
                    occupied[(size_t)ay * words + (ax >> 6)] |= 1ULL << (ax & 63);
                    --freeCells[ay];
                }

                result.placements[id] = {bestOrient, bestX, y};
                placed = true;

                while (firstNonFull < H && freeCells[firstNonFull] == 0) {
                    ++firstNonFull;
                }
            }
        }
    }

    int minX = W;
    int maxX = -1;
    for (int i = 0; i < n; ++i) {
        const Placement& p = result.placements[i];
        const Orientation& o = pieces[i].orientations[p.orient];
        for (const Cell& c : o.cells) {
            int x = p.x + c.x;
            minX = min(minX, x);
            maxX = max(maxX, x);
        }
    }

    for (Placement& p : result.placements) p.x -= minX;
    result.W = maxX - minX + 1;
    result.H = H;
    return result;
}

struct Orientation {
    int w = 0, h = 0, R = 0, F = 0;
    int minX = 0, minY = 0;
    vector<pair<int,int>> cells;
    vector<unsigned long long> rows;
};

struct Piece {
    int k = 0;
    vector<Orientation> orientations;
};

struct PackResult {
    long long area = LLONG_MAX;
    int W = 0, H = 0;
    vector<array<int,4>> placement;
};

    ios::sync_with_stdio(false);

    int n;

    vector<Piece> pieces(n);
    int totalCells = 0;

    for (int i = 0; i < n; ++i) {

        vector<pair<int,int>> original(k);

        }

        for (int f = 0; f <= 1; ++f) {
            for (int r = 0; r < 4; ++r) {

    auto pack = [&](int width, int mode, int buffer,
                    const vector<int> &initialOrder) -> PackResult {
        vector<unsigned long long> occupied(2 * totalCells + 64, 0);
        vector<int> remaining = initialOrder;

        PackResult result;
        result.placement.resize(n);

        auto ensureRow = [&](int row) {
            if (row >= (int)occupied.size()) {
                occupied.resize(row + 1, 0);
            }

        int cursorX = 0, cursorY = 0;
        int minUsedX = INT_MAX, minUsedY = INT_MAX;
        int maxUsedX = -1, maxUsedY = -1;

        while (!remaining.empty()) {
            ensureRow(cursorY);

            while (true) {
                if (cursorX >= width) {
                    cursorX = 0;
                    ++cursorY;
                    ensureRow(cursorY);
                    continue;
                }
                if (occupied[cursorY] & (1ULL << cursorX)) {
                    ++cursorX;
                    continue;
                }
                break;
            }

            int limit = min(buffer, (int)remaining.size());
            bool found = false;
            long long bestScore = LLONG_MAX;
            int bestIndex = -1, bestOrientation = -1;
            int bestTX = 0, bestTY = 0;

            for (int idx = 0; idx < limit; ++idx) {
                int pieceId = remaining[idx];

                        if (tx < 0 || ty < 0 || tx + o.w > width) continue;

                        ensureRow(ty + o.h);

                        for (int dy = 0; dy < o.h; ++dy) {
                            unsigned long long mask = o.rows[dy] << tx;
                            if (occupied[ty + dy] & mask) {
                                fits = false;
                                break;
                            }
                        }
                        if (!fits) continue;

                        unsigned long long currentRow = occupied[cursorY];
                        if (cursorY >= ty && cursorY < ty + o.h) {
                            currentRow |= o.rows[cursorY - ty] << tx;

                        int run = 0;
                        while (cursorX + run < width &&

                        for (auto [cx, cy] : o.cells) {

                            const int dy[4] = {0, 0, -1, 1};
                            for (int d = 0; d < 4; ++d) {
                                int nx = px + dx[d];
                                int ny = py + dy[d];

                                if (nx < 0 || nx >= width || ny < 0) {
                                    ++contact;
                                } else if (ny < (int)occupied.size() &&
                                           (occupied[ny] & (1ULL << nx))) {
                                    ++contact;
                                }
                            }
                        }

                        int height = ty + o.h - cursorY;
                        long long score;
                        if (mode == 0) {
                            score = (long long)height * 1000000LL
                                  - (long long)run * 10000LL
                                  - (long long)contact * 100LL
                                  - piece.k;
                        } else {
                            score = -(long long)run * 1000000LL
                                  + (long long)height * 10000LL
                                  - (long long)contact * 100LL
                                  - piece.k;
                        }

                        if (score < bestScore) {
                            bestScore = score;
                            found = true;
                            bestIndex = idx;
                            bestOrientation = oi;
                            bestTX = tx;
                            bestTY = ty;
                        }
                    }
                }
            }

            if (!found) {
                occupied[cursorY] |= (1ULL << cursorX);
                ++cursorX;
                continue;
            }

            int pieceId = remaining[bestIndex];
            const Orientation &o = pieces[pieceId].orientations[bestOrientation];

            for (int dy = 0; dy < o.h; ++dy) {
                occupied[bestTY + dy] |= o.rows[dy] << bestTX;
            }

            result.placement[pieceId] = {
                bestTX - o.minX,

            vector<Cell> localA = a;
            vector<Cell> localB = b;
            vector<Cell> geometry = rectangle;

            if (rotation == 1) {
                auto rotateCells = [&](vector<Cell>& cells) {
                    for (auto& [x, y] : cells) {
                        int oldX = x;
                        x = y;
                        y = pair.w - 1 - oldX;
                    }
                    sort(cells.begin(), cells.end());
                };
                rotateCells(localA);
                rotateCells(localB);
                rotateCells(geometry);
            }

            target.cells = std::move(geometry);

            Cell translation;
            int orientation = matchOrientation(p, localA, translation);
            assert(orientation >= 0);
            Membership mp;
            mp.piece = pair.p;
            mp.orientation = orientation;
            mp.translation = translation;
            target.members.push_back(std::move(mp));

            translation = {0, 0};
            orientation = matchOrientation(q, localB, translation);
            assert(orientation >= 0);
            Membership mq;
            mq.piece = pair.q;
            mq.orientation = orientation;
            mq.translation = translation;
            target.members.push_back(std::move(mq));

            prepareObjectOrientation(target);
            object.orientations.push_back(std::move(target));
        }

        used[pair.p] = used[pair.q] = 1;
        objects.push_back(std::move(object));
    }

    for (int i = 0; i < static_cast<int>(pieces.size()); ++i) {
        if (used[i]) continue;

        Object object;
        object.area = pieces[i].k;
        for (int j = 0; j < static_cast<int>(pieces[i].orientations.size()); ++j) {
            const Orientation& source = pieces[i].orientations[j];
            ObjectOrientation target;
            target.cells = source.cells;

            Membership membership;
            membership.piece = i;
            membership.orientation = j;
            membership.translation = {-source.offsetX, -source.offsetY};
            target.members.push_back(std::move(membership));

            prepareObjectOrientation(target);
            object.orientations.push_back(std::move(target));
        }
        objects.push_back(std::move(object));
    }
    return objects;
}

static constexpr int MAX_DX = 20;

class Board {
public:
    int width;
    int words;
    int stride;
    int height = 0;

    int minX = INT_MAX, minY = INT_MAX;
    int maxX = INT_MIN, maxY = INT_MIN;

    vector<ull> occupied;
    vector<ull> shifted;
    vector<vector<ull>> baseMasks;

    Board(int w, int totalCells) : width(w) {
        words = (width + 63) / 64;
        stride = words + 1;

        size_t estimatedRows = max<size_t>(64, 3ULL * totalCells / max(1, width) + 64);
        occupied.reserve(estimatedRows * stride);
        shifted.reserve(estimatedRows * MAX_DX * stride);

        baseMasks.resize(MAX_DX + 1);
        for (int sw = 1; sw <= MAX_DX; ++sw) {
            if (sw > width) continue;
            int maxXPosition = width - sw;
            int count = maxXPosition / 64 + 1;
            baseMasks[sw].assign(words, 0);

            for (int q = 0; q < count; ++q) {
                int bitCount = min(64, maxXPosition - q * 64 + 1);
                if (bitCount == 64) {
                    baseMasks[sw][q] = ~0ULL;
                } else {
                    baseMasks[sw][q] = (1ULL << bitCount) - 1;
                }
            }
        }
        ensureRows(0);
    }

    void ensureRows(int newHeight) {
        if (newHeight <= height) return;
        occupied.resize(static_cast<size_t>(newHeight) * stride, 0);
        shifted.resize(static_cast<size_t>(newHeight) * MAX_DX * stride, 0);
        height = newHeight;
    }

    void refreshRow(int y) {
        const ull* source = occupied.data() + static_cast<size_t>(y) * stride;

        for (int dx = 0; dx < MAX_DX; ++dx) {
            ull* destination = shifted.data() +
                (static_cast<size_t>(y) * MAX_DX + dx) * stride;

            if (dx == 0) {
                for (int q = 0; q < words; ++q) {
                    destination[q] = source[q];
                }
            } else {
                for (int q = 0; q < words; ++q) {
                    destination[q] = (source[q] >> dx) |
                                     (source[q + 1] << (64 - dx));
                }
            }
        }
    }

    void add(const vector<Cell>& cells, int x, int y) {
        for (auto [dx, dy] : cells) {
            int px = x + dx;
            int py = y + dy;
            occupied[static_cast<size_t>(py) * stride + (px >> 6)] |=
                1ULL << (px & 63);
            minX = min(minX, px);
            maxX = max(maxX, px);
            minY = min(minY, py);
            maxY = max(maxY, py);
        }

        int maxRow = 0;
        for (auto [dx, dy] : cells) maxRow = max(maxRow, dy);
        for (int row = 0; row <= maxRow; ++row) refreshRow(y + row);
    }

    bool isOccupied(int x, int y) const {
        if (x < 0 || x >= width || y < 0 || y >= height) return false;
        return (occupied[static_cast<size_t>(y) * stride + (x >> 6)] >>
                (x & 63)) & 1ULL;
    }
};

struct ObjectPlacement {
    int object = -1;
    int orientation = -1;
    int x = 0, y = 0;
};

struct PackResult {
    vector<ObjectPlacement> placements;
    int minX = INT_MAX, minY = INT_MAX;
    int maxX = INT_MIN, maxY = INT_MIN;
    int width = 0, height = 0;
};

struct ObjectStats {
    int area = 0;
    int boxArea = INT_MAX;
    int minHeight = INT_MAX;
    int minWidth = INT_MAX;
    ull randomKey = 0;
};

static vector<int> makeOrder(const vector<Object>& objects, int mode, ull seed) {
    RNG rng(seed);
    vector<ObjectStats> stats(objects.size());

    for (int i = 0; i < static_cast<int>(objects.size()); ++i) {
        stats[i].area = objects[i].area;
        stats[i].randomKey = rng.next();
        for (const ObjectOrientation& o : objects[i].orientations) {
            stats[i].boxArea = min(stats[i].boxArea, o.w * o.h);
            stats[i].minHeight = min(stats[i].minHeight, o.h);
            stats[i].minWidth = min(stats[i].minWidth, o.w);
        }
    }

    vector<int> order(objects.size());
    iota(order.begin(), order.end(), 0);

    sort(order.begin(), order.end(), [&](int a, int b) {
        const ObjectStats& x = stats[a];
        const ObjectStats& y = stats[b];

        if (x.area != y.area) return x.area > y.area;
        if (mode == 0) {
            if (x.boxArea != y.boxArea) return x.boxArea > y.boxArea;
            if (x.minHeight != y.minHeight) return x.minHeight > y.minHeight;
        } else if (mode == 1) {
            if (x.minHeight != y.minHeight) return x.minHeight > y.minHeight;
            if (x.minWidth != y.minWidth) return x.minWidth > y.minWidth;
            if (x.boxArea != y.boxArea) return x.boxArea > y.boxArea;
        } else {
            if (x.randomKey != y.randomKey) return x.randomKey < y.randomKey;
        }
        return x.randomKey < y.randomKey;
    });

    return order;
}

static bool calculateValidPositions(const ObjectOrientation& orientation,
                                    int y,
                                    Board& board,
                                    vector<ull>& valid) {
    const vector<ull>& initial = board.baseMasks[orientation.w];
    if (initial.empty()) return false;

    copy(initial.begin(), initial.end(), valid.begin());

    for (int index : orientation.order) {
        auto [dx, dy] = orientation.cells[index];
        const ull* forbidden = board.shifted.data() +
            (static_cast<size_t>(y + dy) * MAX_DX + dx) * board.stride;

        ull any = 0;
        for (int q = 0; q < board.words; ++q) {
            valid[q] &= ~forbidden[q];
            any |= valid[q];
        }
        if (any == 0) return false;
    }
    return true;
}

static int placementScore(const ObjectOrientation& orientation,
                          int x,
                          int y,
                          const Board& board) {
    int contacts = 0;
    int overhang = 0;

    for (int index : orientation.order) {
        auto [dx, dy] = orientation.cells[index];
        int px = x + dx;
        int py = y + dy;

        if (board.isOccupied(px - 1, py)) ++contacts;
        if (board.isOccupied(px + 1, py)) ++contacts;
        if (board.isOccupied(px, py - 1)) ++contacts;
        if (board.isOccupied(px, py + 1)) ++contacts;

        if (!orientation.hasBelow[index] && py > 0 &&
            !board.isOccupied(px, py - 1)) {
            ++overhang;
        }
    }

    return contacts * 2000 - overhang * 200 - x;
}

static PackResult packObjects(const vector<Object>& objects,
                              const vector<int>& order,
                              int width) {
    Board board(width, 0);
    PackResult result;
    vector<ull> valid(board.words);

    for (int objectIndex : order) {
        const Object& object = objects[objectIndex];
        bool placed = false;

        for (int y = 0; y < board.height && !placed; ++y) {
            bool found = false;
            int bestScore = INT_MIN;
            int bestOrientation = -1;
            int bestX = -1;

            for (int oi = 0; oi < static_cast<int>(object.orientations.size()); ++oi) {
                const ObjectOrientation& orientation = object.orientations[oi];
                if (orientation.w > width || y + orientation.h > board.height) continue;
                if (!calculateValidPositions(orientation, y, board, valid)) continue;

                int collected = 0;
                for (int q = 0; q < board.words && collected < 12; ++q) {
                    ull bits = valid[q];
                    while (bits && collected < 12) {
                        int bit = __builtin_ctzll(bits);
                        bits &= bits - 1;
                        int x = q * 64 + bit;
                        int score = placementScore(orientation, x, y, board);

                        if (!found || score > bestScore ||
                            (score == bestScore && x < bestX)) {
                            found = true;
                            bestScore = score;
                            bestOrientation = oi;
                            bestX = x;
                        }
                        ++collected;
                    }
                }
            }

            if (found) {
                board.add(object.orientations[bestOrientation].cells, bestX, y);
                result.placements.push_back(
                    {objectIndex, bestOrientation, bestX, y});
                placed = true;
            }
        }

        if (!placed) {
            int chosen = -1;
            for (int oi = 0; oi < static_cast<int>(object.orientations.size()); ++oi) {
                const ObjectOrientation& o = object.orientations[oi];
                if (o.w > width) continue;
                if (chosen < 0 ||
                    o.h < object.orientations[chosen].h ||
                    (o.h == object.orientations[chosen].h &&
                     o.w < object.orientations[chosen].w)) {
                    chosen = oi;
                }
            }
            assert(chosen >= 0);

            int y = board.height;
            board.ensureRows(y + object.orientations[chosen].h);
            board.add(object.orientations[chosen].cells, 0, y);
            result.placements.push_back({objectIndex, chosen, 0, y});
        }
    }

    result.minX = board.minX;
    result.minY = board.minY;
    result.maxX = board.maxX;
    result.maxY = board.maxY;
    result.width = result.maxX - result.minX + 1;
    result.height = result.maxY - result.minY + 1;
    return result;
}

struct Answer {
    int w = 0, h = 0;
    vector<int> x, y, r, f;
    bool valid = false;
};

static Answer expandResult(const PackResult& pack,
                            const vector<Object>& objects,
                            const vector<Piece>& pieces) {
    Answer answer;
    answer.w = pack.width;
    answer.h = pack.height;
    answer.x.assign(pieces.size(), 0);
    answer.y.assign(pieces.size(), 0);
    answer.r.assign(pieces.size(), 0);
    answer.f.assign(pieces.size(), 0);

    for (const ObjectPlacement& placement : pack.placements) {
        const Object& object = objects[placement.object];
        const ObjectOrientation& orientation =
            object.orientations[placement.orientation];

        for (const Membership& membership : orientation.members) {
            const Orientation& source =
                pieces[membership.piece].orientations[membership.orientation];

            answer.x[membership.piece] =
                membership.translation.first + placement.x - pack.minX;
            answer.y[membership.piece] =
                membership.translation.second + placement.y - pack.minY;
            answer.r[membership.piece] = source.r;
            answer.f[membership.piece] = source.f;
        }
    }
    return answer;
}

static bool validateAnswer(const Answer& answer, const vector<Piece>& pieces) {
    if (answer.w <= 0 || answer.h <= 0) return false;

    vector<unsigned char> grid(
        static_cast<size_t>(answer.w) * answer.h, 0);

    for (int i = 0; i < static_cast<int>(pieces.size()); ++i) {
        for (auto [x0, y0] : pieces[i].raw) {
            int x = x0, y = y0;
            if (answer.f[i]) x = -x;

            for (int t = 0; t < answer.r[i]; ++t) {
                int nx = y;
                int ny = -x;
                x = nx;
                y = ny;
            }

            x += answer.x[i];
            y += answer.y[i];
            if (x < 0 || x >= answer.w || y < 0 || y >= answer.h) return false;

            size_t id = static_cast<size_t>(y) * answer.w + x;
            if (grid[id]) return false;
            grid[id] = 1;
        }
    }
    return true;
}

static bool betterAnswer(const Answer& a, const Answer& b) {
    if (!b.valid) return true;
    long long areaA = 1LL * a.w * a.h;
    long long areaB = 1LL * b.w * b.h;
    if (areaA != areaB) return areaA < areaB;
    if (a.h != b.h) return a.h < b.h;
    return a.w < b.w;
}

static Answer emergencyAnswer(const vector<Piece>& pieces, int totalCells) {
    Answer answer;
    int width = max(20, static_cast<int>(ceil(sqrt(2.0 * totalCells))) + 20);
    int currentX = 0, currentY = 0, rowHeight = 0;

    answer.w = width;
    answer.x.resize(pieces.size());
    answer.y.resize(pieces.size());
    answer.r.resize(pieces.size());
    answer.f.resize(pieces.size());

    for (int i = 0; i < static_cast<int>(pieces.size()); ++i) {
        int chosen = -1;
        for (int j = 0; j < static_cast<int>(pieces[i].orientations.size()); ++j) {
            const Orientation& o = pieces[i].orientations[j];
            if (chosen < 0 || o.h < pieces[i].orientations[chosen].h ||
                (o.h == pieces[i].orientations[chosen].h &&
                 o.w < pieces[i].orientations[chosen].w)) {
                chosen = j;
            }
        }

        const Orientation& o = pieces[i].orientations[chosen];
        if (currentX + o.w > width) {
            currentY += rowHeight;
            currentX = 0;
            rowHeight = 0;
        }

        answer.x[i] = currentX - o.offsetX;
        answer.y[i] = currentY - o.offsetY;
        answer.r[i] = o.r;
        answer.f[i] = o.f;

        currentX += o.w;
        rowHeight = max(rowHeight, o.h);
    }

    answer.h = currentY + rowHeight;
    answer.valid = validateAnswer(answer, pieces);
    return answer;
}

int main() {
    ios::sync_with_stdio(false);

    int n;

    vector<Piece> pieces(n);

    for (int i = 0; i < n; ++i) {

        for (int f = 0; f < 2; ++f) {
            for (int r = 0; r < 4; ++r) {
                Orientation orientation;
                orientation.f = f;
                orientation.r = r;

                vector<Cell> transformed;
                transformed.reserve(k);
                int minX = INT_MAX, minY = INT_MAX;

                for (auto [x0, y0] : pieces[i].raw) {
                    int x = x0, y = y0;
                    if (f) x = -x;

                    for (int t = 0; t < r; ++t) {
                        int nx = y;
                        int ny = -x;
                        x = nx;
                        y = ny;

                    minX = min(minX, x);
                    minY = min(minY, y);

                orientation.offsetY = minY;

                for (auto& [x, y] : transformed) {
                    x -= minX;
                    y -= minY;
                }
                sort(transformed.begin(), transformed.end());

                orientation.cells = transformed;
                orientation.w = 0;
                orientation.h = 0;
                for (auto [x, y] : transformed) {
                    orientation.w = max(orientation.w, x + 1);
                    orientation.h = max(orientation.h, y + 1);

                bool duplicate = false;
                for (const Orientation& existing : pieces[i].orientations) {
                    if (existing.cells == orientation.cells) {
                        duplicate = true;
                        break;
                    }
                }
                if (!duplicate) {

                }
            }
        }

                   bestKind == 0 ? 1 : 0, factors[1], 0);

                   bestKind, factors[3], 0);
        tryObjects(bestKind == 0 ? pairedObjects : singleObjects,
                   bestKind == 0 ? 1 : 0, factors[3], 1);

    cout << best.w << ' ' << best.h << '\n';
    for (int i = 0; i < n; ++i) {
        cout << best.x[i] << ' ' << best.y[i] << ' '
             << best.r[i] << ' ' << best.f[i] << '\n';
    }

    return 0;
