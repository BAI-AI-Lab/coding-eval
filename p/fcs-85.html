#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    const int MOVE_LIMIT = 100000;

    int initialDeep;
    cin >> initialDeep;
    if (initialDeep == 0) return 0;

    mt19937 rng(
        static_cast<unsigned int>(
            chrono::high_resolution_clock::now().time_since_epoch().count()
        ) ^ random_device{}()
    );

    int movesUsed = 0;

    auto doMove = [&](int c) -> int {
        cout << "move " << c << '\n' << flush;
        ++movesUsed;
        int result;
        cin >> result;
        return result;
    };

    auto doQuery = [&]() -> int {
        cout << "query\n" << flush;
        int result;
        cin >> result;
        return result;
    };

    int depth = initialDeep;
    int incoming = -1;

    // Find the first upward edge. Afterwards, the color of the edge
    // leading back down is known, so only two colors need consideration.
    for (int c = 0; c < 3; ++c) {
        if (doMove(c)) return 0;

        if (c == 2) {
            depth = initialDeep - 1;
            incoming = c;
            break;
        }

        int newDepth = doQuery();
        if (newDepth == initialDeep - 1) {
            depth = newDepth;
            incoming = c;
            break;
        }

        doMove(c); // Return to the initial node.
    }

    while (depth > 0) {
        int t = min(8, depth);

        // Reserve enough moves to finish deterministically, using at most
        // three moves per remaining level.
        while (t > 1 &&
               movesUsed + 3 * t + 3 * (depth - 1) > MOVE_LIMIT) {
            --t;
        }

        vector<int> path;
        path.reserve(t);

        int previous = incoming;
        for (int i = 0; i < t; ++i) {
            int c = static_cast<int>(rng() % 2);
            if (c >= previous) ++c;
            path.push_back(c);
            previous = c;
        }

        bool reachedExit = false;
        for (int c : path) {
            if (doMove(c)) {
                reachedExit = true;
                break;
            }
        }
        if (reachedExit) return 0;

        int newDepth = doQuery();
        int k = (depth + t - newDepth) / 2;

        if (k == t) {
            // Every move was upward.
            depth -= t;
            incoming = path.back();
            continue;
        }

        // The first k moves were upward, move k+1 was downward, and all
        // subsequent moves were also downward.
        int incomingAtFailure = (k == 0 ? incoming : path[k - 1]);
        int forcedUp = 3 - incomingAtFailure - path[k];

        for (int i = t - 1; i >= 0; --i) {
            if (doMove(path[i])) return 0;
        }

        for (int i = 0; i < k; ++i) {
            if (doMove(path[i])) return 0;
        }

        if (doMove(forcedUp)) return 0;

        depth -= k + 1;
        incoming = forcedUp;
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

static constexpr int MOVE_LIMIT = 100000;

int movesUsed = 0;
int queriesUsed = 0;

uint64_t seed = chrono::steady_clock::now().time_since_epoch().count();
mt19937 rng(static_cast<uint32_t>(seed ^ (seed >> 32)));

void performMove(int color) {
    cout << "move " << color << '\n' << flush;

    int response;
    if (!(cin >> response)) exit(0);

    ++movesUsed;
    if (response == 1) exit(0);
}

int performQuery() {
    cout << "query\n" << flush;

    int distance;
    if (!(cin >> distance)) exit(0);

    ++queriesUsed;
    return distance;
}

vector<int> childColors(int parentColor) {
    vector<int> colors;
    for (int color = 0; color < 3; ++color) {
        if (color != parentColor) colors.push_back(color);
    }
    return colors;
}

void exhaustiveSearch(int level, int parentColor, int targetDepth) {
    vector<int> colors = childColors(parentColor);

    for (int color : colors) {
        performMove(color);

        if (level + 1 == targetDepth) {
            performMove(color);
        } else {
            exhaustiveSearch(level + 1, color, targetDepth);
            performMove(color);
        }
    }
}

int partialLimit = 0;
bool partialStopped = false;

bool partialSearch(int level, int parentColor, int targetDepth) {
    if (level == targetDepth) return true;

    vector<int> colors = childColors(parentColor);
    shuffle(colors.begin(), colors.end(), rng);

    bool complete = true;

    for (int color : colors) {
        if (partialStopped) {
            complete = false;
            break;
        }

        performMove(color);
        bool childComplete = partialSearch(level + 1, color, targetDepth);
        performMove(color);

        if (!childComplete) complete = false;

        if (movesUsed >= partialLimit) {
            partialStopped = true;
            break;
        }
    }

    return complete && !partialStopped;
}

void partialSearchFromRoot(int targetDepth, array<bool, 3>& knownChild) {
    vector<int> firstColors{0, 1, 2};
    shuffle(firstColors.begin(), firstColors.end(), rng);

    partialLimit = 98000;
    partialStopped = false;

    for (int color : firstColors) {
        if (partialStopped) break;

        performMove(color);
        bool complete = partialSearch(1, color, targetDepth);
        performMove(color);

        if (complete) knownChild[color] = true;

        if (movesUsed >= partialLimit) partialStopped = true;
    }
}

class Solver {
public:
    int depth;
    array<bool, 3> knownChild{false, false, false};

    explicit Solver(int initialDepth) : depth(initialDepth) {}

    int knownCount() const {
        return static_cast<int>(knownChild[0]) +
               static_cast<int>(knownChild[1]) +
               static_cast<int>(knownChild[2]);
    }

    int onlyUnknownParentColor() const {
        int sum = 0;
        int count = 0;

        for (int color = 0; color < 3; ++color) {
            if (knownChild[color]) {
                sum += color;
                ++count;
            }
        }

        if (count == 1) return 3 - sum;

        for (int color = 0; color < 3; ++color) {
            if (!knownChild[color]) return color;
        }

        return -1;
    }

    void resetKnownChild(int color = -1) {
        knownChild.fill(false);
        if (color >= 0) knownChild[color] = true;
    }

    void finishAtDepthOne() {
        while (depth > 0) {
            vector<int> candidates;
            for (int color = 0; color < 3; ++color) {
                if (!knownChild[color]) candidates.push_back(color);
            }

            for (int color : candidates) {
                performMove(color);
                performMove(color);
            }
        }
    }

    void directAscend() {
        while (depth > 0) {
            int count = knownCount();

            if (count == 2) {
                int parentColor = onlyUnknownParentColor();
                performMove(parentColor);
                --depth;
                resetKnownChild(parentColor);
                continue;
            }

            vector<int> candidates;
            for (int color = 0; color < 3; ++color) {
                if (!knownChild[color]) candidates.push_back(color);
            }

            shuffle(candidates.begin(), candidates.end(), rng);
            int color = candidates.front();

            performMove(color);
            int queriedDepth = performQuery();

            if (queriedDepth == depth - 1) {
                --depth;
                resetKnownChild(color);
            } else {
                performMove(color);

                if (count == 0) {
                    resetKnownChild(color);
                } else {
                    int oldChildColor = 0;
                    for (int c = 0; c < 3; ++c) {
                        if (knownChild[c]) oldChildColor = c;
                    }

                    int parentColor = 3 - oldChildColor - color;
                    performMove(parentColor);
                    --depth;
                    resetKnownChild(parentColor);
                }
            }
        }
    }

    bool canRunProbe(int length) const {
        int minimumGain = (knownCount() >= 1 ? 1 : 0);
        int remainingDepth = depth - minimumGain;

        long long worstProbeMoves = 2LL * length + 1;
        long long directFallbackMoves = 3LL * remainingDepth + 2;

        return movesUsed + worstProbeMoves + directFallbackMoves <= MOVE_LIMIT;
    }

    void probe(int length) {
        int oldKnownCount = knownCount();
        int oldKnownColor = -1;

        for (int color = 0; color < 3; ++color) {
            if (knownChild[color]) oldKnownColor = color;
        }

        vector<int> candidates;
        for (int color = 0; color < 3; ++color) {
            if (!knownChild[color]) candidates.push_back(color);
        }

        shuffle(candidates.begin(), candidates.end(), rng);

        vector<int> path;
        path.reserve(length);

        int previousColor = candidates.front();
        path.push_back(previousColor);
        performMove(previousColor);

        for (int i = 1; i < length; ++i) {
            vector<int> choices;
            for (int color = 0; color < 3; ++color) {
                if (color != previousColor) choices.push_back(color);
            }

            int color = choices[rng() & 1U];
            path.push_back(color);
            performMove(color);
            previousColor = color;
        }

        int queriedDepth = performQuery();
        int numerator = depth + length - queriedDepth;

        if (numerator < 0 || (numerator & 1)) exit(0);

        int upwardSteps = numerator / 2;

        if (upwardSteps == 0) {
            for (int i = length - 1; i >= 0; --i) {
                performMove(path[i]);
            }

            if (oldKnownCount == 1) {
                int parentColor = 3 - oldKnownColor - path[0];
                performMove(parentColor);
                --depth;
                resetKnownChild(parentColor);
            } else {
                resetKnownChild(path[0]);
            }
        } else if (upwardSteps == length) {
            depth -= length;
            resetKnownChild(path.back());
        } else {
            for (int i = length - 1; i >= upwardSteps; --i) {
                performMove(path[i]);
            }

            int parentColor = 3 - path[upwardSteps - 1] - path[upwardSteps];
            performMove(parentColor);
            depth -= upwardSteps + 1;
            resetKnownChild(parentColor);
        }
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int initialDepth;
    if (!(cin >> initialDepth)) return 0;

    if (initialDepth == 0) return 0;

    if (initialDepth <= 14) {
        exhaustiveSearch(0, -1, initialDepth);
        return 0;
    }

    array<bool, 3> knownChild{false, false, false};

    if (15 <= initialDepth && initialDepth <= 20) {
        partialSearchFromRoot(initialDepth, knownChild);
    }

    Solver solver(initialDepth);
    solver.knownChild = knownChild;

    while (solver.depth > 0) {
        if (solver.depth == 1) {
            solver.finishAtDepthOne();
            break;
        }

        if (solver.depth <= 14) {
            long long requiredMoves = 6LL * ((1LL << solver.depth) - 1);
            if (movesUsed + requiredMoves <= MOVE_LIMIT) {
                exhaustiveSearch(0, -1, solver.depth);
                return 0;
            }
        }

        int selectedLength = 0;
        int maximumLength = min(10, solver.depth);

        for (int length = maximumLength; length >= 1; --length) {
            if (solver.canRunProbe(length)) {
                selectedLength = length;
                break;
            }
        }

        if (selectedLength == 0) {
            solver.directAscend();
        } else {
            solver.probe(selectedLength);
        }
    }

    return 0;
}
