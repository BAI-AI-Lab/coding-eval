题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Farmer John has $N$ ($2\le N\le 1000$) cows at distinct locations $l_1,\dots, l_N$ along a circle of circumference $C$ ($0\le l_1 < l_2 < \dots < l_N <C, N\le C\le 10^9$).

FJ will select $k$ pairs of cows, where $1\le k\le \lfloor N/2\rfloor$, and no cow is selected more than once. He wants to select the pairs such that the minimum distance between any two cows in the same pair along the circumference of the circle is maximized.

For each value of $k$, help FJ determine the maximum possible minimum distance.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $N$ and $C$.

The second line contains $l_1\dots l_N$.

OUTPUT FORMAT (print output to the terminal / stdout):

Output a single line with $\lfloor N/2\rfloor$ space-separated integers, with the answers for $k=1\dots \lfloor N/2\rfloor$ in that order.

SAMPLE INPUT:

4 100
0 25 50 75

SAMPLE OUTPUT:

50 50

For $k = 1$, cow 1 can be paired to cow 3, which is distance $50$ away along the circumference of the circle, making the answer $50$.

For $k = 2$, cow 1 can be paired to cow 3, and cow 2 can be paired to cow 4, which is distance $50$ away from it along the circumference of the circle, making the answer still $50$.

SAMPLE INPUT:

4 100
0 1 2 99

SAMPLE OUTPUT:

3 2

For $k = 1$, cow 3 can be paired to cow 4, which is distance $2 + 100 - 99 = 3$ away from it along the circumference of the circle, making the answer $3$.

For $k = 2$, cow 1 can be paired to cow 3 and cow 2 can be paired to cow 4. Each of these pairs contains two cows at a distance of $2$ from each other along the circumference of the circle, making the answer $2$.

SCORING:

  • Inputs 3-4: $2l_N \le C$
  • Inputs 5-6: $N\le 20$
  • Inputs 7-14: $N\le 100$
  • Inputs 15-22: No additional constraints.

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    int64 C;
    cin >> N >> C;

    vector<int64> a(N);
    for (int i = 0; i < N; ++i) {
        cin >> a[i];
    }

    vector<int64> x(2 * N);
    for (int i = 0; i < N; ++i) {
        x[i] = a[i];
        x[i + N] = a[i] + C;
    }

    const int64 INF = (1LL << 60);
    vector<int64> best(N + 1, INF);
    best[1] = 0;

    // For every cyclic interval of consecutive cows, compute its diameter.
    for (int start = 0; start < N; ++start) {
        int64 diameter = 0;

        for (int end = start + 1; end < start + N; ++end) {
            // Find the old cows immediately around the antipode of x[end].
            int lo = start, hi = end;
            const int64 threshold = 2 * x[end] - C;

            while (lo < hi) {
                int mid = (lo + hi) / 2;
                if (2 * x[mid] >= threshold) {
                    hi = mid;
                } else {
                    lo = mid + 1;
                }
            }

            auto update = [&](int idx) {
                if (idx < start || idx >= end) return;
                int64 d = x[end] - x[idx];
                diameter = max(diameter, min(d, C - d));
            };

            update(lo);
            update(lo - 1);

            int len = end - start + 1;
            best[len] = min(best[len], diameter);
        }
    }

    for (int k = 1; k <= N / 2; ++k) {
        if (k > 1) cout << ' ';
        int q = N - k + 1;
        cout << min(best[q], C / 2);
    }
    cout << '\n';

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct Candidate {
    ll d;
    int a;
    bool operator<(const Candidate& other) const {
        return d < other.d;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    ll C;
    cin >> N >> C;

    vector<ll> l(N);
    for (ll& x : l) cin >> x;

    const int K = N / 2;
    const int firstSize = N - K + 1;

    // rel[a][p] is the clockwise distance from cow a to the p-th cow
    // encountered after a in circular order.
    vector<vector<ll>> rel(N, vector<ll>(N));
    for (int a = 0; a < N; a++) {
        for (int p = 0; p < N; p++) {
            int id = (a + p) % N;
            if (id >= a) rel[a][p] = l[id] - l[a];
            else rel[a][p] = C - l[a] + l[id];
        }
    }

    vector<Candidate> candidates;
    candidates.reserve(N * (N - 1) / 2);

    for (int i = 0; i < N; i++) {
        for (int j = i + 1; j < N; j++) {
            ll delta = l[j] - l[i];
            if (2 * delta <= C) {
                candidates.push_back({delta, i});
            } else {
                candidates.push_back({C - delta, j});
            }
        }
    }

    sort(candidates.begin(), candidates.end());

    vector<ll> answerForSize(N + 1, -1);
    int needed = firstSize;

    for (const Candidate& cand : candidates) {
        if (needed > N) break;

        const ll d = cand.d;
        const ll* r = rel[cand.a].data();

        // Every circular distance is at most C/2.
        if (2 * d == C) {
            for (int s = needed; s <= N; s++) answerForSize[s] = d;
            needed = N + 1;
            break;
        }

        // U is the intersection of the two radius-d balls centered at the
        // chosen pair. It consists of at most two circular arcs.
        int nx = int(upper_bound(r, r + N, d) - r);

        int zL = 0, nz = 0;
        ll lo = 0;

        if (3 * d >= C) {
            lo = C - d;
            ll hi = 2 * d;
            zL = int(lower_bound(r + nx, r + N, lo) - r);
            int zR = int(upper_bound(r + zL, r + N, hi) - r);
            nz = zR - zL;
        }

        int total = nx + nz;
        if (total < needed) continue;

        // Within each arc, all pairwise distances are at most d. Between the
        // arcs, bad pairs form a convex bipartite graph. Its maximum matching
        // can be found greedily; removing a minimum vertex cover leaves a clique.
        int limit = total - needed;
        int matching = 0;

        if (nz > 0) {
            const ll w = C - 2 * d;
            int px = 0, pz = 0;

            while (px < nx && pz < nz) {
                ll xv = r[px];
                ll zv = r[zL + pz] - lo;

                if (xv <= zv) {
                    px++;
                } else if (xv - zv >= w) {
                    pz++;
                } else {
                    matching++;
                    if (matching > limit) break;
                    px++;
                    pz++;
                }
            }

            if (matching > limit) continue;
        }

        int cliqueSize = total - matching;
        if (cliqueSize >= needed) {
            for (int s = needed; s <= cliqueSize; s++) {
                answerForSize[s] = d;
            }
            needed = cliqueSize + 1;
        }
    }

    for (int k = 1; k <= K; k++) {
        if (k > 1) cout << ' ';
        cout << answerForSize[N - k + 1];
    }
    cout << '\n';

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct SuccessorDSU {
    vector<int> p;

    void init(int n) {
        p.resize(n + 1);
        iota(p.begin(), p.end(), 0);
    }

    int find(int x) {
        if (p[x] == x) return x;
        return p[x] = find(p[x]);
    }

    void erase(int x) {
        p[x] = find(x + 1);
    }
};

int greedyPairs(const vector<ll>& a, ll d, ll upper, int needed) {
    int n = (int)a.size();
    vector<int> start(n);
    int j = 0;

    for (int i = 0; i < n; ++i) {
        j = max(j, i + 1);
        while (j < n && a[j] - a[i] < d) ++j;
        start[i] = j;
    }

    SuccessorDSU dsu;
    dsu.init(n);

    int pairs = 0;
    while (true) {
        int i = dsu.find(0);
        if (i == n) break;

        int right = dsu.find(start[i]);
        if (right < n && a[right] - a[i] <= upper) {
            ++pairs;
            if (pairs >= needed) return pairs;
            dsu.erase(i);
            dsu.erase(right);
        } else {
            dsu.erase(i);
        }
    }
    return pairs;
}

int N;
ll C;
vector<ll> pos, revPos;

bool canPair(int k, ll d) {
    if (d > C / 2) return false;
    ll upper = C - d;

    if (greedyPairs(pos, d, upper, k) >= k) return true;
    if (greedyPairs(revPos, d, upper, k) >= k) return true;
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> N >> C;
    pos.resize(N);
    for (ll& x : pos) cin >> x;

    revPos.resize(N);
    for (int i = 0; i < N; ++i) {
        revPos[i] = C - pos[N - 1 - i];
    }

    int K = N / 2;
    vector<ll> answer(K);
    ll previousHigh = C / 2;

    for (int k = 1; k <= K; ++k) {
        ll lo = 0, hi = previousHigh;
        while (lo < hi) {
            ll mid = lo + (hi - lo + 1) / 2;
            if (canPair(k, mid)) lo = mid;
            else hi = mid - 1;
        }
        answer[k - 1] = lo;
        previousHigh = lo;
    }

    for (int i = 0; i < K; ++i) {
        if (i) cout << ' ';
        cout << answer[i];
    }
    cout << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    int64 C;
    cin >> N >> C;

    vector<int64> pos(N);
    for (int i = 0; i < N; ++i) cin >> pos[i];

    // d[m] = minimum possible diameter of m cows, where diameter is the
    // maximum circular distance between any two selected cows.
    const int64 INF = (1LL << 62);
    vector<int64> d(N + 1, INF);

    vector<int64> extended(2 * N), scaled(2 * N);
    for (int i = 0; i < 2 * N; ++i) {
        extended[i] = pos[i % N] + (i / N) * C;
        scaled[i] = 2 * extended[i];
    }

    auto circularDistance = [&](int i, int j) -> int64 {
        int64 diff = llabs(pos[i % N] - pos[j % N]);
        return min(diff, C - diff);
    };

    // For large subsets (more than N/2 cows), an optimal subset can always
    // be chosen as cyclically consecutive cows. Build every such window and
    // add its cows one at a time, maintaining its diameter.
    for (int start = 0; start < N; ++start) {
        int64 diameter = 0;

        for (int len = 1; len <= N; ++len) {
            int last = start + len - 1;

            if (len > 1) {
                // For x <= extended[last], circular distance is maximized
                // by the point nearest to extended[last] - C/2.
                int64 target = scaled[last] - C;
                auto it = lower_bound(scaled.begin() + start,
                                      scaled.begin() + last, target);
                int at = int(it - scaled.begin());

                int64 best = 0;
                if (at > start) {
                    best = max(best, circularDistance(last, at - 1));
                }
                if (at < last) {
                    best = max(best, circularDistance(last, at));
                }
                diameter = max(diameter, best);
            }

            d[len] = min(d[len], diameter);
        }
    }

    int K = N / 2;
    for (int k = 1; k <= K; ++k) {
        if (k > 1) cout << ' ';
        cout << d[N - k + 1];
    }
    cout << '\n';

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    long long C;
    cin >> N >> C;

    vector<long long> l(N);
    for (int i = 0; i < N; i++) cin >> l[i];

    const int PERIODS = 6;
    vector<long long> pos;
    pos.reserve(PERIODS * N);
    for (int p = 0; p < PERIODS; p++) {
        for (int i = 0; i < N; i++) {
            pos.push_back(l[i] + 1LL * p * C);
        }
    }

    auto check = [&](int k, long long D) -> bool {
        if (D == 0) return true;

        vector<long long> selected;
        selected.reserve(PERIODS * N);
        deque<long long> q;

        for (long long x : pos) {
            while (!q.empty() && q.front() <= x - D) {
                q.pop_front();
            }
            if ((int)q.size() < k) {
                q.push_back(x);
                selected.push_back(x);
            }
        }

        int need = 2 * k;
        int m = (int)selected.size();
        if (m < need) return false;

        vector<int> bad(m, 0);
        for (int i = 0; i + k < m; i++) {
            long long diff = selected[i + k] - selected[i];
            if (diff < D || diff > C - D) bad[i] = 1;
        }

        vector<int> pref(m + 1, 0);
        for (int i = 0; i < m; i++) {
            pref[i + 1] = pref[i] + bad[i];
        }

        for (int i = 0; i + need <= m; i++) {
            if (selected[i + need - 1] - selected[i] >= C) continue;

            // Need all bad[i .. i+k-1] to be zero.
            if (pref[i + k] - pref[i] == 0) {
                return true;
            }
        }

        return false;
    };

    int K = N / 2;
    vector<long long> ans(K + 1);

    for (int k = 1; k <= K; k++) {
        long long lo = 0, hi = C / 2;

        while (lo < hi) {
            long long mid = (lo + hi + 1) / 2;
            if (check(k, mid)) {
                lo = mid;
            } else {
                hi = mid - 1;
            }
        }

        ans[k] = lo;
    }

    for (int k = 1; k <= K; k++) {
        if (k > 1) cout << ' ';
        cout << ans[k];
    }
    cout << '\n';

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int N;
long long C;
vector<long long> p;

// Maximum matching size for a fixed cut s and threshold D.
// L = p[0..s-1], R = p[s..N-1]
int eval_cut(int s, int D) {
    if (s <= 0 || s >= N) return 0;
    int cnt = 0, i = 0, q = 0;
    long long U = C - D;
    for (int j = s; j < N; ++j) {
        long long r = p[j];
        while (i < s && p[i] <= r - D) ++i;
        while (q < i && p[q] < r - U) ++q;
        if (q < i) {
            ++cnt;
            ++q;
        }
    }
    return cnt;
}

// Uses ternary search assuming f(s) is unimodal. Used only as a fast heuristic.
int max_matching_ternary(int D) {
    if (D == 0) return N / 2;
    if (2LL * D > C) return 0;
    int l = 0, r = N;
    while (r - l > 5) {
        int m1 = l + (r - l) / 3;
        int m2 = r - (r - l) / 3;
        int f1 = eval_cut(m1, D);
        int f2 = eval_cut(m2, D);
        if (f1 < f2) l = m1;
        else r = m2;
    }
    int best = 0;
    for (int s = l; s <= r; ++s)
        best = max(best, eval_cut(s, D));
    return best;
}

// Exact maximum matching size over all cuts.
int max_matching_exact(int D) {
    if (D == 0) return N / 2;
    if (2LL * D > C) return 0;
    int best = 0;
    int half = N / 2;
    long long U = C - D;
    for (int s = 1; s < N; ++s) {
        int max_possible = min(s, N - s);
        if (max_possible <= best) continue;
        int cnt = 0, i = 0, q = 0;
        for (int j = s; j < N; ++j) {
            long long r = p[j];
            while (i < s && p[i] <= r - D) ++i;
            while (q < i && p[q] < r - U) ++q;
            if (q < i) {
                ++cnt;
                ++q;
            }
        }
        if (cnt > best) {
            best = cnt;
            if (best == half) return best;

    int N;
    int64 C;

    vector<int64> loc(N);
    for (int64 &x : loc) cin >> x;

    const int M = N / 2;
    vector<int64> answer(M + 1, 0);

    vector<int64> a(N), bucket(N + 1);
    vector<int64> extended(3 * N);
    vector<int> previousSmaller(3 * N), nextSmaller(3 * N);
    vector<int> st;
    st.reserve(3 * N);

    for (int r = 1; r <= M; ++r) {
        // a[i] is the distance between cow i and cow i+r, cyclically.
        for (int i = 0; i < N; ++i) {
            int j = i + r;
            if (j >= N) j -= N;

            int64 d = abs(loc[i] - loc[j]);
            a[i] = min(d, C - d);
        }

        // Work on three copies to handle cyclic windows.
        for (int i = 0; i < 3 * N; ++i) {
            extended[i] = a[i % N];
        }

        // Nearest strictly smaller element to the left.
        st.clear();
        for (int i = 0; i < 3 * N; ++i) {
            while (!st.empty() && extended[st.back()] >= extended[i]) {
                st.pop_back();
            }
            previousSmaller[i] = st.empty() ? -1 : st.back();
            st.push_back(i);
        }

        // Nearest strictly smaller element to the right.
        st.clear();
        for (int i = 3 * N - 1; i >= 0; --i) {
            while (!st.empty() && extended[st.back()] >= extended[i]) {
                st.pop_back();
            }
            nextSmaller[i] = st.empty() ? 3 * N : st.back();
            st.push_back(i);
        }

        fill(bucket.begin(), bucket.end(), -1);

        for (int i = 0; i < N; ++i) {
            int middle = i + N;
            int span;

            if (previousSmaller[middle] == -1 ||
                nextSmaller[middle] == 3 * N) {
                // A global minimum can be included in a window of any length.
                span = N;
            } else {
                span = nextSmaller[middle] - previousSmaller[middle] - 1;
                span = min(span, N);
            }

            bucket[span] = max(bucket[span], a[i]);
        }

        // best[k] = maximum possible minimum of a cyclic window of length k.
        for (int k = N - 1; k >= 1; --k) {

        for (int k = 1; k <= r; ++k) {
            answer[k] = max(answer[k], bucket[k]);
        }
    }

    for (int k = 1; k <= M; ++k) {

using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    int64 C;
    cin >> N >> C;

    vector<int64> x(2 * N);
    for (int i = 0; i < N; ++i) cin >> x[i];
    sort(x.begin(), x.begin() + N);

    // Duplicate the circle so cyclic blocks become ordinary intervals.
    for (int i = 0; i < N; ++i) {
        x[i + N] = x[i] + C;
    }

    /*
       Returns whether some block of m consecutive cows has
       pairwise circular distances at most D.
    */
    auto existsBlock = [&](int m, int64 D) -> bool {
        int p = -1;
        int maxBad = -1;
        const int64 upper = C - D;

        for (int j = 0; j < m + N - 1; ++j) {
            // p becomes the largest i < j with x[j] - x[i] > D.
            while (p + 1 < j && x[j] - x[p + 1] > D) {
                ++p;
            }

            /*
               A pair i < j is bad when
                   D < x[j] - x[i] < C-D.

               If p is too far away, all earlier indices are also too far,
               so there is no bad pair ending at j.
            */
            if (p >= 0 && x[j] - x[p] < upper) {
                maxBad = max(maxBad, p);
            }

            if (j >= m - 1) {
                int start = j - m + 1;

                // A bad pair is inside this block exactly when maxBad >= start.
                if (maxBad < start) {
                    return true;
                }
            }
        }

        return false;
    };

    vector<int64> answer(N / 2);

    for (int k = 1; k <= N / 2; ++k) {
        int m = N - k + 1;

        int64 low = 0;
        int64 high = C / 2;

        while (low < high) {
            int64 mid = low + (high - low) / 2;
            if (existsBlock(m, mid)) {
                high = mid;
            } else {
                low = mid + 1;
            }
        }

        answer[k - 1] = low;
    }

    for (int i = 0; i < (int)answer.size(); ++i) {
        if (i) cout << ' ';
        cout << answer[i];
    }
    cout << '\n';

}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

class Solver {
private:
    int N;
    int64 C;
    vector<int64> position;

    vector<int> leftmost, rightmost, cutMemo;
    unordered_map<int, int> distanceMemo;

    // Maximum number of valid pairs when every distance is at least d.
    int maximumMatching(int64 d) {
        auto found = distanceMemo.find(static_cast<int>(d));
        if (found != distanceMemo.end()) {
            return found->second;
        }

        const int64 upperDistance = C - d;

        // For row i, valid columns form the interval
        // [leftmost[i], rightmost[i]].
        int ptr = 0;
        for (int i = 0; i < N; ++i) {
            while (ptr < N && position[ptr] - position[i] < d) {
                ++ptr;
            }
            leftmost[i] = ptr;
        }

        ptr = -1;
        for (int i = 0; i < N; ++i) {
            while (ptr + 1 < N &&
                   position[ptr + 1] - position[i] <= upperDistance) {
                ++ptr;
            }
            rightmost[i] = ptr;
        }

        fill(cutMemo.begin(), cutMemo.end(), -1);

        // Maximum bipartite matching across the cut after cow t:
        // left endpoints are at most t, right endpoints are greater than t.
        auto cutCount = [&](int t) -> int {
            int &cached = cutMemo[t];
            if (cached != -1) {
                return cached;
            }

            int nextRight = t + 1;
            int result = 0;

            for (int i = 0; i <= t; ++i) {
                if (nextRight < leftmost[i]) {
                    nextRight = leftmost[i];
                }

                if (nextRight <= rightmost[i]) {
                    ++result;
                    ++nextRight;
                }
            }

            cached = result;
            return cached;
        };

        // The sequence cutCount(t) is concave, so its maximum can be
        // found by binary search.
        int low = 0;
        int high = N - 1;

        while (low < high) {
            int mid = low + (high - low) / 2;

            if (cutCount(mid) < cutCount(mid + 1)) {
                low = mid + 1;
            } else {
                high = mid;
            }
        }

        int answer = cutCount(low);
        distanceMemo.emplace(static_cast<int>(d), answer);
        return answer;
    }

public:
    void solve() {
        cin >> N >> C;

        position.resize(N);
        for (int i = 0; i < N; ++i) {
            cin >> position[i];
        }

        leftmost.resize(N);
        rightmost.resize(N);
        cutMemo.resize(N);

        distanceMemo.reserve(32768);
        distanceMemo.max_load_factor(0.7f);

        const int K = N / 2;
        int64 previousAnswer = C / 2;

        for (int k = 1; k <= K; ++k) {
            int64 low = 1;
            int64 high = previousAnswer;

            while (low < high) {
                int64 mid = low + (high - low + 1) / 2;

                if (maximumMatching(mid) >= k) {
                    low = mid;
                } else {
                    high = mid - 1;
                }
            }

            if (k > 1) {
                cout << ' ';
            }
            cout << low;

            // The optimal minimum distance cannot increase as k increases.
            previousAnswer = low;
        }

        cout << '\n';
    }
};

    cin.tie(nullptr);

    Solver solver;
    solver.solve();

    return 0;
}
