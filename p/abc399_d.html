import sys


def main():
    it = iter(map(int, sys.stdin.buffer.read().split()))
    t = next(it)
    output = []

    for _ in range(t):
        n = next(it)
        m = 2 * n
        a = [next(it) for _ in range(m)]

        # A label is invalid if its two occurrences are adjacent.
        invalid = bytearray(n + 1)
        for i in range(m - 1):
            if a[i] == a[i + 1]:
                invalid[a[i]] = 1

        base = n + 1
        first_seen = {}
        answer = 0

        # A valid label pair must occur in two non-overlapping adjacent
        # bigrams. Conversely, one swap between those bigrams suffices.
        for i in range(m - 1):
            x, y = a[i], a[i + 1]
            if x == y or invalid[x] or invalid[y]:
                continue

            if x > y:
                x, y = y, x
            key = x * base + y

            start = first_seen.get(key)
            if start is None:
                first_seen[key] = i
            elif start >= 0 and i - start >= 2:
                # Mark as counted so later bigrams cannot count it again.
                first_seen[key] = -1
                answer += 1

        output.append(str(answer))

    sys.stdout.write("\n".join(output))


if __name__ == "__main__":
    main()

import sys


def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    idx = 0
    T = data[idx]
    idx += 1

    answers = []

    for _ in range(T):
        N = data[idx]
        idx += 1

        L = 2 * N

        if N == 1:
            idx += L
            answers.append("0")
            continue

        A = [0] * (L + 1)
        left = [0] * (N + 1)
        right = [0] * (N + 1)

        for pos in range(1, L + 1):
            v = data[idx]
            idx += 1
            A[pos] = v

            if left[v] == 0:
                left[v] = pos
            else:
                right[v] = pos

        # key = unordered pair of different labels
        # value = list of starting positions of adjacent pairs having this label pair
        edges = {}

        for p in range(1, L):
            u = A[p]
            v = A[p + 1]

            if u == v:
                continue

            # Both couples must not already be adjacent.
            if right[u] - left[u] == 1 or right[v] - left[v] == 1:
                continue

            if u < v:
                key = (u, v)
            else:
                key = (v, u)

            edges.setdefault(key, []).append(p)

        ans = 0

        for starts in edges.values():
            k = len(starts)
            for i in range(k):
                p = starts[i]
                for j in range(i + 1, k):
                    q = starts[j]
                    # The two adjacent pairs must be disjoint.
                    if p + 1 < q:
                        ans += 1

        answers.append(str(ans))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()

import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    t = data[0]
    idx = 1
    out = []

    for _ in range(t):
        n = data[idx]
        idx += 1
        total = 2 * n

        if n == 1:
            idx += total
            out.append("0")
            continue

        l = [0] * (n + 1)
        r = [0] * (n + 1)
        left_owner = [0] * (total + 2)

        for pos in range(1, total + 1):
            v = data[idx]
            idx += 1
            if l[v] == 0:
                l[v] = pos
                left_owner[pos] = v
            else:
                r[v] = pos

        ans = 0
        for i in range(1, n + 1):
            if r[i] - l[i] <= 1:
                continue

            j = left_owner[l[i] + 1]
            if j != 0 and r[j] - l[j] > 1 and abs(r[i] - r[j]) == 1:
                ans += 1

        out.append(str(ans))

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
