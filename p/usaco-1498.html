using ll = long long;
const ll MOD = 1000000007LL;

ll modPow(ll a, ll e) {
    ll res = 1;
    while (e > 0) {
        if (e & 1) res = res * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return res;
}

struct Run {
    ll start, len;
    int bit;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    ll N;
    int M, Q;
    cin >> N >> M >> Q;

    map<ll, int> diff;
    for (int i = 0; i < M; i++) {
        ll l, r;
        cin >> l >> r;
        diff[l] ^= 1;
        diff[r + 1] ^= 1;
    }

    vector<Run> runs;
    runs.reserve(diff.size() + 1);

    auto addRun = [&](ll start, ll len, int bit) {
        if (len <= 0) return;
        if (!runs.empty() && runs.back().bit == bit) {
            runs.back().len += len;
        } else {
            runs.push_back({start, len, bit});
        }
    };

    ll pos = 1;
    int curBit = 0;
    for (const auto &[x, delta] : diff) {
        if (x > pos) addRun(pos, x - pos, curBit);
        curBit ^= delta;
        pos = x;
    }
    if (pos <= N) addRun(pos, N - pos + 1, curBit);

    int R = (int)runs.size();
    vector<ll> starts(R);
    vector<ll> zeroPref(R + 1, 0);
    vector<ll> valuePref(R + 1, 0);

    for (int i = 0; i < R; i++) {
        starts[i] = runs[i].start;
        zeroPref[i + 1] = zeroPref[i] +
                          (runs[i].bit == 0 ? runs[i].len : 0);

        ll p = modPow(2, runs[i].len);
        ll add = runs[i].bit ? p - 1 : 0;
        valuePref[i + 1] = (valuePref[i] * p + add) % MOD;
    }

    auto runContaining = [&](ll x) {
        return int(upper_bound(starts.begin(), starts.end(), x) -
                   starts.begin() - 1);
    };

    auto zerosUpTo = [&](ll x) -> ll {
        if (x <= 0) return 0;
        int i = runContaining(x);
        ll result = zeroPref[i];
        if (runs[i].bit == 0) result += x - runs[i].start + 1;
        return result;
    };

    // Value of the binary string consisting of global positions 1..x.
    auto prefixValue = [&](ll x) -> ll {
        if (x <= 0) return 0;
        int i = runContaining(x);
        ll taken = x - runs[i].start + 1;
        if (taken == runs[i].len) return valuePref[i + 1];

        ll p = modPow(2, taken);
        ll result = valuePref[i] * p % MOD;
        if (runs[i].bit) result = (result + p - 1) % MOD;
        return result;
    };

    while (Q--) {
        ll l, r, k;
        cin >> l >> r >> k;

        ll length = r - l + 1;
        ll deletions = length - k;
        ll zerosBefore = zerosUpTo(l - 1);
        ll zeros = zerosUpTo(r) - zerosBefore;

        if (deletions >= zeros) {
            // There are enough ones to make the answer k consecutive ones.
            cout << (modPow(2, k) + MOD - 1) % MOD << '\n';
            continue;
        }

        // Delete the first `deletions` zeros in the query. If p is the
        // position immediately after them, the answer consists of all ones
        // before p, followed by the unchanged suffix s[p..r].
        ll p;
        if (deletions == 0) {
            p = l;
        } else {
            ll target = zerosBefore + deletions;
            int j = int(lower_bound(zeroPref.begin(), zeroPref.end(), target) -
                        zeroPref.begin());
            int i = j - 1;
            ll offset = target - zeroPref[i];
            ll zeroPosition = runs[i].start + offset - 1;
            p = zeroPosition + 1;
        }

        ll suffixLength = r - p + 1;
        ll A = prefixValue(p - 1);
        ll B = prefixValue(r);
        ll suffixPow = modPow(2, suffixLength);
        ll answerPow = modPow(2, k);

        ll product = (A + 1) % MOD * suffixPow % MOD;
        ll answer = (answerPow - product + MOD) % MOD;
        answer = (answer + B) % MOD;

        cout << answer << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
const int64 MOD = 1000000007LL;

int64 pow2mod(int64 e) {
    int64 result = 1, base = 2;
    while (e > 0) {
        if (e & 1) result = result * base % MOD;
        base = base * base % MOD;
        e >>= 1;
    }
    return result;
}

    int bit;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int64 N;
    int M, Q;
    cin >> N >> M >> Q;

    vector<pair<int64, int>> events;
    events.reserve(2 * M + 2);
    events.push_back({1, 0});
    events.push_back({N + 1, 0});

    for (int i = 0; i < M; ++i) {
        int64 l, r;
        cin >> l >> r;
        events.push_back({l, 1});
        events.push_back({r + 1, 1});
    }

    sort(events.begin(), events.end());

    vector<Run> runs;
    int parity = 0;

    for (size_t i = 0; i < events.size();) {
        size_t j = i;
        int64 pos = events[i].first;
        int x = 0;

        while (j < events.size() && events[j].first == pos) {
            x ^= events[j].second;
            ++j;
        }
        parity ^= x;

        if (j < events.size()) {
            int64 nextPos = events[j].first;
            int64 end = min(N, nextPos - 1);
            if (pos <= end) {
                if (!runs.empty() && runs.back().bit == parity &&
                    runs.back().r + 1 == pos) {
                    runs.back().r = end;
                } else {
                    runs.push_back({pos, end, parity});
                }
            }
        }
        i = j;
    }

    int R = (int)runs.size();
    vector<int64> ends(R);
    vector<int64> prefOnes(R + 1, 0), prefZeros(R + 1, 0);
    vector<int64> prefHash(R + 1, 0);

    for (int i = 0; i < R; ++i) {
        ends[i] = runs[i].r;
        int64 len = runs[i].r - runs[i].l + 1;
        int64 p = pow2mod(len);

        prefOnes[i + 1] = prefOnes[i] + (runs[i].bit ? len : 0);
        prefZeros[i + 1] = prefZeros[i] + (runs[i].bit ? 0 : len);

        prefHash[i + 1] = prefHash[i] * p % MOD;
        if (runs[i].bit) {
            prefHash[i + 1] = (prefHash[i + 1] + p - 1) % MOD;
        }
    }

    auto prefixCount = [&](int64 x, int bit) -> int64 {
        if (x <= 0) return 0;
        int idx = (int)(lower_bound(ends.begin(), ends.end(), x) - ends.begin());
        int64 taken = x - runs[idx].l + 1;
        if (bit == 1) {
            return prefOnes[idx] + (runs[idx].bit == 1 ? taken : 0);
        }
        return prefZeros[idx] + (runs[idx].bit == 0 ? taken : 0);
    };

    auto prefixHash = [&](int64 x) -> int64 {
        if (x <= 0) return 0;
        int idx = (int)(lower_bound(ends.begin(), ends.end(), x) - ends.begin());
        int64 taken = x - runs[idx].l + 1;
        int64 fullLen = runs[idx].r - runs[idx].l + 1;

        if (taken == fullLen) return prefHash[idx + 1];

        int64 p = pow2mod(taken);
        int64 value = prefHash[idx] * p % MOD;
        if (runs[idx].bit) value = (value + p - 1) % MOD;
        return value;
    };

    while (Q--) {
        int64 l, r, k;
        cin >> l >> r >> k;

        int64 length = r - l + 1;
        int64 deletions = length - k;
        int64 zerosBefore = prefixCount(l - 1, 0);
        int64 zeros = prefixCount(r, 0) - zerosBefore;

        if (zeros <= deletions) {
            int64 answer = pow2mod(k) - 1;
            if (answer < 0) answer += MOD;
            cout << answer << '\n';
            continue;
        }

        int64 targetRank = zerosBefore + deletions + 1;
        int boundary = (int)(lower_bound(prefZeros.begin(), prefZeros.end(),
                                        targetRank) - prefZeros.begin());
        int idx = boundary - 1;

        int64 offset = targetRank - prefZeros[idx];
        int64 p = runs[idx].l + offset - 1;

        int64 leadingOnes = p - l - deletions;
        int64 suffixLength = r - p + 1;

        int64 suffixPow = pow2mod(suffixLength);
        int64 suffixValue = (prefixHash(r) -
                             prefixHash(p - 1) * suffixPow) % MOD;
        if (suffixValue < 0) suffixValue += MOD;

        int64 leadingPow = pow2mod(leadingOnes);
        int64 answer = ((leadingPow - 1) * suffixPow + suffixValue) % MOD;
        if (answer < 0) answer += MOD;

        cout << answer << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

static const long long MOD = 1000000007LL;

long long modPow(long long a, long long e) {
    long long res = 1;
    while (e > 0) {
        if (e & 1) res = res * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return res;
}

struct Run {
    long long l, r;
    int bit;
};

struct Node {
    long long val, pw;
};

Node makeNode(int bit, long long len) {
    long long pw = modPow(2, len);
    return {bit ? pw - 1 : 0, pw};
}

Node combine(const Node &a, const Node &b) {
    return {
        (a.val * b.pw + b.val) % MOD,
        a.pw * b.pw % MOD
    };
}

class ValueTree {
private:
    const vector<Run> &runs;
    int n;
    vector<Node> tree;

    void build(int v, int tl, int tr) {
        if (tl == tr) {
            tree[v] = makeNode(runs[tl].bit, runs[tl].r - runs[tl].l + 1);
            return;
        }
        int tm = (tl + tr) / 2;
        build(v * 2, tl, tm);
        build(v * 2 + 1, tm + 1, tr);
        tree[v] = combine(tree[v * 2], tree[v * 2 + 1]);
    }

    Node query(int v, int tl, int tr, int l, int r) const {
        if (l == tl && r == tr) return tree[v];
        int tm = (tl + tr) / 2;
        if (r <= tm) return query(v * 2, tl, tm, l, r);
        if (l > tm) return query(v * 2 + 1, tm + 1, tr, l, r);
        return combine(
            query(v * 2, tl, tm, l, tm),
            query(v * 2 + 1, tm + 1, tr, tm + 1, r)
        );
    }

public:
    ValueTree(const vector<Run> &runs) : runs(runs), n((int)runs.size()), tree(4 * n) {
        build(1, 0, n - 1);
    }

    Node query(int l, int r) const {
        return query(1, 0, n - 1, l, r);
    }
};

        answer = (answer + suffixValue) % MOD;

    return 0;

#include <bits/stdc++.h>
using namespace std;

const int MOD = 1000000007;

int modpow(long long a, long long e) {
    long long res = 1;
    a %= MOD;
    while (e > 0) {
        if (e & 1) res = res * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return (int)res;
}

struct Node {
    long long len;
    long long ones;
    int val;
    int p2;
};

Node combineNode(const Node& A, const Node& B) {
    if (A.len == 0) return B;
    if (B.len == 0) return A;

    Node C;
    C.len = A.len + B.len;
    C.ones = A.ones + B.ones;
    C.val = ((long long)A.val * B.p2 + B.val) % MOD;
    C.p2 = (long long)A.p2 * B.p2 % MOD;
    return C;
}

struct SegTree {
    int size;
    vector<Node> tr;

    void init(int n) {

// O(1) evaluation of 2^e modulo MOD for 0 <= e <= 1e9.
// Since 2^15 * 2^15 > 1e9, two tables of size 2^15 suffice.
struct PowersOfTwo {
    static constexpr int SHIFT = 15;
    static constexpr int SIZE = 1 << SHIFT;
    static constexpr int MASK = SIZE - 1;

    vector<int> low, high;

    PowersOfTwo() : low(SIZE), high(SIZE) {
        low[0] = 1;
        for (int i = 1; i < SIZE; ++i) {
            low[i] = (2LL * low[i - 1]) % MOD;
        }

        long long step = (2LL * low[SIZE - 1]) % MOD; // 2^SIZE
        high[0] = 1;
        for (int i = 1; i < SIZE; ++i) {
            high[i] = (1LL * high[i - 1] * step) % MOD;
        }
    }

    int get(long long exponent) const {
        int a = int(exponent >> SHIFT);
        int b = int(exponent & MASK);
        return (1LL * high[a] * low[b]) % MOD;
    }
};

// Computes 2^e and 2^(-e) modulo MOD in O(1).
// small tables handle the low bits, large tables handle complete blocks.
class PowerTable {
    static constexpr int B = 1 << 15;

    vector<int> smallPow, largePow;
    vector<int> smallInv, largeInv;

public:
    explicit PowerTable(int maxExponent) {
        smallPow.resize(B);
        smallInv.resize(B);

        smallPow[0] = smallInv[0] = 1;
        const int INV2 = 500'000'004;

        for (int i = 1; i < B; ++i) {
            smallPow[i] = 2LL * smallPow[i - 1] % MOD;
            smallInv[i] = 1LL * smallInv[i - 1] * INV2 % MOD;
        }

        int blocks = maxExponent / B + 2;
        largePow.assign(blocks, 1);
        largeInv.assign(blocks, 1);

        int step = 2LL * smallPow[B - 1] % MOD;
        int inverseStep = 1LL * smallInv[B - 1] * INV2 % MOD;

        for (int i = 1; i < blocks; ++i) {
            largePow[i] = 1LL * largePow[i - 1] * step % MOD;
            largeInv[i] = 1LL * largeInv[i - 1] * inverseStep % MOD;
        }
    }

    int powerOfTwo(long long exponent) const {
        int block = static_cast<int>(exponent / B);
        int offset = static_cast<int>(exponent % B);
        return 1LL * largePow[block] * smallPow[offset] % MOD;
    }

    int inversePowerOfTwo(long long exponent) const {
        int block = static_cast<int>(exponent / B);
        int offset = static_cast<int>(exponent % B);
        return 1LL * largeInv[block] * smallInv[offset] % MOD;
    }
};

    auto zerosPrefix = [&](int position) -> int {
        if (position <= 0) return 0;

        int index = runIndex(position);
        int result = prefixZeros[index];

        if (!runs[index].bit) {
            result += position - runs[index].l + 1;
        }
        return result;
    };

    auto weightPrefix = [&](int position) -> int {
        if (position <= 0) return 0;

        int index = runIndex(position);
        int result = prefixWeight[index];

        if (runs[index].bit) {
            int contribution =
                powers.inversePowerOfTwo(runs[index].l - 1) -
                powers.inversePowerOfTwo(position);

            if (contribution < 0) contribution += MOD;

            result += contribution;
            if (result >= MOD) result -= MOD;
        }
        return result;
    };

    // Binary value of s[l..r], interpreted as a binary number.
    auto rangeValue = [&](int l, int r) -> int {
        if (l > r) return 0;

        int difference = weightPrefix(r) - weightPrefix(l - 1);
        if (difference < 0) difference += MOD;

        return 1LL * powers.powerOfTwo(r) * difference % MOD;
    };

    string output;
    output.reserve(static_cast<size_t>(Q) * 12);

    for (int query = 0; query < Q; ++query) {
        int l, r, k;
        cin >> l >> r >> k;

        int length = r - l + 1;
        int ones = onesPrefix(r) - onesPrefix(l - 1);
        int answer;

        if (ones >= k) {
            // The answer consists of k ones.
            answer = powers.powerOfTwo(k) - 1;
            if (answer < 0) answer += MOD;
        } else {
            int deletedZeros = length - k;

            // If deletedZeros == 0, use the virtual position l-1.
            int p = l - 1;

            if (deletedZeros > 0) {
                // Global rank of the deletedZeros-th zero in [l, r].
                long long target =
                    static_cast<long long>(zerosPrefix(l - 1)) +
                    deletedZeros;

                auto it = lower_bound(
                    prefixZeros.begin(), prefixZeros.end(), target
                );

                int completedRuns =
                    static_cast<int>(it - prefixZeros.begin());
                int index = completedRuns - 1;

                long long offset =
                    target - prefixZeros[index];

                p = runs[index].l + static_cast<int>(offset) - 1;
            }

            int suffixLength = r - p;
            int suffixValue = rangeValue(p + 1, r);

            long long value =
                static_cast<long long>(powers.powerOfTwo(k)) -
                powers.powerOfTwo(suffixLength) +
                suffixValue;

            value %= MOD;
            if (value < 0) value += MOD;
            answer = static_cast<int>(value);
        }

        output += to_string(answer);
        output.push_back('\n');
    }

    cout << output;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
static constexpr int64 MOD = 1'000'000'007LL;

class FastInput {
    static constexpr size_t SIZE = 1 << 16;
    char buffer[SIZE];
    size_t pos = 0, len = 0;

    int getChar() {
        if (pos == len) {
            len = fread(buffer, 1, SIZE, stdin);
            pos = 0;
            if (len == 0) return EOF;
        }
        return static_cast<unsigned char>(buffer[pos++]);
    }

public:
    template <class T>
    bool read(T& value) {
        int c = getChar();
        while (c != EOF && c <= ' ') c = getChar();
        if (c == EOF) return false;

        bool negative = false;
        if (c == '-') {
            negative = true;
            c = getChar();
        }

        T result = 0;
        while (c >= '0' && c <= '9') {
            result = result * 10 + (c - '0');
            c = getChar();
        }

        value = negative ? -result : result;
        return true;
    }
};

class FastOutput {
    static constexpr size_t SIZE = 1 << 16;
    char buffer[SIZE];
    size_t pos = 0;

public:
    ~FastOutput() {
        flush();
    }

    void flush() {
        if (pos) {
            fwrite(buffer, 1, pos, stdout);
            pos = 0;
        }
    }

    void put(char c) {
        if (pos == SIZE) flush();
        buffer[pos++] = c;
    }

    void writeInt(int64 value) {
        if (value == 0) {
            put('0');
            put('\n');
            return;
        }

        char digits[24];
        int count = 0;
        while (value) {
            digits[count++] = char('0' + value % 10);
            value /= 10;
        }
        while (count) put(digits[--count]);
        put('\n');
    }
};

// O(1) lookup for 2^exponent, where exponent <= 1e9 < 2^30.
// Split the exponent into two 15-bit parts.
class PowerOfTwo {
    static constexpr int SHIFT = 15;
    static constexpr int BLOCK = 1 << SHIFT;

    vector<int64> low, high;

public:
    explicit PowerOfTwo(int64 maxExponent) {
        low.resize(BLOCK);
        low[0] = 1;
        for (int i = 1; i < BLOCK; ++i) {
            low[i] = low[i - 1] * 2 % MOD;
        }

        int64 jump = low[BLOCK - 1] * 2 % MOD; // 2^BLOCK

        high.resize(static_cast<size_t>((maxExponent >> SHIFT) + 1));
        high[0] = 1;
        for (size_t i = 1; i < high.size(); ++i) {
            high[i] = high[i - 1] * jump % MOD;
        }
    }

    int64 operator()(int64 exponent) const {
        int lo = static_cast<int>(exponent & (BLOCK - 1));
        int hi = static_cast<int>(exponent >> SHIFT);
        return low[lo] * high[hi] % MOD;
    }
};

struct Run {
    int64 start, end;
    int bit;
    int64 onesBefore;
    int64 zerosBefore;
    int64 valueBefore;
};

struct PrefixInfo {
    int64 ones;
    int64 value;
};

int main() {
    FastInput in;
    FastOutput out;

    int64 N;
    int M, Q;
    if (!in.read(N)) return 0;
    in.read(M);
    in.read(Q);

    // Each flip toggles at l and again at r+1.
    vector<int64> events;
    events.reserve(2 * M);

    for (int i = 0; i < M; ++i) {
        int64 l, r;
        in.read(l);
        in.read(r);
        events.push_back(l);
        events.push_back(r + 1);
    }

    sort(events.begin(), events.end());

    vector<Run> runs;
    runs.reserve(events.size() + 1);

    auto addRun = [&](int64 l, int64 r, int bit) {
        if (l > r) return;

        if (!runs.empty() &&
            runs.back().bit == bit &&
            runs.back().end + 1 == l) {
            runs.back().end = r;
        } else {
            runs.push_back(Run{l, r, bit, 0, 0, 0});
        }
    };

    int64 current = 1;
    int currentBit = 0;

    size_t pos = 0;
    while (pos < events.size()) {
        int64 x = events[pos];
        addRun(current, x - 1, currentBit);

        size_t next = pos;
        while (next < events.size() && events[next] == x) ++next;

        currentBit ^= static_cast<int>((next - pos) & 1ULL);
        current = x;
        pos = next;
    }

    if (current <= N) {
        addRun(current, N, currentBit);
    }

    PowerOfTwo power2(N);

    vector<int64> starts;
    vector<int64> cumulativeZeros;
    starts.reserve(runs.size());
    cumulativeZeros.reserve(runs.size());

    int64 prefixOnes = 0;
    int64 prefixZeros = 0;
    int64 prefixValue = 0;

    for (Run& run : runs) {
        int64 length = run.end - run.start + 1;

        run.onesBefore = prefixOnes;
        run.zerosBefore = prefixZeros;
        run.valueBefore = prefixValue;

        int64 p = power2(length);
        prefixValue = prefixValue * p % MOD;
        if (run.bit) {
            prefixValue = (prefixValue + p - 1 + MOD) % MOD;
        }

        if (run.bit) {
            prefixOnes += length;
        } else {
            prefixZeros += length;
        }

        starts.push_back(run.start);
        cumulativeZeros.push_back(prefixZeros);
    }

    // Information about the prefix S[1..x].
    auto prefixInfo = [&](int64 x) -> PrefixInfo {
        if (x <= 0) return {0, 0};

        int id = static_cast<int>(
            upper_bound(starts.begin(), starts.end(), x) - starts.begin()
        ) - 1;

        int64 length = x - runs[id].start + 1;
        int64 ones = runs[id].onesBefore;
        if (runs[id].bit) ones += length;

        int64 p = power2(length);
        int64 value = runs[id].valueBefore * p % MOD;
        if (runs[id].bit) {
            value = (value + p - 1 + MOD) % MOD;
        }

        return {ones, value};
    };

    for (int qi = 0; qi < Q; ++qi) {
        int64 l, r, k;
        in.read(l);
        in.read(r);
        in.read(k);

        PrefixInfo left = prefixInfo(l - 1);
        PrefixInfo right = prefixInfo(r);

        int64 countOnes = right.ones - left.ones;

        // We can choose k ones, so 111...111 is optimal.
        if (countOnes >= k) {
            out.writeInt((power2(k) - 1 + MOD) % MOD);
            continue;
        }

        int64 substringLength = r - l + 1;
        int64 deletions = substringLength - k;

        // The optimal subsequence retains all ones and the rightmost
        // (k - countOnes) zeros. Thus the first retained zero is the
        // (deletions + 1)-st zero of the queried substring.
        int64 zerosBeforeL = (l - 1) - left.ones;
        int64 targetZero = zerosBeforeL + deletions + 1;

        int id = static_cast<int>(
            lower_bound(cumulativeZeros.begin(), cumulativeZeros.end(),
                        targetZero) - cumulativeZeros.begin()
        );

        const Run& zeroRun = runs[id];
        int64 localZeroIndex =
            targetZero - zeroRun.zerosBefore; // 1-indexed within this run
        int64 x = zeroRun.start + localZeroIndex - 1;

        // All characters before x that remain are ones.
        int64 p = zeroRun.onesBefore - left.ones;

        int64 suffixLength = r - x + 1;
        int64 scale = power2(suffixLength);

        // The run containing x is a zero run, so this is value(S[1..x-1]).
        int64 valueBeforeX =
            zeroRun.valueBefore * power2(x - zeroRun.start) % MOD;

        int64 suffixValue =
            (right.value - valueBeforeX * scale % MOD + MOD) % MOD;

        int64 leadingValue = (power2(p) - 1 + MOD) % MOD;
        int64 answer = (leadingValue * scale + suffixValue) % MOD;

        out.writeInt(answer);
    }

    return 0;
}
