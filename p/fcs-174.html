# Graph 3-Coloring ## Problem You are given an undirected graph with n vertices and m edges. Your task is to assign each vertex one of three colors. The objective is to minimize the number of conflicting edges. An edge is conflicting if its two endpoints have the same color. ## Input Format - Line 1: two integers n and m (1 ≤ n ≤ 1000, 0 ≤ m ≤ n(n−1)/2) - Next m lines: two integers u v (1 ≤ u, v ≤ n, u ≠ v) The graph may be disconnected. There are no multiple edges or self-loops. ## Output Format - Output exactly one line: - n integers c₁ c₂ … cₙ - each cᵢ ∈ {1, 2, 3} ## Scoring All feasibility and interaction limits apply before scoring; invalid output receives zero. Points are averaged over test files. For the logarithmic branch, write x for the original minimizing quantity and a for its physical 60-point reference. The improvement factor is r=a/x, and the upper-band award is P=min(100,60+ln(r)). Here ln is natural: r=1 corresponds to 60 points, and on the uncapped logarithmic upper band, r=exp(P-60). Each additional point multiplies the physical improvement factor by e (divides a minimizing objective by e or multiplies a maximizing objective by e, as applicable). The lower-band, zero-objective, and completion rules below still apply. E is the number of edges whose endpoints have the same color, and m is the number of graph edges. Define the objective x=E, its physical reference a=A=0.4*m, and the lower-band calculation L=100*(1-E/m) when m>0. If m=0, every valid coloring receives 100; the following rule otherwise applies. ```text P = 100, if x = 0 P = L, if L <= 60 and x > 0 P = min(100, 60 + ln(r)), otherwise ``` Zero is tested on the valid original count, cost, or error, before the lower-band test. A zero reference anchor has no positive improvement branch; positive objectives keep their lower-band score. The checker reports P/100.

struct FastRNG {
    uint64_t state;
    explicit FastRNG(uint64_t seed) : state(seed) {}
    uint64_t next() {
        uint64_t x = state;
        x ^= x >> 12;
        x ^= x << 25;
        x ^= x >> 27;
        state = x;
        return x * 2685821657736338717ULL;
    }
    int operator()(int bound) {
        return int(next() % uint64_t(bound));
    }
};

vector<int> greedyColoring(bool randomTie) {
    vector<int> col(n, -1);
    vector<array<int, 3>> gcnt(n);
    for (int i = 0; i < n; ++i) gcnt[i] = {0, 0, 0};
    vector<unsigned char> mask(n, 0);
    vector<int> deg(n);
    for (int v = 0; v < n; ++v) deg[v] = (int)adj[v].size();

    for (int step = 0; step < n; ++step) {
        int best = -1;
        int bestSat = -1;
        int bestDeg = -1;

        for (int v = 0; v < n; ++v) {
            if (col[v] != -1) continue;
            int sat = __builtin_popcount((unsigned)mask[v]);
            if (sat > bestSat ||
                (sat == bestSat && (deg[v] > bestDeg ||
                 (randomTie && deg[v] == bestDeg && (rand() & 1))))) {
                bestSat = sat;
                bestDeg = deg[v];
                best = v;
            }
        }

        int minConf = 1000000000;
        int choices[3];
        int num = 0;
        for (int c = 0; c < 3; ++c) {
            if (gcnt[best][c] < minConf) {
                minConf = gcnt[best][c];
                num = 0;
                choices[num++] = c;
            } else if (gcnt[best][c] == minConf) {
                choices[num++] = c;
            }
        }

        int chosen;
        if (randomTie) chosen = choices[rand() % num];
        else chosen = choices[0];

        col[best] = chosen;

        for (int v : adj[best]) {
            if (col[v] == -1) {
                if (gcnt[v][chosen] == 0) {
                    mask[v] |= (1 << chosen);
                }
                ++gcnt[v][chosen];
            }
        }
    }

    return col;
}

int evaluate(const vector<int>& col) {
    int e = 0;
    for (int v = 0; v < n; ++v) {
        for (int u : adj[v]) {
            if (v < u && col[v] == col[u]) ++e;
        }
    }
    return e;
}

bool checkBipartite(vector<int>& res) {
    res.assign(n, -1);
    queue<int> q;

    for (int s = 0; s < n; ++s) {
        if (res[s] != -1) continue;
        res[s] = 0;
        q.push(s);

        while (!q.empty()) {
            int v = q.front();
            q.pop();

            for (int u : adj[v]) {
                if (res[u] == -1) {
                    res[u] = res[v] ^ 1;
                    q.push(u);
                } else if (res[u] == res[v]) {
                    return false;
                }
            }
        }
    }

    return true;
}

    bool readInt(int &x) {

        do {
            c = getChar();

        } while (c <= ' ');

        }

struct RNG {
    uint64_t state;

    explicit RNG(uint64_t seed) : state(seed) {}

    inline uint64_t next() {
        uint64_t z = (state += 0x9e3779b97f4a7c15ULL);
        z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9ULL;

        return z ^ (z >> 31);
    }

    inline int integer(int n) {
        return n <= 1 ? 0 : (int)(next() % (uint64_t)n);

    for (int i = 0; i < m; ++i) {
        int u, v;
        scanner.readInt(u);
        scanner.readInt(v);
        --u;
        --v;
        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    vector<int> degree(n);
    vector<unsigned char> removed(n, 0);
    vector<int> removalOrder;
    removalOrder.reserve(n);

    for (int v = 0; v < n; ++v) degree[v] = (int)graph[v].size();

    vector<int> queue;
    queue.reserve(n);
    for (int v = 0; v < n; ++v) {
        if (degree[v] < 3) queue.push_back(v);
    }

    for (size_t head = 0; head < queue.size(); ++head) {
        int v = queue[head];
        if (removed[v]) continue;

        removed[v] = 1;
        removalOrder.push_back(v);

        for (int u : graph[v]) {
            if (!removed[u]) {
                --degree[u];
                if (degree[u] == 2) queue.push_back(u);
            }
        }
    }

    vector<unsigned char> visited(n, 0);
    vector<vector<int>> coreComponents;

    for (int root = 0; root < n; ++root) {
        if (removed[root] || visited[root]) continue;

        vector<int> component;
        component.push_back(root);
        visited[root] = 1;

        for (size_t head = 0; head < component.size(); ++head) {
            int v = component[head];
            for (int u : graph[v]) {
                if (!removed[u] && !visited[u]) {
                    visited[u] = 1;
                    component.push_back(u);
                }
            }
        }
        coreComponents.push_back(std::move(component));
    }

    sort(coreComponents.begin(), coreComponents.end(),
         [](const vector<int> &a, const vector<int> &b) {
             return a.size() < b.size();
         });

    int totalCoreVertices = 0;
    for (const auto &component : coreComponents) {
        totalCoreVertices += (int)component.size();
    }

    vector<int> answer(n, -1);
    long long exactBudget = 300000;
    const long long totalLocalIterations = 120000;

    for (size_t componentIndex = 0;
         componentIndex < coreComponents.size();
         ++componentIndex) {
        const auto &component = coreComponents[componentIndex];
        int size = (int)component.size();

        vector<int> localIndex(n, -1);
        for (int i = 0; i < size; ++i) {
            localIndex[component[i]] = i;
        }

        vector<vector<int>> localGraph(size);
        for (int i = 0; i < size; ++i) {
            int v = component[i];
            for (int u : graph[v]) {
                if (!removed[u] && localIndex[u] != -1) {
                    localGraph[i].push_back(localIndex[u]);
                }
            }
        }

        long long exactAllowance = 100000;
        if (!coreComponents.empty()) {
            long long remainingComponents =
                (long long)coreComponents.size() - (long long)componentIndex;
            exactAllowance = min(exactAllowance,
                                 max(5000LL, exactBudget / remainingComponents));
        }

        int localAllowance = 3000;
        if (totalCoreVertices > 0) {
            localAllowance = (int)min(
                120000LL,
                max(3000LL, totalLocalIterations * size / totalCoreVertices)
            );
        }

        uint64_t seed = 0x9e3779b97f4a7c15ULL ^
                        ((uint64_t)size << 32) ^
                        ((uint64_t)componentIndex * 0x94d049bb133111ebULL);

        CoreSolver solver(std::move(localGraph), seed);
        vector<int> localAnswer = solver.solve(
            exactBudget, exactAllowance, localAllowance
        );

        for (int i = 0; i < size; ++i) {
            answer[component[i]] = localAnswer[i];
        }
    }

    for (int i = (int)removalOrder.size() - 1; i >= 0; --i) {
        int v = removalOrder[i];
        array<int, 3> used{0, 0, 0};

        for (int u : graph[v]) {
            if (answer[u] != -1) used[answer[u]] = 1;
        }

        int chosen = -1;
        for (int c = 0; c < 3; ++c) {
            if (!used[c]) {
                chosen = c;
                break;
            }
        }

        if (chosen == -1) {
            array<int, 3> counts{0, 0, 0};
            for (int u : graph[v]) {
                if (answer[u] != -1) ++counts[answer[u]];
            }
            chosen = (int)(min_element(counts.begin(), counts.end()) -
                            counts.begin());
        }
        answer[v] = chosen;
    }

    string output;
    output.reserve((size_t)n * 2);
    for (int v = 0; v < n; ++v) {
        if (v) output.push_back(' ');
        output.push_back(char('0' + answer[v] + 1));
    }
    output.push_back('\n');
    fwrite(output.data(), 1, output.size(), stdout);

    return 0;
}
