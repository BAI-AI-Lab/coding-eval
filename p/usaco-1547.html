#include <bits/stdc++.h>
using namespace std;

static const long long MOD = 1000000007LL;
static const long long INV2 = 500000004LL;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    long long D;
    cin >> N >> D;

    vector<int> qPos(N), qVal(N);
    int head = 0, tail = 0;

    long long total = 0;   // Sum of all DP states.
    long long active = 0;  // Sum of states whose last coach is still in the queue.
    long long mul = 1;     // Common multiplier for queued values.
    long long invMul = 1;

    for (int i = 0; i < N; ++i) {
        long long p;
        int type;
        cin >> p >> type;

        if (type == 1) {
            // Select this coach after any previous state, or start with it alone.
            long long value = (total + 1) % MOD;
            total = (total + value) % MOD;
            active = (active + value) % MOD;

            qPos[tail] = static_cast<int>(p);
            qVal[tail] = static_cast<int>(value * invMul % MOD);
            ++tail;
        } else {
            long long leftLimit = p - D;

            // These coaches are too far left for this and all future campers.
            while (head < tail && static_cast<long long>(qPos[head]) < leftLimit) {
                long long value = static_cast<long long>(qVal[head]) * mul % MOD;
                active -= value;
                if (active < 0) active += MOD;
                ++head;
            }

            // Every active state may either include or exclude this camper.
            total = (total + active) % MOD;
            active = active * 2 % MOD;
            mul = mul * 2 % MOD;
            invMul = invMul * INV2 % MOD;
        }
    }

    cout << total << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

static constexpr int MOD = 1'000'000'007;

class FastScanner {
    static constexpr size_t BUFFER_SIZE = 1 << 16;
    char buffer[BUFFER_SIZE];
    size_t index = 0, length = 0;

    int getChar() {
        if (index == length) {
            length = fread(buffer, 1, BUFFER_SIZE, stdin);
            index = 0;
            if (length == 0) return EOF;
        }
        return static_cast<unsigned char>(buffer[index++]);
    }

public:
    bool readInt(int& value) {
        int c;
        do {
            c = getChar();
            if (c == EOF) return false;
        } while (c <= ' ');

        value = 0;
        while (c >= '0' && c <= '9') {
            value = value * 10 + (c - '0');
            c = getChar();
        }
        return true;
    }
};

struct CoachState {
    int position;
    int creationCount;
    int campersAtCreation;
};

int main() {
    FastScanner input;

    int N, D;
    if (!input.readInt(N)) return 0;
    input.readInt(D);

    vector<int> powerOfTwo(N + 1);
    powerOfTwo[0] = 1;
    for (int i = 1; i <= N; ++i) {
        powerOfTwo[i] = (2LL * powerOfTwo[i - 1]) % MOD;
    }

    vector<CoachState> coachStates;
    coachStates.reserve(N);
    size_t head = 0;

    int nearCount = 0;
    int farCount = 1;  // The empty subset.
    int campersProcessed = 0;

    for (int i = 0; i < N; ++i) {
        int p, type;
        input.readInt(p);
        input.readInt(type);

        // Expire coach states that cannot support a cow at position p.
        // Equality is allowed because the distance limit is inclusive.
        while (head < coachStates.size() &&
               static_cast<long long>(coachStates[head].position) + D < p) {
            const CoachState& state = coachStates[head];

            int currentCount =
                1LL * state.creationCount *
                powerOfTwo[campersProcessed - state.camperAtCreation] % MOD;

            ++head;

            nearCount -= currentCount;
            if (nearCount < 0) nearCount += MOD;

            farCount += currentCount;
            if (farCount >= MOD) farCount -= MOD;
        }

        if (type == 1) {
            // Coach: every previous subset may omit or include this coach.
            int total = nearCount + farCount;
            if (total >= MOD) total -= MOD;

            nearCount += total;
            if (nearCount >= MOD) nearCount -= MOD;

            coachStates.push_back({p, total, campersProcessed});
        } else {
            // Camper: only subsets with a sufficiently close coach may include it.
            nearCount = (2LL * nearCount) % MOD;
            ++campersProcessed;
        }
    }

    int answer = nearCount + farCount;
    if (answer >= MOD) answer -= MOD;

    // Exclude the empty subset.
    --answer;
    if (answer < 0) answer += MOD;

    printf("%d\n", answer);
    return 0;
}

static constexpr long long INV2 = 500'000'004LL;

class FastScanner {
    static constexpr size_t BUFFER_SIZE = 1 << 16;
    char buffer[BUFFER_SIZE];
    size_t index = 0, length = 0;

    int getChar() {
        if (index == length) {
            length = fread(buffer, 1, BUFFER_SIZE, stdin);
            index = 0;
            if (length == 0) return EOF;
        }
        return static_cast<unsigned char>(buffer[index++]);
    }

public:
    template <typename T>
    bool readInt(T& result) {
        int c;
        do {
            c = getChar();
            if (c == EOF) return false;
        } while (c <= ' ');

        bool negative = false;
        if (c == '-') {
            negative = true;
            c = getChar();
        }

        T value = 0;
        while (c >= '0' && c <= '9') {
            value = value * 10 + (c - '0');
            c = getChar();
        }

        result = negative ? -value : value;
        return true;
    }
};

struct CoachState {
    long long position;
    int storedCount;
};

int main() {
    FastScanner scanner;

    int N;
    long long D;
    if (!scanner.readInt(N)) return 0;
    scanner.readInt(D);

    vector<CoachState> states;
    states.reserve(N);
    size_t head = 0;

    // Number of valid subsets of the processed prefix, including the empty set.
    long long total = 1;

    // Actual count of a state = storedCount * scale (mod MOD).
    long long scale = 1;
    long long inverseScale = 1;
    long long sumStored = 0;

    for (int i = 0; i < N; ++i) {
        long long p;
        int type;
        scanner.readInt(p);
        scanner.readInt(type);

        long long cutoff = p - D;

        // A coach exactly D units to the left is still usable.
        while (head < states.size() && states[head].position < cutoff) {
            sumStored -= states[head].storedCount;
            if (sumStored < 0) sumStored += MOD;
            ++head;
        }

        if (type == 1) {
            // Selecting this coach creates one new state for every old valid subset.
            int storedCount =
                static_cast<int>(total * inverseScale % MOD);

            states.push_back({p, storedCount});
            sumStored += storedCount;
            if (sumStored >= MOD) sumStored -= MOD;

            // Either skip or select this coach.
            total = total * 2 % MOD;
        } else {
            // Only states whose latest selected coach is within D may select it.
            long long eligible = sumStored * scale % MOD;
            total += eligible;
            if (total >= MOD) total -= MOD;

            // Every eligible state's count doubles.
            scale = scale * 2 % MOD;
            inverseScale = inverseScale * INV2 % MOD;
        }
    }

    // The empty subset is valid but must not be counted.
    printf("%lld\n", (total + MOD - 1) % MOD);
    return 0;
}

    int result = a + b;
    if (result >= MOD) result -= MOD;
    return result;
}

    int position;
    int base;
    int campersBefore;
};

    for (int i = 1; i <= N; ++i) {
        powerOfTwo[i] = addMod(powerOfTwo[i - 1], powerOfTwo[i - 1]);
    }

    vector<Coach> coaches;
    coaches.reserve(N);
    size_t head = 0;

    int total = 1;       // Includes the empty selection.
    int active = 0;      // Sum of states whose latest coach is still usable.
    int campersSeen = 0;

    for (int i = 0; i < N; ++i) {
        int position, type;
        input.readInt(position);
        input.readInt(type);

        while (head < coaches.size() &&
               static_cast<long long>(coaches[head].position) < minimumPosition) {
            const Coach& coach = coaches[head];

            int exponent = campersSeen - coach.campersBefore;
            int currentWeight = static_cast<int>(
                1LL * coach.base * powerOfTwo[exponent] % MOD
            );

            active -= currentWeight;
            if (active < 0) active += MOD;

            ++head;

            int base = total;

            coaches.push_back({position, base, campersSeen});

            active = addMod(active, base);
            total = addMod(total, total);
        } else {
            // Active states may either omit or select this camper.
            int oldActive = active;

            total = addMod(total, oldActive);
            active = addMod(oldActive, oldActive);

            ++campersSeen;
        }
    }

    int answer = total - 1; // Exclude the empty selection.
    if (answer < 0) answer += MOD;

    printf("%d\n", answer);
    return 0;
}
