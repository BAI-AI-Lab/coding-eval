题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Bessie is experimenting with a powerful hoof implant that has the ability to create massive shock waves. She has $N$ ($2 \leq N \leq 10^5$) tiles lined up in front of her, which require powers of at least $p_0,p_1,\dots,p_{N-1}$ to break, respectively ($0 \leq p_i \leq 10^{18}$).

Bessie can apply power by punching a specific tile but due to the strange nature of her implant, it will not apply any power to the tile she punches. Instead, if she chooses to punch tile $x$ once, where $x$ is an integer in $[0,N-1]$, it applies $|i-x|$ power to tile $i$ for all integers $i$ in the range $[0,N-1]$. This power is also cumulative, so applying $2$ power twice to a tile will apply a total of $4$ power to the tile.

Please determine the fewest number of punches required to break all the tiles.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $T$ ($1 \leq T \leq 100$) representing the number of test cases.

Line $2t$ contains a single integer $N$, the number of tiles in test case $t$.

Line $2t+1$ contains $N$ space separated numbers $p_0,p_1, \ldots, p_{N-1}$ representing that tile $i$ takes $p_i$ power to be broken.

It is guaranteed that the sum of all $N$ in a single input does not exceed $5\cdot 10^5$.

OUTPUT FORMAT (print output to the terminal / stdout):

$T$ lines, the $i$th line representing the answer to the $i$th test case.

SAMPLE INPUT:

6
5
0 2 4 5 8
5
6 5 4 5 6
5
1 1 1 1 1
5
12 10 8 6 4
7
6 1 2 3 5 8 13
2
1000000000000000000 1000000000000000000

SAMPLE OUTPUT:

2
3
2
4
4
2000000000000000000

For the first test, the only way for Bessie to break all the tiles with two punches is to punch $0$ twice, applying total powers of $[0,2,4,6,8]$ respectively.

For the second test, one way for Bessie to break all the tiles with three punches is to punch $0$, $2$, and $4$ one time each, applying total powers of $[6,5,4,5,6]$ respectively.

For the third test, one way for Bessie to break all the tiles with two punches is to punch $0$ and $1$ one time each, applying total powers of $[1,1,3,5,7]$ respectively.

SCORING:

  • Input 2: All $p_i$ are equal.
  • Inputs 3-6: $N\le 100$
  • Inputs 7-14: No additional constraints.

using namespace std;

using int64 = long long;
using i128 = __int128_t;

struct Rat {
    i128 n;
    int64 d;
};

struct Interval {
    bool ok;
    Rat lo, hi;
};

// Interval of real a for which a punches at 0 and K-a punches at N-1
// (fractional punches allowed) cover all requirements.
Interval getInterval(int64 K, const vector<int64>& p) {
    const int N = (int)p.size();
    const int L = N - 1;

    Interval res{true, {0, 1}, {(i128)K, 1}};

    for (int i = 0; i < N; ++i) {
        i128 base = (i128)K * (L - i);
        int64 d = 2LL * i - L;

        if (d == 0) {
            if (base < (i128)p[i]) {
                res.ok = false;
                return res;
            }
        } else if (d > 0) {
            i128 num = (i128)p[i] - base;
            if (num > 0 && num * res.lo.d > res.lo.n * d) {
                res.lo = {num, d};
            }
        } else {
            int64 e = -d;
            i128 num = base - (i128)p[i];
            if (num < 0) {
                res.ok = false;
                return res;
            }
            if (num * res.hi.d < res.hi.n * e) {
                res.hi = {num, e};
            }
        }
    }

    res.ok = (res.lo.n * res.hi.d <= res.hi.n * res.lo.d);
    return res;
}

// Exact feasibility with K integral punches.
bool integerFeasible(int64 K, const vector<int64>& p) {
    const int N = (int)p.size();
    const int L = N - 1;

    Interval in = getInterval(K, p);
    if (!in.ok) return false;

    i128 ceilLo = in.lo.n / in.lo.d;
    if (in.lo.n % in.lo.d != 0) ++ceilLo;
    i128 floorHi = in.hi.n / in.hi.d;

    // Integral endpoint-only solution.
    if (ceilLo <= floorHi) return true;

    // Otherwise only one possible pair of endpoint base counts remains.
    i128 q = in.lo.n / in.lo.d;
    if (q < 0 || q >= (i128)K) return false;

    int64 A = (int64)q;
    int64 B = K - 1 - A;

    // Check A punches at 0, B punches at N-1, and one additional punch x.
    vector<i128> r(N), suffix(N);
    for (int i = 0; i < N; ++i) {
        i128 base = (i128)A * i + (i128)B * (L - i);
        r[i] = (i128)p[i] - base;
    }

    const i128 INF = i128(1) << 120;
    i128 mn = INF;
    for (int i = N - 1; i >= 0; --i) {
        mn = min(mn, (i128)i - r[i]);
        suffix[i] = mn;
    }

    i128 mx = -INF;
    for (int x = 0; x < N; ++x) {
        mx = max(mx, (i128)x + r[x]);
        if (mx <= (i128)x && (i128)x <= suffix[x]) return true;
    }
    return false;
}

// Ceiling of the fractional (linear programming) optimum.
int64 fractionalCeil(const vector<int64>& p) {
    const int N = (int)p.size();
    const int L = N - 1;

    vector<int> hull;
    hull.reserve(N);

    // Upper concave hull of (i,p_i).
    for (int i = 0; i < N; ++i) {
        while ((int)hull.size() >= 2) {
            int a = hull[(int)hull.size() - 2];
            int b = hull.back();
            i128 lhs = ((i128)p[b] - p[a]) * (i - b);
            i128 rhs = ((i128)p[i] - p[b]) * (b - a);
            if (lhs <= rhs) hull.pop_back();
            else break;
        }
        hull.push_back(i);
    }

    // Evaluate the hull at x=L/2.
    int e = 0;
    while (e + 1 < (int)hull.size() && 2LL * hull[e + 1] < L) ++e;

    int x1 = hull[e], x2 = hull[e + 1];
    int64 dx = x2 - x1;

    i128 den = (i128)L * dx;
    i128 num = 2 * (i128)p[x1] * dx
             + ((i128)p[x2] - p[x1]) * (L - 2LL * x1);

    if (num <= 0) return 0;
    return (int64)((num + den - 1) / den);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;
        vector<int64> p(N);
        for (auto &x : p) cin >> x;

        int64 m = fractionalCeil(p);
        int64 answer = integerFeasible(m, p) ? m : m + 1;
        cout << answer << '\n';
    }

    return 0;
}

using int64 = long long;

struct Fraction {
    i128 num;
    int64 den; // always positive
};

static int cmpFraction(const Fraction &a, const Fraction &b) {
    i128 left = a.num * b.den;
    i128 right = b.num * a.den;
    if (left < right) return -1;
    if (left > right) return 1;
    return 0;
}

static i128 floorDiv(i128 a, int64 b) {
    if (a >= 0) return a / b;
    return -((-a + b - 1) / b);
}

static i128 ceilDiv(i128 a, int64 b) {
    if (a >= 0) return (a + b - 1) / b;
    return -((-a) / b);
}

struct TestSolver {
    int n;
    int64 L;
    vector<int64> p;

    bool getRealInterval(int64 m, Fraction &lower, Fraction &upper) const {
        lower = {0, 1};
        upper = {(i128)L * m, 1};

        for (int i = 0; i < n; ++i) {
            i128 mi = (i128)m * i;
            int64 d = L - 2LL * i;

            if (d == 0) {
                if ((i128)p[i] > mi) return false;
                continue;
            }

            i128 num = (i128)L * ((i128)p[i] - mi);
            Fraction f{num, d};
            if (f.den < 0) {
                f.den = -f.den;
                f.num = -f.num;
            }

            if (d > 0) {
                if (cmpFraction(f, lower) > 0) lower = f;
            } else {
                if (cmpFraction(f, upper) < 0) upper = f;
            }

            if (cmpFraction(lower, upper) > 0) return false;
        }
        return true;
    }

    bool feasible(int64 m) const {
        Fraction lower, upper;
        if (!getRealInterval(m, lower, upper)) return false;

        i128 aLo = ceilDiv(lower.num, lower.den);
        i128 aHi = floorDiv(upper.num, upper.den);
        if (aLo > aHi) return false;

        // If A is a multiple of L, punches only at the two endpoints suffice.
        i128 firstMultiple = ceilDiv(aLo, L) * L;
        if (firstMultiple <= aHi) return true;

        // Otherwise all possible A values have the same quotient b=A/L.
        i128 b = aLo / L;
        int64 rLo = (int64)(aLo % L);
        int64 rHi = (int64)(aHi % L);
        if (rLo > rHi || rLo <= 0 || rHi >= L) return false;

        // b punches at L, m-b-1 punches at 0, and one punch at r.
        // At tile i the power is base_i + |i-r|.
        vector<int> diff(L + 2, 0);

        for (int i = 0; i < n; ++i) {
            i128 base = (i128)(m - 1) * i + b * (L - 2LL * i);
            i128 need = (i128)p[i] - base;
            if (need <= 0) continue;

            // Every available r has |i-r| <= L-1.
            if (need >= L) return false;

            int64 t = (int64)need;
            int64 forbiddenL = max(rLo, (int64)i - t + 1);
            int64 forbiddenR = min(rHi, (int64)i + t - 1);

            if (forbiddenL <= forbiddenR) {
                ++diff[forbiddenL];
                --diff[forbiddenR + 1];
            }
        }

        int covered = 0;
        for (int64 r = rLo; r <= rHi; ++r) {
            covered += diff[r];
            if (covered == 0) return true;
        }
        return false;
    }

    int64 solve() const {
        int64 lo = 0, hi = 4000000000000000000LL;
        while (lo < hi) {
            int64 mid = lo + (hi - lo) / 2;
            if (feasible(mid)) hi = mid;
            else lo = mid + 1;
        }
        return lo;
    }
};

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using i128 = __int128_t;

struct Rat {
    i128 n;
    int d; // always positive
};

struct Bounds {
    bool ok;
    Rat lo, hi;
};

// For a fixed K, decide whether fractional punches at the two endpoints
// can cover all tiles. beta is the amount punched at tile L=N-1, and
// K-beta is punched at tile 0.
static Bounds getBounds(ll K, const vector<ll>& p, int L) {
    i128 k = (i128)K;
    Rat lo{0, 1}, hi{(i128)K, 1};
    bool ok = true;

    for (int i = 0; i <= L; ++i) {
        // K*i + beta*(L-2*i) >= p[i]
        i128 num = (i128)p[i] - k * (i128)i;
        int c = L - 2 * i;

        if (c == 0) {
            if (num > 0) {
                ok = false;
                break;
            }
        } else if (c > 0) {
            // beta >= num/c
            if (num * (i128)lo.d > lo.n * (i128)c) {
                lo = {num, c};
            }
        } else {
            int d = -c;
            // beta <= (K*i-p[i])/d
            i128 upperNum = -num;
            if (upperNum * (i128)hi.d < hi.n * (i128)d) {
                hi = {upperNum, d};
            }
        }
    }

    if (ok && lo.n * (i128)hi.d > hi.n * (i128)lo.d) {
        ok = false;
    }
    return {ok, lo, hi};
}

static i128 floorDiv(i128 n, i128 d) {
    i128 q = n / d, r = n % d;
    if (r < 0) --q;
    return q;
}

static i128 ceilDiv(i128 n, i128 d) {
    i128 q = n / d, r = n % d;
    if (r > 0) ++q;
    return q;
}

// Check whether K integral punches suffice.
static bool integralFeasible(ll K, const vector<ll>& p, int L) {
    Bounds bounds = getBounds(K, p, L);
    if (!bounds.ok) return false;

    // An integral beta in this interval gives an endpoint-only solution.
    i128 smallestInt = ceilDiv(bounds.lo.n, bounds.lo.d);
    i128 largestInt = floorDiv(bounds.hi.n, bounds.hi.d);
    if (smallestInt <= largestInt) return true;

    // Otherwise every feasible real beta lies strictly between b and b+1.
    // After spreading punches apart, an integral solution can be assumed
    // to have b punches at L, K-1-b punches at 0, and one punch at r.
    i128 b = floorDiv(bounds.lo.n, bounds.lo.d);
    if (b < 0 || b > (i128)(K - 1)) return false;

    const int N = L + 1;
    const i128 INF = ((i128)1 << 120);
    i128 k = (i128)K;

    vector<i128> R(N), A(N), B(N);
    for (int i = 0; i < N; ++i) {
        i128 c = (i128)(L - 2 * i);
        i128 base = (k - 1) * (i128)i + b * c;
        R[i] = (i128)p[i] - base;

        // For the remaining punch at r:
        // |i-r| >= R[i].
        A[i] = (i128)i + R[i]; // r >= A[i] when i < r
        B[i] = (i128)i - R[i]; // r <= B[i] when i > r
    }

    vector<i128> prefixMax(N), suffixMin(N);
    prefixMax[0] = -INF;
    for (int r = 1; r < N; ++r) {
        prefixMax[r] = max(prefixMax[r - 1], A[r - 1]);
    }

    suffixMin[N - 1] = INF;
    for (int r = N - 2; r >= 0; --r) {
        suffixMin[r] = min(suffixMin[r + 1], B[r + 1]);
    }

    for (int r = 0; r < N; ++r) {
        if (R[r] <= 0 &&
            prefixMax[r] <= (i128)r &&
            suffixMin[r] >= (i128)r) {
            return true;
        }
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;
        vector<ll> p(N);
        ll pmax = 0;
        for (ll& x : p) {
            cin >> x;
            pmax = max(pmax, x);
        }

        if (pmax == 0) {
            cout << 0 << '\n';
            continue;
        }

        int L = N - 1;

        // Two equal integral endpoint allocations give an always-feasible
        // upper bound: C*L >= pmax at every tile.
        ll C = (pmax + L - 1) / L;
        ll lo = 0, hi = 2 * C;

        // M is the smallest integer K feasible for the LP relaxation.
        while (lo < hi) {
            ll mid = lo + (hi - lo) / 2;
            if (getBounds(mid, p, L).ok) {
                hi = mid;
            } else {
                lo = mid + 1;
            }
        }

        ll M = lo;
        // The integral optimum is always either M or M+1.
        ll answer = integralFeasible(M, p, L) ? M : M + 1;
        cout << answer << '\n';
    }

    return 0;
}

        }

        vector<int> diff(N + 2);

        auto feasible = [&](int64 k) -> bool {
            if (k == 0) return maxP == 0;

            const i128 K = (i128)k * L;
            i128 slo = 0, shi = K;

            // If S = q[0], then q[N-1] = k*L-S. Since q is convex,
            // q[i] is at most the linear interpolation of its endpoints.
            for (int i = 0; i < N; i++) {
                const int64 c = L - 2LL * i;
                const i128 R = (i128)L * (p[i] - (i128)i * k);

                if (c > 0) {
                    if (R > 0) {
                        i128 lower = (R + c - 1) / c;
                        slo = max(slo, lower);
                    }
                } else if (c < 0) {
                    if (R > 0) return false;
                    i128 upper = (-R) / (-c);
                    shi = min(shi, upper);
                } else {
                    if (R > 0) return false;
                }

                if (slo > shi || shi < 0 || slo > K) return false;
            }

            // Punches only at the two endpoints. Such a configuration exists
            // exactly when an integer multiple of L lies in [slo, shi].
            i128 v = (slo + L - 1) / L;
            if (v <= k && v * L <= shi) return true;

            if (L == 1) return false;

            // Otherwise [slo,shi] has length < L, so only O(1) values of v
            // need to be considered. A maximally spread configuration has
            // v punches at L, one punch at x, and all remaining punches at 0.
            i128 vs = slo / L - 1;
            i128 ve = shi / L + 1;
            int64 vStart = vs > 0 ? (int64)vs : 0;
            int64 vEnd = ve < (i128)(k - 1) ? (int64)ve : k - 1;
            vStart = max<int64>(vStart, 0);

            for (int64 cntRight = vStart; cntRight <= vEnd; cntRight++) {
                i128 baseS = (i128)cntRight * L;

                i128 xl = slo - baseS;
                i128 xr = shi - baseS;
                xl = max(xl, (i128)1);
                xr = min(xr, (i128)(L - 1));
                if (xl > xr) continue;

                int xlo = (int)xl;
                int xhi = (int)xr;

                fill(diff.begin(), diff.begin() + L + 2, 0);

                for (int i = 0; i < N; i++) {
                    i128 basePower =
                        (i128)(k - cntRight - 1) * (L - i) +
                        (i128)cntRight * i;
                    i128 d = (i128)p[i] - basePower;

                    if (d <= 0) continue;

                    // We need |i-x| >= d. Thus the following interval of x
                    // values is forbidden.
                    i128 left = (i128)i - d + 1;
                    i128 right = (i128)i + d - 1;
                    left = max(left, (i128)xlo);
                    right = min(right, (i128)xhi);

                    if (left <= right) {
                        diff[(int)left]++;
                        diff[(int)right + 1]--;
                    }
                }

                int active = 0;
                for (int x = xlo; x <= xhi; x++) {
                    active += diff[x];
                    if (active == 0) return true;
                }
            }

            return false;
        };

            if (feasible(mid)) high = mid;
            else low = mid + 1;

    return 0;
}

static int compareFractions(const Fraction& x, const Fraction& y) {

/*
    Represents the line
        y = (a*x + b) / d,  d > 0.

    All hulls used in this solution insert lines in strictly decreasing
    slope order. The query direction is fixed for each hull.
*/
struct Line {
    i128 b;
    int64 a, d;

    Line(int64 a_, i128 b_, int64 d_) : b(b_), a(a_), d(d_) {}
};

class MonotoneHull {
    bool maximum;
    vector<Line> lines;
    size_t head = 0;

    static int compareAt(const Line& x, const Line& y, int64 at) {
        i128 nx = x.b + (i128)x.a * at;
        i128 ny = y.b + (i128)y.a * at;
        i128 lhs = nx * y.d;
        i128 rhs = ny * x.d;
        return (lhs > rhs) - (lhs < rhs);
    }

    // x12 is the intersection of l1,l2; x23 is the intersection of l2,l3.
    bool redundant(const Line& l1, const Line& l2, const Line& l3) const {
        i128 n12 = l2.b * l1.d - l1.b * l2.d;
        i128 n23 = l3.b * l2.d - l2.b * l3.d;

        int64 q12 = l1.a * l2.d - l2.a * l1.d;
        int64 q23 = l2.a * l3.d - l3.a * l2.d;

        // Slopes are decreasing, so q12 and q23 are positive.
        if (maximum) {
            // For a maximum hull, x12 must be greater than x23.
            return n12 * q23 <= n23 * q12;
        } else {
            // For a minimum hull with decreasing slopes, x12 < x23.
            return n12 * q23 >= n23 * q12;
        }
    }

        lines.reserve(capacity);

        return lines.size() == head;

            size_t n = lines.size();

/*
    Check whether K punches can break everything using only tiles 0 and L.

    q punches are placed at L, and K-q punches are placed at 0.
*/
static bool endpointFeasible(const vector<int64>& p, int64 K) {

    i128 lower = 0;
    i128 upper = K;

    for (int i = 0; i <= L; ++i) {
        if (2LL * i < L) {
            long long d = L - 2 * i;
            i128 numerator = (i128)p[i] - (i128)K * i;
            i128 needed = ceilDiv(numerator, d);
            if (needed > lower) lower = needed;

            i128 numerator = (i128)K * i - p[i];

            if (allowed < upper) upper = allowed;

            if ((i128)K * i < p[i]) return false;

    int64 M = (maxP + L - 1) / L;

    // Necessary lower bounds for endpoint-only solutions.
    int64 low = (int64)(((i128)p[0] + p[L] + L - 1) / L);
    for (int i = 0; i <= L; ++i) {
        int64 maxDistance = max(i, L - i);
        int64 needed = (p[i] + maxDistance - 1) / maxDistance;
        low = max(low, needed);

    int64 A = (p[L] + L - 1) / L; // punches at 0
    int64 B = (p[0] + L - 1) / L; // punches at L
    i128 maxDeficit = 0;
    for (int i = 0; i <= L; ++i) {
        i128 base = (i128)A * i + (i128)B * (L - i);
        maxDeficit = max(maxDeficit, (i128)p[i] - base);
    }
    int64 extra = (int64)((maxDeficit + L - 1) / L);
    i128 candidate = (i128)A + B + 2 * extra;
    i128 high128 = min((i128)2 * M, candidate);
    int64 high = (int64)high128;

    while (low < high) {
        int64 mid = low + (high - low) / 2;
        if (endpointFeasible(p, mid)) {
            high = mid;
        } else {
            low = mid + 1;

    // The unrestricted optimum is either E or E-1.
    if (interiorFeasible(p, endpointOptimum - 1)) {
        return endpointOptimum - 1;
    }
    return endpointOptimum;
}

/*
    A line represents the rational function

        (slope * x + b) / den,    den > 0.

    The Li Chao tree maintains either the maximum or the minimum.
*/
class LiChao {
    struct Line {
        i128 b;
        int slope;
        int den;
    };

    int n;
    bool maximize;
    vector<Line> lines;
    vector<int> tree;

    bool better(int id1, int id2, int x) const {
        const Line &a = lines[id1];
        const Line &b = lines[id2];

        i128 lhs = (a.b + (i128)a.slope * x) * b.den;
        i128 rhs = (b.b + (i128)b.slope * x) * a.den;

        return maximize ? lhs > rhs : lhs < rhs;
    }

    void insert(int id, int node, int l, int r) {
        if (tree[node] == -1) {
            tree[node] = id;
            return;
        }

        int mid = (l + r) / 2;
        bool leftBetter = better(id, tree[node], l);
        bool midBetter = better(id, tree[node], mid);

        if (midBetter) swap(id, tree[node]);

        if (l == r) return;

        if (leftBetter != midBetter) {
            insert(id, node * 2, l, mid);
        } else {
            insert(id, node * 2 + 1, mid + 1, r);
        }
    }

public:
    struct Bound {
        i128 num = 0;
        int den = 0;  // den == 0 means empty
    };

    LiChao(int domainSize, bool findMaximum)
        : n(domainSize), maximize(findMaximum), tree(4 * domainSize + 5, -1) {
        lines.reserve(domainSize);
    }

    void clear() {
        lines.clear();
        fill(tree.begin(), tree.end(), -1);
    }

    void add(int slope, i128 b, int den) {
        lines.push_back({b, slope, den});
        insert((int)lines.size() - 1, 1, 0, n - 1);
    }

    Bound query(int x) const {
        int node = 1, l = 0, r = n - 1;
        int best = -1;

        while (true) {
            int id = tree[node];
            if (id == -1) break;

            if (best == -1 || better(id, best, x)) best = id;

            if (l == r) break;

            int mid = (l + r) / 2;
            if (x <= mid) {
                node *= 2;
                r = mid;
            } else {
                node = node * 2 + 1;
                l = mid + 1;
            }
        }

        if (best == -1) return {};

        return {
            lines[best].b + (i128)lines[best].slope * x,
            lines[best].den
        };
    }
};

/*
    Is there a fractional endpoint solution with exactly C total punches?

    u punches at 0, v punches at L=N-1, with u+v=C.
    Equivalently, u=C-v.
*/
static bool fractionalEndpointFeasible(int64 C,
                                       const vector<int64> &p,
                                       int L) {
    i128 lowNum = 0;
    int lowDen = 1;

    i128 highNum = C;
    int highDen = 1;

    for (int i = 0; i <= L; ++i) {
        int d = L - 2 * i;

        if (d > 0) {
            // v >= (p[i] - C*i) / d
            i128 num = (i128)p[i] - (i128)C * i;
            if (num * lowDen > lowNum * d) {
                lowNum = num;
                lowDen = d;
            }
        } else if (d < 0) {
            // v <= (C*i - p[i]) / (2*i-L)
            i128 num = (i128)C * i - p[i];
            int den = -d;
            if (num * highDen < highNum * den) {
                highNum = num;
                highDen = den;
            }
        } else {
            if ((i128)C * i < p[i]) return false;
        }
    }

    return lowNum * highDen <= highNum * lowDen;
}

    q punches are at L and C-q punches are at 0.

/*
    Check whether k punches suffice.

    By the canonicalization argument, it is enough to consider:
      - q ordinary punches at L,
      - one special punch at r,
      - k-1-q ordinary punches at 0,

    where 0 <= q <= k-1 and 0 <= r <= L.

    Their power at tile i is

        (k-1)*i + q*(L-2*i) + abs(i-r).
*/
static bool canonicalFeasible(int64 k,
                              const vector<int64> &p,
                              int L) {
    int N = L + 1;

    if (integerEndpointFeasible(k, p, L)) return true;

    LiChao maxLeft(N, true);
    LiChao minRight(N, false);

    vector<LiChao::Bound> prefixMax(N), prefixMin(N);

    i128 km2 = (i128)k - 2;
    i128 kk = (i128)k;

    // Prefix envelopes: indices i <= r.
    for (int r = 0; r <= L; ++r) {
        int i = r;

        if (2 * i < L) {
            int den = L - 2 * i;
            i128 b = (i128)p[i] - km2 * i;
            maxLeft.add(-1, b, den);
        } else if (2 * i > L) {
            int den = 2 * i - L;
            i128 b = km2 * i - p[i];
            minRight.add(1, b, den);
        }

        prefixMax[r] = maxLeft.query(r);
        prefixMin[r] = minRight.query(r);
    }

    maxLeft.clear();
    minRight.clear();

    // Suffix envelopes: indices i > r.
    for (int r = L; r >= 0; --r) {
        LiChao::Bound suffixMax = maxLeft.query(r);
        LiChao::Bound suffixMin = minRight.query(r);

        // Add tile r to the suffix for the next iteration.
        int i = r;
        if (2 * i < L) {
            int den = L - 2 * i;
            i128 b = (i128)p[i] - kk * i;
            maxLeft.add(1, b, den);
        } else if (2 * i > L) {
            int den = 2 * i - L;
            i128 b = kk * i - p[i];
            minRight.add(-1, b, den);
        }

        LiChao::Bound lowerBound = prefixMax[r];
        if (suffixMax.den != 0 &&
            (lowerBound.den == 0 ||
             fractionGreater(suffixMax, lowerBound))) {
            lowerBound = suffixMax;
        }

        LiChao::Bound upperBound = prefixMin[r];
        if (suffixMin.den != 0 &&
            (upperBound.den == 0 ||
             fractionLess(suffixMin, upperBound))) {
            upperBound = suffixMin;
        }

        if (lowerBound.den == 0 || upperBound.den == 0) continue;

        // The middle tile imposes a condition independent of q.
        if (L % 2 == 0) {
            int mid = L / 2;
            i128 power = (i128)(k - 1) * mid + abs(mid - r);
            if (power < p[mid]) continue;
        }

        i128 low = ceilDiv(lowerBound.num, lowerBound.den);
        i128 high = floorDiv(upperBound.num, upperBound.den);

        if (low < 0) low = 0;
        if (high > (i128)k - 1) high = (i128)k - 1;

        if (low <= high) return true;
    }

    return false;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using i128 = __int128_t;

/*
    A line represents
        f(x) = (a*x + b) / den,  den > 0.

    Slopes are always inserted in increasing order.
*/
struct Line {
    i128 b;
    int a;
    int den;
};

struct Fraction {
    i128 num;
    int den;
};

static i128 ceil_div(i128 a, i128 b) {
    if (a >= 0) return (a + b - 1) / b;
    return a / b;
}

static i128 floor_div(i128 a, i128 b) {
    if (a >= 0) return a / b;
    return -((-a + b - 1) / b);
}

/*
    Monotone convex hull.

    isMax = true:  maximum envelope
    isMax = false: minimum envelope

    pointerMode = true:
        Queries move toward lines with larger slopes.

    pointerMode = false:
        Queries move toward lines with smaller slopes, so dominated
        lines can be permanently removed from the back.
*/
class MonotoneHull {
    bool isMax;
    bool pointerMode;
    vector<Line> hull;
    size_t ptr = 0;

    static bool value_ge(const Line& x, const Line& y, int pos) {
        i128 nx = (i128)x.a * pos + x.b;
        i128 ny = (i128)y.a * pos + y.b;
        return nx * y.den >= ny * x.den;
    }

    static bool value_le(const Line& x, const Line& y, int pos) {
        i128 nx = (i128)x.a * pos + x.b;
        i128 ny = (i128)y.a * pos + y.b;
        return nx * y.den <= ny * x.den;
    }

    // Intersection of consecutive lines, with positive denominator.
    static void intersection(const Line& x, const Line& y,
                             i128& numerator, i128& denominator) {
        denominator = (i128)y.a * x.den - (i128)x.a * y.den;
        numerator = x.b * y.den - y.b * x.den;
    }

    static bool bad_max(const Line& x, const Line& y, const Line& z) {
        i128 n12, d12, n23, d23;
        intersection(x, y, n12, d12);
        intersection(y, z, n23, d23);

        // The middle line is unnecessary if intersection(x,y) >= intersection(y,z).
        return n12 * d23 >= n23 * d12;
    }

    static bool bad_min(const Line& x, const Line& y, const Line& z) {
        i128 n12, d12, n23, d23;
        intersection(x, y, n12, d12);
        intersection(y, z, n23, d23);

        // The middle line is unnecessary if intersection(y,z) >= intersection(x,y).
        return n23 * d12 >= n12 * d23;
    }

public:
    MonotoneHull(bool maximum, bool use_pointer)
        : isMax(maximum), pointerMode(use_pointer) {}

    void reserve(int n) {
        hull.reserve(n);
    }

    void clear() {
        hull.clear();
        ptr = 0;
    }

    bool empty() const {
        return hull.empty();
    }

    void add(Line line) {
        while (hull.size() >= 2) {
            bool bad = isMax
                ? bad_max(hull[hull.size() - 2], hull.back(), line)
                : bad_min(hull[hull.size() - 2], hull.back(), line);

            if (!bad) break;
            hull.pop_back();
        }

        hull.push_back(line);

        if (pointerMode && ptr >= hull.size()) {
            ptr = hull.size() - 1;
        }
    }

    const Line& get(int x) {
        if (pointerMode) {
            while (ptr + 1 < hull.size()) {
                bool next_better = isMax
                    ? value_ge(hull[ptr + 1], hull[ptr], x)
                    : value_le(hull[ptr + 1], hull[ptr], x);

                if (!next_better) break;
                ++ptr;
            }
            return hull[ptr];
        }

        while (hull.size() >= 2) {
            const Line& previous = hull[hull.size() - 2];
            const Line& last = hull.back();

            bool remove_last = isMax
                ? value_ge(previous, last, x)
                : value_le(previous, last, x);

            if (!remove_last) break;
            hull.pop_back();
        }

        return hull.back();
    }
};
