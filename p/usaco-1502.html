    int64 sum = 0;

public:

    vector<int64> values;

        tree.reserve(static_cast<size_t>(n) * 20 + 1);
        tree.push_back(Node{});

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct Item {
    ll w, a, b;
    int cb;
};

struct Node {
    int l = 0, r = 0, cnt = 0;
    ll sum = 0;
};

struct PersistentSegTree {
    vector<Node> tr;

    explicit PersistentSegTree(int n) {
        tr.reserve(n * 20 + 5);
        tr.push_back(Node{});
    }

    int update(int prev, int l, int r, int pos, ll val) {
        int u = (int)tr.size();
        tr.push_back(tr[prev]);
        tr[u].cnt++;
        tr[u].sum += val;

        if (l != r) {
            int mid = (l + r) >> 1;
            if (pos <= mid) {
                tr[u].l = update(tr[prev].l, l, mid, pos, val);
            } else {
                tr[u].r = update(tr[prev].r, mid + 1, r, pos, val);
            }
        }
        return u;
    }

    ll sumSmallest(int u, int k, const vector<ll>& coords) const {
        ll result = 0;
        int l = 0, r = (int)coords.size() - 1;

        while (l < r) {
            if (k == tr[u].cnt) return result + tr[u].sum;

            int mid = (l + r) >> 1;
            int left = tr[u].l;
            int leftCnt = tr[left].cnt;

            if (k <= leftCnt) {
                u = left;
                r = mid;
            } else {
                k -= leftCnt;
                result += tr[left].sum;
                u = tr[u].r;
                l = mid + 1;
            }
        }

        return result + 1LL * k * coords[l];
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    vector<Item> items(N);
    vector<ll> coords;
    coords.reserve(N);

    for (int i = 0; i < N; i++) {
        ll a, b;
        cin >> a >> b;
        items[i] = {a + b, a, b, 0};
        coords.push_back(b);
    }

    vector<int> queries(Q);
    for (int& k : queries) cin >> k;

    sort(coords.begin(), coords.end());
    coords.erase(unique(coords.begin(), coords.end()), coords.end());

    for (Item& it : items) {
        it.cb = (int)(lower_bound(coords.begin(), coords.end(), it.b) - coords.begin());
    }

    sort(items.begin(), items.end(), [](const Item& x, const Item& y) {
        return x.w > y.w;
    });

    int M = (int)coords.size();
    PersistentSegTree pst(N);

    vector<int> roots(N + 1, 0);
    vector<ll> prefixA(N + 1, 0);

    for (int i = 1; i <= N; i++) {
        const Item& it = items[i - 1];
        roots[i] = pst.update(roots[i - 1], 0, M - 1, it.cb, it.b);
        prefixA[i] = prefixA[i - 1] + it.a;
    }

    const ll totalA = prefixA[N];
    vector<ll> answer(N + 1, 0);
    answer[0] = totalA;

    auto valueAt = [&](int r, int k) -> ll {
        ll safe = totalA - prefixA[r];
        ll flipped = pst.sumSmallest(roots[r], k, coords);
        return safe - flipped;
    };

    vector<int> positiveK;
    positiveK.reserve(Q);
    for (int k : queries) {
        if (k > 0) positiveK.push_back(k);
    }
    sort(positiveK.begin(), positiveK.end());

    function<void(int, int, int, int)> solve =
        [&](int ql, int qr, int rl, int rr) {
            if (ql > qr) return;

            int qm = (ql + qr) >> 1;
            int k = positiveK[qm];
            int lo = max(rl, k);

            int bestR = lo;
            ll bestValue = valueAt(lo, k);

            for (int r = lo + 1; r <= rr; r++) {
                ll cur = valueAt(r, k);
                if (cur > bestValue) {
                    bestValue = cur;
                    bestR = r;
                }
            }

            answer[k] = bestValue;
            solve(ql, qm - 1, rl, bestR);
            solve(qm + 1, qr, bestR, rr);
        };

    if (!positiveK.empty()) {
        solve(0, (int)positiveK.size() - 1, 1, N);
    }

    for (int k : queries) {
        cout << answer[k] << '\n';
    }

    return 0;
}

        Node copy = tree[previous];
        tree.push_back(copy);

        ++tree[current].count;
        tree[current].sum += value;

            int mid = (lo + hi) / 2;
            if (position <= mid) {
                int child = insertImpl(tree[previous].left, lo, mid,
                                       position, value);
                tree[current].left = child;
            } else {
                int child = insertImpl(tree[previous].right, mid + 1, hi,
                                       position, value);
                tree[current].right = child;

        return current;
    }

        tree.reserve(1 + (size_t)n * (height + 1));
        tree.push_back(Node{});

    vector<int> root(N + 1, 0);
    for (int p = N - 1; p >= 0; --p) {
        root[p] = pst.insert(root[p + 1], rank[p], items[p].b);
    }

    const int64 NEG_INF = numeric_limits<int64>::min() / 4;

    /*
       Matrix row = k, column c represents cut p = N - c.
       Columns are therefore in reverse order of cuts.

       M(k, p) = prefix[p] - sum of the k smallest b-values in suffix [p, N).
    */
    auto getValue = [&](int row, int col) -> int64 {
        int p = N - col;
        if (p > N - row) return NEG_INF;
        return prefix[p] - pst.sumSmallest(root[p], row);
    };

    vector<int64> answer(N + 1);
    vector<int> optimumColumn(N + 1, -1);

    // SMAWK for row maxima of a totally monotone matrix.
    auto smawk = [&](auto&& self, const vector<int>& rows,
                     const vector<int>& columns) -> void {
        if (rows.empty()) return;

        vector<int> reduced;
        reduced.reserve(min(rows.size(), columns.size()));

        // Column reduction.
        for (int col : columns) {
            while (!reduced.empty()) {
                int row = rows[(int)reduced.size() - 1];
                if (getValue(row, col) >= getValue(row, reduced.back())) {
                    reduced.pop_back();
                } else {
                    break;
                }
            }
            if (reduced.size() < rows.size()) {
                reduced.push_back(col);
            }
        }

        vector<int> oddRows;
        oddRows.reserve(rows.size() / 2);
        for (int i = 1; i < (int)rows.size(); i += 2) {
            oddRows.push_back(rows[i]);
        }

        self(self, oddRows, reduced);

        // Interpolate the even-indexed rows.
        int start = 0;
        for (int i = 0; i < (int)rows.size(); i += 2) {
            int stop;
            if (i + 1 < (int)rows.size()) {
                int target = optimumColumn[rows[i + 1]];
                stop = lower_bound(reduced.begin(), reduced.end(), target)
                       - reduced.begin();
            } else {
                stop = (int)reduced.size() - 1;
            }

            int row = rows[i];
            int bestColumn = reduced[start];
            int64 bestValue = getValue(row, bestColumn);

            for (int j = start + 1; j <= stop; ++j) {
                int64 candidate = getValue(row, reduced[j]);
                if (candidate >= bestValue) {
                    bestValue = candidate;
                    bestColumn = reduced[j];
                }
            }

            optimumColumn[row] = bestColumn;
            answer[row] = bestValue;
            start = stop;
        }
    };

    vector<int> rows(N + 1), columns(N + 1);
    iota(rows.begin(), rows.end(), 0);
    iota(columns.begin(), columns.end(), 0);

    smawk(smawk, rows, columns);

    for (int k : queries) {
        cout << answer[k] << '\n';
    }

    return 0;
}

/*
 * Wavelet matrix over compressed b-values.
 * sumSmallest(p, k) returns the sum of the k smallest b-values
 * among the first p elements.
 */
class WaveletMatrix {
private:
    int levels;
    vector<int> values;
    vector<vector<int>> prefixZeroCount;
    vector<vector<int64>> prefixZeroSum;
    vector<int> zeroCount;
    vector<int64> prefixTotal;

public:
    WaveletMatrix(const vector<int>& coordinates, const vector<int>& data)
        : values(coordinates) {
        int n = (int)data.size();
        int m = (int)values.size();

        levels = 0;
        while ((1LL << levels) < m) ++levels;

        vector<int> current(n), next(n);
        for (int i = 0; i < n; ++i) {
            current[i] = int(lower_bound(values.begin(), values.end(), data[i])
                             - values.begin());
        }

        prefixTotal.assign(n + 1, 0);
        for (int i = 0; i < n; ++i) {
            prefixTotal[i + 1] = prefixTotal[i] + data[i];
        }

        prefixZeroCount.resize(levels);
        prefixZeroSum.resize(levels);
        zeroCount.resize(levels);

        for (int level = 0; level < levels; ++level) {
            int bit = levels - 1 - level;

            prefixZeroCount[level].assign(n + 1, 0);
            prefixZeroSum[level].assign(n + 1, 0);

            for (int i = 0; i < n; ++i) {
                bool isZero = ((current[i] >> bit) & 1) == 0;

                prefixZeroCount[level][i + 1] =
                    prefixZeroCount[level][i] + (isZero ? 1 : 0);

                prefixZeroSum[level][i + 1] =
                    prefixZeroSum[level][i]
                    + (isZero ? (int64)values[current[i]] : 0LL);
            }

            int totalZero = prefixZeroCount[level][n];
            zeroCount[level] = totalZero;

            int zeroPos = 0;
            int onePos = totalZero;

            for (int i = 0; i < n; ++i) {
                if (((current[i] >> bit) & 1) == 0) {
                    next[zeroPos++] = current[i];
                } else {
                    next[onePos++] = current[i];
                }
            }

            current.swap(next);
        }
    }

    int64 sumSmallest(int prefixLength, int k) const {
        if (k == 0) return 0;
        if (k == prefixLength) return prefixTotal[prefixLength];

        int left = 0;
        int right = prefixLength;
        int rank = 0;
        int64 result = 0;

        for (int level = 0; level < levels; ++level) {
            const auto& counts = prefixZeroCount[level];

            int zerosLeft = counts[left];
            int zerosRight = counts[right];
            int count = zerosRight - zerosLeft;

            if (k <= count) {
                // All selected values are in the zero branch.
                left = zerosLeft;
                right = zerosRight;
            } else {
                // Take every value in the zero branch, then continue in the
                // one branch for the remaining values.
                result += prefixZeroSum[level][right]
                        - prefixZeroSum[level][left];
                k -= count;

                rank |= (1 << (levels - 1 - level));

                int totalZero = zeroCount[level];
                left = totalZero + left - zerosLeft;
                right = totalZero + right - zerosRight;
            }
        }

        return result + (int64)k * values[rank];
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    vector<Question> questions(N);
    for (int i = 0; i < N; ++i) {
        int64 a, b;
        cin >> a >> b;
        questions[i] = {a, b, a + b};
    }

    vector<int> queries(Q);
    for (int& k : queries) cin >> k;

    // Elsie sabotages questions with the largest a+b values.
    sort(questions.begin(), questions.end(),
         [](const Question& x, const Question& y) {
             if (x.c != y.c) return x.c > y.c;
             if (x.b != y.b) return x.b < y.b;
             return x.a > y.a;
         });

    vector<int> bValues(N);
    for (int i = 0; i < N; ++i) {
        bValues[i] = (int)questions[i].b;
    }

    vector<int> coordinates = bValues;
    sort(coordinates.begin(), coordinates.end());
    coordinates.erase(unique(coordinates.begin(), coordinates.end()),
                      coordinates.end());

    WaveletMatrix wavelet(coordinates, bValues);

    // suffixA[p] = sum of a_i for positions i >= p.
    vector<int64> suffixA(N + 2, 0);
    for (int i = N; i >= 1; --i) {
        suffixA[i] = suffixA[i + 1] + questions[i - 1].a;
    }

    vector<int64> answer(N + 1);
    answer[0] = suffixA[1];

    /*
     * For each k, find the best prefix length p.
     * The optimal prefix lengths are nondecreasing in k, so divide and
     * conquer over k while restricting the possible prefix lengths.
     */
    auto solve = [&](auto&& self, int kLeft, int kRight,
                     int pLeft, int pRight) -> void {
        if (kLeft > kRight) return;

        int kMid = (kLeft + kRight) / 2;
        int firstPrefix = max(pLeft, kMid);

        int64 bestValue = LLONG_MIN;
        int bestPrefix = firstPrefix;

        for (int p = firstPrefix; p <= pRight; ++p) {
            int64 value =
                suffixA[p + 1] - wavelet.sumSmallest(p, kMid);

            // Scanning in increasing order keeps the smallest maximizing p.
            if (value > bestValue) {
                bestValue = value;
                bestPrefix = p;
            }
        }

        answer[kMid] = bestValue;

        if (kLeft < kMid) {
            self(self, kLeft, kMid - 1, pLeft, bestPrefix);
        }
        if (kMid < kRight) {
            self(self, kMid + 1, kRight, bestPrefix, pRight);
        }
    };

    solve(solve, 1, N, 1, N);

    for (int k : queries) {
        cout << answer[k] << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

struct Item {
    int64 a, b, c;
};

class PersistentOrderStats {
    struct Node {
        int left = 0;
        int right = 0;
        int count = 0;
        int64 sum = 0;
    };

    vector<Node> nodes;
    vector<int> values;
    int m;

    int cloneNode(int from) {
        Node copy = nodes[from];
        nodes.push_back(copy);
        return (int)nodes.size() - 1;
    }

public:
    PersistentOrderStats(const vector<int>& compressedValues, int updates)
        : values(compressedValues), m((int)compressedValues.size()) {
        int height = 0;
        while ((1LL << height) < m) ++height;

        nodes.reserve(1 + (size_t)updates * (height + 1));
        nodes.push_back(Node{}); // Empty node.
    }

    int insert(int oldRoot, int position, int value) {
        int newRoot = cloneNode(oldRoot);
        int current = newRoot;
        int lo = 0, hi = m - 1;

        while (true) {
            ++nodes[current].count;
            nodes[current].sum += value;

            if (lo == hi) break;

            int mid = (lo + hi) / 2;
            if (position <= mid) {
                int oldChild = nodes[current].left;
                int newChild = cloneNode(oldChild);
                nodes[current].left = newChild;
                current = newChild;
                hi = mid;
            } else {
                int oldChild = nodes[current].right;
                int newChild = cloneNode(oldChild);
                nodes[current].right = newChild;
                current = newChild;
                lo = mid + 1;
            }
        }

        return newRoot;
    }

    // Sum of the k smallest values in this version.
    int64 sumSmallest(int root, int k) const {
        if (k == 0) return 0;

        int current = root;
        int lo = 0, hi = m - 1;
        int64 result = 0;

        while (true) {
            if (k == nodes[current].count) {
                return result + nodes[current].sum;
            }

            if (lo == hi) {
                return result + (int64)k * values[lo];
            }

            int left = nodes[current].left;
            int leftCount = nodes[left].count;
            int mid = (lo + hi) / 2;

            if (k <= leftCount) {
                current = left;
                hi = mid;
            } else {
                result += nodes[left].sum;
                k -= leftCount;
                current = nodes[current].right;
                lo = mid + 1;
            }
        }
    }
};
