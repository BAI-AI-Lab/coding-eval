import sys
import heapq


def prefix_scores(a):
    """res[2*t] = (sum of t largest) - (sum of t smallest) in each prefix."""
    n = len(a)
    res = [0] * (n + 1)

    low = []   # max-heap (stored negated): smaller half
    high = []  # min-heap: larger half
    sum_low = 0
    sum_high = 0

    for i, x in enumerate(a, 1):
        if not low or x <= -low[0]:
            heapq.heappush(low, -x)
            sum_low += x
        else:
            heapq.heappush(high, x)
            sum_high += x

        if len(low) > len(high) + 1:
            y = -heapq.heappop(low)
            sum_low -= y
            heapq.heappush(high, y)
            sum_high += y
        elif len(low) < len(high):
            y = heapq.heappop(high)
            sum_high -= y
            heapq.heappush(low, -y)
            sum_low += y

        if i % 2 == 0:
            res[i] = sum_high - sum_low

    return res


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n = data[0]
    a = data[1:]

    if n % 2 == 0:
        b = sorted(a)
        m = n // 2
        answer = sum(b[m:]) - sum(b[:m])
        print(answer)
        return

    # If the final remaining element is a[left] (0-based), left is even.
    # The elements before and after it must be removed independently.
    pref = prefix_scores(a)
    suff = prefix_scores(a[::-1])

    answer = 0
    for left in range(0, n, 2):
        right_len = n - 1 - left
        answer = max(answer, pref[left] + suff[right_len])

    print(answer)


if __name__ == "__main__":
    main()

import sys


class Fenwick:
    __slots__ = ("n", "bit", "bit_sum", "values")

    def __init__(self, values):
        self.n = len(values)
        self.bit = [0] * (self.n + 1)
        self.bit_sum = [0] * (self.n + 1)
        self.values = values

    def add(self, i, value):
        n = self.n
        bit = self.bit
        bit_sum = self.bit_sum
        while i <= n:
            bit[i] += 1
            bit_sum[i] += value
            i += i & -i

    def sum_smallest(self, k):
        if k <= 0:
            return 0

        n = self.n
        bit = self.bit
        bit_sum = self.bit_sum

        idx = 0
        cnt = 0
        s = 0
        step = 1 << (n.bit_length() - 1)

        while step:
            nxt = idx + step
            if nxt <= n and cnt + bit[nxt] < k:
                idx = nxt
                cnt += bit[nxt]
                s += bit_sum[nxt]
            step >>= 1

        return s + (k - cnt) * self.values[idx]


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    a = data[1:1 + n]

    values = sorted(set(a))
    comp = {v: i + 1 for i, v in enumerate(values)}

    pre = [0] * (n + 1)
    fw = Fenwick(values)
    total = 0

    for i, v in enumerate(a, 1):
        fw.add(comp[v], v)
        total += v

        if i % 2 == 0:
            k = i // 2
            small_sum = fw.sum_smallest(k)
            pre[i] = total - 2 * small_sum

    if n % 2 == 0:
        print(pre[n])
        return

    suf = [0] * (n + 1)
    fw = Fenwick(values)
    total = 0

    for i in range(n - 1, -1, -1):
        v = a[i]
        fw.add(comp[v], v)
        total += v

        length = n - i
        if length % 2 == 0:
            k = length // 2
            small_sum = fw.sum_smallest(k)
            suf[i] = total - 2 * small_sum

    ans = 0

    for i in range(0, n, 2):
        ans = max(ans, pre[i] + suf[i + 1])

    print(ans)


if __name__ == "__main__":
    solve()

import sys
import heapq

def prefix_even(arr):
    n = len(arr)
    res = [0] * (n + 1)

    low = []   # max-heap (stored as negatives), smaller half
    high = []  # min-heap, larger half
    sum_low = 0
    total = 0

    heappush = heapq.heappush
    heappop = heapq.heappop

    for i, x in enumerate(arr, 1):
        total += x

        if not low or x <= -low[0]:
            heappush(low, -x)
            sum_low += x
        else:
            heappush(high, x)

        desired_low = (i + 1) // 2

        while len(low) > desired_low:
            v = -heappop(low)
            sum_low -= v
            heappush(high, v)

        while len(low) < desired_low:
            v = heappop(high)
            heappush(low, -v)
            sum_low += v

        if i % 2 == 0:
            res[i] = total - 2 * sum_low

    return res


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    A = data[1:1 + N]

    if N % 2 == 0:
        pref = prefix_even(A)
        ans = pref[N]
    else:
        pref = prefix_even(A)
        rev = prefix_even(A[::-1])

        ans = 0
        # The final remaining element must have an odd 1-indexed position.
        for i in range(1, N + 1, 2):
            val = pref[i - 1] + rev[N - i]
            if val > ans:
                ans = val

    print(ans)


if __name__ == "__main__":
    solve()
