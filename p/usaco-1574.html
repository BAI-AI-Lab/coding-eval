题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Farmer Nhoj has trapped Bessie on a rooted tree with $N$ ($2 \le N \le 2 \cdot 10^5$) nodes, where node $1$ is the root. Scared and alone, Bessie makes the following move each second:

  • If Bessie's current node has no children, then she will move to a random ancestor of the current node (excluding the node itself).
  • Otherwise, Bessie will move to a random child of the current node.

Initially, Bessie is at node $x$, and her only way out is the exit located at node $y$ ($1\le x,y\le N$). For $Q$ ($1 \le Q \le 2 \cdot 10^5$) independent queries of $x$ and $y$, compute the expected number of seconds it would take Bessie to reach node $y$ for the first time if she started at node $x$, modulo $10^9+7$.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $N$ and $Q$.

The next line contains $N-1$ integers $p_2, \ldots p_N$ describing the tree ($1\le p_i<i$). For each $2 \le i \le N$, there is an edge between nodes $i$ and $p_i$.

Each of the next $Q$ lines contains integers $x$ and $y$ representing the nodes for that query.

OUTPUT FORMAT (print output to the terminal / stdout):

For each query, output the expected number of seconds for Bessie to reach node $y$ for the first time starting at node $x$, modulo $10^9+7$.

static const long long MOD = 1000000007LL;

long long modpow(long long a, long long e) {
    long long r = 1;
    while (e) {
        if (e & 1) r = r * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return r;
}

inline long long norm(long long x) {
    x %= MOD;
    if (x < 0) x += MOD;
    return x;
}

            r = r * invU[y] % MOD;
        }

        long long ans = norm(
            F[x] - F[y] + norm(1 - r) * targetC[y]
        );

        cout << ans << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

static const int MOD = 1000000007;
using int64 = long long;

int modpow(int a, int e = MOD - 2) {
    int64 r = 1, x = a;
    while (e > 0) {
        if (e & 1) r = r * x % MOD;
        x = x * x % MOD;
        e >>= 1;
    }
    return (int)r;
}

// Transformation: z -> a*z + b*k + c (mod MOD).
struct Aff {
    int a = 1, b = 0, c = 0;
};

// Returns f(g(z)).
Aff compose(const Aff &f, const Aff &g) {
    Aff h;
    h.a = (int64)f.a * g.a % MOD;
    h.b = ((int64)f.a * g.b + f.b) % MOD;
    h.c = ((int64)f.a * g.c + f.c) % MOD;
    return h;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    vector<int> parent(N + 1), depth(N + 1);
    vector<vector<int>> children(N + 1);

    for (int i = 2; i <= N; ++i) {
        cin >> parent[i];
        children[parent[i]].push_back(i);
        depth[i] = depth[parent[i]] + 1;
    }

    vector<int> inv(N + 1);
    inv[1] = 1;
    for (int i = 2; i <= N; ++i) {
        inv[i] = (int64)(MOD - MOD / i) * inv[MOD % i] % MOD;
    }

    // For every non-root v:
    // H(v,target) = A[v] + B[v] * (sum of H at strict ancestors of v),
    // provided the target is outside subtree(v).
    vector<int> A(N + 1), B(N + 1);
    for (int v = N; v >= 2; --v) {
        if (children[v].empty()) {
            A[v] = 1;
            B[v] = inv[depth[v]];
            continue;
        }

        int64 sumA = 0, sumB = 0;
        for (int c : children[v]) {
            sumA += A[c];
            sumB += B[c];
        }

        int k = (int)children[v].size();
        int avgA = sumA % MOD * inv[k] % MOD;
        int avgB = sumB % MOD * inv[k] % MOD;
        int den = 1 - avgB;
        if (den < 0) den += MOD;
        int invDen = modpow(den);

        A[v] = (int64)(1 + avgA) * invDen % MOD;
        B[v] = (int64)avgB * invDen % MOD;
    }

    int LOG = 1;
    while ((1 << LOG) <= N) ++LOG;

    vector<vector<int>> up(LOG, vector<int>(N + 1));
    for (int v = 2; v <= N; ++v) up[0][v] = parent[v];
    for (int k = 1; k < LOG; ++k) {
        for (int v = 1; v <= N; ++v) {
            up[k][v] = up[k - 1][up[k - 1][v]];
        }
    }

    // tr[k][v] describes the 2^k transformations on the path ending at v,
    // composed from top to bottom.
    vector<vector<Aff>> tr(LOG, vector<Aff>(N + 1));
    for (int v = 2; v <= N; ++v) {
        tr[0][v] = {(B[v] + 1) % MOD, B[v], A[v]};
    }

    for (int k = 1; k < LOG; ++k) {
        int len = 1 << k;
        for (int v = 2; v <= N; ++v) {
            if (depth[v] < len) continue;
            int mid = up[k - 1][v];
            // Bottom half is applied after the top half.
            tr[k][v] = compose(tr[k - 1][v], tr[k - 1][mid]);
        }
    }

    // P[v]: expected time to reach v from the root.
    // C[v]: sum of expected hitting times to v from all strict ancestors of v.
    vector<int> P(N + 1), C(N + 1), ancestorSumP(N + 1), hitParent(N + 1);

    for (int v = 1; v <= N; ++v) {
        if (children[v].empty()) continue;

        int64 sumR = 0;
        for (int c : children[v]) {
            hitParent[c] = (A[c] + (int64)B[c] * C[v]) % MOD;
            sumR += hitParent[c];
        }
        sumR %= MOD;

        int k = (int)children[v].size();
        for (int c : children[v]) {
            int edgeTime = (k + sumR - hitParent[c]) % MOD;
            if (edgeTime < 0) edgeTime += MOD;

            P[c] = (P[v] + edgeTime) % MOD;
            ancestorSumP[c] = (ancestorSumP[v] + P[v]) % MOD;

            C[c] = ((int64)depth[c] * P[c] - ancestorSumP[c]) % MOD;
            if (C[c] < 0) C[c] += MOD;
        }
    }

    auto lca = [&](int a, int b) {
        if (depth[a] < depth[b]) swap(a, b);
        int diff = depth[a] - depth[b];
        for (int k = LOG - 1; k >= 0; --k) {
            if ((diff >> k) & 1) a = up[k][a];
        }
        if (a == b) return a;

        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }
        return up[0][a];
    };

    // Sum of hitting values strictly between ancestor a and descendant x,
    // including parent(x), expressed using the boundary value C[a].
    auto intermediateSum = [&](int x, int a) {
        int len = depth[x] - depth[a] - 1;
        if (len <= 0) return 0;

        int cur = parent[x];
        Aff result; // identity

        for (int k = 0; k < LOG; ++k) {
            if ((len >> k) & 1) {
                // The new block is above all previously processed blocks.
                result = compose(result, tr[k][cur]);
                cur = up[k][cur];
            }
        }

        return (int)(((int64)result.b * C[a] + result.c) % MOD);
    };

    // Expected time to reach ancestor a from descendant x.
    auto hitAncestor = [&](int x, int a) {
        int z = intermediateSum(x, a);
        int total = C[a] + z;
        if (total >= MOD) total -= MOD;
        return (int)((A[x] + (int64)B[x] * total) % MOD);
    };

    while (Q--) {
        int x, y;
        cin >> x >> y;

        int w = lca(x, y);
        int answer;

        if (w == y) {
            answer = (x == y) ? 0 : hitAncestor(x, y);
        } else if (w == x) {
            answer = P[y] - P[x];
            if (answer < 0) answer += MOD;
        } else {
            answer = ((int64)hitAncestor(x, w) + P[y] - P[w]) % MOD;
            if (answer < 0) answer += MOD;
        }

        cout << answer << '\n';
    }

    return 0;
}

    par[1] = 1;

        cin >> par[v];
        depth[v] = depth[par[v]] + 1;
        ++deg[par[v]];

    // P_v is the sum of F values on the root-to-v path.

        return normalize(1LL * pathA[v] * root + pathB[v]);

                int xValue = normalize(
                    1LL * messageA[x] * parentValue + messageB[x]
                );
                answer = normalize(xValue - parentValue);
            }
        }

// Affine 2 x 3 matrix:
// [ a b c ]
// [ d e f ]
struct Mat {
    int a, b, c, d, e, f;
};

static const Mat ID{1, 0, 0, 0, 1, 0};

inline Mat compose(const Mat &A, const Mat &B) {
    Mat C;
    C.a = (1ULL * A.a * B.a + 1ULL * A.b * B.d) % MOD;
    C.b = (1ULL * A.a * B.b + 1ULL * A.b * B.e) % MOD;
    C.c = (1ULL * A.a * B.c + 1ULL * A.b * B.f + A.c) % MOD;
    C.d = (1ULL * A.d * B.a + 1ULL * A.e * B.d) % MOD;
    C.e = (1ULL * A.d * B.b + 1ULL * A.e * B.e) % MOD;
    C.f = (1ULL * A.d * B.c + 1ULL * A.e * B.f + A.f) % MOD;
    return C;
}

// A state whose two components are affine functions of one variable h:
// component 0 = c0 * h + d0
// component 1 = c1 * h + d1

};

inline void applyMatrix(const Mat &M, AffState &S) {
    int nc0 = (1ULL * M.a * S.c0 + 1ULL * M.b * S.c1) % MOD;
    int nd0 = (1ULL * M.a * S.d0 + 1ULL * M.b * S.d1 + M.c) % MOD;
    int nc1 = (1ULL * M.d * S.c0 + 1ULL * M.e * S.c1) % MOD;
    int nd1 = (1ULL * M.d * S.d0 + 1ULL * M.e * S.d1 + M.f) % MOD;
    S = {nc0, nd0, nc1, nd1};
}
