#include <bits/stdc++.h>
using namespace std;

int main() {
    unsigned long long x;
    cin >> x;

    if (x == 1) {
        cout << 1 << "\n1\n";
        return 0;
    }

    int L = 0;
    for (unsigned long long v = x; v; v >>= 1) ++L;

    int n = 2 * L - 1;
    vector<vector<int>> g(n, vector<int>(n, 0));

    auto open = [&](int r, int c) {
        g[r - 1][c - 1] = 1;
    };

    for (int c = 1; c <= n; ++c) open(1, c);

    open(2, 1);
    open(2, 2);
    open(3, 2);
    open(3, 3);

    if ((x >> (L - 2)) & 1ULL) open(3, 1);

    for (int t = 1; t <= L - 2; ++t) {
        open(2 * t + 2, 2 * t + 1);
        open(2 * t + 1, 2 * t + 2);
        open(2 * t + 2, 2 * t + 2);
        open(2 * t + 3, 2 * t + 2);
        open(2 * t + 3, 2 * t + 3);

        int bitIndex = L - 2 - t;
        if ((x >> bitIndex) & 1ULL) {
            for (int r = 2; r <= 2 * t; ++r) {
                open(r, 2 * t + 2);
            }
        }
    }

    cout << n << '\n';
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (j) cout << ' ';
            cout << g[i][j];
        }
        cout << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    unsigned long long x;
    cin >> x;

    if (x == 1) {
        cout << 1 << "\n1\n";
        return 0;
    }

    int k = 63 - __builtin_clzll(x);
    int n = 2 + 2 * k;
    vector<vector<int>> a(n + 1, vector<int>(n + 1, 0));

    for (int j = 1; j <= n; ++j) a[1][j] = 1;
    a[2][2] = 1;

    int r = 2;
    for (int b = k - 1; b >= 0; --b) {
        if ((x >> b) & 1ULL) {
            for (int i = 1; i <= r - 1; ++i) {
                a[i][r + 1] = 1;
            }
        }

        a[r][r + 1] = 1;
        a[r + 1][r] = 1;
        a[r + 1][r + 1] = 1;
        a[r + 2][r + 1] = 1;
        a[r + 2][r + 2] = 1;
        r += 2;
    }

    cout << n << '\n';
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            cout << a[i][j] << (j == n ? '\n' : ' ');
        }
    }

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    
    unsigned long long x;
    cin >> x;
    
    vector<int> bits;
    while (x > 0) {
        bits.push_back(x & 1ULL);
        x >>= 1ULL;
    }
    reverse(bits.begin(), bits.end());
    
    int G = (int)bits.size() - 1;
    int n = 3 + 2 * G;
    
    vector<vector<int>> grid(n + 1, vector<int>(n + 1, 0));
    
    // Top wire
    for (int j = 1; j <= n; ++j) grid[1][j] = 1;
    
    int r0 = 3, c0 = 2;
    
    // Entry path to A0
    for (int i = 1; i <= r0; ++i) grid[i][1] = 1;
    for (int j = 1; j <= c0; ++j) grid[r0][j] = 1;
    
    // Gadgets for remaining bits
    for (int idx = 1; idx < (int)bits.size(); ++idx) {
        int bit = bits[idx];
        int t = idx - 1;
        int r = r0 + 2 * t;
        int c = c0 + 2 * t;
        
        grid[r + 1][c] = 1;
        grid[r][c + 1] = 1;
        grid[r + 1][c + 1] = 1;
        grid[r + 2][c + 1] = 1;
        grid[r + 2][c + 2] = 1;
        
        if (bit) {
            int C = c + 1;
            for (int row = 2; row <= r; ++row) {
                grid[row][C] = 1;
            }
        }
    }
    
    // Connect final A to (n,n)
    grid[n][n] = 1;
    
    cout << n << '\n';
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            cout << grid[i][j] << (j == n ? '\n' : ' ');
        }
    }
    
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    unsigned long long x;
    cin >> x;

    if (x == 1) {
        cout << "1\n1\n";
        return 0;
    }

    int L = 64 - __builtin_clzll(x);
    int stages = L - 2;
    int n = 2 * stages + 4;

    vector<vector<int>> a(n, vector<int>(n, 0));

    // The initial gadget has either 2 or 3 paths.
    int initial = int((x >> (L - 2)) & 3ULL);

    a[0][0] = 1;
    a[1][0] = 1;

    a[2][0] = a[2][1] = a[2][2] = 1;
    if (initial == 3) {
        a[2][3] = 1;
    }
    a[3][1] = a[3][2] = a[3][3] = 1;

    for (int stage = 0; stage < stages; ++stage) {
        int p = 3 + 2 * stage;

        // Boundary of a square: exactly two disjoint routes
        // from (p,p) to (p+2,p+2).
        for (int c = p; c <= p + 2; ++c) {
            a[p][c] = 1;
            a[p + 2][c] = 1;
        }
        for (int r = p; r <= p + 2; ++r) {
            a[r][p] = 1;
            a[r][p + 2] = 1;
        }

        int bitIndex = L - 3 - stage;
        int bit = int((x >> bitIndex) & 1ULL);

        if (bit == 1) {
            // A unique extra route from the upper rail into this stage.
            for (int r = 0; r <= p; ++r) {
                a[r][p + 2] = 1;
            }
            for (int c = 0; c <= p + 2; ++c) {
                a[0][c] = 1;
            }
        }
    }

    cout << n << '\n';
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (j) cout << ' ';
            cout << a[i][j];
        }
        cout << '\n';
    }

    return 0;
}
