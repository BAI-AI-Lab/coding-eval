#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using cd = complex<double>;

static constexpr double PI = 3.141592653589793238462643383279502884;

struct FFT {
    int n;
    vector<int> rev;
    vector<cd> roots;

    explicit FFT(int m) {
        n = 1;
        while (n < 2 * m) n <<= 1;
        rev.resize(n);
        for (int i = 1; i < n; ++i) {
            rev[i] = (rev[i >> 1] >> 1) | ((i & 1) ? (n >> 1) : 0);
        }
        roots.resize(n / 2);
        for (int i = 0; i < n / 2; ++i) {
            double ang = -2.0 * PI * i / n;
            roots[i] = cd(cos(ang), sin(ang));
        }
    }

    void transform(vector<cd>& a, bool inverse) const {
        for (int i = 0; i < n; ++i) {
            if (i < rev[i]) swap(a[i], a[rev[i]]);
        }
        for (int len = 2; len <= n; len <<= 1) {
            int step = n / len;
            int half = len >> 1;
            for (int i = 0; i < n; i += len) {
                for (int j = 0; j < half; ++j) {
                    cd w = roots[j * step];
                    if (inverse) w = conj(w);
                    cd u = a[i + j];
                    cd v = a[i + j + half] * w;
                    a[i + j] = u + v;
                    a[i + j + half] = u - v;
                }
            }
        }
        if (inverse) {
            double inv = 1.0 / n;
            for (cd& x : a) x *= inv;
        }
    }
};

struct Solver {
    int M;
    ll budget;
    vector<ll> cost;
    vector<int> w, d;
    FFT fft;
    chrono::steady_clock::time_point startTime;
    mt19937 rng{123456789};

    ll totalW = 0, upperBound = 0;
    vector<char> bestA;
    ll bestValue = -1, bestSpent = 0;

    vector<char> A;
    vector<int> c;
    ll value = 0, spent = 0;
    int k = 0;

    Solver(int m, ll b, const vector<ll>& co, const vector<int>& ww,
           const vector<int>& dd, chrono::steady_clock::time_point st)
        : M(m), budget(b), cost(co), w(ww), d(dd), fft(m), startTime(st) {
        for (int x : w) totalW += x;
        for (int i = 0; i < M; ++i) upperBound += (ll)w[i] * d[i];
    }

    double elapsed() const {
        return chrono::duration<double>(chrono::steady_clock::now() - startTime).count();
    }

    inline ll utility(int r, int cnt) const {
        return (ll)w[r] * min(cnt, d[r]);
    }

    void consider() {
        if (value > bestValue || (value == bestValue && spent < bestSpent)) {
            bestValue = value;
            bestSpent = spent;
            bestA = A;
        }
    }

    void computeCounts() {
        vector<int> S;
        bool useA = (k <= M - k);
        S.reserve(min(k, M - k));
        for (int i = 0; i < M; ++i) {
            if ((bool)A[i] == useA) S.push_back(i);
        }

        int s = (int)S.size();
        c.assign(M, s);

        if (s == 0) return;

        if (1LL * s * s <= 4LL * M || s <= 32) {
            for (int x : S) {
                for (int y : S) {
                    --c[(x + y) % M];
                }
            }
            return;
        }

        vector<cd> fa(fft.n, cd(0, 0));
        for (int x : S) fa[x] = cd(1, 0);
        fft.transform(fa, false);
        vector<cd> y(fft.n);
        for (int i = 0; i < fft.n; ++i) y[i] = fa[i] * fa[i];
        fft.transform(y, true);

        for (int r = 0; r < M; ++r) {
            ll t = llround(y[r].real() + y[r + M].real());
            int cr = s - (int)t;
            if (cr < 0) cr = 0;
            if (cr > s) cr = s;
            c[r] = cr;
        }
    }

    void initializeState(const vector<char>& start) {
        A = start;
        spent = 0;
        k = 0;
        for (int i = 0; i < M; ++i) {
            if (A[i]) {
                ++k;
                spent += cost[i];
            }
        }
        computeCounts();
        value = 0;
        for (int r = 0; r < M; ++r) value += utility(r, c[r]);
        consider();
    }

    ll flipPort(int p) {
        bool removing = A[p];
        ll delta = 0;

        if (!removing) {
            for (int q = 0; q < M; ++q) {
                if (q == p) continue;
                int r = (p + q) % M;
                int dc = A[q] ? -1 : 1;
                int old = c[r];
                int nw = old + dc;
                delta += utility(r, nw) - utility(r, old);
                c[r] = nw;
            }
            A[p] = 1;
            ++k;
            spent += cost[p];
        } else {
            for (int q = 0; q < M; ++q) {
                if (q == p) continue;
                int r = (p + q) % M;
                int dc = A[q] ? 1 : -1;
                int old = c[r];
                int nw = old + dc;
                delta += utility(r, nw) - utility(r, old);
                c[r] = nw;
            }
            A[p] = 0;
            --k;
            spent -= cost[p];
        }

        value += delta;
        return delta;
    }

    void computeGains(vector<ll>& gain) {
        vector<ll> G(M), L(M);
        ll sumG = 0, sumL = 0;
        for (int r = 0; r < M; ++r) {
            G[r] = (c[r] < d[r]) ? w[r] : 0;
            L[r] = (c[r] <= d[r]) ? w[r] : 0;
            sumG += G[r];
            sumL += L[r];
        }

        bool useA = (k <= M - k);
        vector<int> S;
        S.reserve(min(k, M - k));
        for (int i = 0; i < M; ++i) {
            if ((bool)A[i] == useA) S.push_back(i);
        }

        vector<ll> corrG(M, 0), corrL(M, 0);
        int s = (int)S.size();

        if (1LL * s * M <= 3000000LL) {
            for (int p = 0; p < M; ++p) {
                ll sg = 0, sl = 0;
                for (int q : S) {
                    int r = p + q;
                    if (r >= M) r -= M;
                    sg += G[r];
                    sl += L[r];
                }
                corrG[p] = sg;
                corrL[p] = sl;
            }
        } else {
            vector<cd> fs(fft.n, cd(0, 0)), z(fft.n, cd(0, 0));
            for (int q : S) fs[q] = cd(1, 0);
            for (int r = 0; r < M; ++r) {
                z[r] = cd((double)G[r], (double)L[r]);
            }
            fft.transform(fs, false);
            fft.transform(z, false);
            for (int i = 0; i < fft.n; ++i) z[i] = conj(fs[i]) * z[i];
            fft.transform(z, true);

            for (int p = 0; p < M; ++p) {
                int other = fft.n + p - M;
                double sg = z[p].real() + z[other].real();
                double sl = z[p].imag() + z[other].imag();
                ll cg = llround(sg);
                ll cl = llround(sl);
                cg = max(0LL, min(sumG, cg));
                cl = max(0LL, min(sumL, cl));
                corrG[p] = cg;
                corrL[p] = cl;
            }
        }

        gain.resize(M);
        for (int p = 0; p < M; ++p) {
            int twice = (int)((2LL * p) % M);
            if (useA) {
                if (A[p]) {
                    gain[p] = corrG[p] - G[twice] - (sumL - corrL[p]);
                } else {
                    gain[p] = sumG - corrG[p] - G[twice] - corrL[p];
                }
            } else {
                if (A[p]) {
                    gain[p] = sumG - corrG[p] - G[twice] - corrL[p];
                } else {
                    gain[p] = corrG[p] - G[twice] - (sumL - corrL[p]);
                }
            }
        }
    }

    ll evaluateSwap(int rem, int add) {
        ll removeDelta = 0;

        for (int q = 0; q < M; ++q) {
            if (q == rem) continue;
            int r = (rem + q) % M;
            int dc = A[q] ? 1 : -1;
            int old = c[r];
            int nw = old + dc;
            removeDelta += utility(r, nw) - utility(r, old);
            c[r] = nw;
        }

        A[rem] = 0;
        ll addDelta = 0;

        for (int q = 0; q < M; ++q) {
            if (q == add) continue;
            int r = (add + q) % M;
            if (A[q]) {
                if (c[r] <= d[r]) addDelta -= w[r];
            } else {
                if (c[r] < d[r]) addDelta += w[r];
            }
        }

        for (int q = 0; q < M; ++q) {
            if (q == rem) continue;
            int r = (rem + q) % M;
            int dc = A[q] ? 1 : -1;
            c[r] -= dc;
        }
        A[rem] = 1;

        return removeDelta + addDelta;
    }

    pair<int, int> findSwap(const vector<ll>& gain, double deadline) {
        if (k == 0 || k == M) return {-1, -1};

        ll maxCostA = 0;
        vector<int> additions;
        additions.reserve(M - k);

        for (int i = 0; i < M; ++i) {
            if (A[i]) maxCostA = max(maxCostA, cost[i]);
        }
        for (int j = 0; j < M; ++j) {
            if (!A[j] && spent - maxCostA + cost[j] <= budget) {
                additions.push_back(j);
            }
        }
        if (additions.empty()) return {-1, -1};

        sort(additions.begin(), additions.end(), [&](int x, int y) {
            if (gain[x] != gain[y]) return gain[x] > gain[y];
            if (cost[x] != cost[y]) return cost[x] < cost[y];
            return x < y;
        });
        if ((int)additions.size() > 28) additions.resize(28);

        struct Candidate {
            ll approximate;
            int rem, add;
        };
        vector<Candidate> candidates;

        for (int add : additions) {
            int best1 = -1, best2 = -1;
            for (int rem = 0; rem < M; ++rem) {
                if (!A[rem]) continue;
                if (spent - cost[rem] + cost[add] > budget) continue;

                auto better = [&](int x, int y) {
                    if (y == -1) return true;
                    if (gain[x] != gain[y]) return gain[x] > gain[y];
                    return cost[x] > cost[y];
                };

                if (better(rem, best1)) {
                    best2 = best1;
                    best1 = rem;
                } else if (better(rem, best2)) {
                    best2 = rem;
                }
            }
            if (best1 != -1) {
                candidates.push_back({gain[best1] + gain[add], best1, add});
            }
            if (best2 != -1) {
                candidates.push_back({gain[best2] + gain[add], best2, add});
            }
        }

        sort(candidates.begin(), candidates.end(), [](const Candidate& x, const Candidate& y) {
            return x.approximate > y.approximate;
        });

        ll bestDelta = 0;
        pair<int, int> answer{-1, -1};
        int evaluated = 0;

        for (const Candidate& cand : candidates) {
            if (evaluated >= 24) break;
            if ((evaluated & 3) == 0 && elapsed() >= deadline) break;
            ++evaluated;

            ll delta = evaluateSwap(cand.rem, cand.add);
            if (delta > bestDelta) {
                bestDelta = delta;
                answer = {cand.rem, cand.add};
            }
        }
        return answer;
    }

    void localSearch(const vector<char>& start, double deadline) {
        if (elapsed() >= deadline) return;
        initializeState(start);
        if (bestValue >= upperBound) return;

        vector<ll> gain;

        while (elapsed() < deadline) {
            computeGains(gain);

            int bestPort = -1;
            ll bestGain = 0;

            for (int p = 0; p < M; ++p) {
                if (!A[p] && spent + cost[p] > budget) continue;
                ll g = gain[p];
                if (g <= 0) continue;

                bool take = false;
                if (g > bestGain) {
                    take = true;
                } else if (g == bestGain && bestPort != -1) {
                    bool curRemove = A[p], oldRemove = A[bestPort];
                    if (curRemove != oldRemove) {
                        take = !curRemove;
                    } else if (!curRemove) {
                        take = cost[p] < cost[bestPort];
                    } else {
                        take = cost[p] > cost[bestPort];
                    }
                } else if (bestPort == -1) {
                    take = true;
                }

                if (take) {
                    bestGain = g;
                    bestPort = p;
                }
            }

            if (bestPort != -1) {
                ll oldValue = value;
                flipPort(bestPort);
                if (value <= oldValue) {
                    flipPort(bestPort);
                    return;
                }
                consider();
                if (bestValue >= upperBound) return;
                continue;
            }

            pair<int, int> sw = findSwap(gain, deadline);
            if (sw.first == -1) return;

            ll oldValue = value;
            flipPort(sw.first);
            flipPort(sw.second);

            if (value <= oldValue) {
                flipPort(sw.second);
                flipPort(sw.first);
                return;
            }

            consider();
            if (bestValue >= upperBound) return;
        }
    }

    void printAnswer() const {
        int count = 0;
        for (char x : bestA) count += x;
        cout << count << '\n';
        bool first = true;
        for (int i = 0; i < M; ++i) {
            if (bestA[i]) {
                if (!first) cout << ' ';
                first = false;
                cout << i;
            }
        }
        if (count) cout << '\n';
    }

    void run() {
        const double hardDeadline = 7.15;

        vector<char> singleA(M, 0);
        int singlePort = -1;
        ll singleValue = -1;

        for (int p = 0; p < M; ++p) {
            if (cost[p] > budget) continue;
            ll v = totalW - w[(2LL * p) % M];
            if (v > singleValue ||
                (v == singleValue && (singlePort == -1 || cost[p] < cost[singlePort]))) {
                singleValue = v;
                singlePort = p;
            }
        }

        singleA[singlePort] = 1;
        bestA = singleA;
        bestValue = singleValue;
        bestSpent = cost[singlePort];

        if (bestValue >= upperBound) {
            printAnswer();
            return;
        }

        vector<int> baseOrder(M);
        iota(baseOrder.begin(), baseOrder.end(), 0);
        sort(baseOrder.begin(), baseOrder.end(), [&](int x, int y) {
            if (cost[x] != cost[y]) return cost[x] < cost[y];
            return x < y;
        });

        vector<char> baseline(M, 0);
        ll baselineSpent = 0;
        for (int p : baseOrder) {
            if (baselineSpent + cost[p] <= budget) {
                baseline[p] = 1;
                baselineSpent += cost[p];
            }
        }

        localSearch(baseline, hardDeadline);

        if (bestValue >= upperBound || elapsed() >= hardDeadline) {
            printAnswer();
            return;
        }

        double now = elapsed();
        double remaining = hardDeadline - now;
        if (remaining > 0.08) {
            localSearch(singleA, now + remaining * 0.45);
        }

        int restart = 0;
        while (bestValue < upperBound && elapsed() + 0.05 < hardDeadline) {
            vector<char> start;
            ll startSpent = 0;

            if ((restart & 1) == 0) {
                start = bestA;
                vector<int> emitters;
                for (int i = 0; i < M; ++i) {
                    if (start[i]) {
                        emitters.push_back(i);
                        startSpent += cost[i];
                    }
                }

                if ((int)emitters.size() > 1) {
                    shuffle(emitters.begin(), emitters.end(), rng);
                    int maxRemove = min((int)emitters.size() - 1, 6);
                    int removeCount = 1 + (int)(rng() % maxRemove);
                    for (int t = 0; t < removeCount; ++t) {
                        int p = emitters[t];
                        start[p] = 0;
                        startSpent -= cost[p];
                    }
                } else if ((rng() & 1) && M > 1) {
                    for (int attempt = 0; attempt < 40; ++attempt) {
                        int p = rng() % M;
                        if (cost[p] <= budget) {
                            fill(start.begin(), start.end(), 0);
                            start[p] = 1;
                            startSpent = cost[p];
                            break;
                        }
                    }
                }

                int randomAdds = rng() % 3;
                int added = 0;
                for (int attempt = 0; attempt < 40 && added < randomAdds; ++attempt) {
                    int p = rng() % M;
                    if (!start[p] && startSpent + cost[p] <= budget) {
                        start[p] = 1;
                        startSpent += cost[p];
                        ++added;
                    }
                }
            } else {
                vector<int> order = baseOrder;
                int block = 16 + rng() % 32;
                for (int l = 0; l < M; l += block) {
                    int r = min(M, l + block);
                    shuffle(order.begin() + l, order.begin() + r, rng);
                }

                start.assign(M, 0);
                for (int p : order) {
                    if (startSpent + cost[p] <= budget) {
                        start[p] = 1;
                        startSpent += cost[p];
                    }
                }
            }

            now = elapsed();
            remaining = hardDeadline - now;
            if (remaining <= 0.04) break;
            double slice = min(remaining, M > 50000 ? 0.85 : 0.45);
            localSearch(start, now + slice);
            ++restart;
        }

        printAnswer();
    }
};

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using i128 = __int128_t;

static const int MOD = 998244353;
static const int G = 3;

static int modpow(int a, int64 e) {
    int64 r = 1, x = a;
    while (e) {
        if (e & 1) r = r * x % MOD;
        x = x * x % MOD;
        e >>= 1;
    }
    return (int)r;
}

static void ntt(vector<int>& a, bool invert) {
    int n = (int)a.size();
    for (int i = 1, j = 0; i < n; ++i) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    for (int len = 2; len <= n; len <<= 1) {
        int wlen = modpow(G, (MOD - 1) / len);
        if (invert) wlen = modpow(wlen, MOD - 2);
        int half = len >> 1;
        for (int i = 0; i < n; i += len) {
            int64 w = 1;
            for (int j = 0; j < half; ++j) {
                int u = a[i + j];
                int v = (int)(a[i + j + half] * w % MOD);
                int x = u + v;
                if (x >= MOD) x -= MOD;
                int y = u - v;
                if (y < 0) y += MOD;
                a[i + j] = x;
                a[i + j + half] = y;
                w = w * wlen % MOD;
            }
        }
    }
    if (invert) {
        int inv = modpow(n, MOD - 2);
        for (int& x : a) x = (int)((int64)x * inv % MOD);
    }
}

struct NTT {
    int n;
    vector<int> rev, roots;

    explicit NTT(int n_) : n(n_), rev(n), roots(n) {
        for (int i = 1; i < n; ++i) {
            rev[i] = (rev[i >> 1] >> 1) | ((i & 1) ? (n >> 1) : 0);
        }

        for (int len = 1; len < n; len <<= 1) {
            int w = modPow(PRIMITIVE_ROOT, (MOD - 1) / (len << 1));
            roots[len] = 1;
            for (int j = 1; j < len; ++j) {
                roots[len + j] = int64(roots[len + j - 1]) * w % MOD;
            }
        }
    }

        if (inverse) {
            reverse(a.begin() + 1, a.end());
        }

        for (int i = 0; i < n; ++i) {
            if (i < rev[i]) swap(a[i], a[rev[i]]);

        for (int len = 1; len < n; len <<= 1) {
            for (int i = 0; i < n; i += len << 1) {

bool randomSwap(State& s, mt19937_64& rng) {
    if (s.A.empty() || s.R.empty()) return false;

    int p = s.A[rng() % s.A.size()];
    int chosen = -1;

    for (int attempt = 0; attempt < 150; ++attempt) {
        int q = s.R[rng() % s.R.size()];
        if (s.totalCost - cost[p] + cost[q] <= budget) {
            chosen = q;
            break;
        }
    }

    if (chosen == -1) {
        int seen = 0;
        for (int q : s.R) {
            if (s.totalCost - cost[p] + cost[q] > budget) continue;
            ++seen;
            if (rng() % seen == 0) chosen = q;
        }
    }

    if (chosen == -1) return false;

    doRemove(s, p);
    doAdd(s, chosen);
    return true;
}

bool randomAdd(State& s, mt19937_64& rng) {
    int chosen = -1;
    int seen = 0;
    for (int p : s.R) {
        if (s.totalCost + cost[p] > budget) continue;
        ++seen;
        if (rng() % seen == 0) chosen = p;
    }
    if (chosen == -1) return false;
    doAdd(s, chosen);
    return true;
}

bool randomRemove(State& s, mt19937_64& rng) {
    if (s.A.empty()) return false;
    int p = s.A[rng() % s.A.size()];
    doRemove(s, p);
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    auto start = chrono::steady_clock::now();
    deadline = start + chrono::milliseconds(7200);

    int64 B;
    cin >> M >> B;
    budget = B;

    cost.resize(M);
    w.resize(M);
    d.resize(M);

    for (int& x : cost) {
        cin >> x;
        totalCostAll += x;
    }
    for (int& x : w) {
        cin >> x;
        sumW += x;
    }
    for (int& x : d) cin >> x;

    int transformSize = 1;
    while (transformSize < 2 * M) transformSize <<= 1;
    ntt = make_unique<NTT>(transformSize);

    costOrder.resize(M);
    iota(costOrder.begin(), costOrder.end(), 0);
    sort(costOrder.begin(), costOrder.end(), [](int a, int b) {
        if (cost[a] != cost[b]) return cost[a] < cost[b];
        return a < b;
    });

    vector<int> baselineChosen;
    int64 baselineCost = 0;
    for (int p : costOrder) {
        if (baselineCost + cost[p] <= budget) {
            baselineCost += cost[p];
            baselineChosen.push_back(p);
        }
    }

    State baseline = stateFromChosen(baselineChosen);
    State global = baseline;

    if (totalCostAll - baseline.totalCost <= budget) {
        State complement = complementState(baseline);
        updateGlobal(complement, global);
    }

    mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());

    for (int attempt = 0; attempt < 4; ++attempt) {
        if (chrono::steady_clock::now() > start + chrono::milliseconds(1300)) break;
        int k = (attempt & 1) ? M / 2 : M / 4;
        State candidate = randomBalancedState(k, rng);
        updateGlobal(candidate, global);
    }

    if (chrono::steady_clock::now() <= start + chrono::milliseconds(1600)) {
        vector<int> order(M);
        iota(order.begin(), order.end(), 0);
        shuffle(order.begin(), order.end(), rng);

        vector<int> chosen;
        int64 currentCost = 0;
        for (int p : order) {
            if (currentCost + cost[p] <= budget) {
                currentCost += cost[p];
                chosen.push_back(p);
            }
        }
        State candidate = stateFromChosen(chosen);
        updateGlobal(candidate, global);
    }

    int bestAnyPair = -1, bestAnyQ = -1;
    int bestFeasiblePair = -1, bestFeasibleQ = -1;
    int64 bestAnyPairValue = -1;
    int64 bestFeasiblePairValue = -1;

    vector<int> sumOrder(M), diagOrder(M);
    iota(sumOrder.begin(), sumOrder.end(), 0);
    iota(diagOrder.begin(), diagOrder.end(), 0);

    sort(sumOrder.begin(), sumOrder.end(), [](int a, int b) {
        int64 va = int64(w[a]) * min(2, d[a]);
        int64 vb = int64(w[b]) * min(2, d[b]);
        if (va != vb) return va < vb;
        return a < b;
    });

    sort(diagOrder.begin(), diagOrder.end(), [](int a, int b) {
        int64 va = d[addMod(a, a)] >= 2 ? w[addMod(a, a)] : 0;
        int64 vb = d[addMod(b, b)] >= 2 ? w[addMod(b, b)] : 0;
        if (va != vb) return va < vb;
        return a < b;
    });

    auto examinePair = [&](int p, int q) {
        if (p == q) return;
        int64 value = pairValue(p, q);
        int64 pc = int64(cost[p]) + cost[q];

        if (value > bestAnyPairValue ||
            (value == bestAnyPairValue && p < bestAnyPair)) {
            bestAnyPairValue = value;
            bestAnyPair = p;
            bestAnyQ = q;
        }

        bool feasible = pc <= budget || totalCostAll - pc <= budget;
        if (feasible &&
            (value > bestFeasiblePairValue ||
             (value == bestFeasiblePairValue && p < bestFeasiblePair))) {
            bestFeasiblePairValue = value;
            bestFeasiblePair = p;
            bestFeasibleQ = q;
        }
    };

    auto pairSearchEnd = min(deadline, start + chrono::milliseconds(2200));
    for (int p = 0; p < M; ++p) {
        if ((p & 511) == 0 && chrono::steady_clock::now() > pairSearchEnd) break;

        for (int t = 0; t < min(48, M); ++t) {
            int r = sumOrder[t];
            int q = r - p;
            if (q < 0) q += M;
            examinePair(p, q);
        }

        for (int t = 0; t < min(48, M); ++t) {
            examinePair(p, diagOrder[t]);
        }
    }

    State pairState;
    bool havePairState = false;
    if (bestFeasiblePair != -1) {
        int p = bestFeasiblePair;
        int q = bestFeasibleQ;
        int64 pc = int64(cost[p]) + cost[q];
        int64 cc = totalCostAll - pc;

        vector<int> chosen;
        if (pc <= budget && (cc > budget || pc <= cc)) {
            chosen = {p, q};
        } else {
            vector<uint8> excluded(M, 0);
            excluded[p] = excluded[q] = 1;
            for (int i = 0; i < M; ++i) {
                if (!excluded[i]) chosen.push_back(i);
            }
        }

        pairState = makeState(chosen, pairCounts(p, q));
        havePairState = true;
        updateGlobal(pairState, global);
    }

    int bestSingleton = -1;
    int64 bestSingletonValue = -1;
    int bestFeasibleSingleton = -1;
    int64 bestFeasibleSingletonValue = -1;

    for (int p = 0; p < M; ++p) {
        int64 value = sumW - w[addMod(p, p)];
        if (value > bestSingletonValue ||
            (value == bestSingletonValue && p < bestSingleton)) {
            bestSingletonValue = value;
            bestSingleton = p;
        }

        bool feasible = cost[p] <= budget || totalCostAll - cost[p] <= budget;
        if (feasible &&
            (value > bestFeasibleSingletonValue ||
             (value == bestFeasibleSingletonValue && p < bestFeasibleSingleton))) {
            bestFeasibleSingletonValue = value;
            bestFeasibleSingleton = p;
        }
    }

    State singletonState;
    bool haveSingletonState = false;
    if (bestFeasibleSingleton != -1) {
        int p = bestFeasibleSingleton;
        int64 sc = cost[p];
        int64 cc = totalCostAll - sc;

        vector<int> chosen;
        if (sc <= budget && (cc > budget || sc <= cc)) {
            chosen = {p};
        } else {
            vector<uint8> excluded(M, 0);
            excluded[p] = 1;
            for (int i = 0; i < M; ++i) {
                if (!excluded[i]) chosen.push_back(i);
            }
        }

        singletonState = makeState(chosen, singletonCounts(p));
        haveSingletonState = true;
        updateGlobal(singletonState, global);
    }

    LocalOptimizer optimizer(rng);

    optimizer.improve(
        baseline, global,
        min(deadline, start + chrono::milliseconds(2800)), true
    );

    if (haveSingletonState) {
        optimizer.improve(
            singletonState, global,
            min(deadline, start + chrono::milliseconds(3400)), false
        );
    }

    if (havePairState) {
        optimizer.improve(
            pairState, global,
            min(deadline, start + chrono::milliseconds(4100)), true
        );
    }

    State current = global;
    optimizer.improve(
        current, global,
        min(deadline, start + chrono::milliseconds(4600)), false
    );

    int iteration = 0;
    while (chrono::steady_clock::now() < deadline) {
        if (iteration % 4 == 0) current = global;

        int kicks = 1 + int(rng() % 4);
        for (int k = 0; k < kicks; ++k) {
            int type = int(rng() % 10);
            if (type < 5) {
                randomSwap(current, rng);
            } else if (type < 8) {
                randomRemove(current, rng);
            } else {
                randomAdd(current, rng);
            }
        }

        auto endTime = min(deadline, chrono::steady_clock::now() +
                                           chrono::milliseconds(700));
        optimizer.improve(current, global, endTime, false);
        ++iteration;
    }

    vector<int> answer = global.A;
    sort(answer.begin(), answer.end());

    cout << answer.size() << '\n';
    if (!answer.empty()) {
        for (int i = 0; i < int(answer.size()); ++i) {
            if (i) cout << ' ';
            cout << answer[i];
        }
        cout << '\n';
    }

    return 0;
}
