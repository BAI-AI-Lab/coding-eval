Declining Invitations

USACO 2026 Second · 银组 · 2026-01 · L2 中等 · Python 3.11 · 时限 4 秒 · 测试点等分

N 名选手按排名依次排列，第 i 名满足若干条标准；按标准 1…C 依次从未邀请者中邀请满足该标准的前 fᵢ 名；对每个 i = 0…N−1，求排列 p 中前 i 人退出后被邀请者的名次之和。

import sys


def read_ints():
    data = sys.stdin.buffer.read()
    i = 0
    length = len(data)
    while i < length:
        while i < length and data[i] <= 32:
            i += 1
        if i >= length:
            break
        value = 0
        while i < length and data[i] > 32:
            value = value * 10 + data[i] - 48
            i += 1
        yield value


def main():
    it = read_ints()
    n = next(it)
    c = next(it)

    f = [0] * (c + 1)
    for i in range(1, c + 1):
        f[i] = next(it)

    order = [next(it) for _ in range(n)]

    criteria = [[] for _ in range(c + 1)]
    for rank in range(1, n + 1):
        count = next(it)
        for _ in range(count):
            criteria[next(it)].append(rank)
    del it

    # state[r] = 0: uninvited, -1: declined, otherwise invitation criterion.
    state = [0] * (n + 1)
    pointers = [0] * (c + 1)
    answer = 0

    for q in range(1, c + 1):
        contestants = criteria[q]
        index = 0
        chosen = 0
        limit = f[q]
        length = len(contestants)

        while chosen < limit and index < length:
            rank = contestants[index]
            index += 1
            if state[rank] == 0:
                state[rank] = q
                answer += rank
                chosen += 1

        pointers[q] = index

    output = [str(answer)]

    for i in range(n - 1):
        declined = order[i]
        q = state[declined]
        state[declined] = -1

        if q > 0:
            answer -= declined

            # Fill the vacancy. If the replacement was invited at a later
            # criterion, that criterion in turn receives a vacancy.
            while True:
                contestants = criteria[q]
                index = pointers[q]
                length = len(contestants)
                replacement = 0

                while index < length:
                    rank = contestants[index]
                    index += 1
                    old_criterion = state[rank]
                    if old_criterion == 0 or old_criterion > q:
                        replacement = rank
                        break

                pointers[q] = index

                if replacement == 0:
                    break

                old_criterion = state[replacement]
                state[replacement] = q

                if old_criterion == 0:
                    answer += replacement
                    break

                q = old_criterion

        output.append(str(answer))

    sys.stdout.write("\n".join(output))


if __name__ == "__main__":
    main()

import sys
import heapq


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    idx = 0

    N = data[idx]
    C = data[idx + 1]
    idx += 2

    f = data[idx:idx + C]
    idx += C

    p = data[idx:idx + N]
    idx += N

    heaps = [[] for _ in range(C + 1)]

    for contestant in range(1, N + 1):
        n = data[idx]
        idx += 1
        for _ in range(n):
            c = data[idx]
            idx += 1
            heaps[c].append(contestant)

    del data

    assigned = [0] * (N + 1)
    total = 0
    heappop = heapq.heappop

    # Initial invitation process.
    for c in range(1, C + 1):
        h = heaps[c]
        limit = f[c - 1]
        cnt = 0

        while cnt < limit and h:
            x = heappop(h)
            if assigned[x] == 0:
                assigned[x] = c
                total += x
                cnt += 1

    active = bytearray(b'\x01') * (N + 1)
    answers = [str(total)]

    # Process the first N-1 declinations.
    for i in range(N - 1):
        x = p[i]

        if active[x] and assigned[x] != 0:
            old = assigned[x]
            active[x] = 0
            total -= x

            v = old
            while True:
                h = heaps[v]

                while h:
                    y = h[0]
                    if active[y] and (assigned[y] == 0 or assigned[y] > v):
                        break
                    heappop(h)

                if not h:
                    break

                y = heappop(h)
                old_y = assigned[y]

                if old_y == 0:
                    assigned[y] = v
                    total += y
                    break
                else:
                    assigned[y] = v
                    v = old_y
        else:
            active[x] = 0

        answers.append(str(total))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

import sys
from itertools import islice


def solve():
    data = map(int, sys.stdin.buffer.read().split())
    try:
        N = next(data)
    except StopIteration:
        return
    C = next(data)

    f = [0] + list(islice(data, C))
    p = list(islice(data, N))

    # Initially, pool[c] contains everyone satisfying criterion c.
    # Since ranks are read in increasing order, every pool is rank-sorted.
    pool = [[] for _ in range(C + 1)]
    for rank in range(1, N + 1):
        k = next(data)
        for _ in range(k):
            criterion = next(data)
            pool[criterion].append(rank)
    del data

    # assigned[rank]:
    #   0  -> not currently invited
    #  >0  -> criterion that currently invites this contestant
    #  -1  -> has declined
    assigned = [0] * (N + 1)
    total = 0

    # Simulate the original process.
    # Afterwards, pool[c] contains the contestants who satisfy c but were
    # not selected by c or by any earlier criterion.
    for criterion in range(1, C + 1):
        slots = f[criterion]
        candidates = []

        for rank in pool[criterion]:
            if assigned[rank] == 0:
                if slots:
                    assigned[rank] = criterion
                    slots -= 1
                    total += rank
                else:
                    candidates.append(rank)

        pool[criterion] = candidates

    # Each replacement pool is static and sorted, so use a moving head.
    head = [0] * (C + 1)
    answer = []

    for position, declined in enumerate(p):
        # Exactly p[:position] have declined at this point.
        answer.append(str(total))

        # The last requested state has N-1 declinations.
        if position == N - 1:
            break

        vacancy = assigned[declined]
        assigned[declined] = -1

        if vacancy == 0:
            # Removing a contestant who was not invited creates no vacancy.
            continue

        total -= declined

        # Repair the vacancy and any vacancy it creates.
        while vacancy:
            candidates = pool[vacancy]
            pos = head[vacancy]
            size = len(candidates)

            # A candidate is usable if it is still uninvited, or is currently
            # invited by a later criterion. Candidates selected by an earlier
            # criterion can never become usable again.
            while pos < size:
                candidate = candidates[pos]
                state = assigned[candidate]
                if state == 0 or state > vacancy:
                    break
                pos += 1

            head[vacancy] = pos
            if pos == size:
                break

            candidate = candidates[pos]
            head[vacancy] = pos + 1

            previous = assigned[candidate]
            assigned[candidate] = vacancy

            if previous == 0:
                # Previously uninvited contestant is now invited.
                total += candidate
                break

            # The contestant was promoted from a later criterion, which now
            # has a vacancy of its own.
            vacancy = previous

    sys.stdout.write("\n".join(answer) + "\n")


if __name__ == "__main__":
    solve()
