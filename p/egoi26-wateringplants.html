#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, D;
    cin >> N >> D;

    vector<long long> t(N);
    for (int i = 0; i < N; ++i) {
        cin >> t[i];
    }

    // Edge r represents resident r watering resident r - 1.
    vector<long long> count(N, 0), lastSync(N, 0);
    vector<bool> active(N, false);

    for (int r = 1; r < N; ++r) {
        active[r] = t[r] < t[r - 1];
    }

    auto sync = [&](int r, int daysElapsed) {
        if (active[r]) {
            count[r] += daysElapsed - lastSync[r];
        }
        lastSync[r] = daysElapsed;
    };

    for (int day = 0; day < D; ++day) {
        char type;
        cin >> type;
        int daysElapsed = day + 1;

        if (type == '?') {
            int r;
            cin >> r;
            sync(r, daysElapsed);
            cout << count[r] << '\n';
        } else {
            int r;
            long long x;
            cin >> r >> x;

            // Finalize the affected edges using the old schedule.
            for (int e = r; e <= r + 1; ++e) {
                if (e >= 1 && e < N) {
                    sync(e, daysElapsed);
                }
            }

            t[r] = x;

            // The new schedule takes effect from the next day.
            for (int e = r; e <= r + 1; ++e) {
                if (e >= 1 && e < N) {
                    active[e] = t[e] < t[e - 1];
                }
            }
        }
    }

    return 0;
}

        }

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, D;
    cin >> N >> D;

    vector<long long> t(N);
    for (int i = 0; i < N; ++i) cin >> t[i];

    vector<char> active(N, 0);
    vector<long long> sum(N, 0), last(N, -1);

    for (int i = 1; i < N; ++i) {
        active[i] = (t[i] < t[i - 1]);
    }

    auto updateEdge = [&](int e, int day) {
        if (e < 1 || e >= N) return;

        bool newActive = (t[e] < t[e - 1]);
        if (newActive != (bool)active[e]) {
            if (active[e]) {
                sum[e] += (long long)day - last[e];
            }
            last[e] = day;
            active[e] = newActive;
        }
    };

    for (int day = 0; day < D; ++day) {
        char type;
        cin >> type;

        if (type == '?') {
            int r;
            cin >> r;

            long long ans = sum[r];
            if (active[r]) {
                ans += (long long)day - last[r];
            }

            cout << ans << '\n';
        } else {
            int r;
            long long x;
            cin >> r >> x;

            t[r] = x;

            updateEdge(r, day);
            updateEdge(r + 1, day);
        }
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, D;
    cin >> N >> D;

    vector<int> t(N);
    for (int &x : t) cin >> x;

    // Edge e represents resident e watering for resident e - 1.
    vector<char> good(N, false);
    vector<int> segmentStart(N, 0);
    vector<long long> completed(N, 0);

    for (int e = 1; e < N; ++e) {
        good[e] = (t[e] < t[e - 1]);
    }

    // Close the current constant segment of edge e, which lasts
    // from segmentStart[e] through day.
    auto closeSegment = [&](int e, int day) {
        if (good[e]) {
            completed[e] += static_cast<long long>(day) - segmentStart[e] + 1;
        }
    };

    for (int day = 0; day < D; ++day) {
        char type;
        int r;
        cin >> type >> r;

        if (type == '!') {
            int x;
            cin >> x;

            // Changing t[r] can affect only edges r and r + 1.
            int e1 = (r >= 1 ? r : -1);
            int e2 = (r + 1 < N ? r + 1 : -1);

            // The old values apply through the current day.
            if (e1 != -1) closeSegment(e1, day);
            if (e2 != -1) closeSegment(e2, day);

            t[r] = x;

            // The new values apply starting from the next day.
            if (e1 != -1) {
                good[e1] = (t[e1] < t[e1 - 1]);
                segmentStart[e1] = day + 1;
            }
            if (e2 != -1) {
                good[e2] = (t[e2] < t[e2 - 1]);
                segmentStart[e2] = day + 1;
            }
        } else { // type == '?'
            long long answer = completed[r];
            if (good[r]) {
                answer += static_cast<long long>(day) - segmentStart[r] + 1;
            }
            cout << answer << '\n';
        }
    }

    return 0;
}

    // active[e] is true when resident e waters resident e-1.

    vector<int> nextDay(N, 0);

        active[e] = (t[e] < t[e - 1]);

    auto closeEdge = [&](int e, int day) {
        answer[e] += 1LL * active[e] * (day - nextDay[e] + 1);
        nextDay[e] = day + 1;

            cin >> x;

#include <bits/stdc++.h>
using namespace std;

struct Change {
    int start;  // First day on which this value is used
    int value;
};

struct Query {
    int day;
    int id;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, D;
    cin >> N >> D;

    vector<int> initial(N);
    for (int &x : initial) {
        cin >> x;
    }

    vector<vector<Change>> changes(N);
    for (int r = 0; r < N; ++r) {
        changes[r].push_back({0, initial[r]});
    }

    vector<vector<Query>> queries(N);
    int queryCount = 0;

    for (int day = 0; day < D; ++day) {
        char type;
        cin >> type;

        if (type == '!') {
            int r, x;
            cin >> r >> x;

            // The new return time starts on the next day.
            changes[r].push_back({day + 1, x});
        } else {
            int r;
            cin >> r;
            queries[r].push_back({day, queryCount++});
        }
    }

    vector<long long> answer(queryCount);
    constexpr int INF = numeric_limits<int>::max();

    for (int r = 1; r < N; ++r) {
        if (queries[r].empty()) {
            continue;
        }

        const auto &lowerChanges = changes[r - 1];
        const auto &upperChanges = changes[r];

        size_t lowerPos = 1;  // Skip the initial value
        size_t upperPos = 1;

        int lowerValue = initial[r - 1];
        int upperValue = initial[r];

        int segmentStart = 0;
        long long completed = 0;

        for (const Query &query : queries[r]) {
            // Apply all schedule changes effective by this query day.
            while (true) {
                int lowerBoundary =
                    lowerPos < lowerChanges.size()
                        ? lowerChanges[lowerPos].start
                        : INF;

                int upperBoundary =
                    upperPos < upperChanges.size()
                        ? upperChanges[upperPos].start
                        : INF;

                int boundary = min(lowerBoundary, upperBoundary);
                if (boundary > query.day) {
                    break;
                }

                // Close the segment [segmentStart, boundary).
                if (upperValue < lowerValue) {
                    completed += boundary - segmentStart;
                }

                segmentStart = boundary;

                if (lowerBoundary == boundary) {
                    lowerValue = lowerChanges[lowerPos].value;
                    ++lowerPos;
                }

                if (upperBoundary == boundary) {
                    upperValue = upperChanges[upperPos].value;
                    ++upperPos;
                }
            }

            long long result = completed;
            if (upperValue < lowerValue) {
                result += query.day - segmentStart + 1;
            }

            answer[query.id] = result;
        }
    }

    for (long long result : answer) {
        cout << result << '\n';
    }

    return 0;
}
