import sys
import math

MAX_X = 10**6

# Count distinct prime factors for every integer up to MAX_X.
prime_factor_count = [0] * (MAX_X + 1)
for p in range(2, MAX_X + 1):
    if prime_factor_count[p] == 0:  # p is prime
        for multiple in range(p, MAX_X + 1, p):
            prime_factor_count[multiple] += 1

# A 400 number is exactly x^2, where x has exactly two distinct
# prime factors. Precompute the largest such x not exceeding each index.
nearest = [0] * (MAX_X + 1)
last = 0
for x in range(2, MAX_X + 1):
    if prime_factor_count[x] == 2:
        last = x
    nearest[x] = last

data = list(map(int, sys.stdin.buffer.read().split()))
answers = []
for a in data[1:]:
    limit = math.isqrt(a)
    x = nearest[limit]
    answers.append(str(x * x))

sys.stdout.write("\n".join(answers) + "\n")

import sys
from bisect import bisect_right

MAX_M = 10**6

omega = [0] * (MAX_M + 1)

for p in range(2, MAX_M + 1):
    if omega[p] == 0:  # p is prime
        for x in range(p, MAX_M + 1, p):
            omega[x] += 1

numbers = []
for m in range(2, MAX_M + 1):
    if omega[m] == 2:
        numbers.append(m * m)

data = list(map(int, sys.stdin.buffer.read().split()))
q = data[0]
queries = data[1:1 + q]

answers = []
for a in queries:
    idx = bisect_right(numbers, a) - 1
    answers.append(str(numbers[idx]))

sys.stdout.write("\n".join(answers))

import sys
import math
from bisect import bisect_right

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    q = data[0]
    queries = data[1:1 + q]

    max_a = max(queries)
    max_m = math.isqrt(max_a)

    # omega[i] = number of distinct prime factors of i
    omega = bytearray(max_m + 1)
    for p in range(2, max_m + 1):
        if omega[p] == 0:  # p is prime
            for multiple in range(p, max_m + 1, p):
                omega[multiple] += 1

    # M^2 is a 400 number iff M has exactly two distinct prime factors
    valid_m = [m for m in range(2, max_m + 1) if omega[m] == 2]

    out = []
    for a in queries:
        x = math.isqrt(a)
        idx = bisect_right(valid_m, x) - 1
        m = valid_m[idx]
        out.append(str(m * m))

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
