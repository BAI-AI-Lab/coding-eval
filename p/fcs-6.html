#include <bits/stdc++.h>
using namespace std;

namespace WM {

struct Item {
    int w, h;
    int child;  // -1 for a single-cell pocket
    int color;  // used only for pockets
};

struct Placement {
    int x, y, idx;
};

struct Layout {
    int w = INT_MAX, h = INT_MAX;
    vector<Placement> p;
};

struct Region {
    int w = 1, h = 1;
    vector<Item> items;
    vector<Placement> p;
};

static Layout packShelf(const vector<Item>& items, const vector<int>& order,
                        int limit) {
    Layout out;
    out.w = out.h = 0;
    out.p.clear();

    int x = 1, y = 1;
    int rowWidth = 0, rowHeight = 0;
    int maxWidth = 0, totalHeight = 0;

    for (int idx : order) {
        int pw = items[idx].w + 1;
        int ph = items[idx].h + 1;

        if (rowWidth > 0 && rowWidth + pw > limit) {
            maxWidth = max(maxWidth, rowWidth);
            totalHeight += rowHeight;
            y += rowHeight;
            x = 1;
            rowWidth = rowHeight = 0;
        }

        out.p.push_back({x, y, idx});
        x += pw;
        rowWidth += pw;
        rowHeight = max(rowHeight, ph);
    }

    maxWidth = max(maxWidth, rowWidth);
    totalHeight += rowHeight;
    out.w = maxWidth + 1;
    out.h = totalHeight + 1;
    return out;
}

static bool betterLayout(const Layout& a, const Layout& b) {
    auto keyA = tuple<int, int, long long>(
        max(a.w, a.h), abs(a.w - a.h), 1LL * a.w * a.h);
    auto keyB = tuple<int, int, long long>(
        max(b.w, b.h), abs(b.w - b.h), 1LL * b.w * b.h);
    return keyA < keyB;
}

static Region makeRegion(vector<Item> items) {
    Region r;
    r.items = move(items);

    if (r.items.empty()) {
        r.w = r.h = 1;
        return r;
    }

    if (r.items.size() == 1) {
        r.w = r.items[0].w + 2;
        r.h = r.items[0].h + 2;
        r.p.push_back({1, 1, 0});
        return r;
    }

    int n = (int)r.items.size();
    int maxPW = 0, sumPW = 0;
    long long paddedArea = 0;

    for (const auto& it : r.items) {
        int pw = it.w + 1;
        int ph = it.h + 1;
        maxPW = max(maxPW, pw);
        sumPW += pw;
        paddedArea += 1LL * pw * ph;
    }

    vector<int> limits;
    auto addLimit = [&](int x) {
        if (x >= maxPW && x <= sumPW) limits.push_back(x);
    };

    int range = sumPW - maxPW + 1;
    if (range <= 600) {
        for (int x = maxPW; x <= sumPW; ++x) addLimit(x);
    } else {
        int step = max(1, range / 300);
        for (int x = maxPW; x < sumPW; x += step) addLimit(x);
        addLimit(sumPW);
        int guess = max(1, (int)sqrt((long double)paddedArea));
        for (int d = -20; d <= 20; ++d) addLimit(guess + d);
    }

    sort(limits.begin(), limits.end());
    limits.erase(unique(limits.begin(), limits.end()), limits.end());

    vector<vector<int>> orders(4);
    for (int t = 0; t < 4; ++t) {
        orders[t].resize(n);
        iota(orders[t].begin(), orders[t].end(), 0);
    }

    sort(orders[1].begin(), orders[1].end(), [&](int a, int b) {
        const auto& A = r.items[a];
        const auto& B = r.items[b];
        int ma = max(A.w, A.h), mb = max(B.w, B.h);
        if (ma != mb) return ma > mb;
        return 1LL * A.w * A.h > 1LL * B.w * B.h;
    });

    sort(orders[2].begin(), orders[2].end(), [&](int a, int b) {
        const auto& A = r.items[a];
        const auto& B = r.items[b];
        if (A.h != B.h) return A.h > B.h;
        return A.w > B.w;
    });

    sort(orders[3].begin(), orders[3].end(), [&](int a, int b) {
        const auto& A = r.items[a];
        const auto& B = r.items[b];
        if (A.w != B.w) return A.w > B.w;
        return A.h > B.h;
    });

    Layout best;
    for (const auto& ord : orders) {
        for (int lim : limits) {
            Layout cur = packShelf(r.items, ord, lim);
            if (best.w == INT_MAX || betterLayout(cur, best)) {
                best = move(cur);
            }
        }
    }

    r.w = best.w;
    r.h = best.h;
    r.p = move(best.p);
    return r;
}

struct Builder {
    int n, root;
    const vector<vector<int>>& children;
    const vector<vector<int>>& pockets;
    const vector<int>& order;
    vector<Region> regions;

    Builder(int n_, int root_, const vector<vector<int>>& children_,
            const vector<vector<int>>& pockets_, const vector<int>& order_)
        : n(n_), root(root_), children(children_), pockets(pockets_),
          order(order_) {
        regions.resize(n);
    }

    void draw(int v, int r0, int c0, vector<vector<int>>& C) {
        const Region& r = regions[v];

        for (int i = 0; i < r.h; ++i) {
            for (int j = 0; j < r.w; ++j) {
                C[r0 + i][c0 + j] = v + 1;
            }
        }

        for (const auto& pl : r.p) {
            const Item& it = r.items[pl.idx];
            if (it.child >= 0) {
                draw(it.child, r0 + pl.y, c0 + pl.x, C);
            } else {
                C[r0 + pl.y][c0 + pl.x] = it.color + 1;
            }
        }
    }

    vector<vector<int>> run() {
        for (int oi = (int)order.size() - 1; oi >= 0; --oi) {
            int v = order[oi];
            vector<Item> items;

            for (int c : children[v]) {
                items.push_back({regions[c].w, regions[c].h, c, -1});
            }
            for (int c : pockets[v]) {
                items.push_back({1, 1, -1, c});
            }

            regions[v] = makeRegion(move(items));
        }

        int K = max(regions[root].w, regions[root].h);
        vector<vector<int>> C(K, vector<int>(K, root + 1));
        draw(root, 0, 0, C);
        return C;
    }
};

}  // namespace WM

vector<vector<int>> create_map(int N, int M, vector<int> A, vector<int> B) {
    if (N == 1) return {{1}};

    for (int& x : A) --x;
    for (int& x : B) --x;

    vector<vector<char>> adj(N, vector<char>(N, false));
    vector<vector<pair<int, int>>> g(N);

    for (int i = 0; i < M; ++i) {
        int u = A[i], v = B[i];
        adj[u][v] = adj[v][u] = true;
        g[u].push_back({v, i});
        g[v].push_back({u, i});
    }

    auto valid = [&](const vector<vector<int>>& C) {
        int K = (int)C.size();
        if (K <= 0 || K > 240) return false;

        for (const auto& row : C) {
            if ((int)row.size() != K) return false;
            for (int x : row) {
                if (x < 1 || x > N) return false;
            }
        }

        vector<char> present(N, false);
        vector<vector<char>> seen(N, vector<char>(N, false));

        for (int i = 0; i < K; ++i) {
            for (int j = 0; j < K; ++j) {
                present[C[i][j] - 1] = true;
            }
        }

        for (int i = 0; i < K; ++i) {
            for (int j = 0; j < K; ++j) {
                if (i + 1 < K) {
                    int u = C[i][j] - 1, v = C[i + 1][j] - 1;
                    if (u != v) {
                        if (!adj[u][v]) return false;
                        seen[u][v] = seen[v][u] = true;
                    }
                }
                if (j + 1 < K) {
                    int u = C[i][j] - 1, v = C[i][j + 1] - 1;
                    if (u != v) {
                        if (!adj[u][v]) return false;
                        seen[u][v] = seen[v][u] = true;
                    }
                }
            }
        }

        for (int v = 0; v < N; ++v) {
            if (!present[v]) return false;
        }
        for (int i = 0; i < M; ++i) {
            if (!seen[A[i]][B[i]]) return false;
        }
        return true;
    };

    vector<vector<int>> best;
    auto consider = [&](const vector<vector<int>>& C) {
        if ((best.empty() || C.size() < best.size()) && valid(C)) {
            best = C;
        }
    };

    int lowerBound = 1;
    while (lowerBound * lowerBound < N ||
           2 * lowerBound * (lowerBound - 1) < M) {
        ++lowerBound;
    }

    // Short walks covering all edges give diagonal-stripe maps.
    vector<vector<int>> dist(N, vector<int>(N, -1));
    vector<vector<int>> nxt(N, vector<int>(N, -1));

    for (int t = 0; t < N; ++t) {
        queue<int> q;
        q.push(t);
        dist[t][t] = 0;
        nxt[t][t] = t;

        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (auto [v, id] : g[u]) {
                if (dist[v][t] == -1) {
                    dist[v][t] = dist[u][t] + 1;
                    nxt[v][t] = u;
                    q.push(v);
                }
            }
        }
    }

    vector<int> bestWalk;

    for (int start = 0; start < N; ++start) {
        vector<char> used(M, false);
        int covered = 0;
        int cur = start;
        vector<int> walk = {start};

        auto hasUnused = [&](int v) {
            for (auto [to, id] : g[v]) {
                if (!used[id]) return true;
            }
            return false;
        };

        while (covered < M) {
            int chosen = -1, to = -1;
            for (auto [v, id] : g[cur]) {
                if (!used[id]) {
                    chosen = id;
                    to = v;
                    break;
                }
            }

            if (chosen != -1) {
                used[chosen] = true;
                ++covered;
                cur = to;
                walk.push_back(cur);
                continue;
            }

            int target = -1;
            for (int v = 0; v < N; ++v) {
                if (!hasUnused(v)) continue;
                if (target == -1 ||
                    dist[cur][v] < dist[cur][target]) {
                    target = v;
                }
            }

            while (cur != target) {
                cur = nxt[cur][target];
                walk.push_back(cur);
            }
        }

        if (bestWalk.empty() || walk.size() < bestWalk.size()) {
            bestWalk = move(walk);
        }
    }

    if (!bestWalk.empty()) {
        int L = (int)bestWalk.size();
        int K = (L + 2) / 2;

        if (K <= 240 && (best.empty() || K < (int)best.size())) {
            vector<vector<int>> C(K, vector<int>(K));
            for (int i = 0; i < K; ++i) {
                for (int j = 0; j < K; ++j) {
                    C[i][j] = bestWalk[min(i + j, L - 1)] + 1;
                }
            }
            consider(C);
        }
    }

    if (!best.empty() && (int)best.size() == lowerBound) return best;

    // A particularly compact construction for complete graphs.
    if (M == N * (N - 1) / 2) {
        auto prime = [](int x) {
            if (x < 2) return false;
            for (int d = 2; d * d <= x; ++d) {
                if (x % d == 0) return false;
            }
            return true;
        };

        int p = N;
        while (!prime(p)) ++p;

        if (p <= 240) {
            vector<vector<int>> C(p, vector<int>(p));
            for (int i = 0; i < p; ++i) {
                for (int j = 0; j < p; ++j) {
                    int c = (i * j) % p;
                    if (c >= N) c = 0;
                    C[i][j] = c + 1;
                }
            }
            consider(C);
        }
    }

    if (!best.empty() && (int)best.size() == lowerBound) return best;

    auto nestedMap = [&](int root, int mode) {
        vector<int> parent(N, -1), depth(N, 0), order;
        vector<vector<int>> children(N);
        queue<int> q;

        parent[root] = root;
        q.push(root);

        while (!q.empty()) {
            int u = q.front();
            q.pop();
            order.push_back(u);

            for (auto [v, id] : g[u]) {
                if (parent[v] == -1) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    children[u].push_back(v);
                    q.push(v);
                }
            }
        }

        vector<vector<int>> pockets(N);
        vector<int> load(N, 0);

        for (int i = 0; i < M; ++i) {
            int u = A[i], v = B[i];
            if (parent[u] == v || parent[v] == u) continue;

            int host, other;

            auto balancedChoice = [&]() {
                if (load[u] < load[v]) {
                    host = u;
                    other = v;
                } else if (load[v] < load[u]) {
                    host = v;
                    other = u;
                } else {
                    host = u;
                    other = v;
                }
            };

            if (mode == 0 && depth[u] != depth[v]) {
                if (depth[u] > depth[v]) {
                    host = u;
                    other = v;
                } else {
                    host = v;
                    other = u;
                }
            } else if (mode == 1 && depth[u] != depth[v]) {
                if (depth[u] < depth[v]) {
                    host = u;
                    other = v;
                } else {
                    host = v;
                    other = u;
                }
            } else {
                balancedChoice();
            }

            pockets[host].push_back(other);
            ++load[host];
        }

        WM::Builder builder(N, root, children, pockets, order);
        return builder.run();
    };

    int root = 0;
    for (int v = 1; v < N; ++v) {
        if (g[v].size() > g[root].size()) root = v;
    }

    vector<vector<int>> fallback;

    for (int mode = 0; mode < 3; ++mode) {
        auto C = nestedMap(root, mode);
        if (fallback.empty() || C.size() < fallback.size()) fallback = C;
        consider(C);

        if (!best.empty() && (int)best.size() == lowerBound) return best;
        if (M == N - 1) break;
    }

    // This is only needed for unusually unfavourable rectangle layouts.
    if (best.empty()) {
        for (int r = 0; r < N; ++r) {
            for (int mode = 0; mode < 3; ++mode) {
                auto C = nestedMap(r, mode);
                if (fallback.empty() || C.size() < fallback.size()) {
                    fallback = C;
                }
                consider(C);
                if (!best.empty()) break;
            }
            if (!best.empty()) break;
        }
    }

    if (!best.empty()) return best;
    return fallback;
}

#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> create_map(int N, int M, vector<int> A, vector<int> B) {
    if (N == 1) return {{1}};

    vector<vector<int>> g(N + 1);
    for (int i = 0; i < M; ++i) {
        g[A[i]].push_back(B[i]);
        g[B[i]].push_back(A[i]);
    }

    auto normalize = [](int a, int b) {
        if (a > b) swap(a, b);
        return make_pair(a, b);
    };

    auto make_walk = [&](int root) {
        vector<int> walk;
        vector<bool> vis(N + 1, false);
        int seen = 0;

        function<void(int)> dfs = [&](int v) {
            vis[v] = true;
            ++seen;
            walk.push_back(v);
            if (seen == N) return;

            for (int to : g[v]) {
                if (!vis[to]) {
                    dfs(to);
                    if (seen == N) return;
                    walk.push_back(v);
                }
            }
        };

        dfs(root);
        return walk;
    };

    auto build_from_walk = [&](const vector<int>& walk) {
        int L = (int)walk.size();
        vector<int> first(N + 1, -1);
        for (int i = 0; i < L; ++i) {
            if (first[walk[i]] == -1) first[walk[i]] = i;
        }

        set<pair<int,int>> baseEdges;
        for (int i = 1; i < L; ++i) {
            baseEdges.insert(normalize(walk[i - 1], walk[i]));
        }

        vector<vector<int>> targets(N + 1);
        for (int i = 0; i < M; ++i) {
            int u = A[i], v = B[i];
            if (baseEdges.count(normalize(u, v))) continue;
            if (first[u] < first[v]) targets[u].push_back(v);
            else targets[v].push_back(u);
        }

        auto generate = [&](vector<int> h, vector<int> w) {
            int sh = accumulate(h.begin(), h.end(), 0);
            int sw = accumulate(w.begin(), w.end(), 0);
            if (sh != sw) return vector<vector<int>>{};

            vector<int> ro(L + 1, 0), co(L + 1, 0);
            for (int i = 0; i < L; ++i) {
                ro[i + 1] = ro[i] + h[i];
                co[i + 1] = co[i] + w[i];
            }

            int K = sh;
            vector<vector<int>> C(K, vector<int>(K));
            for (int i = 0; i < L; ++i) {
                for (int j = 0; j < L; ++j) {
                    int color = walk[min(i, j)];
                    for (int di = 0; di < h[i]; ++di) {
                        for (int dj = 0; dj < w[j]; ++dj) {
                            C[ro[i] + di][co[j] + dj] = color;
                        }
                    }
                }
            }

            vector<vector<pair<int,int>>> slots(N + 1);
            for (int v = 1; v <= N; ++v) {
                int p = first[v];

                if (h[p] == 3) {
                    for (int j = p + 1; j < L; j += 2) {
                        slots[v].push_back({ro[p] + 1, co[j]});
                    }
                }
                if (w[p] == 3) {
                    for (int i = p + 1; i < L; i += 2) {
                        slots[v].push_back({ro[i], co[p] + 1});
                    }
                }
            }

            for (int v = 1; v <= N; ++v) {
                if (slots[v].size() < targets[v].size()) {
                    return vector<vector<int>>{};
                }
                for (int i = 0; i < (int)targets[v].size(); ++i) {
                    auto [r, c] = slots[v][i];
                    C[r][c] = targets[v][i];
                }
            }
            return C;
        };

        vector<int> fullH(L, 3), fullW(L, 3);
        auto fullMap = generate(fullH, fullW);

        vector<int> h(L, 1), w(L, 1);
        for (int v = 1; v <= N; ++v) {
            int need = (int)targets[v].size();
            if (need == 0) continue;

            int p = first[v];
            int q = L - p - 1;
            if (need <= (q + 1) / 2) {
                h[p] = 3;
            } else {
                h[p] = w[p] = 3;
            }
        }

        int sh = accumulate(h.begin(), h.end(), 0);
        int sw = accumulate(w.begin(), w.end(), 0);

        while (sh < sw) {
            int p = 0;
            while (p < L && h[p] == 3) ++p;
            if (p == L) break;
            h[p] = 3;
            sh += 2;
        }
        while (sw < sh) {
            int p = 0;
            while (p < L && w[p] == 3) ++p;
            if (p == L) break;
            w[p] = 3;
            sw += 2;
        }

        auto selectiveMap = generate(h, w);
        if (!selectiveMap.empty() &&
            (fullMap.empty() || selectiveMap.size() < fullMap.size())) {
            return selectiveMap;
        }
        return fullMap;
    };

    vector<vector<int>> answer;
    for (int root = 1; root <= N; ++root) {
        vector<int> walk = make_walk(root);
        if ((int)walk.size() < N) continue;

        auto candidate = build_from_walk(walk);
        if (!candidate.empty() &&
            (answer.empty() || candidate.size() < answer.size())) {
            answer = move(candidate);
        }
    }

    return answer;
}

using Mask = uint64_t;

class MaxCliqueSolver {
    int n;
    vector<Mask> adj;
    int bestSize = 0;
    Mask bestSet = 0;

    void colorSort(Mask candidates, vector<int>& order, vector<int>& colors) const {
        order.clear();
        colors.clear();

        Mask uncolored = candidates;
        int color = 0;

        while (uncolored) {
            ++color;
            Mask available = uncolored;

            while (available) {
                int v = __builtin_ctzll(available);
                Mask bit = Mask(1) << v;

                order.push_back(v);
                colors.push_back(color);

                uncolored &= ~bit;
                available &= ~bit;
                available &= ~adj[v];
            }
        }
    }

    void expand(Mask candidates, int size, Mask chosen) {
        if (size > bestSize) {
            bestSize = size;
            bestSet = chosen;
        }

        if (!candidates) return;

        vector<int> order, colors;
        colorSort(candidates, order, colors);

        for (int i = static_cast<int>(order.size()) - 1; i >= 0; --i) {
            if (size + colors[i] <= bestSize) return;

            int v = order[i];
            Mask bit = Mask(1) << v;
            Mask next = candidates & adj[v];

            expand(next, size + 1, chosen | bit);
            candidates &= ~bit;
        }
    }

public:
    explicit MaxCliqueSolver(vector<Mask> graph)
        : n(static_cast<int>(graph.size())), adj(std::move(graph)) {}

    Mask solve() {
        bestSize = 0;
        bestSet = 0;
        Mask all = (Mask(1) << n) - 1;
        expand(all, 0, 0);
        return bestSet;
    }
};

static vector<int> minimumVertexCover(const vector<Mask>& graph) {
    int n = static_cast<int>(graph.size());
    if (n == 0) return {};

    Mask all = (Mask(1) << n) - 1;
    Mask edgeCount = 0;

    for (Mask neighbors : graph) {
        edgeCount += __builtin_popcountll(neighbors);
    }

    if ((edgeCount & 1ULL) == 0 && edgeCount == 0) return {};

    vector<Mask> complement(n);
    for (int v = 0; v < n; ++v) {
        Mask bit = Mask(1) << v;
        complement[v] = all & ~(graph[v] | bit);
    }

    MaxCliqueSolver clique(std::move(complement));
    Mask independentSet = clique.solve();
    Mask cover = all & ~independentSet;

    vector<int> result;
    for (int v = 0; v < n; ++v) {
        if (cover & (Mask(1) << v)) result.push_back(v);
    }
    return result;
}

            q.push(source);

                        parent[u] = v;
                        q.push(u);
                    }
                }

            for (int v = 0; v < n; ++v) {
                if (distance[v] > distance[farthest]) farthest = v;

        };

        auto result = bfs(endpoint);

            path.push_back(other);
            other = parent[other];

        for (int v : cover) coverMask |= Mask(1) << v;

        struct Edge {
            int u, v;
            int options;

        for (int u = 0; u < n; ++u) {

                int options = 0;
                if (coverMask & (Mask(1) << u)) ++options;
                if (coverMask & (Mask(1) << v)) ++options;

                if (options > 0) edgeList.push_back({u, v, options});
            }

             [](const Edge& a, const Edge& b) {

            bool canU = coverMask & (Mask(1) << u);
            bool canV = coverMask & (Mask(1) << v);

            if (!canU && !canV) return {};

            int chosen;
            if (!canU) {
                chosen = v;
            } else if (!canV) {
                chosen = u;
            } else if (load[u] != load[v]) {
                chosen = load[u] < load[v] ? u : v;
            } else {
                int du = __builtin_popcountll(edges[u]);
                int dv = __builtin_popcountll(edges[v]);
                chosen = du >= dv ? u : v;
            }

            assigned[i] = chosen;
            ++load[chosen];
        }

        bool changed = true;
        while (changed) {
            changed = false;

            for (int i = 0; i < static_cast<int>(edgeList.size()); ++i) {
                int a = assigned[i];
                int b = edgeList[i].u ^ edgeList[i].v ^ a;

                if ((coverMask & (Mask(1) << b)) && load[a] > load[b] + 1) {
                    --load[a];
                    ++load[b];
                    assigned[i] = b;
                    changed = true;
                }

        Orientation result;
        result.load.assign(n, 0);
        result.targets.resize(n);

        for (int i = 0; i < static_cast<int>(edgeList.size()); ++i) {
            int a = assigned[i];
            int b = edgeList[i].u ^ edgeList[i].v ^ a;
            result.targets[a].push_back(b);
            ++result.load[a];
        }

        return result;

    bool findHamiltonianPath(mt19937& rng, vector<int>& path) const {

        if (leaves > 2) return false;

        vector<int> starts(n);
        iota(starts.begin(), starts.end(), 0);
        shuffle(starts.begin(), starts.end(), rng);

            vector<char> visited(n, false);
            path.clear();
            path.push_back(start);
            visited[start] = true;
            int nodes = 0;
            const int LIMIT = 300000;

            function<bool(int)> dfs = [&](int v) {
                if (++nodes > LIMIT) return false;
                if (static_cast<int>(path.size()) == n) return true;

                vector<pair<int, int>> candidates;
                for (int u : adjacency[v]) {
                    if (visited[u]) continue;

                    int remaining = 0;
                    for (int w : adjacency[u]) {
                        if (!visited[w]) ++remaining;
                    }
                    candidates.push_back({remaining, u});
                }

                sort(candidates.begin(), candidates.end());

                for (auto [remaining, u] : candidates) {
                    if (nodes > LIMIT) break;
                    visited[u] = true;
                    path.push_back(u);

                    if (dfs(u)) return true;

                    path.pop_back();
                    visited[u] = false;
                }

                return false;
            };

            if (dfs(start)) return true;
        }

        return false;
    }

static vector<pair<int, int>> roundRobinMatching(int size, int round) {
    vector<int> arrangement(size - 1);
    iota(arrangement.begin(), arrangement.end(), 0);

    int fixed = size - 1;
    vector<pair<int, int>> matching;
    matching.push_back({fixed, arrangement[round]});

    for (int i = 1; i <= (size - 2) / 2; ++i) {
        int a = arrangement[(round + i) % (size - 1)];
        int b = arrangement[(round - i + size - 1) % (size - 1)];
        matching.push_back({a, b});
    }

    return matching;
}

static vector<vector<int>> completeGraphMap(int n) {
    vector<vector<int>> result(n, vector<int>(n));

    if (n % 2 == 0) {
        for (int round = 0; round < n - 1; ++round) {
            auto matching = roundRobinMatching(n, round);
            int column = 0;

            for (auto [a, b] : matching) {
                result[round][column++] = a;
                result[round][column++] = b;
            }
        }

        for (int column = 0; column < n; ++column) {
            result[n - 1][column] = 0;
        }
    } else {
        for (int round = 0; round < n; ++round) {
            auto matching = roundRobinMatching(n + 1, round);
            int column = 0;
            int singleton = -1;

            for (auto [a, b] : matching) {
                if (a == n) {
                    singleton = b;
                } else if (b == n) {
                    singleton = a;
                } else {
                    result[round][column++] = a;
                    result[round][column++] = b;
                }
            }

            if (singleton != -1) result[round][column++] = singleton;
        }
    }

    return result;
}

static vector<vector<int>> starMap(int n) {
    int side = 2;
    while ((side * side) / 2 < n - 1) ++side;

    vector<vector<int>> result(side, vector<int>(side, 0));
    int nextLeaf = 1;

    for (int r = 0; r < side; ++r) {
        for (int c = 0; c < side; ++c) {
            if ((r + c) & 1) {
                result[r][c] = nextLeaf;
                if (++nextLeaf == n) nextLeaf = 1;
            }
        }
    }

    return result;
}

std::vector<std::vector<int>> create_map(int N, int M,
                                          std::vector<int> A,
                                          std::vector<int> B) {
    vector<Mask> adjacency(N, 0);

    for (int i = 0; i < M; ++i) {
        int u = A[i] - 1;
        int v = B[i] - 1;
        adjacency[u] |= Mask(1) << v;
        adjacency[v] |= Mask(1) << u;
    }

    bool complete = (M == N * (N - 1) / 2);
    if (complete) {
        for (int u = 0; u < N && complete; ++u) {
            if (__builtin_popcountll(adjacency[u]) != N - 1) complete = false;
        }
    }

    if (complete) {
        vector<vector<int>> answer = completeGraphMap(N);
        for (auto& row : answer) {
            for (int& color : row) ++color;
        }
        return answer;
    }

    int maximumDegree = 0;
    for (int v = 0; v < N; ++v) {
        maximumDegree = max(maximumDegree,
                            __builtin_popcountll(adjacency[v]));
    }

    if (N >= 3 && M == N - 1 && maximumDegree == N - 1) {
        vector<vector<int>> answer = starMap(N);
        for (auto& row : answer) {
            for (int& color : row) ++color;
        }
        return answer;
    }

    GeneralMapSolver solver(N, adjacency);
    vector<vector<int>> answer = solver.solve();

    for (auto& row : answer) {
        for (int& color : row) ++color;
    }

    return answer;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N, M;
        cin >> N >> M;

        vector<int> A(M), B(M);
        for (int i = 0; i < M; ++i) {
            cin >> A[i] >> B[i];
        }

        vector<vector<int>> C = create_map(N, M, A, B);
        int K = static_cast<int>(C.size());

        cout << K << "\n\n";

        for (int r = 0; r < K; ++r) {
            for (int c = 0; c < K; ++c) {
                if (c) cout << ' ';
                cout << C[r][c];
            }
            cout << '\n';
        }
    }

    return 0;
}
