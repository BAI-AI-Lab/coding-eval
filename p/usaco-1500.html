using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1);
    int maxValue = 0;
    for (int i = 1; i <= N; ++i) {
        cin >> a[i];
        maxValue = max(maxValue, a[i]);
    }

    // Positions are bucketed by value.
    vector<int> head(maxValue + 1, -1), nextInBucket(N + 1, -1);
    for (int i = 1; i <= N; ++i) {
        nextInBucket[i] = head[a[i]];
        head[a[i]] = i;
    }

    auto gapEven = [](int g) -> int64 {
        return 1LL * (g / 2) * ((g + 1) / 2);
    };

    auto gapOdd = [&](int g) -> int64 {
        return 1LL * g * (g + 1) / 2 - gapEven(g);
    };

    // Number of even-length subarrays containing p, with left endpoint in
    // [P+1,p] and right endpoint in [p,Q-1].
    auto oneActiveEven = [](int p, int P, int Q) -> int64 {
        int64 leftEven = p / 2 - P / 2;
        int64 leftOdd = (p + 1) / 2 - (P + 1) / 2;
        int64 rightEven = (Q - 1) / 2 - (p - 1) / 2;
        int64 rightOdd = Q / 2 - p / 2;
        return leftEven * rightOdd + leftOdd * rightEven;
    };

    const int64 totalSubarrays = 1LL * N * (N + 1) / 2;
    const int64 totalEven = gapEven(N);
    const int64 totalOdd = totalSubarrays - totalEven;

    // Initially every position is active.
    vector<int> left(N + 2), right(N + 2);
    for (int i = 1; i <= N; ++i) {
        left[i] = i - 1;
        right[i] = i + 1;
    }
    right[0] = 1;
    left[N + 1] = N;

    int64 zeroEven = 0, zeroOdd = 0, exactlyOneEven = 0;
    int64 sumMaximumOdd = 0, sumSecondMaximumEven = 0;
    int previousValue = 0;

    // Process thresholds increasingly, removing positions below the next threshold.
    for (int value = 1; value <= maxValue; ++value) {
        if (head[value] == -1) continue;

        int64 withActiveOdd = totalOdd - zeroOdd;
        int64 withTwoActiveEven = totalEven - zeroEven - exactlyOneEven;

        int64 delta = value - previousValue;
        sumMaximumOdd += delta * withActiveOdd;
        sumSecondMaximumEven += delta * withTwoActiveEven;
        previousValue = value;

        for (int p = head[value]; p != -1; p = nextInBucket[p]) {
            int P = left[p], Q = right[p];

            exactlyOneEven -= oneActiveEven(p, P, Q);

            if (P != 0) {
                int PP = left[P];
                exactlyOneEven -= oneActiveEven(P, PP, p);
                exactlyOneEven += oneActiveEven(P, PP, Q);
            }
            if (Q != N + 1) {
                int QQ = right[Q];
                exactlyOneEven -= oneActiveEven(Q, p, QQ);
                exactlyOneEven += oneActiveEven(Q, P, QQ);
            }

            int oldGapLeft = p - P - 1;
            int oldGapRight = Q - p - 1;
            int mergedGap = Q - P - 1;

            zeroEven += gapEven(mergedGap)
                      - gapEven(oldGapLeft) - gapEven(oldGapRight);
            zeroOdd += gapOdd(mergedGap)
                     - gapOdd(oldGapLeft) - gapOdd(oldGapRight);

            right[P] = Q;
            left[Q] = P;
        }
    }

    /*
       For odd lengths at least 5, the answer is the subarray maximum.
       For even lengths at least 8, it is the second maximum.
       Lengths 1 and 2 already agree with these formulas.
       Only lengths 3, 4, and 6 need corrections.
    */
    int64 correction = 0;

    for (int i = 1; i + 2 <= N; ++i) {
        int x = a[i], y = a[i + 1], z = a[i + 2];
        int rangeMaximum = max(x, max(y, z));
        int answer = max(max(min(x, y), z), max(x, min(y, z)));
        correction += answer - rangeMaximum;
    }

    for (int i = 1; i + 3 <= N; ++i) {
        int v[4] = {a[i], a[i + 1], a[i + 2], a[i + 3]};
        sort(v, v + 4);
        int second = v[2], third = v[1];

        if (second > third) {
            int positions[2], count = 0;
            for (int j = 0; j < 4; ++j) {
                if (a[i + j] >= second) positions[count++] = j + 1;
            }
            int p = positions[0], q = positions[1];
            bool losing = (p == 1 && q == 3)
                       || (p == 2 && q == 3)
                       || (p == 2 && q == 4);
            if (losing) correction += (int64)third - second;
        }
    }

    for (int i = 1; i + 5 <= N; ++i) {
        int v[6] = {a[i], a[i + 1], a[i + 2],
                    a[i + 3], a[i + 4], a[i + 5]};
        sort(v, v + 6);
        int second = v[4], third = v[3];

            for (int j = 0; j < 6; ++j) {
                if (a[i + j] >= second) positions[count++] = j + 1;

            int p = positions[0], q = positions[1];
            bool losing = (p == 2 && q == 4)
                       || (p == 2 && q == 5)
                       || (p == 3 && q == 5);
            if (losing) correction += (int64)third - second;
        }
    }

    cout << sumMaximumOdd + sumSecondMaximumEven + correction << '\n';
    return 0;

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

static ll smallExcluded[7][7];

// Number of subarrays of an inactive gap having lengths
// first, first+2, first+4, ... .
static inline ll gapCount(ll d, int first) {
    if (d < first) return 0;
    ll k = (d - first) / 2 + 1;
    return k * (d - first + 1) - k * (k - 1);
}

// A specified active position is at distance A from the active position
// on its left and B from the one on its right. Count even-length (>=8)
// subarrays containing this position and no other active position.
static inline ll oneActiveCount(ll A, ll B) {
    ll evenA = (A + 1) / 2, oddA = A / 2;
    ll evenB = (B + 1) / 2, oddB = B / 2;

    // If u+v is odd, the corresponding subarray has even length.
    ll result = evenA * oddB + oddA * evenB;

    int ia = A >= 6 ? 6 : (int)A;
    int ib = B >= 6 ? 6 : (int)B;

    // Exclude lengths 2, 4 and 6.
    result -= smallExcluded[ia][ib];
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1);
    int maxA = 0;
    for (int i = 1; i <= N; ++i) {
        cin >> a[i];
        maxA = max(maxA, a[i]);
    }

    // Precompute exclusions for sums u+v = 1,3,5. Only A,B <= 6 matter.
    for (int A = 1; A <= 6; ++A) {
        for (int B = 1; B <= 6; ++B) {
            for (int s : {1, 3, 5}) {
                int lo = max(0, s - (B - 1));
                int hi = min(s, A - 1);
                if (lo <= hi) smallExcluded[A][B] += hi - lo + 1;
            }
        }
    }

    ll answer = 0;

    // Length 1.
    for (int i = 1; i <= N; ++i) answer += a[i];

    // Length 3: the answer is the maximum of the two endpoints.
    for (int i = 1; i + 2 <= N; ++i) {
        answer += max(a[i], a[i + 2]);
    }

    // Lengths 2, 4 and 6. For two distinguished elements p<q, let the
    // three gaps around/between them have sizes L,M,R. The pair works
    // exactly when the number of nonempty gaps is at most len/2-1.
    auto addEvenLength = [&](int len) {
        if (len > N) return;

        int rounds = len / 2 - 1;
        vector<pair<int, int>> pairs;

        for (int p = 0; p < len; ++p) {
            for (int q = p + 1; q < len; ++q) {
                int nonemptyGaps =
                    (p > 0) +
                    (q - p - 1 > 0) +
                    (len - 1 - q > 0);

                if (nonemptyGaps <= rounds) {
                    pairs.emplace_back(p, q);
                }
            }
        }

        for (int start = 1; start + len - 1 <= N; ++start) {
            int best = 0;
            for (auto [p, q] : pairs) {
                best = max(best, min(a[start + p], a[start + q]));
            }
            answer += best;
        }
    };

    addEvenLength(2);
    addEvenLength(4);
    addEvenLength(6);

    /*
       Remaining subarrays:
         - odd length >= 5: answer is the subarray maximum;
         - even length >= 8: answer is the subarray second maximum.

       For a threshold x, call positions with value >= x active. We need:
         - odd long subarrays containing at least one active position;
         - even long subarrays containing at least two active positions.

       Summing these counts over x=1..max(a) gives the required sum.
       Active positions are maintained under deletions in a linked list.
    */

    const ll totalOdd = gapCount(N, 5);
    const ll totalEven = gapCount(N, 8);

    ll zeroOdd = 0;       // Qualifying odd subarrays with no active position.
    ll zeroEven = 0;      // Qualifying even subarrays with no active position.
    ll exactlyOne = 0;    // Qualifying even subarrays with exactly one active.

    vector<int> prv(N + 2), nxt(N + 2);
    for (int i = 0; i <= N + 1; ++i) {
        prv[i] = i - 1;
        nxt[i] = i + 1;
    }

    // Buckets of positions by value, using linked lists.
    vector<int> head(N + 1, -1), bucketNext(N + 1, -1);
    for (int i = 1; i <= N; ++i) {
        bucketNext[i] = head[a[i]];
        head[a[i]] = i;
    }

    for (int x = 1; x <= maxA; ++x) {
        answer += totalOdd - zeroOdd;
        answer += totalEven - zeroEven - exactlyOne;

        if (x == maxA) break;

        // Delete positions whose value is x, preparing threshold x+1.
        for (int i = head[x]; i != -1; i = bucketNext[i]) {
            int p = prv[i];
            int q = nxt[i];

            ll A = i - p;
            ll B = q - i;

            // Merge the two inactive gaps, with position i joining them.
            zeroOdd += gapCount(A + B - 1, 5)
                     - gapCount(A - 1, 5)
                     - gapCount(B - 1, 5);

            zeroEven += gapCount(A + B - 1, 8)
                      - gapCount(A - 1, 8)
                      - gapCount(B - 1, 8);

            // Remove i's exactly-one-active contribution.
            exactlyOne -= oneActiveCount(A, B);

            // Only the nearest active neighbors change their contributions.
            if (p != 0) {
                ll leftDistance = p - prv[p];
                exactlyOne -= oneActiveCount(leftDistance, A);
                exactlyOne += oneActiveCount(leftDistance, A + B);
            }

            if (q != N + 1) {
                ll rightDistance = nxt[q] - q;
                exactlyOne -= oneActiveCount(B, rightDistance);
                exactlyOne += oneActiveCount(A + B, rightDistance);
            }

            nxt[p] = q;
            prv[q] = p;
        }
    }

    cout << answer << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

static int lowPairs[7][7];

// Number of odd-length subarrays of length at least 5 in a segment of length g.
static inline int64 oddFamily(int g) {
    if (g < 5) return 0;
    int64 h = (g + 1) / 2;
    int64 allOdd = h * (g + 1 - h);
    return allOdd - g - (g - 2);
}

// Number of even-length subarrays of length at least 8 in a segment of length g.
static inline int64 evenFamily(int g) {
    if (g < 8) return 0;
    int64 h = g / 2;
    int64 allEven = h * (g - h);
    return allEven - (g - 1) - (g - 3) - (g - 5);
}

/*
   Count subarrays containing one distinguished active position, where:
   L left endpoints and R right endpoints are available.
   The subarray must have even length at least 8.

   If x and y are the numbers of positions used on the two sides, then
   x+y must be odd and at least 7.
*/
static inline int64 oneActiveEven(int L, int R) {
    if (L + R <= 8) return 0;

    int64 evenL = (L + 1LL) / 2;
    int64 oddL = L / 2;
    int64 evenR = (R + 1LL) / 2;
    int64 oddR = R / 2;

    int64 result = evenL * oddR + oddL * evenR;
    result -= lowPairs[min(L, 6)][min(R, 6)];
    return result;
}

                for (int y = 0; y <= 5; ++y) {
                    if (x < L && y < R && ((x + y) & 1) && x + y < 7) {
                        ++lowPairs[L][R];
                    }
                }
            }
        }
    }

    int N;
    if (!(cin >> N)) return 0;

    vector<int> a(N + 1);
    vector<int> head(N + 1, -1), bucketNext(N + 1, -1);

    int maxA = 0;

        cin >> a[i];
        maxA = max(maxA, a[i]);
        bucketNext[i] = head[a[i]];
        head[a[i]] = i;
    }

    int64 answer = 0;

    // Length 1.
    for (int i = 1; i <= N; ++i) {
        answer += a[i];
    }

    // Length 2.
    for (int i = 1; i + 1 <= N; ++i) {
        answer += min(a[i], a[i + 1]);
    }

    // Length 3.
    for (int i = 1; i + 2 <= N; ++i) {
        answer += max(a[i], a[i + 2]);

    // Length 4.
    for (int i = 1; i + 3 <= N; ++i) {
        int cur = min(a[i], a[i + 1]);
        cur = max(cur, min(a[i], a[i + 3]));
        cur = max(cur, min(a[i + 2], a[i + 3]));
        answer += cur;
    }

    /*
       Length 6. A pair of positions can supply the final two numbers iff:
       - it contains the first position, or
       - it contains the last position, or
       - its two positions are adjacent.

    for (int i = 1; i + 5 <= N; ++i) {

        int maxBeforeLast = max(max(x0, x1), max(x2, x3));
        maxBeforeLast = max(maxBeforeLast, x4);

        adjacent = max(adjacent, min(x2, x3));
        adjacent = max(adjacent, min(x3, x4));
        adjacent = max(adjacent, min(x4, x5));

        int cur = max(min(x0, maxAfterFirst),
                      min(maxBeforeLast, x5));
        cur = max(cur, adjacent);
        answer += cur;
    }

    /*
       For odd lengths at least 5, the answer is the subarray maximum.
       For even lengths at least 8, it is the second maximum.

       Sweep the threshold v. Positions with value at least v are active.
       A maximum is at least v iff a subarray contains an active position;
       a second maximum is at least v iff it contains at least two.
    */
    if (N >= 5) {
        const int64 totalOdd = oddFamily(N);
        const int64 totalEven = evenFamily(N);
        const bool needEven = (totalEven != 0);

        vector<int> prev(N + 2), next(N + 2);
        for (int i = 0; i <= N + 1; ++i) {
            prev[i] = i - 1;
            next[i] = i + 1;
        }

        int64 zeroOdd = 0;
        int64 zeroEven = 0;
        int64 exactlyOneEven = 0;

        for (int v = 1; v <= maxA; ++v) {
            answer += totalOdd - zeroOdd;
            if (needEven) {
                answer += totalEven - zeroEven - exactlyOneEven;
            }

            // No further threshold is needed, so avoid the final deletions.
            if (v == maxA) break;

            for (int i = head[v]; i != -1; i = bucketNext[i]) {
                int p = prev[i];
                int q = next[i];

                int gapLeft = i - p - 1;
                int gapRight = q - i - 1;
                int mergedGap = q - p - 1;

                zeroOdd += oddFamily(mergedGap)
                         - oddFamily(gapLeft)
                         - oddFamily(gapRight);

                if (needEven) {
                    zeroEven += evenFamily(mergedGap)
                              - evenFamily(gapLeft)
                              - evenFamily(gapRight);

                    // Remove the term belonging to position i.
                    exactlyOneEven -= oneActiveEven(i - p, q - i);

                    // p's next active position changes from i to q.
                    if (p != 0) {
                        int leftChoices = p - prev[p];
                        exactlyOneEven +=
                            oneActiveEven(leftChoices, q - p)
                          - oneActiveEven(leftChoices, i - p);
                    }

                    // q's previous active position changes from i to p.
                    if (q != N + 1) {
                        int rightChoices = next[q] - q;
                        exactlyOneEven +=
                            oneActiveEven(q - p, rightChoices)
                          - oneActiveEven(q - i, rightChoices);
                    }
                }

                next[p] = q;
                prev[q] = p;
            }
        }
    }

    cout << answer << '\n';
    return 0;
}

struct FastSet {

    int n;
    vector<int> sz;
    vector<vector<uint64_t>> bits;

        int cur = n;

            sz.push_back(cur);
            bits.emplace_back((cur + 63) / 64, 0);
            if (cur == 1) break;
            cur = (cur + 63) / 64;
        }
    }

    void insert(int x) {
        for (int lev = 0; lev < (int)bits.size(); ++lev) {
            int w = x >> 6;
            uint64_t mask = 1ULL << (x & 63);
            uint64_t old = bits[lev][w];
            bits[lev][w] = old | mask;
            if (old != 0 || lev + 1 == (int)bits.size()) break;
            x = w;
        }
    }

    int nextAt(int lev, int x) const {
        if (x >= sz[lev]) return INF;
        int w = x >> 6;
        int b = x & 63;
        uint64_t mask = bits[lev][w] & (~0ULL << b);

        sum[v] = 1LL * (r - l + 1) * x;
        setv[v] = x;
        addv[v] = 0;

            applySet(v << 1 | 1, m + 1, r, setv[v]);
            setv[v] = 0;

            applyAdd(v << 1 | 1, m + 1, r, addv[v]);

        }

    for (int i = 1; i <= n; ++i) {
        ParityCounts left = parityCounts(L1[i] + 1, i);
        ParityCounts right = parityCounts(i, R1[i] - 1);

        // An odd-length interval has endpoints of the same parity.
        int64 count = left.even * right.even + left.odd * right.odd;
        answer += 1LL * a[i] * count;
    }

    // Sum of second maxima over all even-length subarrays.
    for (int i = 1; i <= n; ++i) {
        // The maximum is to the left of the second maximum.
        if (L1[i] != 0) {
            ParityCounts left = parityCounts(L2[i] + 1, L1[i]);
            ParityCounts right = parityCounts(i, R1[i] - 1);

            // An even-length interval has endpoints of opposite parity.
            int64 count = left.even * right.odd + left.odd * right.even;
            answer += 1LL * a[i] * count;
        }

        // The maximum is to the right of the second maximum.
        if (R1[i] != n + 1) {
            ParityCounts left = parityCounts(L1[i] + 1, i);
            ParityCounts right = parityCounts(R1[i], R2[i] - 1);

            int64 count = left.even * right.odd + left.odd * right.even;
            answer += 1LL * a[i] * count;
        }
    }

    // Correct length-3 subarrays:
    // actual answer = max(first, third), rather than max(first, second, third).
    for (int i = 1; i + 2 <= n; ++i) {
        int allThree = max({a[i], a[i + 1], a[i + 2]});
        int endpoints = max(a[i], a[i + 2]);
        answer -= allThree - endpoints;
    }

    // Correct length-4 subarrays.
    for (int i = 1; i + 3 <= n; ++i) {
        int special = max({
            min(a[i], a[i + 1]),
            min(a[i], a[i + 3]),
            min(a[i + 2], a[i + 3])
        });

        int baseline = secondLargest(a, i, 4);
        answer -= baseline - special;
    }

    // Correct length-6 subarrays.
    static constexpr int allowedPairs[12][2] = {
        {1, 2}, {1, 3}, {1, 4}, {1, 5}, {1, 6},
        {2, 3}, {2, 6},
        {3, 4}, {3, 6},
        {4, 5}, {4, 6},
        {5, 6}
    };

    for (int i = 1; i + 5 <= n; ++i) {
        int special = 0;
        for (const auto& p : allowedPairs) {
            special = max(special, min(a[i + p[0] - 1], a[i + p[1] - 1]));
        }

        int baseline = secondLargest(a, i, 6);
        answer -= baseline - special;
    }

    cout << answer << '\n';
    return 0;
}
