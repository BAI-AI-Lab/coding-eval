Cow Checkups

USACO 2025 January · 银组 · 2025-01 · L2 中等 · Python 3.11 · 时限 4 秒 · 测试点等分

import sys


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    a = data[1:n + 1]
    b = data[n + 1:n + 1 + 2 * n]
    del data

    total_operations = n * (n + 1) // 2
    answer = 0

    count_a = [0] * (n + 1)
    count_b = [0] * (n + 1)

    # Contributions from cows outside the reversed interval.
    for i in range(n):
        x = a[i]
        y = b[i]
        if x == y:
            position = i + 1
            answer += total_operations - position * (n - position + 1)
        count_a[x] += 1
        count_b[y] += 1

    # current is the number of equal-species pairs (i in b, j in a)
    # inside the current interval [t, n + 1 - t].
    current = 0
    for species in range(1, n + 1):
        current += count_a[species] * count_b[species]

    last_t = (n + 1) // 2
    for t in range(1, last_t + 1):
        answer += current
        if t == last_t:
            break

        # Shrink the interval by removing its two endpoints.
        left = t - 1
        right = n - t

        x = a[left]
        current -= count_b[x]
        count_a[x] -= 1
        y = b[left]
        current -= count_a[y]
        count_b[y] -= 1

        x = a[right]
        current -= count_b[x]
        count_a[x] -= 1
        y = b[right]
        current -= count_a[y]
        count_b[y] -= 1

    sys.stdout.write(str(answer) + "\n")


if __name__ == "__main__":
    solve()

import sys
from bisect import bisect_right


def main() -> None:
    input = sys.stdin.buffer.readline

    line = input()
    if not line:
        return
    N = int(line)

    a = list(map(int, input().split()))
    b = list(map(int, input().split()))

    # Group positions of a by species in a flat sorted-by-species array.
    count = [0] * (N + 2)
    for x in a:
        count[x] += 1

    start = [0] * (N + 2)
    total = 0
    for v in range(1, N + 1):
        start[v] = total
        total += count[v]
    start[N + 1] = total

    cursor = start[:]
    pos = [0] * N

    for i, x in enumerate(a, 1):
        p = cursor[x]
        pos[p] = i
        cursor[x] = p + 1

    del count, cursor

    # Prefix sums over the flat position array.
    pref = [0] * (N + 1)
    acc = 0
    for idx, p in enumerate(pos):
        acc += p
        pref[idx + 1] = acc

    num_operations = N * (N + 1) // 2

    # Contributions from positions that are not reversed.
    total_matches = 0
    removed = 0
    for i in range(1, N + 1):
        if a[i - 1] == b[i - 1]:
            total_matches += 1
            removed += i * (N - i + 1)

    answer = total_matches * num_operations - removed

    # Contributions from positions inside the reversed segment.
    N1 = N + 1
    for i, species in enumerate(b, 1):
        lo = start[species]
        hi = start[species + 1]

        if lo == hi:
            continue

        other = N1 - i
        if i < other:
            t1 = i
            tmax = other
        else:
            t1 = other
            tmax = i

        idx1 = bisect_right(pos, t1, lo, hi)
        idx2 = bisect_right(pos, tmax, idx1, hi)

        # j <= t1: contribution is j
        answer += pref[idx1] - pref[lo]

        # t1 < j <= tmax: contribution is t1
        answer += t1 * (idx2 - idx1)

        # j > tmax: contribution is N + 1 - j
        cnt = hi - idx2
        if cnt:
            answer += N1 * cnt - (pref[hi] - pref[idx2])

    sys.stdout.write(str(answer))


if __name__ == "__main__":
    main()

import sys


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    a = data[1:n + 1]
    b = data[n + 1:2 * n + 1]
    del data

    # Initially every position is active.
    active_count = [0] * (n + 1)
    total_depth = [0] * (n + 1)

    for i, species in enumerate(a):
        depth = i + 1
        if n - i < depth:
            depth = n - i

        active_count[species] += 1
        total_depth[species] += depth

    active_sum = total_depth.copy()

    total_intervals = n * (n + 1) // 2
    answer = 0

    # The two positions of depth k are k-1 and n-k (0-based).
    for k in range(1, (n + 1) // 2 + 1):
        left = k - 1
        right = n - k

        # Contributions for the left position.
        species = b[left]
        answer += (
            total_depth[species]
            - active_sum[species]
            + k * active_count[species]
        )

        # Reversals that do not contain this position.
        if a[left] == species:
            answer += total_intervals - (left + 1) * (n - left)

        # Contributions for the right position.
        if right != left:
            species = b[right]
            answer += (
                total_depth[species]
                - active_sum[species]
                + k * active_count[species]
            )

            if a[right] == species:
                answer += total_intervals - (right + 1) * (n - right)

        # Remove all positions whose depth is exactly k.
        species = a[left]
        active_count[species] -= 1
        active_sum[species] -= k

        if right != left:
            species = a[right]
            active_count[species] -= 1
            active_sum[species] -= k

    print(answer)


if __name__ == "__main__":
    solve()
