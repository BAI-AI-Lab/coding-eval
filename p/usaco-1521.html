#include <bits/stdc++.h>
using namespace std;

static const long long MOD = 1000000007LL;

long long modpow(long long a, long long e) {
    long long r = 1;
    while (e > 0) {
        if (e & 1) r = r * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return r;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long K, N, L;
    string T;
    cin >> K >> N >> L >> T;

    long long fact = 1;
    for (long long i = 2; i <= K; ++i) {
        fact = fact * i % MOD;
    }
    long long invFact = modpow(fact, MOD - 2);

    long long coefficient = 1;
    long long degree = 0;

    for (char c : T) {
        if (c == 'M') {
            coefficient = coefficient * invFact % MOD;
            degree += K;
        } else {
            coefficient = coefficient * (degree % MOD) % MOD;
            --degree;
        }
    }

    cout << modpow(coefficient, L) << '\n';
    return 0;
}

using namespace std;

static const long long MOD = 1000000007LL;

long long modpow(long long a, long long e) {
    long long r = 1;
    while (e > 0) {
        if (e & 1) r = r * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return r;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long K, L;
    int N;
    cin >> K >> N >> L;

    string T;
    cin >> T;

    long long mCount = 0, oCount = 0;
    long long prod = 1;
    long long Kmod = K % MOD;

    for (char c : T) {
        if (c == 'M') {
            ++mCount;
        } else {
            long long factor = (Kmod * (mCount % MOD) % MOD - (oCount % MOD) + MOD) % MOD;
            prod = prod * factor % MOD;
            ++oCount;
        }
    }

    // A decomposition is possible only if the total number of Os is K times
    // the total number of Ms.
    __int128 lhs = (__int128)K * mCount;
    if (lhs != (__int128)oCount) {
        cout << 0 << '\n';
        return 0;
    }

    int kk = (int)K;

    long long fact = 1;
    for (int i = 2; i <= kk; ++i) {
        fact = fact * i % MOD;
    }

    long long invFact = modpow(fact, MOD - 2);

    // Ways for one copy of T.
    long long waysT = prod * modpow(invFact, mCount) % MOD;

    // The copies are independent because each period uses all capacity before
    // the next period starts.
    long long answer = modpow(waysT, L);

    cout << answer << '\n';
    return 0;
}

static const long long MOD = 1000000007LL;

long long modpow(long long a, long long e) {
    long long r = 1 % MOD;
    a %= MOD;
    while (e > 0) {
        if (e & 1) r = r * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return r;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long K, L;
    int N;
    cin >> K >> N >> L;

    string T;
    cin >> T;

    long long M = 0;
    for (char ch : T) {
        if (ch == 'M') ++M;
    }
    long long O = N - M;

    // A valid decomposition must use every character:
    // total Os = K * total Ms.
    if (K > N || K * M != O) {
        cout << 0 << '\n';
        return 0;
    }

    long long available = 0;
    long long numerator = 1;
    long long kMod = K % MOD;

    for (char ch : T) {
        if (ch == 'M') {
            available += kMod;
            if (available >= MOD) available -= MOD;
        } else {
            numerator = numerator * available % MOD;
            --available;
            if (available < 0) available += MOD;
        }
    }

    long long factK = 1;
    for (long long i = 2; i <= K; ++i) {
        factK = factK * i % MOD;
    }

    long long invFactK = modpow(factK, MOD - 2);
    long long denominatorInverse = modpow(invFactK, M);

    long long perPeriod = numerator * denominatorInverse % MOD;
    long long answer = modpow(perPeriod, L);

    cout << answer << '\n';
    return 0;
}

    int64 result = 1;

        if (exp & 1) result = result * base % MOD;
        base = base * base % MOD;
        exp >>= 1;
    }
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int64 K, L;
    int N;
    cin >> K >> N >> L;

    string T;
    cin >> T;

    int64 seenM = 0, seenO = 0;
    int64 numerator = 1;

    for (char c : T) {
        if (c == 'M') {
            ++seenM;
        } else {
            ++seenO;
            int64 choices = K * seenM - (seenO - 1);
            choices %= MOD;
            if (choices < 0) choices += MOD;
            numerator = numerator * choices % MOD;
        }
    }

    // A nonzero decomposition requires #O(T) = K * #M(T).
    if (seenO != K * seenM) {
        cout << 0 << '\n';
        return 0;

    for (int64 i = 1; i <= K; ++i) {
        factorial = factorial * i % MOD;

    int64 oneCopy = numerator * modPow(denominator, MOD - 2) % MOD;

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using uint64 = unsigned long long;

constexpr int64 MOD = 1'000'000'007LL;

int64 modPow(int64 base, uint64 exponent) {
    int64 result = 1;
    while (exponent > 0) {
        if (exponent & 1ULL) {
            result = result * base % MOD;
        }
        base = base * base % MOD;
        exponent >>= 1ULL;
    }
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int64 K;
    int N;
    uint64 L;
    cin >> K >> N >> L;

    string T;
    cin >> T;

    int64 mSeen = 0;
    int64 oBefore = 0;
    int64 numerator = 1;

    for (char c : T) {
        if (c == 'O') {
            ++oBefore;
        } else {
            ++mSeen;

            // This M contributes C(K * mSeen - oBefore, K).
            // Accumulate the numerators of these binomial coefficients.
            int64 available = K * mSeen - oBefore;
            for (int64 q = 0; q < K; ++q) {
                int64 factor = (available - q) % MOD;
                if (factor < 0) factor += MOD;
                numerator = numerator * factor % MOD;
            }
        }
    }

    int64 factorialK = 1;
    for (int64 i = 2; i <= K; ++i) {
        factorialK = factorialK * i % MOD;
    }

    // Divide by (K!)^mSeen.
    int64 inverseFactorialK = modPow(factorialK, MOD - 2);
    int64 inverseDenominator = modPow(inverseFactorialK, mSeen);

    int64 waysPerCopy = numerator * inverseDenominator % MOD;
    cout << modPow(waysPerCopy, L) << '\n';

    return 0;
}
