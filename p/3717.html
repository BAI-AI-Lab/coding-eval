from typing import List


class _Fenwick:
    __slots__ = ("n", "count", "sums", "step")

    def __init__(self, n: int):
        self.n = n
        self.count = [0] * (n + 1)
        self.sums = [0] * (n + 1)
        self.step = 1 << (n.bit_length() - 1)

    def update(self, index: int, count_delta: int, sum_delta: int) -> None:
        i = index + 1
        while i <= self.n:
            self.count[i] += count_delta
            self.sums[i] += sum_delta
            i += i & -i

    def prefix(self, index: int):
        """Count and sum over compressed coordinates [0, index]."""
        i = index + 1
        count = 0
        total = 0
        while i > 0:
            count += self.count[i]
            total += self.sums[i]
            i -= i & -i
        return count, total

    def kth(self, k: int) -> int:
        """Return the 0-based compressed coordinate of the kth element."""
        index = 0
        bit = self.step
        while bit:
            nxt = index + bit
            if nxt <= self.n and self.count[nxt] < k:
                index = nxt
                k -= self.count[nxt]
            bit >>= 1
        return index


class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)

        # Coordinate compression for a sliding-window frequency/sum BIT.
        values = sorted(set(nums))
        compressed = {value: i for i, value in enumerate(values)}
        bit = _Fenwick(len(values))

        window_sum = sum(nums[:x])
        for value in nums[:x]:
            bit.update(compressed[value], 1, value)

        window_count = n - x + 1
        costs = [0] * window_count
        median_rank = (x + 1) // 2

        for start in range(window_count):
            median_index = bit.kth(median_rank)
            median = values[median_index]

            count_le, sum_le = bit.prefix(median_index)
            left_cost = median * count_le - sum_le
            right_cost = (window_sum - sum_le) - median * (x - count_le)
            costs[start] = left_cost + right_cost

            if start + 1 < window_count:
                outgoing = nums[start]
                incoming = nums[start + x]
                if outgoing != incoming:
                    bit.update(compressed[outgoing], -1, -outgoing)
                    bit.update(compressed[incoming], 1, incoming)
                    window_sum += incoming - outgoing

        # prev[i]: minimum cost using the previous number of subarrays
        # within the prefix nums[:i].
        prev = [0] * (n + 1)
        inf = 10**30

        for chosen in range(1, k + 1):
            cur = [inf] * (n + 1)

            # Fewer than chosen*x elements cannot contain enough subarrays.
            for end in range(chosen * x, n + 1):
                skip = cur[end - 1]
                take = prev[end - x] + costs[end - x]
                cur[end] = take if take < skip else skip

            prev = cur

        return prev[n]

from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)

        class Fenwick:
            def __init__(self, size: int):
                self.n = size
                self.tree = [0] * (size + 1)

            def add(self, idx: int, delta: int) -> None:
                i = idx + 1
                while i <= self.n:
                    self.tree[i] += delta
                    i += i & -i

            def prefix(self, idx: int) -> int:
                if idx < 0:
                    return 0
                i = idx + 1
                res = 0
                while i > 0:
                    res += self.tree[i]
                    i -= i & -i
                return res

            def kth(self, rank: int) -> int:
                idx = 0
                bit = 1 << (self.n.bit_length() - 1)
                while bit:
                    nxt = idx + bit
                    if nxt <= self.n and self.tree[nxt] < rank:
                        idx = nxt
                        rank -= self.tree[nxt]
                    bit >>= 1
                return idx

        values = sorted(set(nums))
        value_to_idx = {v: i for i, v in enumerate(values)}
        m = len(values)

        count_bit = Fenwick(m)
        sum_bit = Fenwick(m)

        total = 0
        for i in range(x):
            v = nums[i]
            idx = value_to_idx[v]
            count_bit.add(idx, 1)
            sum_bit.add(idx, v)
            total += v

        costs = [0] * (n - x + 1)
        median_rank = (x + 1) // 2

        for start in range(n - x + 1):
            if start > 0:
                out_val = nums[start - 1]
                in_val = nums[start + x - 1]

                out_idx = value_to_idx[out_val]
                count_bit.add(out_idx, -1)
                sum_bit.add(out_idx, -out_val)

                in_idx = value_to_idx[in_val]
                count_bit.add(in_idx, 1)
                sum_bit.add(in_idx, in_val)

                total += in_val - out_val

            med_idx = count_bit.kth(median_rank)
            median = values[med_idx]

            if med_idx > 0:
                less_count = count_bit.prefix(med_idx - 1)
                less_sum = sum_bit.prefix(med_idx - 1)
            else:
                less_count = 0
                less_sum = 0

            le_count = count_bit.prefix(med_idx)
            le_sum = sum_bit.prefix(med_idx)

            greater_count = x - le_count
            greater_sum = total - le_sum

            costs[start] = (
                median * less_count
                - less_sum
                + greater_sum
                - median * greater_count
            )

        INF = 10**30
        size = x + 1
        rows = [None] * size

        row0 = [INF] * (k + 1)
        row0[0] = 0
        rows[0] = row0

        for i in range(1, n + 1):
            prev = rows[(i - 1) % size]
            curr = prev.copy()
            curr[0] = 0

            if i >= x:
                start = i - x
                base = rows[start % size]
                cost = costs[start]

                for cnt in range(1, k + 1):
                    candidate = base[cnt - 1] + cost
                    if candidate < curr[cnt]:
                        curr[cnt] = candidate

            rows[i % size] = curr

        return rows[n % size][k]

from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)

        # Coordinate compression for Fenwick trees
        vals = sorted(set(nums))
        comp = {v: i + 1 for i, v in enumerate(vals)}
        N = len(vals)

        class BIT:
            def __init__(self, n: int):
                self.n = n
                self.tree = [0] * (n + 1)

            def add(self, i: int, delta: int) -> None:
                n = self.n
                tree = self.tree
                while i <= n:
                    tree[i] += delta
                    i += i & -i

            def query(self, i: int) -> int:
                s = 0
                tree = self.tree
                while i > 0:
                    s += tree[i]
                    i -= i & -i
                return s

            def kth(self, k: int) -> int:
                idx = 0
                bitmask = 1 << (self.n.bit_length() - 1)
                tree = self.tree
                n = self.n
                while bitmask:
                    nxt = idx + bitmask
                    if nxt <= n and tree[nxt] < k:
                        idx = nxt
                        k -= tree[nxt]
                    bitmask >>= 1
                return idx + 1

        bit_count = BIT(N)
        bit_sum = BIT(N)

        total = 0
        for v in nums[:x]:
            total += v
            idx = comp[v]
            bit_count.add(idx, 1)
            bit_sum.add(idx, v)

        M = n - x + 1
        costs = [0] * M
        p = (x + 1) // 2  # lower median position

        for i in range(M):
            if i > 0:
                rem = nums[i - 1]
                total -= rem
                idx = comp[rem]
                bit_count.add(idx, -1)
                bit_sum.add(idx, -rem)

                addv = nums[i + x - 1]
                total += addv
                idx = comp[addv]
                bit_count.add(idx, 1)
                bit_sum.add(idx, addv)

            idx_med = bit_count.kth(p)
            med = vals[idx_med - 1]

            less_count = bit_count.query(idx_med - 1)
            less_sum = bit_sum.query(idx_med - 1)

            costs[i] = total - 2 * less_sum + (2 * less_count - x) * med

        INF = 10**30
        prev = [0] * M  # dp for 0 intervals

        for j in range(1, k + 1):
            curr = [INF] * M
            for i in range(M):
                if i > 0:
                    curr[i] = curr[i - 1]

                if j == 1:
                    best = costs[i]
                elif i >= x and prev[i - x] < INF:
                    best = costs[i] + prev[i - x]
                else:
                    best = INF

                if best < curr[i]:
                    curr[i] = best

            prev = curr

        return prev[M - 1]
