#include <bits/stdc++.h>
using namespace std;

int ask(int v, long long x) {
    cout << "? " << v << " " << x << '\n' << flush;
    int response;
    cin >> response;
    return response;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    if (!(cin >> n)) return 0;

    for (int tc = 0; tc < n; ++tc) {
        int start = ask(1, 1);

        vector<int> cache(1025, -1);
        cache[0] = start;

        auto getValue = [&](int x) -> int {
            if (cache[x] == -1) cache[x] = ask(start, x);
            return cache[x];
        };

        auto attempt = [&](int baby, int giants) -> int {
            unordered_map<int, int> position;
            position.reserve(2 * baby + 10);

            for (int i = 0; i < baby; ++i) {
                int w = getValue(i);
                auto it = position.find(w);
                if (it != position.end()) return i - it->second;
                position.emplace(w, i);
            }

            for (int k = 1; k <= giants; ++k) {
                int x = k * baby;
                int w = getValue(x);
                auto it = position.find(w);
                if (it != position.end()) return x - it->second;
            }
            return -1;
        };

        int answer = attempt(2, 4);
        if (answer == -1) answer = attempt(8, 4);
        if (answer == -1) answer = attempt(32, 32);

        if (answer == -1) {
            unordered_map<int, int> position;
            position.reserve(2048);

            for (int i = 0; i < 1000; ++i) {
                int w = getValue(i);
                auto it = position.find(w);
                if (it != position.end()) {
                    answer = i - it->second;
                    break;
                }
                position.emplace(w, i);
            }

            int k = 2;
            while (answer == -1) {
                int x = k * 1000;
                int w = ask(start, x);
                auto it = position.find(w);
                if (it != position.end()) answer = x - it->second;
                ++k;
            }
        }

        cout << "! " << answer << '\n' << flush;
        int verdict;
        if (!(cin >> verdict) || verdict == -1) return 0;
    }

    return 0;
}

static const int MAXV = 1000000;
static const int B = 1000;

long long ask(long long v, long long x) {
    cout << "? " << v << " " << x << endl;
    cout.flush();
    long long res;
    if (!(cin >> res)) exit(0);
    return res;
}

bool answer(long long s) {
    cout << "! " << s << endl;
    cout.flush();
    int verdict;
    if (!(cin >> verdict)) exit(0);
    return verdict == 1;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    if (!(cin >> n)) return 0;

    vector<int> seen(MAXV + 1, -1);

    for (int g = 0; g < n; ++g) {
        long long a = ask(1, 1);
        long long s = -1;

        seen[a] = 0;

        for (int i = 1; i <= B; ++i) {
            long long r = ask(a, i);
            if (r == a) {
                s = i;
                break;
            }
            seen[r] = i;
        }

        if (s == -1) {
            for (long long j = 2; ; ++j) {
                long long x = j * B;
                long long r = ask(a, x);
                if (seen[r] != -1) {
                    s = x - seen[r];
                    break;
                }
            }
        }

        if (!answer(s)) return 0;

        seen[a] = -1;
        for (int i = 1; i <= min<long long>(B, s - 1); ++i) {
            // Entries are cleared below by reconstructing only when necessary.
        }
        fill(seen.begin(), seen.end(), -1);
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

constexpr int MAX_VERTEX = 1'000'000;
constexpr int QUERY_LIMIT = 2500;
constexpr int EXTRACTION_RESERVE = 32;
constexpr int EXPONENT_RANGE = 1'000'000'000;

static int firstExponent[MAX_VERTEX + 1];

int64 randomExponent(mt19937_64& rng) {
    return uniform_int_distribution<int>(1, EXPONENT_RANGE)(rng);
}

vector<pair<int64, int>> factorSmallPrimes(int64 value) {
    vector<pair<int64, int>> factors;
    int64 remaining = value;

    for (int64 p = 2; p * p <= remaining; ++p) {
        if (remaining % p != 0) continue;

        int exponent = 0;
        do {
            remaining /= p;
            ++exponent;
        } while (remaining % p == 0);

        if (p <= MAX_VERTEX) {
            factors.emplace_back(p, exponent);
        }
    }

    if (remaining > 1 && remaining <= MAX_VERTEX) {
        factors.emplace_back(remaining, 1);
    }

    return factors;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    if (!(cin >> n)) return 0;

    uint64_t seed = chrono::high_resolution_clock::now().time_since_epoch().count();
    seed ^= uint64_t(random_device{}()) << 32;
    mt19937_64 rng(seed);

    int queries = 0;

    auto ask = [&](int vertex, int64 exponent) -> int {
        cout << "? " << vertex << ' ' << exponent << endl;
        if (!(cin >> queries)) {
            exit(0);
        }
        ++queries;

        int result;
        if (!(cin >> result)) exit(0);
        return result;
    };

    auto recoverPeriod = [&](int64 multiple, int cycleVertex) -> int64 {
        auto factors = factorSmallPrimes(multiple);
        int64 answer = 1;

        for (auto [prime, exponent] : factors) {
            int64 fullPower = 1;
            for (int i = 0; i < exponent; ++i) {
                fullPower *= prime;
            }

            int64 remainingMultiple = multiple / fullPower;

            int low = 0;
            int high = exponent;

            while (low < high) {
                int mid = (low + high) / 2;

                int64 primePower = 1;
                for (int i = 0; i < mid; ++i) {
                    primePower *= prime;
                }

                int64 testExponent = remainingMultiple * primePower;
                bool isPeriod = (ask(cycleVertex, testExponent) == cycleVertex);

                if (isPeriod) {
                    high = mid;
                } else {
                    low = mid + 1;
                }
            }

            for (int i = 0; i < low; ++i) {
                answer *= prime;
            }
        }

        return answer;
    };

    while (n--) {
        fill(firstExponent, firstExponent + MAX_VERTEX + 1, -1);

        int cycleVertex = ask(1, 1);
        firstExponent[cycleVertex] = 0;

        int64 difference = 0;

        while (difference == 0 && queries < QUERY_LIMIT - EXTRACTION_RESERVE) {
            int64 exponent = randomExponent(rng);
            int result = ask(cycleVertex, exponent);

            if (firstExponent[result] == -1) {
                firstExponent[result] = static_cast<int>(exponent);
            } else {
                int64 previous = firstExponent[result];
                if (previous != exponent) {
                    difference = exponent > previous
                                   ? exponent - previous
                                   : previous - exponent;
                }
            }
        }

        if (difference == 0) {
            int64 exponent = randomExponent(rng);
            int result = ask(cycleVertex, exponent);

            if (firstExponent[result] != -1) {
                int64 previous = firstExponent[result];
                if (previous != exponent) {
                    difference = exponent > previous
                                   ? exponent - previous
                                   : previous - exponent;
                }
            }
        }

        if (difference == 0) {
            int result = ask(cycleVertex, MAX_VERTEX);
            if (result == cycleVertex) {
                difference = MAX_VERTEX;
            } else {
                constexpr int64 FALLBACK_MULTIPLE = 223092870;
                result = ask(cycleVertex, FALLBACK_MULTIPLE);
                if (result == cycleVertex) {
                    difference = FALLBACK_MULTIPLE;
                }
            }
        }

        int64 answer = (difference != 0)
                           ? recoverPeriod(difference, cycleVertex)
                           : MAX_VERTEX;

        cout << "! " << answer << endl;

        int verdict;
        if (!(cin >> verdict) || verdict != 1) return 0;
    }

    return 0;
}
