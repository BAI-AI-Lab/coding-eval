import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n = data[0]
    a = data[1:]

    count = [0] * (n + 1)
    for x in a:
        count[x] += 1

    missing = 0
    answers = []

    for i in range(n + 1):
        answers.append(str(max(count[i], missing)))
        if count[i] == 0:
            missing += 1

    sys.stdout.write("\n".join(answers))

if __name__ == "__main__":
    main()

import sys


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n = data[0]

    freq = [0] * (n + 2)
    for v in data[1:1 + n]:
        freq[v] += 1

    out = []
    missing = 0  # number of absent values among 0..i-1

    for i in range(n + 1):
        c = freq[i]
        out.append(str(c if c > missing else missing))
        if c == 0:
            missing += 1  # value i is absent, so it counts as missing for i+1

    sys.stdout.write("\n".join(out) + "\n")


if __name__ == "__main__":
    solve()
