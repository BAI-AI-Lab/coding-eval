import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))

    counts = {}
    for x in a:
        counts[x] = counts.get(x, 0) + 1

    answer = -1
    max_value = -1
    for i, x in enumerate(a, 1):
        if counts[x] == 1 and x > max_value:
            max_value = x
            answer = i

    print(answer)

if __name__ == "__main__":
    main()
