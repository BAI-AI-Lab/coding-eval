#include <bits/stdc++.h>
using namespace std;

struct Query {
    int a, b, res;
};

int n;
map<pair<int, int>, int> cache;
vector<Query> history;

int ask(int a, int b) {
    if (a > b) swap(a, b);
    auto key = make_pair(a, b);
    auto it = cache.find(key);
    if (it != cache.end()) return it->second;

    cout << "? " << a + 1 << " " << b + 1 << endl;
    cout.flush();

    int res;
    cin >> res;
    if (res == -1) exit(0);

    cache[key] = res;
    history.push_back({a, b, res});
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;

    vector<int> candidates(n);
    iota(candidates.begin(), candidates.end(), 0);

    vector<int> answer(n, -1);
    int zero = -1;

    while (zero == -1) {
        if (candidates.size() == 1) {
            zero = candidates[0];
            break;
        }

        int a = candidates[0];
        int value = ~0;
        vector<pair<int, int>> results;

        for (int i = 1; i < (int)candidates.size(); ++i) {
            int b = candidates[i];
            int res = ask(a, b);
            value &= res;
            results.push_back({b, res});
        }

        answer[a] = value;

        if (value == 0) {
            zero = a;
            break;
        }

        vector<int> nextCandidates;
        for (auto [b, res] : results) {
            if (res == value) nextCandidates.push_back(b);
        }
        candidates.swap(nextCandidates);
    }

    answer[zero] = 0;

    for (const auto &q : history) {
        if (q.a == zero) answer[q.b] = q.res;
        if (q.b == zero) answer[q.a] = q.res;
    }

    for (int i = 0; i < n; ++i) {
        if (answer[i] == -1) {
            answer[i] = ask(zero, i);
        }
    }

    cout << "!";
    for (int i = 0; i < n; ++i) {
        cout << " " << answer[i];
    }
    cout << endl;
    cout.flush();

    return 0;
}

    vector<int> candidates(n), known(n, -1), answer(n, -1);

        shuffle(candidates.begin(), candidates.end(), rng);

        vector<int> result(n, -1);

            }

        }

        zero = candidates[0];
        answer[zero] = 0;
        known[zero] = 0;

#include <bits/stdc++.h>
using namespace std;

static int n;
static vector<int> perm;
static mt19937 rng((unsigned)chrono::steady_clock::now().time_since_epoch().count());

static int ask(int i, int j) {
    cout << "? " << (i + 1) << ' ' << (j + 1) << endl;
    int r;
    if (!(cin >> r)) exit(0);
    if (r == -1) exit(0);
    return r;
}

/*
   Invariant: the leaves in G hold exactly the values
              { u : u is a submask of W } \ {W}
   (so |G| = 2^popcount(W) - 1 and 0 belongs to the set).
   Returns the leaf whose hidden value equals 0.
*/
static int findZero(const vector<int>& G) {
    if (G.size() == 1) return G[0];

    const int m = (int)G.size();
    const int x = G[(int)(rng() % m)];

    vector<int> av(m, -1);
    int w = -1;
    for (int i = 0; i < m; i++) {
        if (G[i] == x) continue;
        av[i] = ask(x, G[i]);
        w = (w < 0) ? av[i] : (w & av[i]);
    }
    /* w == p_x  (AND trick: every bit is missing from at least two values) */
    if (w == 0) {
        for (int i = 0; i < m; i++)
            if (G[i] != x) perm[G[i]] = av[i];
        return x;
    }

    vector<int> G2;
    for (int i = 0; i < m; i++) {
        if (G[i] == x) continue;
        if ((av[i] & ~w) == 0) G2.push_back(G[i]);
    }
    return findZero(G2);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    if (!(cin >> n)) return 0;
    perm.assign(n, -1);

    if (n <= 4) {
        /* tiny instances: ask everything and solve by backtracking */
        vector<vector<int>> q(n, vector<int>(n, -1));
        for (int i = 0; i < n; i++)
            for (int j = i + 1; j < n; j++) { q[i][j] = q[j][i] = ask(i, j); }
        vector<int> p(n, -1);
        vector<char> used(n, 0);
        function<bool(int)> rec = [&](int i) -> bool {
            if (i == n) return true;
            for (int v = 0; v < n; v++) {
                if (used[v]) continue;
                bool ok = true;
                for (int j = 0; j < i; j++)
                    if ((v | p[j]) != q[i][j]) { ok = false; break; }
                if (!ok) continue;
                p[i] = v; used[v] = 1;
                if (rec(i + 1)) return true;
                used[v] = 0;
            }
            return false;
        };
        rec(0);
        cout << "!";
        for (int i = 0; i < n; i++) cout << ' ' << p[i];
        cout << endl;
        return 0;
    }

    /* step 1: a star around vertex 0 -> p_0 and p_j & ~p_0 for every other j */
    vector<int> a(n, -1);
    int c = -1;
    for (int j = 1; j < n; j++) {
        a[j] = ask(0, j);
        c = (c < 0) ? a[j] : (c & a[j]);
    }
    perm[0] = c;

    int z;
    if (c == 0) {
        /* p_0 == 0 : the star already revealed every value */
        z = 0;
        for (int j = 1; j < n; j++) perm[j] = a[j];
    } else {
        /* leaves whose value is a submask of c contain the value 0 */
        vector<int> G0;
        for (int j = 1; j < n; j++)
            if ((a[j] & ~c) == 0) G0.push_back(j);
        z = findZero(G0);
    }
    perm[z] = 0;

    /* step 2: read every remaining value using the zero vertex */
    for (int j = 0; j < n; j++) {
        if (j == 0 || j == z || perm[j] != -1) continue;
        perm[j] = ask(z, j);
    }

    cout << "!";
    for (int i = 0; i < n; i++) cout << ' ' << perm[i];
    cout << endl;
    return 0;
}
