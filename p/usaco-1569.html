Balancing the Barns

题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Farmer John has $N$ ($1\le N\le 5\cdot 10^4$) barns arranged along a road. The $i$-th barn contains $a_i$ bales of hay and $b_i$ bags of feed $(0\le a_i,b_i\le 10^9$).

Bessie has been complaining about the inequality between barns. She defines the "imbalance" of the farm as the difference between the maximum hay in any barn and the minimum feed in any barn. Formally, the imbalance is $\max(a) - \min(b)$.

To address Bessie's concerns, Farmer John can perform exactly $K$ ($1\le K\le 10^{18}$) transfers. In each transfer, he selects a barn $i$, sells one of its haybales, and buys it a new bag of feed for the same barn. Note that there can be negative amounts in his farm (he is not afraid of debt). Formally, $K$ times, you choose an index $i\in [1,N]$, decrement $a_i$, and increment $b_i$.

Help Farmer John determine the minimum possible imbalance after performing exactly $K$ transfers.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $T$ ($1 \leq T \leq 10^3$), the number of independent test cases.

The first line of each test case contains $N$ and $K$.

The following line contains $a_1\dots a_N$.

The following line contains $b_1\dots b_N$.

The sum of $N$ over all test cases is at most $5 \cdot 10^4$.

OUTPUT FORMAT (print output to the terminal / stdout):

For each test case, output a single integer, the minimum possible value of $\max(a) - \min(b)$ after performing $K$ operations.

SAMPLE INPUT:

4
1 10
5
3
2 6
100 96
0 4
3 3
1 1 2
0 0 1
3 3
1 2 2
0 1 1

SAMPLE OUTPUT:

-18
90
0
0

In the first test case, Farmer John can transfer $10$ haybales from barn $1$ into bags of feed. This leaves $a = [-5]$ and $b = [13]$. The imbalance is $\max(a) - \min(b) = -5 - 13 = -18$.

In the second test case, Farmer john can transfer $5$ haybales from barn $1$ and $1$ haybale from barn $2$. This leaves $a = [95, 95]$ and $b = [5, 5]$. The imbalance is $95 - 5 = 90$. This is the minimum imbalance Farmer John can achieve.

SCORING:

  • Inputs 2-4: $K\le 500$, sum of $N$ over all testcases is $\le 500$
  • Inputs 5-8: Sum of $N$ over all testcases is $\le 500$
  • Inputs 9-13: No additional constraints.

using namespace std;

using int64 = long long;
using i128 = __int128_t;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        int64 K;
        cin >> N >> K;

        vector<int64> a(N), b(N);
        for (auto &x : a) cin >> x;
        for (auto &x : b) cin >> x;

        int64 maxA = *max_element(a.begin(), a.end());
        int64 minB = *min_element(b.begin(), b.end());
        int64 initial = maxA - minB;

        vector<int64> events(2 * N);

        auto requiredOperations = [&](int64 t, int64 D) -> int64 {
            i128 total = 0;

            for (int i = 0; i < N; ++i) {
                i128 need1 = (i128)t - b[i];
                i128 need2 = (i128)a[i] - t - D;
                i128 need = max<i128>(0, max(need1, need2));

                total += need;
                if (total > K) return K + 1;
            }

            return (int64)total;
        };

        auto feasible = [&](int64 D) -> bool {
            for (int i = 0; i < N; ++i) {
                int64 c = a[i] - D;

                if (c <= b[i]) {
                    events[2 * i] = 2 * c;
                    events[2 * i + 1] = 2 * b[i];
                } else {
                    events[2 * i] = b[i] + c;
                    events[2 * i + 1] = b[i] + c;
                }
            }

            nth_element(events.begin(), events.begin() + (N - 1), events.end());
            int64 median2 = events[N - 1];

            int64 t0 = median2 / 2;
            if (median2 < 0 && median2 % 2 != 0) --t0;
            int64 t1 = t0 + 1;

            return requiredOperations(t0, D) <= K ||
                   requiredOperations(t1, D) <= K;
        };

        int64 low = initial - 2 * K - 1; // Impossible.
        int64 high = initial;            // Always possible.

        while (high - low > 1) {
            int64 mid = low + (high - low) / 2;
            if (feasible(mid)) high = mid;
            else low = mid;
        }

        cout << high << '\n';
    }

    return 0;
}

    int T;
    cin >> T;

    while (T--) {
        int N;
        int64 K;
        cin >> N >> K;

        vector<int64> a(N), b(N);
        for (auto &x : a) cin >> x;
        for (auto &x : b) cin >> x;

        const int64 minA = *min_element(a.begin(), a.end());
        const int64 maxA = *max_element(a.begin(), a.end());
        const int64 minB = *min_element(b.begin(), b.end());

        vector<int64> points;
        points.reserve(2 * N);

        auto feasible = [&](int64 D) {
            points.clear();

            // For fixed D and final minimum feed m, barn i needs
            // max(0, m-b[i], a[i]-D-m) operations.
            // Each such function is distance to an interval, or an absolute
            // value around its midpoint. A weighted median minimizes their sum.
            for (int i = 0; i < N; ++i) {
                const int64 c = a[i] - D;
                const int64 d = b[i];

                if (c <= d) {
                    points.push_back(2 * c);
                    points.push_back(2 * d);
                } else {
                    points.push_back(c + d);
                    points.push_back(c + d);
                }
            }

            nth_element(points.begin(), points.begin() + (N - 1), points.end());
            const int64 median2 = points[N - 1];

            int64 base = median2 / 2;
            if (median2 < 0 && median2 % 2 != 0) --base;

            auto enough = [&](int64 m) {
                m = max(m, minB);
                int64 used = 0;

                for (int i = 0; i < N; ++i) {
                    const int64 need = max({0LL, m - b[i], a[i] - D - m});
                    if (need > K - used) return false;
                    used += need;
                }
                return true;
            };

            return enough(base) || enough(base + 1);
        };

        int64 low = minA - minB - 2 * K - 1; // impossible
        int64 high = maxA - minB;            // always possible

        while (high - low > 1) {
            const int64 mid = low + (high - low) / 2;
            if (feasible(mid)) high = mid;
            else low = mid;
        }

        cout << high << '\n';

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using i128 = __int128_t;

static int64 floorDiv2(int64 x) {
    int64 q = x / 2;
    if (x < 0 && x % 2 != 0) --q;
    return q;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        int64 K;
        cin >> N >> K;

        vector<int64> a(N), b(N);
        int64 maxA = LLONG_MIN;
        int64 minB = LLONG_MAX;

        for (int i = 0; i < N; ++i) {
            cin >> a[i];
            maxA = max(maxA, a[i]);
        }
        for (int i = 0; i < N; ++i) {
            cin >> b[i];
            minB = min(minB, b[i]);
        }

        const int64 initial = maxA - minB;

        vector<int64> events;
        events.reserve(2 * N);

        // Returns whether an imbalance of at most D is achievable.
        auto possible = [&](int64 D) -> bool {
            events.clear();

            for (int i = 0; i < N; ++i) {
                int64 c = b[i] + D;

                if (a[i] >= c) {
                    // The turning point is (a[i] + c) / 2,
                    // where the slope increases by 2.
                    int64 doubledPoint = a[i] + c;
                    events.push_back(doubledPoint);
                    events.push_back(doubledPoint);
                } else {
                    // The function is zero on [a[i], c].
                    // Its slope increases by 1 at each endpoint.
                    events.push_back(2 * a[i]);
                    events.push_back(2 * c);
                }
            }

            // There are 2N unit slope-change events. The N-th event
            // is where the total slope changes from negative to
            // nonnegative, so it identifies a real minimizer.
            auto middle = events.begin() + (N - 1);
            nth_element(events.begin(), middle, events.end());
            int64 doubledPoint = *middle;

            int64 A1 = floorDiv2(doubledPoint);
            int64 A2 = A1 + (doubledPoint % 2 != 0);

            auto withinBudget = [&](int64 A) -> bool {
                i128 used = 0;

                for (int i = 0; i < N; ++i) {
                    i128 c = (i128)b[i] + D;
                    i128 needed = (i128)a[i] - A;

                    i128 other = (i128)A - c;
                    if (other > needed) needed = other;
                    if (needed < 0) needed = 0;

                    used += needed;
                    if (used > (i128)K) return false;
                }

                return true;
            };

            return withinBudget(A1) || withinBudget(A2);
        };

        // One operation can decrease the imbalance by at most 2.
        int64 low = initial - 2LL * K;
        int64 high = initial;

        while (low < high) {
            int64 mid = low + (high - low) / 2;

            if (possible(mid)) {
                high = mid;
            } else {
                low = mid + 1;
            }
        }

        cout << low << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using i128 = __int128_t;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        int64 K;
        cin >> N >> K;

        vector<int64> a(N), b(N);
        int64 A = LLONG_MIN;
        int64 B = LLONG_MAX;

        for (int i = 0; i < N; ++i) {
            cin >> a[i];
            A = max(A, a[i]);
        }
        for (int i = 0; i < N; ++i) {
            cin >> b[i];
            B = min(B, b[i]);
        }

        vector<int64> breakpoints;
        breakpoints.reserve(2 * N);

        // Is min_b >= y-D and max_a <= y achievable using at most K transfers?
        auto costOK = [&](int64 y, int64 D) -> bool {
            i128 total = 0;

            for (int i = 0; i < N; ++i) {
                i128 reduceHay = (i128)a[i] - y;
                i128 increaseFeed = (i128)y - D - b[i];

                i128 needed = 0;
                if (reduceHay > needed) needed = reduceHay;
                if (increaseFeed > needed) needed = increaseFeed;

                total += needed;
                if (total > (i128)K) return false;
            }

            return true;
        };

        auto feasible = [&](int64 D) -> bool {
            breakpoints.clear();

            for (int i = 0; i < N; ++i) {
                int64 c = b[i] + D;

                // All breakpoint coordinates are doubled.
                if (a[i] <= c) {
                    breakpoints.push_back(2LL * a[i]);
                    breakpoints.push_back(2LL * c);
                } else {
                    int64 midpointTwice = a[i] + c;
                    breakpoints.push_back(midpointTwice);
                    breakpoints.push_back(midpointTwice);
                }
            }

            // The continuous minimum begins at the N-th smallest breakpoint.
            nth_element(
                breakpoints.begin(),
                breakpoints.begin() + (N - 1),
                breakpoints.end()
            );

            int64 z = breakpoints[N - 1];

            // Actual breakpoint is z / 2. Compute floor(z / 2).
            int64 yFloor = z / 2;
            if (z < 0 && z % 2 != 0) --yFloor;

            int64 yCeil = yFloor + (z % 2 != 0);

            if (costOK(yFloor, D)) return true;
            if (yCeil != yFloor && costOK(yCeil, D)) return true;

            return false;
        };

        // Every result is at least A-B-2K.
        // A-B is always feasible because transfers cannot increase imbalance.
        int64 low = A - B - 2LL * K - 1;
        int64 high = A - B;

        while (high - low > 1) {
            int64 mid = low + (high - low) / 2;

            if (feasible(mid)) {
                high = mid;
            } else {
                low = mid;
            }
        }

        cout << high << '\n';
    }

    return 0;
}

        breakpoints.reserve(2 * N);

        // Minimum transfers needed for the chosen L, capped at K + 1.
        auto transfersNeeded = [&](int64 D, int64 L) -> int64 {
            i128 total = 0;

            for (int i = 0; i < N; ++i) {
                i128 needHay = (i128)a[i] - (i128)D - (i128)L;
                i128 needFeed = (i128)L - (i128)b[i];

                i128 need = 0;
                if (needHay > need) need = needHay;
                if (needFeed > need) need = needFeed;

                total += need;
                if (total > (i128)K) return K + 1;
            }

            return (int64)total;
        };

        auto feasible = [&](int64 D) -> bool {
            breakpoints.clear();

            for (int i = 0; i < N; ++i) {
                int64 u = a[i] - D;
                int64 d = u - b[i];

                if (d >= 0) {
                    int64 q = u + b[i];
                    breakpoints.push_back(q);
                    breakpoints.push_back(q);
                } else {
                    breakpoints.push_back(2LL * u);
                    breakpoints.push_back(2LL * b[i]);
                }
            }

            // There are 2N breakpoints. Select the lower median.
            nth_element(breakpoints.begin(),
                        breakpoints.begin() + (N - 1),
                        breakpoints.end());

            int64 median = breakpoints[N - 1];
            int64 L1 = floorHalf(median);
            int64 L2 = ceilHalf(median);

            if (transfersNeeded(D, L1) <= K) return true;
            if (L2 != L1 && transfersNeeded(D, L2) <= K) return true;
            return false;
        };

        int64 low = initial - 2LL * K;
        int64 high = initial;

        while (low < high) {
            int64 mid = low + (high - low) / 2;
            if (feasible(mid)) {
                high = mid;
            } else {
                low = mid + 1;
            }
        }

        cout << low << '\n';
    }

    return 0;
}
