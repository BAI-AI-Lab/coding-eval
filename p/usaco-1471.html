Farmer John's Favorite Operation

USACO 2025 January · 银组 · 2025-01 · L2 中等 · Python 3.11 · 时限 4 秒 · 测试点等分

查看原题跳到结果

题面

来源：USACO 官网。模型收到的是同一份题面的文字。

It is another cold and boring day on Farmer John's farm. To pass the time, Farmer John has invented a fun leisure activity involving performing operations on an integer array.

Farmer John has an array $a$ of $N$ ($1 \leq N \leq 2 \cdot 10^5$) non-negative integers and an integer $M$ ($1 \leq M \leq 10^9$). Then, FJ will ask Bessie for an integer $x$. In one operation, FJ can pick an index $i$ and subtract or add $1$ to $a_i$. FJ's boredom value is the minimum number of operations he must perform so that $a_i-x$ is divisible by $M$ for all $1 \leq i \leq N$.

Among all possible $x$, output FJ's minimum possible boredom value.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $T$ ($1 \leq T \leq 10$), the number of independent test cases to solve.

The first line of each test case contains $N$ and $M$.

The second line of each test case contains $a_1, a_2, ..., a_N$ ($0 \leq a_i \leq 10^9$).

It is guaranteed that the sum of $N$ over all test cases does not exceed $5 \cdot 10^5$.

OUTPUT FORMAT (print output to the terminal / stdout):

For each test case, output an integer on a new line containing FJ's minimum possible boredom value among all possible values of $x$.

SAMPLE INPUT:

2
5 9
15 12 18 3 8
3 69
1 988244353 998244853

SAMPLE OUTPUT:

10
21

In the first test case, one optimal choice of $x$ is $3$. FJ can perform $10$ operations to make $a = [12, 12, 21, 3, 12]$.

SCORING:

  • Input 2: $N \le 1000$ and $M \le 1000$.
  • Input 3: $N\le 1000$.
  • Inputs 4-5: $M\le 10^5$.
  • Inputs 6-16: No additional constraints.

Problem credits: Chongtian Ma

import sys
from bisect import bisect_right


def minimum_boredom(residues, m):
    residues.sort()
    n = len(residues)

    # Two copies allow every cyclic distance to be treated as a forward distance.
    extended = residues + [r + m for r in residues]
    prefix = [0]
    for value in extended:
        prefix.append(prefix[-1] + value)

    half = m // 2
    answer = None

    # An optimal x can always be chosen congruent to one of the array elements.
    for i, x in enumerate(residues):
        end = i + n
        split = bisect_right(extended, x + half, i, end)

        # Points at forward distance at most floor(M/2).
        count = split - i
        cost = prefix[split] - prefix[i] - count * x

        # Remaining points are closer in the backward direction.
        count = end - split
        cost += count * (m + x) - (prefix[end] - prefix[split])

        if answer is None or cost < answer:
            answer = cost

    return answer


def solve():
    data = sys.stdin.buffer.read().split()
    iterator = iter(data)
    test_cases = int(next(iterator))
    output = []

    for _ in range(test_cases):
        n = int(next(iterator))
        m = int(next(iterator))
        residues = [int(next(iterator)) % m for _ in range(n)]
        output.append(str(minimum_boredom(residues, m)))

    sys.stdout.write("\n".join(output))


if __name__ == "__main__":
    solve()

import sys


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    t = data[0]
    ptr = 1
    answers = []

    for _ in range(t):
        n = data[ptr]
        m = data[ptr + 1]
        ptr += 2

        if m == 1:
            ptr += n
            answers.append("0")
            continue

        k = m // 2
        odd = m & 1

        f0 = 0          # f(0) = sum distance(r_i, 0)
        d0_sum = 0      # f(1) - f(0)
        events = []
        append_event = events.append

        for _ in range(n):
            a = data[ptr]
            ptr += 1
            r = a % m

            # distance from r to 0 on a cycle of length m
            other = m - r
            if r < other:
                d0 = r
            else:
                d0 = other
            f0 += d0

            # distance from r to 1
            if r > 0:
                diff = r - 1
            else:
                diff = 1

            other = m - diff
            if diff < other:
                d1 = diff
            else:
                d1 = other

            d0_sum += d1 - d0

            if odd:
                # Changes of the derivative at t = 0, k, k + 1
                x = r
                if x != 0:
                    append_event((x, 2))

                x = r + k
                if x >= m:
                    x -= m
                if x != 0:
                    append_event((x, -1))

                x = r + k + 1
                if x >= m:
                    x -= m
                if x != 0:
                    append_event((x, -1))
            else:
                # Changes of the derivative at t = 0, k
                x = r
                if x != 0:
                    append_event((x, 2))

                x = r + k
                if x >= m:
                    x -= m
                if x != 0:
                    append_event((x, -2))

        events.sort()

        ans = f0
        f = f0
        derivative = d0_sum
        pos = 0

        i = 0
        length = len(events)

        while i < length:
            x = events[i][0]

            # Move from pos to x using the current derivative.
            next_f = f + derivative * (x - pos)

            if derivative < 0 and next_f < ans:
                ans = next_f

            f = next_f
            pos = x

            delta = 0
            while i < length and events[i][0] == x:
                delta += events[i][1]
                i += 1

            derivative += delta

        # Last interval from pos to m - 1.
        if derivative < 0:
            last_f = f + derivative * (m - 1 - pos)
            if last_f < ans:
                ans = last_f
        else:
            if f < ans:
                ans = f

        answers.append(str(ans))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()

import sys


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    t = next(it)
    answers = []

    for _ in range(t):
        n = next(it)
        m = next(it)

        residues = [next(it) % m for _ in range(n)]
        residues.sort()

        # Duplicate the circle so every cyclic segment becomes contiguous.
        c = residues + [value + m for value in residues]

        # prefix[k] = sum(c[:k])
        prefix = [0] * (2 * n + 1)
        total = 0
        for i, value in enumerate(c):
            total += value
            prefix[i + 1] = total

        half = m // 2
        split = 0
        best = 10**30

        for i in range(n):
            x = residues[i]
            end = i + n
            limit = x + half

            if split < i:
                split = i

            # c[split] is the first value more than half a circle ahead.
            while split < end and c[split] <= limit:
                split += 1

            # For c[j] - x <= floor(M / 2), the distance is c[j] - x.
            close_count = split - i
            cost = (
                prefix[split]
                - prefix[i]
                - close_count * x
            )

            # Otherwise, the distance is M - (c[j] - x).
            far_count = end - split
            cost += (
                far_count * (m + x)
                - (prefix[end] - prefix[split])
            )

            if cost < best:
                best = cost
                if best == 0:
                    break

        answers.append(str(best))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    solve()
