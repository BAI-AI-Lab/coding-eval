#include <bits/stdc++.h>
using namespace std;

class Solver {
    static constexpr int H = 50;
    static constexpr int W = 50;
    static constexpr int dr[4] = {-1, 1, 0, 0};
    static constexpr int dc[4] = {0, 0, -1, 1};

    int si, sj, startCell, M = 0;
    int id[H][W], val[H][W];
    vector<int> tileWeight;
    vector<vector<int>> adj;

    vector<unsigned char> vis;
    vector<int> pos;
    vector<int> path, bestPath;
    int bestScore = 0;

    vector<int> seen, bfsQueue;
    int bfsStamp = 0;

    uint32_t rngState = 0x9e3779b9u;
    chrono::steady_clock::time_point deadline;

    uint32_t rnd() {
        rngState ^= rngState << 13;
        rngState ^= rngState >> 17;
        rngState ^= rngState << 5;
        return rngState;
    }

    int randint(int n) {
        return n <= 0 ? 0 : int(rnd() % uint32_t(n));
    }

    bool expired() const {
        return chrono::steady_clock::now() >= deadline;
    }

    inline bool inside(int r, int c) const {
        return 0 <= r && r < H && 0 <= c && c < W;
    }

    int onwardDegree(int cell, int enteringTile) const {
        int r = cell / W, c = cell % W;
        int degree = 0;
        for (int d = 0; d < 4; ++d) {
            int nr = r + dr[d], nc = c + dc[d];
            if (!inside(nr, nc)) continue;
            int nt = id[nr][nc];
            if (nt != enteringTile && !vis[nt]) ++degree;
        }
        return degree;
    }

    void reachableFrom(int cell, int enteringTile, int &count, int &weightSum) {
        vis[enteringTile] = 1;
        ++bfsStamp;
        int qh = 0, qt = 0;
        count = 0;
        weightSum = 0;

        int r = cell / W, c = cell % W;
        for (int d = 0; d < 4; ++d) {
            int nr = r + dr[d], nc = c + dc[d];
            if (!inside(nr, nc)) continue;
            int nt = id[nr][nc];
            if (!vis[nt] && seen[nt] != bfsStamp) {
                seen[nt] = bfsStamp;
                bfsQueue[qt++] = nt;
            }
        }

        while (qh < qt) {
            int t = bfsQueue[qh++];
            ++count;
            weightSum += tileWeight[t];
            for (int u : adj[t]) {
                if (!vis[u] && seen[u] != bfsStamp) {
                    seen[u] = bfsStamp;
                    bfsQueue[qt++] = u;
                }
            }
        }
        vis[enteringTile] = 0;
    }

    void considerPath(bool allowEqualRandom = false) {
        int score = 0;
        for (int cell : path) score += val[cell / W][cell % W];

        bool replace = false;
        if (score > bestScore) replace = true;
        else if (score == bestScore && path.size() > bestPath.size()) replace = true;
        else if (allowEqualRandom && score == bestScore &&
                 path.size() == bestPath.size() && randint(8) == 0) replace = true;

        if (replace) {
            bestScore = score;
            bestPath = path;
        }
    }

    bool applyBackbite(int &stallLimit) {
        const int n = int(path.size());
        if (n < 3) return false;

        int endpoint = path.back();
        int er = endpoint / W, ec = endpoint % W;

        int candidateK[4];
        long long candidateKey[4];
        int candidateCount = 0;

        for (int d = 0; d < 4; ++d) {
            int nr = er + dr[d], nc = ec + dc[d];
            if (!inside(nr, nc)) continue;

            int t = id[nr][nc];
            if (!vis[t]) continue;
            int k = pos[t];
            if (k < 0 || k > n - 3) continue;

            int newEndpoint = path[k + 1];
            int rr = newEndpoint / W, cc = newEndpoint % W;
            int mobility = 0, bestNextValue = 0;

            for (int e = 0; e < 4; ++e) {
                int tr = rr + dr[e], tc = cc + dc[e];
                if (!inside(tr, tc)) continue;
                int nt = id[tr][tc];
                if (!vis[nt]) {
                    ++mobility;
                    bestNextValue = max(bestNextValue, val[tr][tc]);
                }
            }

            long long key;
            if (mobility > 0) {
                key = 1000000LL + bestNextValue * 100LL + mobility * 20LL + randint(256);
            } else {
                key = randint(10000);
            }

            candidateK[candidateCount] = k;
            candidateKey[candidateCount] = key;
            ++candidateCount;
        }

        if (candidateCount == 0) return false;

        int chosen = 0;
        for (int i = 1; i < candidateCount; ++i) {
            if (candidateKey[i] > candidateKey[chosen]) chosen = i;
        }

        int k = candidateK[chosen];
        reverse(path.begin() + k + 1, path.end());
        for (int i = k + 1; i < n; ++i) {
            int cell = path[i];
            pos[id[cell / W][cell % W]] = i;
        }
        return true;
    }

    void runAttempt(bool fromBest, int cut, int style, bool useConnectivity, int rotationLimit) {
        path.clear();
        if (fromBest && !bestPath.empty()) {
            path = bestPath;
            cut = min(cut, int(path.size()) - 1);
            path.resize(path.size() - cut);
        } else {
            path.push_back(startCell);
        }

        fill(vis.begin(), vis.end(), 0);
        fill(pos.begin(), pos.end(), -1);

        int score = 0;
        for (int i = 0; i < int(path.size()); ++i) {
            int cell = path[i];
            int t = id[cell / W][cell % W];
            vis[t] = 1;
            pos[t] = i;
            score += val[cell / W][cell % W];
        }

        int stall = 0;
        const int stepCap = useConnectivity ? 6500 : 10000;

        for (int step = 0; step < stepCap; ++step) {
            if ((step & 7) == 0 && expired()) break;

            int endpoint = path.back();
            int r = endpoint / W, c = endpoint % W;

            int bestCell = -1, bestTile = -1;
            long long bestKey = numeric_limits<long long>::min();

            for (int d = 0; d < 4; ++d) {
                int nr = r + dr[d], nc = c + dc[d];
                if (!inside(nr, nc)) continue;

                int nt = id[nr][nc];
                if (vis[nt]) continue;

                int nextCell = nr * W + nc;
                int degree = onwardDegree(nextCell, nt);
                int squareValue = val[nr][nc];
                long long key = 0;

                if (useConnectivity) {
                    int reach = 0, reachWeight = 0;
                    if (degree > 0) {
                        reachableFrom(nextCell, nt, reach, reachWeight);
                    }

                    if ((style & 1) == 0) {
                        key = (reach + 1) * 100000LL
                            - degree * 1000LL
                            + squareValue * 8LL
                            + randint(128);
                    } else {
                        key = (reachWeight + tileWeight[nt]) * 1000LL
                            - degree * 180LL
                            + squareValue * 20LL
                            + randint(256);
                    }
                } else {
                    int terminalPenalty = (degree == 0 ? 200000 : 0);
                    switch (style % 6) {
                        case 0:
                            key = -degree * 2000LL + squareValue * 5LL + randint(256);
                            break;
                        case 1:
                            key = -degree * 600LL + squareValue * 15LL + randint(512);
                            break;
                        case 2:
                            key = -degree * 200LL + squareValue * 5LL + randint(3000);
                            break;
                        case 3:
                            key = degree * 350LL + squareValue * 8LL + randint(1200);
                            break;
                        case 4:
                            key = -degree * 1000LL + squareValue * 2LL + randint(1800);
                            break;
                        default:
                            key = -degree * 800LL + squareValue * 25LL + randint(256);
                            break;
                    }
                    key -= terminalPenalty;
                }

                if (key > bestKey) {
                    bestKey = key;
                    bestCell = nextCell;
                    bestTile = nt;
                }
            }

            if (bestCell >= 0) {
                vis[bestTile] = 1;
                pos[bestTile] = int(path.size());
                path.push_back(bestCell);
                score += val[bestCell / W][bestCell % W];
                stall = 0;

                if (score > bestScore && (path.size() & 31ULL) == 0) {
                    bestScore = score;
                    bestPath = path;
                }
            } else {
                if (stall >= rotationLimit) break;
                if (!applyBackbite(rotationLimit)) break;
                ++stall;
            }
        }

        considerPath(true);
    }

public:
    void solve() {
        auto beginTime = chrono::steady_clock::now();
        deadline = beginTime + chrono::milliseconds(1850);

        cin >> si >> sj;
        for (int i = 0; i < H; ++i) {
            for (int j = 0; j < W; ++j) {
                cin >> id[i][j];
                M = max(M, id[i][j] + 1);
            }
        }
        for (int i = 0; i < H; ++i) {
            for (int j = 0; j < W; ++j) {
                cin >> val[i][j];
            }
        }

        startCell = si * W + sj;
        vector<int> tileSum(M, 0), tileCells(M, 0);
        for (int i = 0; i < H; ++i) {
            for (int j = 0; j < W; ++j) {
                int t = id[i][j];
                tileSum[t] += val[i][j];
                ++tileCells[t];
            }
        }

        tileWeight.resize(M);
        for (int t = 0; t < M; ++t) {
            tileWeight[t] = tileSum[t] / max(1, tileCells[t]);
        }

        adj.assign(M, {});
        for (int i = 0; i < H; ++i) {
            for (int j = 0; j < W; ++j) {
                int a = id[i][j];
                if (i + 1 < H) {
                    int b = id[i + 1][j];
                    if (a != b) {
                        adj[a].push_back(b);
                        adj[b].push_back(a);
                    }
                }
                if (j + 1 < W) {
                    int b = id[i][j + 1];
                    if (a != b) {
                        adj[a].push_back(b);
                        adj[b].push_back(a);
                    }
                }
            }
        }
        for (auto &v : adj) {
            sort(v.begin(), v.end());
            v.erase(unique(v.begin(), v.end()), v.end());
        }

        vis.assign(M, 0);
        pos.assign(M, -1);
        seen.assign(M, 0);
        bfsQueue.resize(M);

        bestPath = {startCell};
        bestScore = val[si][sj];

        int attempt = 0;
        while (!expired()) {
            bool fromBest = attempt >= 2 && randint(100) < 74;
            int cut = 0;

            if (fromBest) {
                int maxCut = int(bestPath.size()) - 1;
                if (maxCut > 0) {
                    int type = randint(12);
                    if (type < 2) {
                        cut = 0;
                    } else if (type < 6) {
                        cut = 1 + randint(min(maxCut, 24));
                    } else if (type < 9) {
                        cut = 1 + randint(min(maxCut, 160));
                    } else {
                        cut = 1 + randint(maxCut);
                    }
                }
            }

            int style = attempt < 2 ? attempt : randint(6);
            bool useConnectivity = attempt < 2 || randint(100) < (fromBest ? 58 : 40);
            int rotationLimit = 35 + randint(100);
            if (fromBest && cut == 0) rotationLimit += 100;

            runAttempt(fromBest, cut, style, useConnectivity, rotationLimit);
            ++attempt;
        }

        string answer;
        answer.reserve(bestPath.size());
        for (int i = 1; i < int(bestPath.size()); ++i) {
            int a = bestPath[i - 1], b = bestPath[i];
            if (b == a - W) answer.push_back('U');
            else if (b == a + W) answer.push_back('D');
            else if (b == a - 1) answer.push_back('L');
            else answer.push_back('R');
        }
        cout << answer << '\n';
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    Solver solver;
    solver.solve();
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using Clock = chrono::steady_clock;

struct XorShift {
    uint64_t x = 0x9e3779b97f4a7c15ULL;
    uint32_t next() {
        x ^= x << 7;
        x ^= x >> 9;
        x ^= x << 8;
        return static_cast<uint32_t>(x >> 32);
    }
    int nextInt(int n) { return static_cast<int>(next() % n); }
};

struct Node {
    int parent;
    char dir;
};

struct State {
    array<uint64_t, 40> used{};
    int pos = 0;
    int score = 0;
    int node = 0;
};

struct Candidate {
    int rank;
    int score;
    int pos;
    int tile;
    int parent;
    char dir;
};

struct RankConfig {
    int lookahead;
    int mobility;
    uint32_t noiseMask;
};

const int H = 50, W = 50;
const int dr[4] = {-1, 1, 0, 0};
const int dc[4] = {0, 0, -1, 1};
const char dchar[4] = {'U', 'D', 'L', 'R'};

int startI, startJ;
int tileAt[H][W];
int tileWeight[2500];
int startPos;
XorShift rng;
Clock::time_point deadline;

int bestScore = -1;
string bestPath;

int charToDir(char c) {
    if (c == 'U') return 0;
    if (c == 'D') return 1;
    if (c == 'L') return 2;
    return 3;
}

void beamSearch(const string &prefix, int width, const RankConfig &cfg) {
    vector<Node> nodes;
    nodes.reserve(1024);
    nodes.push_back({-1, 0});

    State initial;
    initial.used.fill(0);
    initial.pos = startPos;
    int currentTile = tileAt[startPos / W][startPos % W];
    initial.used[currentTile >> 6] |= 1ULL << (currentTile & 63);
    initial.score = tileWeight[currentTile];
    initial.node = 0;

    int pos = startPos;
    int currentNode = 0;
    for (char c : prefix) {
        int d = charToDir(c);
        int r = pos / W + dr[d];
        int col = pos % W + dc[d];
        pos = r * W + col;
        int t = tileAt[r][col];
        initial.used[t >> 6] |= 1ULL << (t & 63);
        initial.score += tileWeight[t];
        nodes.push_back({currentNode, c});
        currentNode = static_cast<int>(nodes.size()) - 1;
    }
    initial.pos = pos;
    initial.node = currentNode;

    vector<State> current, next;
    vector<Candidate> candidates;
    current.reserve(width);
    next.reserve(width);
    candidates.reserve(width * 4);
    current.push_back(initial);

    auto buildPath = [&](int node, char extra) {
        string path;
        while (nodes[node].parent != -1) {
            path.push_back(nodes[node].dir);
            node = nodes[node].parent;
        }
        reverse(path.begin(), path.end());
        path.push_back(extra);
        return path;
    };

    while (!current.empty()) {
        if (Clock::now() >= deadline) return;
        candidates.clear();
        bool stopped = false;

        for (int i = 0; i < static_cast<int>(current.size()); ++i) {
            if ((i & 63) == 0 && Clock::now() >= deadline) {
                stopped = true;
                break;
            }

            const State &state = current[i];
            int r = state.pos / W;
            int c = state.pos % W;

            for (int d = 0; d < 4; ++d) {
                int nr = r + dr[d];
                int nc = c + dc[d];
                if (nr < 0 || nr >= H || nc < 0 || nc >= W) continue;

                int np = nr * W + nc;
                int nt = tileAt[nr][nc];
                if ((state.used[nt >> 6] >> (nt & 63)) & 1ULL) continue;

                int ns = state.score + tileWeight[nt];
                if (ns > bestScore) {
                    bestScore = ns;
                    bestPath = buildPath(state.node, dchar[d]);
                }

                int degree = 0;
                int lookahead = 0;
                for (int e = 0; e < 4; ++e) {
                    int rr = nr + dr[e];
                    int cc = nc + dc[e];
                    if (rr < 0 || rr >= H || cc < 0 || cc >= W) continue;
                    int tt = tileAt[rr][cc];
                    if (tt == nt) continue;
                    if ((state.used[tt >> 6] >> (tt & 63)) & 1ULL) continue;
                    ++degree;
                    lookahead += tileWeight[tt];
                }

                if (degree == 0) continue;

                int rank = ns * 64
                         + lookahead * cfg.lookahead
                         + degree * cfg.mobility;
                if (cfg.noiseMask) rank += static_cast<int>(rng.next() & cfg.noiseMask);

                candidates.push_back({rank, ns, np, nt, i, dchar[d]});
            }
        }

        if (stopped) return;
        if (candidates.empty()) break;

        int keep = min(width, static_cast<int>(candidates.size()));
        if (keep < static_cast<int>(candidates.size())) {
            nth_element(candidates.begin(), candidates.begin() + keep, candidates.end(),
                        [](const Candidate &a, const Candidate &b) {
                            return a.rank > b.rank;
                        });
            candidates.resize(keep);
        }

        next.clear();
        for (const Candidate &cand : candidates) {
            int newNode = static_cast<int>(nodes.size());
            nodes.push_back({current[cand.parent].node, cand.dir});

            State state = current[cand.parent];
            state.used[cand.tile >> 6] |= 1ULL << (cand.tile & 63);
            state.pos = cand.pos;
            state.score = cand.score;
            state.node = newNode;
            next.push_back(state);
        }

        current.swap(next);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> startI >> startJ;
    startPos = startI * W + startJ;

    int maxTile = 0;
    for (int i = 0; i < H; ++i) {
        for (int j = 0; j < W; ++j) {
            cin >> tileAt[i][j];
            maxTile = max(maxTile, tileAt[i][j]);
        }
    }

    fill(tileWeight, tileWeight + maxTile + 1, 0);
    for (int i = 0; i < H; ++i) {
        for (int j = 0; j < W; ++j) {
            int p;
            cin >> p;
            tileWeight[tileAt[i][j]] += p;
        }
    }

    rng.x ^= static_cast<uint64_t>(startPos + 1) * 0xbf58476d1ce4e5b9ULL;
    bestScore = tileWeight[tileAt[startI][startJ]];
    bestPath.clear();

    deadline = Clock::now() + chrono::milliseconds(1850);

    beamSearch("", 1800, {8, 80, 0});

    int iteration = 0;
    while (Clock::now() < deadline) {
        string base = bestPath;
        int len = static_cast<int>(base.size());
        int cut = 0;

        auto remaining = duration_cast<chrono::milliseconds>(
            deadline - Clock::now()).count();

        if (len > 1) {
            int lo, hi;
            if (remaining < 180) {
                lo = len * 4 / 5;
                hi = len * 19 / 20;
            } else {
                lo = len / 4;
                hi = len * 3 / 4;
            }
            hi = min(hi, len - 1);
            lo = min(lo, hi);
            cut = lo + rng.nextInt(hi - lo + 1);
        }

        if (iteration % 7 == 6 && remaining > 650) cut = 0;
        string prefix = base.substr(0, cut);

        RankConfig cfg;
        cfg.lookahead = rng.nextInt(13);
        cfg.mobility = rng.nextInt(201) - 50;
        cfg.noiseMask = (1U << (5 + rng.nextInt(8))) - 1;

        int width = (cut == 0 ? 1000 : 650);
        if (remaining < 180) width = 400;
        beamSearch(prefix, width, cfg);
        ++iteration;
    }

    cout << bestPath << '\n';
    return 0;
}

void considerCurrentDFSPath() {

        (dfsPathScore == initialBestScore &&
         static_cast<int>(dfsPath.size()) > static_cast<int>(initialBestPath.size()))) {

void constructDFS(int current) {
    if (dfsNodeCount >= dfsNodeLimit ||
        ((dfsNodeCount & 2047) == 0 && elapsedSeconds() > 0.245)) {
        considerCurrentDFSPath();
        return;
    }

    ++dfsNodeCount;
    int t = tileId[current];
    tileUsed[t] = true;
    dfsPath.push_back(current);
    dfsPathScore += cellValue[current];

    struct Candidate {
        int cell;
        int exits;
        int noise;
    };
    Candidate candidates[4];
    int candidateCount = 0;

    for (int next : neighbors[current]) {
        if (tileUsed[tileId[next]]) continue;

        int exits = 0;
        for (int exit : neighbors[next]) {
            if (!tileUsed[tileId[exit]] && tileId[exit] != tileId[next]) {
                ++exits;
            }
        }

        candidates[candidateCount++] = {
            next, exits, static_cast<int>(random32() & 15ULL)
        };
    }

    for (int i = 1; i < candidateCount; ++i) {
        Candidate x = candidates[i];
        int j = i - 1;
        while (j >= 0) {
            bool xBefore = x.exits < candidates[j].exits ||
                           (x.exits == candidates[j].exits &&
                            x.noise < candidates[j].noise);
            bool sameNoise = x.exits == candidates[j].exits &&
                             x.noise == candidates[j].noise;
            if (!xBefore && !(sameNoise &&
                              cellValue[x.cell] > cellValue[candidates[j].cell])) {
                break;
            }
            candidates[j + 1] = candidates[j];
            --j;
        }
        candidates[j + 1] = x;
    }

    if (candidateCount == 0) {
        considerCurrentDFSPath();
    } else {
        for (int i = 0; i < candidateCount; ++i) {
            if (dfsNodeCount >= dfsNodeLimit || elapsedSeconds() > 0.245) break;
            constructDFS(candidates[i].cell);
        }
    }

    dfsPathScore -= cellValue[current];
    dfsPath.pop_back();
    tileUsed[t] = false;
}

struct Treap {
    struct Node {
        int left = 0;
        int right = 0;
        int parent = 0;
        int size = 0;
        int sum = 0;
        int cell = -1;
        uint32_t priority = 0;
        bool reversed = false;
    };

    vector<Node> nodes;
    vector<int> freeNodes;
    int cellNode[CELLS];
    int root = 0;

    Treap() {
        nodes.reserve(CELLS + 16);
        reset();
    }

    void reset() {
        nodes.clear();
        nodes.emplace_back();
        freeNodes.clear();
        fill(cellNode, cellNode + CELLS, 0);
        root = 0;
    }

    int nodeSize(int t) const {
        return t ? nodes[t].size : 0;
    }

    int nodeSum(int t) const {
        return t ? nodes[t].sum : 0;
    }

    void pull(int t) {
        int l = nodes[t].left;
        int r = nodes[t].right;
        nodes[t].size = 1 + nodeSize(l) + nodeSize(r);
        nodes[t].sum = cellValue[nodes[t].cell] + nodeSum(l) + nodeSum(r);
        if (l) nodes[l].parent = t;
        if (r) nodes[r].parent = t;
    }

    void applyReverse(int t) {
        if (!t) return;
        swap(nodes[t].left, nodes[t].right);
        nodes[t].reversed = !nodes[t].reversed;
    }

    void push(int t) {
        if (!t || !nodes[t].reversed) return;
        applyReverse(nodes[t].left);
        applyReverse(nodes[t].right);
        nodes[t].reversed = false;
    }

    int newNode(int cell) {
        int id;
        if (freeNodes.empty()) {
            id = static_cast<int>(nodes.size());
            nodes.emplace_back();
        } else {
            id = freeNodes.back();
            freeNodes.pop_back();
            nodes[id] = Node();
        }

        nodes[id].cell = cell;
        nodes[id].size = 1;
        nodes[id].sum = cellValue[cell];
        nodes[id].priority = random32();
        cellNode[cell] = id;
        return id;
    }

    pair<int, int> split(int t, int count) {
        if (!t) return {0, 0};
        push(t);

        int leftSize = nodeSize(nodes[t].left);
        if (count <= leftSize) {
            auto [a, b] = split(nodes[t].left, count);
            nodes[t].left = b;
            pull(t);
            nodes[t].parent = 0;
            if (a) nodes[a].parent = 0;
            return {a, t};
        } else {
            auto [a, b] = split(nodes[t].right, count - leftSize - 1);
            nodes[t].right = a;
            pull(t);
            nodes[t].parent = 0;
            if (b) nodes[b].parent = 0;
            return {t, b};
        }
    }

    int merge(int a, int b) {
        if (!a) {
            if (b) nodes[b].parent = 0;
            return b;
        }
        if (!b) {
            nodes[a].parent = 0;
            return a;
        }

        if (nodes[a].priority < nodes[b].priority) {
            push(a);
            nodes[a].right = merge(nodes[a].right, b);
            pull(a);
            nodes[a].parent = 0;
            return a;
        } else {
            push(b);
            nodes[b].left = merge(a, nodes[b].left);
            pull(b);
            nodes[b].parent = 0;
            return b;
        }
    }

    int size() const {
        return nodeSize(root);
    }

    int nodeAt(int rank) {
        int t = root;
        while (t) {
            push(t);
            int leftSize = nodeSize(nodes[t].left);
            if (rank < leftSize) {
                t = nodes[t].left;
            } else if (rank == leftSize) {
                return t;
            } else {
                rank -= leftSize + 1;
                t = nodes[t].right;
            }
        }
        return 0;
    }

    int getRank(int node) {
        int chain[256];
        int chainLength = 0;
        for (int t = node; t; t = nodes[t].parent) {
            chain[chainLength++] = t;
        }
        for (int i = chainLength - 1; i >= 0; --i) push(chain[i]);

        int rank = nodeSize(nodes[node].left);
        int t = node;
        while (nodes[t].parent) {
            int p = nodes[t].parent;
            if (nodes[p].right == t) {
                rank += nodeSize(nodes[p].left) + 1;
            }
            t = p;
        }
        return rank;
    }

    int prefixSum(int count) {
        int result = 0;
        int t = root;
        while (t && count > 0) {
            push(t);
            int leftSize = nodeSize(nodes[t].left);
            if (count <= leftSize) {
                t = nodes[t].left;
            } else {
                result += nodeSum(nodes[t].left) + cellValue[nodes[t].cell];
                count -= leftSize + 1;
                t = nodes[t].right;
            }
        }
        return result;
    }

    int rangeSum(int left, int right) {
        if (left > right) return 0;
        return prefixSum(right + 1) - prefixSum(left);
    }

    void collect(int t, vector<int>& result) {
        if (!t) return;
        push(t);
        collect(nodes[t].left, result);
        result.push_back(nodes[t].cell);
        collect(nodes[t].right, result);
    }

    void collect(vector<int>& result) {
        result.clear();
        collect(root, result);
    }

    void releaseTree(int t) {
        if (!t) return;
        push(t);
        int l = nodes[t].left;
        int r = nodes[t].right;
        releaseTree(l);
        releaseTree(r);
        int cell = nodes[t].cell;
        tileUsed[tileId[cell]] = false;
        cellNode[cell] = 0;
        freeNodes.push_back(t);
    }
} treap;

bool buildDetour(int u, int v, int side, int height, Detour& d) {
    d.count = 0;
    int ur = u / SIDE, uc = u % SIDE;
    int vr = v / SIDE, vc = v % SIDE;

    if (ur == vr) {
        if (side > 0 && ur + height >= SIDE) return false;
        if (side < 0 && ur - height < 0) return false;

        for (int k = 1; k <= height; ++k) {
            d.cells[d.count++] = (ur + side * k) * SIDE + uc;
        }
        d.cells[d.count++] = (ur + side * height) * SIDE + vc;
        for (int k = height - 1; k >= 1; --k) {
            d.cells[d.count++] = (ur + side * k) * SIDE + vc;
        }
    } else {
        if (side > 0 && uc + height >= SIDE) return false;
        if (side < 0 && uc - height < 0) return false;

        for (int k = 1; k <= height; ++k) {
            d.cells[d.count++] = ur * SIDE + uc + side * k;
        }
        d.cells[d.count++] = vr * SIDE + uc + side * height;
        for (int k = height - 1; k >= 1; --k) {
            d.cells[d.count++] = vr * SIDE + uc + side * k;
        }
    }
    return true;
}

            oppositeU = (ur + side) * SIDE + uc;
            oppositeV = (ur + side) * SIDE + vc;
        } else {
            oppositeU = ur * SIDE + uc + side;
            oppositeV = vr * SIDE + uc + side;
        }

        if (oppositeU < 0 || oppositeU >= CELLS ||
            oppositeV < 0 || oppositeV >= CELLS) {
            continue;
        }

        int nodeU = treap.cellNode[oppositeU];
        int nodeV = treap.cellNode[oppositeV];
        if (!nodeU || !nodeV) continue;

        int rankU = treap.getRank(nodeU);
        int rankV = treap.getRank(nodeV);
        if (rankV != rankU + 1) continue;

        applyTwoOpt(rank, rankU);
        return true;
    }
    return false;
}

    if (n < 3) return false;

    int endpoint = treap.nodes[treap.nodeAt(n - 1)].cell;
    int candidates[4];
    int candidateCount = 0;

    for (int cell : neighbors[endpoint]) {
        int node = treap.cellNode[cell];
        if (!node) continue;

        int rank = treap.getRank(node);
        if (rank >= n - 2) continue;
        candidates[candidateCount++] = rank;
    }

    if (candidateCount == 0) return false;
    int rank = candidates[randomInt(candidateCount)];

    auto [a, suffix] = treap.split(treap.root, rank + 1);
    treap.applyReverse(suffix);
    treap.root = treap.merge(a, suffix);
    return true;
}

int main() {
    startTime = Clock::now();

    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int si, sj;
    cin >> si >> sj;

    for (int i = 0; i < CELLS; ++i) cin >> tileId[i];
    for (int i = 0; i < CELLS; ++i) cin >> cellValue[i];

    for (int r = 0; r < SIDE; ++r) {
        for (int c = 0; c < SIDE; ++c) {
            int cell = r * SIDE + c;
            if (r > 0) neighbors[cell].push_back(cell - SIDE);
            if (r + 1 < SIDE) neighbors[cell].push_back(cell + SIDE);
            if (c > 0) neighbors[cell].push_back(cell - 1);
            if (c + 1 < SIDE) neighbors[cell].push_back(cell + 1);
        }
    }

    rngState ^= static_cast<uint64_t>(
        chrono::duration_cast<chrono::nanoseconds>(
            Clock::now().time_since_epoch()).count());
    random32();

    int startCell = si * SIDE + sj;

    for (int run = 0; run < 200 && elapsedSeconds() < 0.225; ++run) {
        fill(tileUsed, tileUsed + CELLS, false);
        dfsPath.clear();
        dfsPathScore = 0;
        dfsNodeCount = 0;
        dfsNodeLimit = 25000 + run * 1000;
        constructDFS(startCell);
        if (elapsedSeconds() > 0.245) break;
    }

    if (initialBestPath.empty()) {
        initialBestPath.push_back(startCell);
        initialBestScore = cellValue[startCell];
    }

    buildTreapFrom(initialBestPath);
    bestPath = initialBestPath;
    bestScore = currentScore;

    constexpr double GROW_END = 0.35;
    while (elapsedSeconds() < GROW_END) {
        for (int iteration = 0; iteration < 32 && elapsedSeconds() < GROW_END; ++iteration) {
            int operation = randomInt(100);
            bool changed = false;

            if (operation < 50) {
                changed = insertRectangleDetour();
            } else if (operation < 75) {
                changed = insertAtEndpoint();
            } else if (operation < 88) {
                changed = tryTwoOpt();
            } else {
                changed = tryRotation();
            }

            if (changed) rememberBest();
        }
    }

    constexpr double SA_END = 1.66;
    constexpr double FINAL_END = 1.82;
    const double saStart = elapsedSeconds();

    while (elapsedSeconds() < SA_END) {
        double progress = (elapsedSeconds() - saStart) / max(0.001, SA_END - saStart);
        progress = clamp(progress, 0.0, 1.0);
        double temperature = 120.0 * pow(0.02, progress) + 0.3;

        for (int iteration = 0; iteration < 64 && elapsedSeconds() < SA_END; ++iteration) {
            int operation = randomInt(100);

            if (operation < 38) {
                if (insertRectangleDetour()) rememberBest();
            } else if (operation < 56) {
                if (insertAtEndpoint()) rememberBest();
            } else if (operation < 76) {
                RangeProposal proposal;
                if (proposeChordDeletion(proposal)) {
                    bool force = (randomInt(256) == 0);
                    if (acceptLoss(proposal.loss, temperature, force)) {
                        deleteRange(proposal.left, proposal.right);
                    }
                }
            } else if (operation < 83) {
                RangeProposal proposal;
                if (proposeEndpointDeletion(proposal)) {
                    bool force = (randomInt(256) == 0);
                    if (acceptLoss(proposal.loss, temperature, force)) {
                        deleteRange(proposal.left, proposal.right);
                    }
                }
            } else if (operation < 92) {
                tryTwoOpt();
            } else {
                tryRotation();
            }
        }
    }

    while (elapsedSeconds() < FINAL_END) {
        for (int iteration = 0; iteration < 64 && elapsedSeconds() < FINAL_END; ++iteration) {
            int operation = randomInt(100);

            if (operation < 48) {
                if (insertRectangleDetour()) rememberBest();
            } else if (operation < 73) {
                if (insertAtEndpoint()) rememberBest();
            } else if (operation < 82) {
                RangeProposal proposal;
                if (proposeChordDeletion(proposal) && proposal.loss == 0) {
                    deleteRange(proposal.left, proposal.right);
                }
            } else if (operation < 92) {
                tryTwoOpt();
            } else {
                tryRotation();
            }
        }
    }

    vector<int> answer = bestPath;
    if (answer.empty()) answer.push_back(startCell);

    string directions;
    directions.reserve(answer.size());
    for (int i = 1; i < static_cast<int>(answer.size()); ++i) {
        int a = answer[i - 1], b = answer[i];
        int ar = a / SIDE, ac = a % SIDE;
        int br = b / SIDE, bc = b % SIDE;

        if (br < ar) directions.push_back('U');
        else if (br > ar) directions.push_back('D');
        else if (bc < ac) directions.push_back('L');
        else directions.push_back('R');
    }

    cout << directions << '\n';
    return 0;
}
