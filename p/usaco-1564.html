import sys


def main():
    it = iter(map(int, sys.stdin.buffer.read().split()))
    n = next(it)
    k = next(it)

    # Aggregate identical mooves.
    weights = {}
    nn = n * n
    for _ in range(k):
        x = next(it) - 1
        y = next(it) - 1
        z = next(it) - 1
        key = (x * n + y) * n + z
        weights[key] = weights.get(key, 0) + 1
    del it

    # For a set S of cells containing M, a moove contributes
    # w * [x in S] * [y not in S] * [z not in S].
    # Expanding gives linear, pairwise, and triple terms.
    linear = [0] * n
    pair = [[0] * n for _ in range(n)]
    triple = [[[0] * n for _ in range(n)] for _ in range(n)]

    for key, w in weights.items():
        x = key // nn
        rem = key - x * nn
        y, z = divmod(rem, n)

        linear[x] += w

        pair[x][y] -= w
        pair[y][x] -= w
        pair[x][z] -= w
        pair[z][x] -= w

        triple[x][y][z] += w
        triple[x][z][y] += w
        triple[y][x][z] += w
        triple[y][z][x] += w
        triple[z][x][y] += w
        triple[z][y][x] += w

    size = 1 << n
    scores = [0] * size

    # Fast conversion from a power of two to its bit index.
    bit_index = [0] * size
    for i in range(n):
        bit_index[1 << i] = i

    # Process masks according to their least significant set bit.
    # If v is that bit, all other set bits are above v. Scores of
    # masks containing only those higher bits have already been computed.
    for v in range(n - 1, -1, -1):
        base = v + 1
        m = n - base
        subsets = 1 << m

        # delta[mask] contains all pair and triple terms involving v
        # and cells in mask, excluding the linear term.
        delta = [0] * subsets
        pair_v = pair[v]
        triple_v = triple[v]
        linear_v = linear[v]
        v_bit = 1 << v

        scores[v_bit] = linear_v

        for mask in range(1, subsets):
            bit = mask & -mask
            u = base + bit_index[bit]
            rest = mask ^ bit

            value = delta[rest] + pair_v[u]
            row = triple_v[u]

            rr = rest
            while rr:
                b = rr & -rr
                value += row[base + bit_index[b]]
                rr ^= b

            delta[mask] = value

            higher_mask = mask << base
            scores[higher_mask | v_bit] = (
                scores[higher_mask] + linear_v + value
            )

    best = max(scores)
    boards = scores.count(best)
    sys.stdout.write(f"{best} {boards}")


if __name__ == "__main__":
    main()

import sys


def main() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    k = data[1]

    size = 1 << n
    coeff = [0] * size

    idx = 2
    for _ in range(k):
        x = data[idx] - 1
        y = data[idx + 1] - 1
        z = data[idx + 2] - 1
        idx += 3

        bx = 1 << x
        by = 1 << y
        bz = 1 << z

        # m_x * (1 - m_y) * (1 - m_z)
        # = m_x - m_x*m_y - m_x*m_z + m_x*m_y*m_z
        coeff[bx] += 1
        coeff[bx | by] -= 1
        coeff[bx | bz] -= 1
        coeff[bx | by | bz] += 1

    # Subset zeta transform:
    # after this, coeff[mask] = score for the board whose M positions are exactly mask
    arr = coeff
    for i in range(n):
        bit = 1 << i
        step = bit << 1

        for start in range(0, size, step):
            left = start
            right = start + bit

            for j in range(bit):
                arr[right + j] += arr[left + j]

    best = -10**18
    count = 0

    for value in arr:
        if value > best:
            best = value
            count = 1
        elif value == best:
            count += 1

    sys.stdout.write(f"{best} {count}\n")


if __name__ == "__main__":
    main()

import sys


def solve():
    read = sys.stdin.buffer.readline
    n, k = map(int, read().split())

    size = 1 << n
    cell_bit = [1 << i for i in range(n)]
    dp = [0] * size

    for _ in range(k):
        x, y, z = map(int, read().split())

        pair_mask = cell_bit[y - 1] | cell_bit[z - 1]
        triple_mask = pair_mask | cell_bit[x - 1]

        dp[pair_mask] += 1
        dp[triple_mask] -= 1

    # Subset zeta transform.
    # After this, dp[mask] is the score when exactly the cells
    # in mask contain O.
    step = 1
    while step < size:
        jump = step << 1
        for start in range(step, size, jump):
            end = start + step
            for mask in range(start, end):
                dp[mask] += dp[mask - step]
        step = jump

    maximum_score = max(dp)
    board_count = dp.count(maximum_score)

    print(maximum_score, board_count)


if __name__ == "__main__":
    solve()
