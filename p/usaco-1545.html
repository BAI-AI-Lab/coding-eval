题面

来源：USACO 官网。模型收到的是同一份题面的文字。

There are $N$ ($1\le N\le 2\cdot 10^5$) cows labeled $1\dots N$ on Farmer John's farm, where each cow lives in its own barn. Each cow $i$ has a best friend $a_i$ ($1\le a_i\le N$). Cows can be best friends with themselves, and multiple cows can have the same best friend. The cows love to party, so they have decided to throw a party for $M$ ($1\le M\le 2\cdot 10^5$) consecutive nights.

On night $i$, cow $c_i$ will decide to throw a party of type $t_i$ at its barn, where $t_i\in \texttt{"COW"}$. This party will exist for all future nights as well, until cow $c_i$ decides to throw a party of a different type.

Every night, each cow will attempt to go to a party. If a cow is not hosting a party, they will check their best friend's barn, and if there is no party there, will follow their best friend to wherever they are going (who might also follow their best friend and so on). It is possible that a cow might never find a party and will then give up for the night.

Compute for each night, the number of cows that end up at a party of type $C$, $O$, and $W$ respectively.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $N$, the number of cows.

The second line contains $a_1,\dots, a_N$, where $a_i$ is cow $i$'s best friend.

The third line contains $M$, the number of nights.

The next $M$ lines each contain an integer $c_i$ ($1\le c_i\le N$) and a character $v_i$, representing the cow that is throwing the party and the party type respectively.

OUTPUT FORMAT (print output to the terminal / stdout):

Output $M$ lines, where the $i$th consists of $3$ space-separated integers, the number of cows going to parties of type $C$, $O$, and $W$ on the $i$th night, respectively.

SAMPLE INPUT:

5
2 3 4 5 4
4
2 C
4 C
4 W
2 O

SAMPLE OUTPUT:

2 0 0
5 0 0
2 0 3
0 2 3

On night $1$, there is only one party of type $C$ at barn $2$, which only cows $1$ and $2$ attend.

On night $2$, there is a new party of type $C$ at barn $4$, which cows $3$, $4$, and $5$ can now reach.

On night $3$, the party at barn $4$ is changed to type $W$, affecting cows $3$, $4$ and $5$.

On night $4$, the party at barn $2$ is changed to type $O$, affecting cows $1$ and $2$.

SCORING:

  • Input 2: $N, M\leq 100$
  • Inputs 3-4: $N, M\leq 4000$
  • Inputs 5-9: $\{a_i\}$ is a permutation of $\{1,\dots, N\}$
  • Inputs 10-21: No additional constraints

#include <bits/stdc++.h>

struct Fenwick {
    int n = 0;
    vector<long long> bit;

    Fenwick() {}
    explicit Fenwick(int n_) { init(n_); }

    void init(int n_) {
        n = n_;
        bit.assign(n + 1, 0);
    }

    void add(int i, long long v) {
        for (; i <= n; i += i & -i) bit[i] += v;
    }

    long long sum(int i) const {
        long long res = 0;
        for (; i > 0; i -= i & -i) res += bit[i];
        return res;
    }

    // Range addition, point queries.
    void rangeAdd(int l, int r, long long v) {
        if (l > r) return;
        add(l, v);
        if (r + 1 <= n) add(r + 1, -v);
    }

    long long pointQuery(int i) const {
        return sum(i);
    }
};

struct CycleComponent {
    vector<int> nodes;
    int offset = 0;
    set<int> parties; // Local positions on the cycle.
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int> a(N + 1);
    for (int i = 1; i <= N; i++) cin >> a[i];

    int M;
    cin >> M;

    // Remove non-cycle vertices using in-degree peeling.
    vector<int> deg(N + 1, 0);
    for (int i = 1; i <= N; i++) deg[a[i]]++;

    queue<int> q;
    for (int i = 1; i <= N; i++) {
        if (deg[i] == 0) q.push(i);
    }

    while (!q.empty()) {
        int v = q.front();
        q.pop();
        if (--deg[a[v]] == 0) q.push(a[v]);
    }

    vector<char> isCycle(N + 1, false);
    for (int i = 1; i <= N; i++) {
        if (deg[i] > 0) isCycle[i] = true;
    }

    vector<CycleComponent> comps;
    vector<int> compOf(N + 1, -1), cyclePos(N + 1, -1);
    vector<char> seen(N + 1, false);
    vector<int> roots;

    int totalCycleVertices = 0;

    for (int i = 1; i <= N; i++) {
        if (!isCycle[i] || seen[i]) continue;

        int id = (int)comps.size();
        CycleComponent comp;
        comp.offset = totalCycleVertices;

        int v = i, p = 0;
        do {
            seen[v] = true;
            compOf[v] = id;
            cyclePos[v] = p++;
            comp.nodes.push_back(v);
            roots.push_back(v);
            v = a[v];
        } while (v != i);

        totalCycleVertices += p;
        comps.push_back(std::move(comp));
    }

    // Each cycle vertex roots a tree directed towards that cycle vertex.
    vector<vector<int>> children(N + 1);
    for (int i = 1; i <= N; i++) {
        if (!isCycle[i]) children[a[i]].push_back(i);
    }

    vector<int> parent(N + 1, 0), treeRoot(N + 1, 0);
    vector<int> order;
    order.reserve(N);

    vector<int> st;
    for (int r : roots) {
        parent[r] = 0;
        treeRoot[r] = r;
        st.push_back(r);

        while (!st.empty()) {
            int v = st.back();
            st.pop_back();
            order.push_back(v);

            for (int u : children[v]) {
                parent[u] = v;
                treeRoot[u] = r;
                st.push_back(u);
            }
        }
    }

    vector<long long> subtreeSize(N + 1, 1);
    for (auto it = order.rbegin(); it != order.rend(); ++it) {
        int v = *it;
        if (parent[v]) subtreeSize[parent[v]] += subtreeSize[v];
    }

    // Heavy-light decomposition of the rooted forest.
    vector<int> heavy(N + 1, 0);
    for (int v = 1; v <= N; v++) {
        for (int u : children[v]) {
            if (!heavy[v] || subtreeSize[u] > subtreeSize[heavy[v]]) {
                heavy[v] = u;
            }
        }
    }

    vector<int> head(N + 1, 0), pos(N + 1, 0), rev(N + 1, 0);
    int timer = 1;

    vector<pair<int, int>> decomposeStack;
    for (int r : roots) decomposeStack.push_back({r, r});

    while (!decomposeStack.empty()) {
        auto [start, h] = decomposeStack.back();
        decomposeStack.pop_back();

        int v = start;
        while (v) {
            head[v] = h;
            pos[v] = timer;
            rev[timer] = v;
            timer++;

            for (int u : children[v]) {
                if (u != heavy[v]) decomposeStack.push_back({u, u});
            }
            v = heavy[v];
        }
    }

    // F(v): if v is unmarked, the sum of subtree sizes of the topmost
    // parties below v; if v is marked, F(v) = subtreeSize[v].
    Fenwick treeDelta(N);
    vector<set<int>> chainMarks(N + 1);

    auto nearestMarkedAncestor = [&](int v) {
        while (v) {
            auto &marks = chainMarks[head[v]];
            auto it = marks.upper_bound(pos[v]);
            if (it != marks.begin()) {
                --it;
                return rev[*it];
            }
            v = parent[head[v]];
        }
        return 0;
    };

    // Add delta on the path from v upwards, stopping just below p.
    // p = 0 means update through the root.
    auto addVisiblePath = [&](int v, int p, long long delta) {
        if (p == 0) {
            while (v) {
                treeDelta.rangeAdd(pos[head[v]], pos[v], delta);
                v = parent[head[v]];
            }
            return;
        }

        while (head[v] != head[p]) {
            treeDelta.rangeAdd(pos[head[v]], pos[v], delta);
            v = parent[head[v]];
        }
        treeDelta.rangeAdd(pos[p] + 1, pos[v], delta);
    };

    // Number of cows from each cycle tree that still reach the cycle.
    Fenwick cycleWeights(totalCycleVertices);
    for (const auto &comp : comps) {
        for (int i = 0; i < (int)comp.nodes.size(); i++) {
            cycleWeights.add(comp.offset + i + 1,
                             subtreeSize[comp.nodes[i]]);
        }
    }

    auto cycleRangeSum = [&](const CycleComponent &comp, int l, int r) {
        if (l > r) return 0LL;
        return cycleWeights.sum(comp.offset + r + 1)
             - cycleWeights.sum(comp.offset + l);
    };

    // For a marked vertex, sum of subtree sizes of its topmost marked
    // descendants in the virtual party tree.
    vector<long long> below(N + 1, 0);
    vector<int> color(N + 1, 0); // 0 = no party, 1 = C, 2 = O, 3 = W.
    long long answer[4] = {0, 0, 0, 0};

    auto insertParty = [&](int v, int newColor) {
        long long previousTopmost = treeDelta.pointQuery(pos[v]);
        int p = nearestMarkedAncestor(v);
        long long delta = subtreeSize[v] - previousTopmost;

        addVisiblePath(v, p, delta);
        below[v] = previousTopmost;
        if (p) below[p] += delta;

        color[v] = newColor;
        chainMarks[head[v]].insert(pos[v]);

        if (p) {
            // The new party takes delta attendees from its nearest party ancestor.
            answer[color[p]] -= delta;
            answer[newColor] += delta;
            return;
        }

        if (!isCycle[v]) {
            // The residual weight of this cycle tree decreases.
            int r = treeRoot[v];
            CycleComponent &comp = comps[compOf[r]];
            int j = cyclePos[r];

            if (!comp.parties.empty()) {
                auto it = comp.parties.lower_bound(j);
                if (it == comp.parties.end()) it = comp.parties.begin();
                int nextParty = comp.nodes[*it];
                answer[color[nextParty]] -= delta;
            }

            answer[newColor] += delta;
            cycleWeights.add(comp.offset + j + 1, -delta);
            return;
        }

        // Inserting a party directly on a cycle.
        CycleComponent &comp = comps[compOf[v]];
        int i = cyclePos[v];
        int len = (int)comp.nodes.size();
        long long gained;

        if (comp.parties.empty()) {
            gained = cycleRangeSum(comp, 0, len - 1);
        } else {
            auto it = comp.parties.lower_bound(i);
            if (it == comp.parties.begin()) it = comp.parties.end();
            --it;
            int previous = *it;

            auto jt = comp.parties.upper_bound(i);
            if (jt == comp.parties.end()) jt = comp.parties.begin();
            int nextParty = comp.nodes[*jt];

            // Residual weights in the cyclic arc (previous, i].
            if (previous < i) {
                gained = cycleRangeSum(comp, previous + 1, i);
            } else {
                gained = cycleRangeSum(comp, previous + 1, len - 1)
                       + cycleRangeSum(comp, 0, i);
            }

            answer[color[nextParty]] -= gained;
        }

        answer[newColor] += gained;

        // The old residual weight at this root was exactly delta.
        cycleWeights.add(comp.offset + i + 1, -delta);
        comp.parties.insert(i);
    };

    auto cyclePartyBasin = [&](int v) {
        CycleComponent &comp = comps[compOf[v]];
        int i = cyclePos[v];
        int len = (int)comp.nodes.size();

        auto it = comp.parties.lower_bound(i);
        if (it == comp.parties.begin()) it = comp.parties.end();
        --it;
        int previous = *it;

        // Residual weights strictly between previous and i on the cycle.
        long long residual;
        if (previous < i) {
            residual = cycleRangeSum(comp, previous + 1, i - 1);
        } else {
            residual = cycleRangeSum(comp, previous + 1, len - 1)
                     + cycleRangeSum(comp, 0, i - 1);
        }

        return subtreeSize[v] - below[v] + residual;
    };

    while (M--) {
        int v;
        char type;
        cin >> v >> type;

        int newColor = (type == 'C' ? 1 : (type == 'O' ? 2 : 3));

        if (color[v] == 0) {
            insertParty(v, newColor);
        } else if (color[v] != newColor) {
            long long basin;
            if (isCycle[v]) basin = cyclePartyBasin(v);
            else basin = subtreeSize[v] - below[v];

            answer[color[v]] -= basin;
            answer[newColor] += basin;
            color[v] = newColor;
        }

        cout << answer[1] << ' ' << answer[2] << ' ' << answer[3] << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

struct Info {
    int sz = 0;
    int cnt[3] = {0, 0, 0};
};

static inline void addInfo(Info &a, const Info &b, int sign = 1) {
    a.sz += sign * b.sz;
    for (int t = 0; t < 3; ++t) a.cnt[t] += sign * b.cnt[t];
}

struct LinkCutTree {
    int n;
    vector<array<int, 2>> ch;
    vector<int> fa, stk;
    vector<unsigned char> rev;
    vector<Info> base, vir, sum;

    explicit LinkCutTree(int n_) : n(n_), ch(n_ + 1), fa(n_ + 1, 0),
        stk(n_ + 1), rev(n_ + 1, 0), base(n_ + 1), vir(n_ + 1), sum(n_ + 1) {
        for (int i = 1; i <= n; ++i) {
            ch[i] = {0, 0};
            base[i].sz = 1;
            pull(i);
        }
    }

    bool isRoot(int x) const {
        int p = fa[x];
        return p == 0 || (ch[p][0] != x && ch[p][1] != x);
    }

    void pull(int x) {
        sum[x] = base[x];
        addInfo(sum[x], vir[x]);
        if (ch[x][0]) addInfo(sum[x], sum[ch[x][0]]);
        if (ch[x][1]) addInfo(sum[x], sum[ch[x][1]]);
    }

    void push(int x) {
        if (!rev[x]) return;
        swap(ch[x][0], ch[x][1]);
        if (ch[x][0]) rev[ch[x][0]] ^= 1;
        if (ch[x][1]) rev[ch[x][1]] ^= 1;
        rev[x] = 0;
    }

    void rotate(int x) {
        int y = fa[x], z = fa[y];
        int dx = (ch[y][1] == x);
        int dy = (ch[z][1] == y);

        if (!isRoot(y)) ch[z][dy] = x;
        fa[x] = z;

        int b = ch[x][dx ^ 1];
        ch[y][dx] = b;
        if (b) fa[b] = y;

        ch[x][dx ^ 1] = y;
        fa[y] = x;

        pull(y);
        pull(x);
    }

    void splay(int x) {
        int top = 0, y = x;
        stk[top++] = y;
        while (!isRoot(y)) {
            y = fa[y];
            stk[top++] = y;
        }
        while (top) push(stk[--top]);

        while (!isRoot(x)) {
            y = fa[x];
            int z = fa[y];
            if (!isRoot(y)) {
                bool zigzig = ((ch[y][0] == x) == (ch[z][0] == y));
                rotate(zigzig ? y : x);
            }
            rotate(x);
        }
    }

    void access(int x) {
        int y = 0;
        for (int u = x; u; u = fa[u]) {
            splay(u);
            if (ch[u][1]) addInfo(vir[u], sum[ch[u][1]], 1);
            if (y) addInfo(vir[u], sum[y], -1);
            ch[u][1] = y;
            pull(u);
            y = u;
        }
        splay(x);
    }

    void makeRoot(int x) {
        access(x);
        rev[x] ^= 1;
    }

    int findRoot(int x) {
        access(x);
        while (true) {
            push(x);
            if (!ch[x][0]) break;
            x = ch[x][0];
        }
        splay(x);
        return x;
    }

    bool link(int x, int y) {
        makeRoot(x);
        if (findRoot(y) == x) return false;
        access(y);
        fa[x] = y;
        addInfo(vir[y], sum[x], 1);
        pull(y);
        return true;
    }

    void cut(int x, int y) {
        makeRoot(x);
        access(y);
        // x and y are directly adjacent, with x the left child of y.
        ch[y][0] = 0;
        fa[x] = 0;
        pull(y);
    }

    Info componentInfo(int x) {
        return sum[findRoot(x)];
    }

    void changeParty(int x, int oldType, int newType) {
        access(x);
        if (oldType >= 0) --base[x].cnt[oldType];
        ++base[x].cnt[newType];
        pull(x);
    }
};

struct DSU {
    vector<int> p, sz;
    explicit DSU(int n) : p(n + 1), sz(n + 1, 1) {
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) {
        return p[x] == x ? x : p[x] = find(p[x]);
    }
    void unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return;
        if (sz[a] < sz[b]) swap(a, b);
        p[b] = a;
        sz[a] += sz[b];
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;
    vector<int> a(N + 1);
    for (int i = 1; i <= N; ++i) cin >> a[i];

    DSU dsu(N);
    vector<int> indeg(N + 1, 0);
    for (int i = 1; i <= N; ++i) {
        dsu.unite(i, a[i]);
        ++indeg[a[i]];
    }

    // Mark vertices lying on directed cycles by removing all tree vertices.
    queue<int> q;
    for (int i = 1; i <= N; ++i) {
        if (indeg[i] == 0) q.push(i);
    }
    while (!q.empty()) {
        int v = q.front(); q.pop();
        if (--indeg[a[v]] == 0) q.push(a[v]);
    }
    vector<unsigned char> onCycle(N + 1, 0);
    for (int i = 1; i <= N; ++i) onCycle[i] = (indeg[i] > 0);

    vector<int> comp(N + 1);
    for (int i = 1; i <= N; ++i) comp[i] = dsu.find(i);

    LinkCutTree lct(N);

    // Each component initially has one extra edge closing its unique cycle.
    // All other edges are stored in the link-cut forest.
    vector<int> extra(N + 1, 0);
    for (int i = 1; i <= N; ++i) {
        if (i == a[i]) {
            extra[comp[i]] = i;
        } else if (!lct.link(i, a[i])) {
            extra[comp[i]] = i;
        }
    }

    int M;
    cin >> M;

    vector<int> partyType(N + 1, -1);
    long long answer[3] = {0, 0, 0};

    auto typeOf = [](char c) {
        if (c == 'C') return 0;
        if (c == 'O') return 1;
        return 2;
    };

    while (M--) {
        int v;
        char vc;
        cin >> v >> vc;
        int nt = typeOf(vc);

        if (partyType[v] != -1) {
            int ot = partyType[v];
            int attendees = lct.componentInfo(v).sz;
            answer[ot] -= attendees;
            answer[nt] += attendees;
            lct.changeParty(v, ot, nt);
            partyType[v] = nt;
        } else {
            Info before = lct.componentInfo(v);
            int ot = -1;
            for (int t = 0; t < 3; ++t) {
                if (before.cnt[t] > 0) ot = t;
            }

            int id = comp[v];
            int ex = extra[id];

            if (ot == -1) {
                // The component still contains its original directed cycle.
                if (v == ex) {
                    // Removing the already omitted edge breaks the cycle.
                    extra[id] = 0;
                } else if (onCycle[v]) {
                    // Replace the removed cycle edge by the omitted one,
                    // preserving undirected connectivity.
                    lct.cut(v, a[v]);
                    lct.link(ex, a[ex]);
                    extra[id] = 0;
                } else {
                    // A tree edge is cut; the cycle remains in the other part.
                    lct.cut(v, a[v]);
                }
            } else {
                // This component already has a party at its root.
                lct.cut(v, a[v]);
            }

            int attendees = lct.componentInfo(v).sz;
            if (ot != -1) answer[ot] -= attendees;
            answer[nt] += attendees;

            lct.changeParty(v, -1, nt);
            partyType[v] = nt;
        }

        cout << answer[0] << ' ' << answer[1] << ' ' << answer[2] << '\n';
    }

    return 0;
}

    }

    }

        siz[x] = 1 + vir[x] + siz[ch[x][0]] + siz[ch[x][1]];
    }

        int dx = (ch[y][1] == x);
        int b = ch[x][dx ^ 1];

            else if (ch[z][1] == y) ch[z][1] = x;

        ch[x][dx ^ 1] = y;
        fa[y] = x;

            int y = fa[x], z = fa[y];
            if (!isRoot(y)) {
                if ((ch[y][0] == x) ^ (ch[z][0] == y)) rotate(x);

            }
            rotate(x);

            splay(u);

        access(x);

struct Fenwick {
    int n;
    vector<long long> bit;

    Fenwick() : n(0) {}
    Fenwick(int n) { init(n); }

    void init(int n_) {
        n = n_;
        bit.assign(n + 1, 0);
    }

    void add(int idx, long long delta) {
        for (++idx; idx <= n; idx += idx & -idx) bit[idx] += delta;
    }

    long long sumPrefix(int idx) const {
        long long res = 0;
        for (; idx > 0; idx -= idx & -idx) res += bit[idx];
        return res;
    }

    long long rangeSum(int l, int r) const {
        if (l > r) return 0;
        return sumPrefix(r + 1) - sumPrefix(l);
    }
};

struct SegTree {
    int n;
    vector<pair<int, int>> seg;

    SegTree() : n(1) {}
    SegTree(int sz) { init(sz); }

    void init(int sz) {
        n = 1;
        while (n < max(1, sz)) n <<= 1;
        seg.assign(2 * n, {-1, -1});
    }

        int p = idx + n;
        seg[p] = {dep, id};
        for (p >>= 1; p; p >>= 1) {
            seg[p] = max(seg[p << 1], seg[p << 1 | 1]);
        }
    }

    pair<int, int> query(int l, int r) const {
        if (l > r) return {-1, -1};
        l += n;
        r += n;
        pair<int, int> res = {-1, -1};
        while (l <= r) {
            if (l & 1) res = max(res, seg[l++]);
            if (!(r & 1)) res = max(res, seg[r--]);
            l >>= 1;
            r >>= 1;
        }
        return res;
    }

class MinCountSegmentTree {
    int n;
    vector<int> mn, cnt, lazy;

    static pair<int, int> combine(pair<int, int> a, pair<int, int> b) {
        if (a.first < b.first) return a;
        if (b.first < a.first) return b;
        return {a.first, a.second + b.second};
    }

    void build(int p, int l, int r) {
        cnt[p] = r - l + 1;
        if (l == r) return;
        int mid = (l + r) / 2;
        build(p * 2, l, mid);
        build(p * 2 + 1, mid + 1, r);
    }

    void apply(int p, int value) {
        mn[p] += value;
        lazy[p] += value;
    }

    void push(int p) {
        if (lazy[p] == 0) return;
        apply(p * 2, lazy[p]);
        apply(p * 2 + 1, lazy[p]);
        lazy[p] = 0;
    }

    void pull(int p) {
        mn[p] = min(mn[p * 2], mn[p * 2 + 1]);
        cnt[p] = 0;
        if (mn[p * 2] == mn[p]) cnt[p] += cnt[p * 2];
        if (mn[p * 2 + 1] == mn[p]) cnt[p] += cnt[p * 2 + 1];
    }

    void add(int p, int l, int r, int ql, int qr, int value) {
        if (ql <= l && r <= qr) {
            apply(p, value);
            return;
        }
        push(p);
        int mid = (l + r) / 2;
        if (ql <= mid) add(p * 2, l, mid, ql, qr, value);
        if (qr > mid) add(p * 2 + 1, mid + 1, r, ql, qr, value);
        pull(p);
    }

    pair<int, int> query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return {mn[p], cnt[p]};

        push(p);
        int mid = (l + r) / 2;
        if (qr <= mid) return query(p * 2, l, mid, ql, qr);
        if (ql > mid) return query(p * 2 + 1, mid + 1, r, ql, qr);

        return combine(
            query(p * 2, l, mid, ql, qr),
            query(p * 2 + 1, mid + 1, r, ql, qr)
        );
    }

public:
    explicit MinCountSegmentTree(int n)
        : n(n), mn(4 * n, 0), cnt(4 * n, 0), lazy(4 * n, 0) {
        build(1, 0, n - 1);
    }

    void add(int l, int r, int value) {
        add(1, 0, n - 1, l, r, value);
    }

    pair<int, int> query(int l, int r) {
        return query(1, 0, n - 1, l, r);
    }
};

/*
 * Range "chmax" updates and point maximum queries.
 * Each update inserts an active vertex over its subtree interval.
 * The deepest active vertex containing a point is its nearest active ancestor.
 */
class RangeMaxPointQuery {
    int base;
    const vector<int>& depth;
    vector<int> tag;

    void relax(int& best, int candidate) const {
        if (candidate < 0) return;
        if (best < 0 || depth[candidate] > depth[best]) {
            best = candidate;
        }
    }

public:
    explicit RangeMaxPointQuery(const vector<int>& depth) : depth(depth) {
        base = 1;
        while (base < (int)depth.size()) base <<= 1;
        tag.assign(2 * base, -1);
    }

    void update(int l, int r, int vertex) {
        l += base;
        r += base + 1;
        while (l < r) {
            if (l & 1) {
                relax(tag[l], vertex);
                ++l;
            }
            if (r & 1) {
                --r;
                relax(tag[r], vertex);
            }
            l >>= 1;
            r >>= 1;
        }
    }

    int query(int position) const {
        int best = -1;
        for (int p = position + base; p > 0; p >>= 1) {
            relax(best, tag[p]);
        }
        return best;
    }
};

            int j = i + (i & -i);
            if (j <= n) bit[j] += bit[i];

        for (int i = index + 1; i <= n; i += i & -i) {
            bit[i] += delta;
        }

    int prefix(int r) const {
        int result = 0;
        for (int i = r; i > 0; i -= i & -i) result += bit[i];
        return result;
    }

    int rangeSum(int l, int r) const {

using ll = long long;

struct Fenwick {
    int n;
    vector<ll> bit;

    explicit Fenwick(int n = 0) : n(n), bit(n + 1, 0) {}

    void build(const vector<ll>& values) {
        n = (int)values.size();
        bit.assign(n + 1, 0);
        for (int i = 1; i <= n; ++i) {
            bit[i] += values[i - 1];
            int parent = i + (i & -i);
            if (parent <= n) bit[parent] += bit[i];
        }
    }

    void add(int pos, ll delta) {
        for (int i = pos + 1; i <= n; i += i & -i) {
            bit[i] += delta;
        }
    }

    // Sum of [0, end).
    ll prefixSum(int end) const {
        ll result = 0;
        for (int i = end; i > 0; i -= i & -i) {
            result += bit[i];
        }
        return result;
    }

    // Sum of [left, right], inclusive.
    ll rangeSum(int left, int right) const {
        if (left > right) return 0;
        return prefixSum(right + 1) - prefixSum(left);
    }
};

// Supports inserting an active vertex's subtree and finding the deepest
// active ancestor of a given Euler-tour position.
struct AncestorMax {
    int base;
    const vector<int>& depth;
    vector<int> best;

    AncestorMax(int n, const vector<int>& depth) : depth(depth) {
        base = 1;
        while (base < n) base <<= 1;
        best.assign(base << 1, -1);
    }

    bool better(int a, int b) const {
        if (a == -1) return false;
        if (b == -1) return true;
        if (depth[a] != depth[b]) return depth[a] > depth[b];
        return a > b;
    }

    void relax(int index, int candidate) {
        if (better(candidate, best[index])) {
            best[index] = candidate;
        }
    }

    void update(int left, int right, int candidate) {
        left += base;
        right += base;
        while (left <= right) {
            if (left & 1) relax(left++, candidate);
            if (!(right & 1)) relax(right--, candidate);
            left >>= 1;
            right >>= 1;
        }
    }

    int query(int pos) const {
        int answer = -1;
        for (int index = base + pos; index > 0; index >>= 1) {
            if (better(best[index], answer)) {
                answer = best[index];
            }
        }
        return answer;
    }
};

struct Fenwick {
    int n;
    vector<ll> bit;

    explicit Fenwick(int n = 0) : n(n), bit(n + 1, 0) {}

    void build(const vector<ll>& values) {
        n = (int)values.size();
        bit.assign(n + 1, 0);
        for (int i = 1; i <= n; ++i) bit[i] = values[i - 1];
        for (int i = 1; i <= n; ++i) {
            int j = i + (i & -i);
            if (j <= n) bit[j] += bit[i];
        }
    }

    // Indices are 0-based.
    void add(int index, ll value) {
        for (int i = index + 1; i <= n; i += i & -i) {
            bit[i] += value;
        }
    }

    ll prefixSum(int index) const {
        ll result = 0;
        for (int i = index + 1; i > 0; i -= i & -i) {
            result += bit[i];
        }
        return result;
    }

    ll rangeSum(int left, int right) const {
        if (left > right || right < 0 || left >= n) return 0;
        left = max(left, 0);
        right = min(right, n - 1);
        return prefixSum(right) - prefixSum(left - 1);
    }
};

// Supports range chmax updates and point maximum queries.
// No pushes are necessary because updates only take maxima.
struct RangeMaxPointQuery {
    int size;
    vector<ll> tag;

    explicit RangeMaxPointQuery(int n) {
        size = 1;
        while (size < n) size <<= 1;
        tag.assign(size << 1, 0);
    }

    // Left and right are 1-based and inclusive.
    void update(int left, int right, ll value) {
        left += size - 1;
        right += size - 1;

        while (left <= right) {
            if (left & 1) {
                tag[left] = max(tag[left], value);
                ++left;
            }
            if (!(right & 1)) {
                tag[right] = max(tag[right], value);
                --right;
            }
            left >>= 1;
            right >>= 1;
        }
    }

    // Position is 1-based.
    ll query(int position) const {
        ll result = 0;
        for (int p = position + size - 1; p > 0; p >>= 1) {
            result = max(result, tag[p]);
        }
        return result;
    }
};
