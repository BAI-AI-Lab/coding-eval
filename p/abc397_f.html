import sys


def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))
    del data

    # Positions of the strict right-to-left maxima of P[i] - i,
    # where P[i] is the number of distinct values in the prefix of length i.
    seen = bytearray(n + 1)
    record_pos = []
    for i in range(n - 1):
        x = a[i]
        if seen[x]:
            record_pos.append(i)
        else:
            seen[x] = 1
    record_pos.append(n - 1)

    # suffix[j] = distinct values in a[j:]
    suffix = [0] * n
    seen = bytearray(n + 1)
    distinct = 0
    for i in range(n - 1, -1, -1):
        x = a[i]
        if not seen[x]:
            seen[x] = 1
            distinct += 1
        suffix[i] = distinct

    k = len(record_pos)

    # before[p]: number of record positions strictly smaller than p.
    before = [0] * n
    ri = 0
    for pos in range(n):
        while ri < k and record_pos[ri] < pos:
            ri += 1
        before[pos] = ri

    # DSU for finding the nearest still-active record to the left.
    parent = list(range(k + 1))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    last = [0] * (n + 1)
    first = 1                 # Records use 1-based DSU indices.
    current_max = 0
    answer = 0

    for j in range(1, n):
        x = a[j - 1]
        previous = last[x]
        last[x] = j

        if previous <= record_pos[first - 1]:
            current_max += 1
        else:
            # Remove the nearest active record before `previous`.
            target = find(before[previous])
            parent[target] = find(target - 1)

            if target == first:
                first += 1
                while parent[first] != first:
                    first += 1

        if j >= 2:
            candidate = current_max + suffix[j]
            if candidate > answer:
                answer = candidate

    print(answer)


if __name__ == "__main__":
    main()

import sys


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    A = [0] + data[1:1 + N]

    # P[i] = number of distinct values in A[1..i]
    P = [0] * (N + 1)
    seen = bytearray(N + 1)
    cnt = 0
    for i in range(1, N + 1):
        x = A[i]
        if not seen[x]:
            seen[x] = 1
            cnt += 1
        P[i] = cnt

    # S[i] = number of distinct values in A[i..N]
    S = [0] * (N + 2)
    seen = bytearray(N + 1)
    cnt = 0
    for i in range(N, 0, -1):
        x = A[i]
        if not seen[x]:
            seen[x] = 1
            cnt += 1
        S[i] = cnt

    size = 1
    log = 0
    while size < N + 2:
        size <<= 1
        log += 1

    NEG = -10**9
    seg = [NEG] * (2 * size)
    lazy = [0] * size

    last = [0] * (N + 1)
    ans = 0

    for j in range(1, N):
        x = A[j]
        p = last[x]

        left = p if p > 1 else 1
        right = j - 2

        # Existing candidates i with i >= p gain +1.
        if left <= right:
            l = left + size
            r = right + 1 + size
            l0, r0 = l, r

            # Push lazy values on boundary paths.
            for h in range(log, 0, -1):
                if ((l0 >> h) << h) != l0:
                    k = l0 >> h
                    z = lazy[k]
                    if z:
                        lc = k << 1
                        rc = lc | 1
                        seg[lc] += z
                        if lc < size:
                            lazy[lc] += z
                        seg[rc] += z
                        if rc < size:
                            lazy[rc] += z
                        lazy[k] = 0

                if ((r0 >> h) << h) != r0:
                    k = (r0 - 1) >> h
                    z = lazy[k]
                    if z:
                        lc = k << 1
                        rc = lc | 1
                        seg[lc] += z
                        if lc < size:
                            lazy[lc] += z
                        seg[rc] += z
                        if rc < size:
                            lazy[rc] += z
                        lazy[k] = 0

            # Range add +1.
            while l < r:
                if l & 1:
                    seg[l] += 1
                    if l < size:
                        lazy[l] += 1
                    l += 1
                if r & 1:
                    r -= 1
                    seg[r] += 1
                    if r < size:
                        lazy[r] += 1
                l >>= 1
                r >>= 1

            # Rebuild boundary paths.
            for h in range(1, log + 1):
                if ((l0 >> h) << h) != l0:
                    k = l0 >> h
                    v = seg[k << 1]
                    w = seg[(k << 1) | 1]
                    if w > v:
                        v = w
                    seg[k] = v

                if ((r0 >> h) << h) != r0:
                    k = (r0 - 1) >> h
                    v = seg[k << 1]
                    w = seg[(k << 1) | 1]
                    if w > v:
                        v = w
                    seg[k] = v

        # Add new candidate i = j - 1.
        if j >= 2:
            idx = j - 1
            pos = idx + size
            seg[pos] = P[idx] + 1
            pos >>= 1

            while pos:
                v = seg[pos << 1]
                w = seg[(pos << 1) | 1]
                if w > v:
                    v = w
                seg[pos] = v
                pos >>= 1

            best_middle = seg[1]
            total = best_middle + S[j + 1]
            if total > ans:
                ans = total

        last[x] = j

    print(ans)


if __name__ == "__main__":
    solve()

import sys

class LazySegTree:
    __slots__ = ('n', 'size', 'log', 'data', 'lazy')

    def __init__(self, arr):
        n = len(arr)
        size = 1
        log = 0
        while size < n:
            size <<= 1
            log += 1

        self.n = n
        self.size = size
        self.log = log

        data = [-10**15] * (2 * size)
        lazy = [0] * (2 * size)

        for i, v in enumerate(arr):
            data[size + i] = v

        for i in range(size - 1, 0, -1):
            a = data[2 * i]
            b = data[2 * i + 1]
            data[i] = a if a >= b else b

        self.data = data
        self.lazy = lazy

    def range_add(self, l, r, x):
        if l >= r:
            return

        size = self.size
        log = self.log
        data = self.data
        lazy = self.lazy

        l += size
        r += size
        l0, r0 = l, r

        for i in range(log, 0, -1):
            if ((l0 >> i) << i) != l0:
                k = l0 >> i
                z = lazy[k]
                if z:
                    data[2 * k] += z
                    lazy[2 * k] += z
                    data[2 * k + 1] += z
                    lazy[2 * k + 1] += z
                    lazy[k] = 0
            if ((r0 >> i) << i) != r0:
                k = r0 >> i
                z = lazy[k]
                if z:
                    data[2 * k] += z
                    lazy[2 * k] += z
                    data[2 * k + 1] += z
                    lazy[2 * k + 1] += z
                    lazy[k] = 0

        while l < r:
            if l & 1:
                data[l] += x
                lazy[l] += x
                l += 1
            if r & 1:
                r -= 1
                data[r] += x
                lazy[r] += x
            l >>= 1
            r >>= 1

        l, r = l0, r0
        for i in range(1, log + 1):
            if ((l >> i) << i) != l:
                k = l >> i
                a = data[2 * k]
                b = data[2 * k + 1]
                data[k] = a if a >= b else b
            if ((r >> i) << i) != r:
                k = r >> i
                a = data[2 * k]
                b = data[2 * k + 1]
                data[k] = a if a >= b else b

    def range_max(self, l, r):
        if l >= r:
            return -10**15

        size = self.size
        log = self.log
        data = self.data
        lazy = self.lazy

        l += size
        r += size
        l0, r0 = l, r

        for i in range(log, 0, -1):
            if ((l0 >> i) << i) != l0:
                k = l0 >> i
                z = lazy[k]
                if z:
                    data[2 * k] += z
                    lazy[2 * k] += z
                    data[2 * k + 1] += z
                    lazy[2 * k + 1] += z
                    lazy[k] = 0
            if ((r0 >> i) << i) != r0:
                k = r0 >> i
                z = lazy[k]
                if z:
                    data[2 * k] += z
                    lazy[2 * k] += z
                    data[2 * k + 1] += z
                    lazy[2 * k + 1] += z
                    lazy[k] = 0

        res = -10**15
        while l < r:
            if l & 1:
                v = data[l]
                if v > res:
                    res = v
                l += 1
            if r & 1:
                r -= 1
                v = data[r]
                if v > res:
                    res = v
            l >>= 1
            r >>= 1

        return res


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    A = [0] + data[1:1 + N]

    pre = [0] * (N + 1)
    seen = [0] * (N + 1)
    cnt = 0
    for i in range(1, N + 1):
        x = A[i]
        if seen[x] == 0:
            cnt += 1
            seen[x] = 1
        pre[i] = cnt

    suf = [0] * (N + 2)
    seen = [0] * (N + 1)
    cnt = 0
    for i in range(N, 0, -1):
        x = A[i]
        if seen[x] == 0:
            cnt += 1
            seen[x] = 1
        suf[i] = cnt

    seg = LazySegTree(pre[:N])

    last = [0] * (N + 1)
    ans = 0

    for r in range(1, N):
        x = A[r]
        p = last[x]

        # For l <= p, A[r] is already in [l, r-1],
        # so distinct(l, r) does not increase.
        # For l > p, it increases by 1.
        seg.range_add(p, r, 1)
        last[x] = r

        if r >= 2:
            mx = seg.range_max(1, r)  # l = 2..r
            total = mx + suf[r + 1]
            if total > ans:
                ans = total

    print(ans)


if __name__ == "__main__":
    main()
