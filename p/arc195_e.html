import sys

MOD = 998244353


def main():
    it = iter(map(int, sys.stdin.buffer.read().split()))
    n = next(it)
    q = next(it)

    a = [0] * (n + 1)
    for i in range(2, n + 1):
        a[i] = next(it) % MOD

    # Modular inverses of 1, ..., n+1.
    inv = [0] * (n + 2)
    inv[1] = 1
    for i in range(2, n + 2):
        inv[i] = MOD - (MOD // i) * inv[MOD % i] % MOD

    # prefix_low[k]: sum of coefficients for edges below u.
    # prefix_mid[k]: sum of A_i / i for i <= k.
    prefix_low = [0] * (n + 1)
    prefix_mid = [0] * (n + 1)

    for k in range(2, n + 1):
        low = a[k] * (2 * (k - 1)) % MOD
        low = low * inv[k] % MOD * inv[k + 1] % MOD
        prefix_low[k] = (prefix_low[k - 1] + low) % MOD
        prefix_mid[k] = (prefix_mid[k - 1] + a[k] * inv[k]) % MOD

    total_trees = 1
    for i in range(1, n):
        total_trees = total_trees * i % MOD

    answers = []
    for _ in range(q):
        u = next(it)
        v = next(it)

        if u == 1:
            expected = (prefix_mid[v - 1] + a[v]) % MOD
        else:
            expected = prefix_low[u - 1]
            expected += a[u] * (u - 1) % MOD * inv[u] % MOD
            expected += prefix_mid[v - 1] - prefix_mid[u]
            expected += a[v]
            expected %= MOD

        answers.append(str(expected * total_trees % MOD))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

        A[i] = next(it) % MOD

    inv = [0] * (N + 2)
    inv[1] = 1
    for i in range(2, N + 2):
        inv[i] = MOD - (MOD // i) * inv[MOD % i] % MOD

    # P[x] = sum over edges i of A[i] * Pr(vertex x is in subtree of i)
    P = [0] * (N + 1)
    prefP = 0
    for x in range(2, N + 1):
        P[x] = (prefP + A[x]) % MOD
        prefP = (prefP + A[x] * inv[x]) % MOD

    # Q[x] = sum over edges i of A[i] * Pr(both endpoints are in subtree of i)
    # when the smaller endpoint is x.
    Qarr = [0] * (N + 1)
    prefQ = 0
    for x in range(2, N + 1):
        Qarr[x] = (prefQ + A[x] * inv[x]) % MOD
        if x < N:
            term = A[x] * 2 % MOD
            term = term * inv[x] % MOD
            term = term * inv[x + 1] % MOD
            prefQ = (prefQ + term) % MOD

    fact = 1
    for i in range(1, N):
        fact = fact * i % MOD

    answers = []

        v = next(it)

        if u > v:
            u, v = v, u

        value = (P[u] + P[v] - 2 * Qarr[u]) % MOD
        ans = fact * value % MOD
        answers.append(str(ans))

import sys

MOD = 998244353

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N, Q = data[0], data[1]
    A = [0] * (N + 2)

    idx = 2
    for i in range(2, N + 1):
        A[i] = data[idx] % MOD
        idx += 1

    # modular inverses up to N+2
    inv = [0] * (N + 3)
    inv[1] = 1
    for i in range(2, N + 3):
        inv[i] = MOD - (MOD // i) * inv[MOD % i] % MOD

    # S_D[i] = sum_{k=2..i} A_k / k
    # S_B[i] = sum_{k=2..i} A_k * 2*(k-1)/(k*(k+1))
    # C[i] = A_i * (i-1)/i
    S_D = [0] * (N + 1)
    S_B = [0] * (N + 1)
    C = [0] * (N + 1)

    for i in range(2, N + 1):
        D = A[i] * inv[i] % MOD
        S_D[i] = (S_D[i - 1] + D) % MOD

        B = A[i] * 2 * (i - 1) % MOD * inv[i] % MOD * inv[i + 1] % MOD
        S_B[i] = (S_B[i - 1] + B) % MOD

        C[i] = A[i] * (i - 1) % MOD * inv[i] % MOD

    # total number of possible P: (N-1)!
    fact = 1
    for i in range(1, N):
        fact = fact * i % MOD

    out = []
    for _ in range(Q):
        u = data[idx]
        v = data[idx + 1]
        idx += 2

        if u > v:
            u, v = v, u

        if u == 1:
            ans = S_D[v - 1] + A[v]
        else:
            ans = (
                S_B[u - 1]
                + C[u]
                + (S_D[v - 1] - S_D[u])
                + A[v]
            )

        ans %= MOD
        out.append(str(ans * fact % MOD))

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
