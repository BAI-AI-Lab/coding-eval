Bakery Survey This is an interactive problem. Your city has n bakeries (where n is a power of 2), and bakery i specializes in one type of cake a_i. You want to determine d — the number of distinct cake types available in the city. You don't know the values of a_1, ..., a_n. However, your friend can help you by tasting cakes. Your friend has a memory capacity of k (where k is also a power of 2), which works as follows: Your friend's memory is a queue S. You can perform two types of operations: 1. Query operation: Ask your friend to taste the cake from bakery c. This will: - Tell you whether a_c is already in S (the last k cake types tasted) - Add a_c to the end of S - If |S| > k, remove the front element from S 2. Reset operation: Clear your friend's memory, making S empty. Your goal is to find d while minimizing the total cost of operations. This problem is graded based on the total cost of operations. The cost is calculated as: Total Cost = (number of resets) × n + (number of queries) + 1 You must use at most 100,000 operations in total. Input The first line contains two integers n and k (1 ≤ k ≤ n ≤ 1024, both k and n are powers of 2). Interaction To perform an operation, output one line in one of the following formats: ? c — Ask your friend to taste the cake from bakery c (1 ≤ c ≤ n). R — Reset your friend's memory. After a query operation, read a single character: - Y (Yes) if a_c is in the memory S - N (No) if a_c is not in the memory S When you have found the answer, output: ! d — where d is the number of distinct cake types. After printing the answer, your program should terminate immediately. To flush your output, use: - fflush(stdout) or cout.flush() in C++ - System.out.flush() in Java - stdout.flush() in Python Example Input: 4 2 N N Y N N N N Output: ? 1 ? 2 ? 3 ? 4 R ? 4 ? 1 ? 2 ! 3 Time limit: 4 seconds Memory limit: 512 MB ## Scoring All feasibility and interaction limits apply before scoring; invalid output receives zero. Points are averaged over test files. For the logarithmic branch, write x for the original minimizing quantity and a for its physical 60-point reference. The improvement factor is r=a/x, and the upper-band award is P=min(100,60+ln(r)). Here ln is natural: r=1 corresponds to 60 points, and on the uncapped logarithmic upper band, r=exp(P-60). Each additional point multiplies the physical improvement factor by e (divides a minimizing objective by e or multiplies a maximizing objective by e, as applicable). The lower-band, zero-objective, and completion rules below still apply. C=n*reset_count+query_count+1 is the total operation cost, and R>0 is the stored reference cost. Define the objective x=C, its physical reference a=A=R, and the lower-band calculation L=60*R/C. ```text P = 100, if x = 0 P = L, if L <= 60 and x > 0 P = min(100, 60 + ln(r)), otherwise ``` Zero is tested on the valid original count, cost, or error, before the lower-band test. A zero reference anchor has no positive improvement branch; positive objectives keep their lower-band score. The +1 in C is part of the defined operation cost, so C is always positive. The checker reports P/100.

        for (int j = 0; j < k; j++) {
            int id = block + j + 1;
            if (ask(id) == 'N') cand.push_back(id);

        if (reps.empty()) {
            reps = cand;
            continue;

        vector<int> comparedCount(q, 0);

            if (!compared[b].test(g)) {
                compared[b].set(g);
                comparedCount[b]++;
                if (state[b] == 0 && comparedCount[b] == p) {
                    state[b] = 2;
                    resolved++;

                            needed = true;
                            break;
                        }
                    }
                    if (!needed) break;

                    int ge = min(p, gs + h);
                    int gsize = ge - gs;

                    if (gs > 0) {
                        int padding = k - gsize - dsize + 1;
                        for (int i = 0; i < padding; i++) ask(reps[0]);

                    for (int g = gs; g < ge; g++) ask(reps[g]);

                    for (int di = ds; di < de; di++) {
                        int b = unresolved[di];
                        if (ask(cand[b]) == 'Y') state[b] = 1;
                    }
                }
            }
        }

        for (int b = 0; b < q; b++) {
            if (state[b] != 1) reps.push_back(cand[b]);
        }
    }

    cout << "! " << reps.size() << '\n';
    cout.flush();
    return 0;
}

using namespace std;

int n, k;

char ask(int c) {
    cout << "? " << c << endl;
    char response;
    cin >> response;
    return response;
}

struct DSU {
    vector<int> parent, size;
    explicit DSU(int n) : parent(n), size(n, 1) {
        iota(parent.begin(), parent.end(), 0);
    }
    int find(int x) {
        if (parent[x] == x) return x;
        return parent[x] = find(parent[x]);
    }
    void unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return;
        if (size[a] < size[b]) swap(a, b);
        parent[b] = a;
        size[a] += size[b];
    }
};

int solveWithCapacityOne(const vector<int>& items) {
    int m = (int)items.size();
    if (m <= 1) return m;

    vector<vector<pair<int, int>>> adjacency(m);
    vector<char> used;
    int edgeCount = 0;

    auto addEdge = [&](int u, int v) {
        int id = edgeCount++;
        adjacency[u].push_back({v, id});
        adjacency[v].push_back({u, id});
    };

    for (int i = 0; i < m; ++i) {
        for (int j = i + 1; j < m; ++j) {
            addEdge(i, j);
        }
    }

    if (m % 2 == 0) {
        for (int i = 0; i < m; i += 2) {
            addEdge(i, i + 1);
        }
    }

    used.assign(edgeCount, 0);
    vector<int> nextEdge(m, 0), st, circuit;
    st.push_back(0);

    while (!st.empty()) {
        int v = st.back();
        while (nextEdge[v] < (int)adjacency[v].size() &&
               used[adjacency[v][nextEdge[v]].second]) {
            ++nextEdge[v];
        }

        if (nextEdge[v] == (int)adjacency[v].size()) {
            circuit.push_back(v);
            st.pop_back();
        } else {
            auto [to, id] = adjacency[v][nextEdge[v]++];
            if (used[id]) continue;
            used[id] = 1;
            st.push_back(to);
        }
    }

    reverse(circuit.begin(), circuit.end());

    DSU dsu(m);
    ask(items[circuit[0]]);
    for (int i = 1; i < (int)circuit.size(); ++i) {
        int previous = circuit[i - 1];
        int current = circuit[i];
        if (ask(items[current]) == 'Y') {
            dsu.unite(previous, current);
        }
    }

    int answer = 0;
    for (int i = 0; i < m; ++i) {
        if (dsu.find(i) == i) ++answer;
    }
    return answer;
}

int solveWithSmallCapacity(const vector<int>& items) {
    vector<int> representatives;

    for (int item : items) {
        bool duplicate = false;

        for (size_t start = 0;
             start < representatives.size() && !duplicate;
             start += k) {
            int length = (int)min((size_t)k,
                                  representatives.size() - start);

            for (int t = 0; t < k; ++t) {
                ask(representatives[start + t % length]);
            }
            if (ask(item) == 'Y') duplicate = true;
        }

        if (!duplicate) representatives.push_back(item);
    }

    return (int)representatives.size();
}

long long estimatedMergeCost(int a, int b) {
    if (a == 0 || b == 0) return 0;
    long long aChunks = (a + k - 1) / k;
    long long bBatches = (b + k - 1) / k;
    return 2LL * aChunks * (bBatches * k + b);
}

vector<int> mergeSets(const vector<int>& first,
                      const vector<int>& second) {
    if (first.empty()) return second;
    if (second.empty()) return first;

    const vector<int>* A = &first;
    const vector<int>* B = &second;

    if (estimatedMergeCost((int)second.size(), (int)first.size()) <
        estimatedMergeCost((int)first.size(), (int)second.size())) {
        swap(A, B);
    }

    vector<int> result = *A;
    result.reserve(A->size() + B->size());

    for (int bStart = 0; bStart < (int)B->size(); bStart += k) {
        int bLength = min(k, (int)B->size() - bStart);
        vector<char> duplicate(bLength, 0);

        for (int aStart = 0; aStart < (int)A->size(); aStart += k) {
            bool allFound = true;
            for (char value : duplicate) {
                if (!value) {
                    allFound = false;
                    break;
                }
            }
            if (allFound) break;

            int aLength = min(k, (int)A->size() - aStart);
            vector<int> padded(k);
            for (int t = 0; t < k; ++t) {
                padded[t] = (*A)[aStart + t % aLength];
            }

            for (int t = 0; t < k; ++t) ask(padded[t]);
            for (int j = 0; j < bLength; ++j) {
                if (ask((*B)[bStart + j]) == 'Y') {
                    duplicate[j] = 1;
                }
            }

            for (int t = k - 1; t >= 0; --t) ask(padded[t]);
            for (int j = bLength - 1; j >= 0; --j) {
                if (ask((*B)[bStart + j]) == 'Y') {
                    duplicate[j] = 1;
                }
            }
        }

        for (int j = 0; j < bLength; ++j) {
            if (!duplicate[j]) result.push_back((*B)[bStart + j]);
        }
    }

    return result;
}

vector<int> solveBlocks(const vector<vector<int>>& blocks, int left, int right) {
    if (right - left == 1) return blocks[left];
    int middle = (left + right) / 2;
    vector<int> a = solveBlocks(blocks, left, middle);
    vector<int> b = solveBlocks(blocks, middle, right);
    return mergeSets(a, b);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> k;

    vector<vector<int>> blocks(n / k);
    vector<int> candidates;

    for (int i = 1; i <= n; ++i) {
        if (ask(i) == 'N') {
            blocks[(i - 1) / k].push_back(i);
            candidates.push_back(i);
        }
    }

    int answer;
    if (k == n || candidates.size() <= 1) {
        answer = (int)candidates.size();
    } else if (k == 1) {
        answer = solveWithCapacityOne(candidates);
    } else if (k == 2) {
        answer = solveWithSmallCapacity(candidates);
    } else {
        answer = (int)solveBlocks(blocks, 0, (int)blocks.size()).size();
    }

    cout << "! " << answer << endl;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

class BakerySurvey {
private:
    int n, k;
    long long queryCount = 0;
    long long resetCount = 0;
    int nextLeaf = 0;
    int blockSize;

    bool ask(int c) {
        cout << "? " << c << '\n';
        cout.flush();

        char response;
        if (!(cin >> response)) exit(0);
        if (response != 'Y' && response != 'N') exit(0);

        ++queryCount;
        return response == 'Y';
    }

    void resetMemory() {
        cout << "R\n";
        cout.flush();
        ++resetCount;
    }

    vector<int> buildLeaf(int left, int right) {
        if (nextLeaf++ > 0) resetMemory();

        vector<int> representatives;
        for (int c = left + 1; c <= right; ++c) {
            if (!ask(c)) representatives.push_back(c);
        }
        return representatives;
    }

    vector<int> mergeSets(const vector<int>& source,
                          const vector<int>& target) {
        int a = static_cast<int>(source.size());
        int b = static_cast<int>(target.size());

        if (a == 0) return target;
        if (b == 0) return source;

        if (a + b <= k + 1) {
            for (int c : source) ask(c);

            vector<char> matched(b, false);
            for (int i = 0; i < b; ++i) {
                if (ask(target[i])) matched[i] = true;
            }

            vector<int> result = source;
            result.reserve(a + b);
            for (int i = 0; i < b; ++i) {
                if (!matched[i]) result.push_back(target[i]);
            }
            return result;
        }

        long long bestOperations = LLONG_MAX;
        int bestSourceChunk = 1;
        int bestTargetChunk = k;

        for (int s = 1; s <= k; ++s) {
            int t = k + 1 - s;
            long long sourceChunks = (a + s - 1) / s;
            long long targetChunks = (b + t - 1) / t;
            long long operations = sourceChunks * targetChunks * (k + 1LL);

            if (operations < bestOperations) {
                bestOperations = operations;
                bestSourceChunk = s;
                bestTargetChunk = t;
            }
        }

        int s = bestSourceChunk;
        int t = bestTargetChunk;

        vector<char> matched(b, false);

        for (int sourceLeft = 0; sourceLeft < a; sourceLeft += s) {
            int sourceRight = min(a, sourceLeft + s);

            for (int targetLeft = 0; targetLeft < b; targetLeft += t) {
                int targetRight = min(b, targetLeft + t);

                for (int i = sourceLeft; i < sourceRight; ++i) {
                    ask(source[i]);
                }
                for (int i = sourceLeft; i < sourceRight; ++i) {
                    ask(source[i]);
                }

                for (int i = targetLeft; i < targetRight; ++i) {
                    if (ask(target[i])) matched[i] = true;
                }

                int padding = t - (targetRight - targetLeft);
                for (int i = 0; i < padding; ++i) {
                    ask(source[sourceLeft]);
                }
            }
        }

        vector<int> result = source;
        result.reserve(a + b);
        for (int i = 0; i < b; ++i) {
            if (!matched[i]) result.push_back(target[i]);
        }
        return result;
    }

    vector<int> solve(int left, int right) {
        if (right - left == 1) {
            return buildLeaf(left * blockSize,
                             min(n, (left + 1) * blockSize) - 1);
        }

        int middle = (left + right) / 2;
        vector<int> first = solve(left, middle);
        vector<int> second = solve(middle, right);
        return mergeSets(second, first);
    }

public:
    BakerySurvey(int n, int k) : n(n), k(k) {
        blockSize = k + 1;
    }

    int run() {
        int leafCount = (n + blockSize - 1) / blockSize;
        vector<int> representatives = solve(0, leafCount);
        return static_cast<int>(representatives.size());
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    if (!(cin >> n >> k)) return 0;

    BakerySurvey solver(n, k);
    int answer = solver.run();

    cout << "! " << answer << '\n';
    cout.flush();
    return 0;
}
