import sys

MOD = 998244353


def main():
    data = sys.stdin.buffer.read().split()
    T = int(data[0])
    pos = 1
    answers = []

    for _ in range(T):
        H = int(data[pos])
        W = int(data[pos + 1])
        pos += 2

        n = H + W
        parent = list(range(n))
        size = [1] * n
        parity = [0] * n  # value(x) XOR value(parent[x])

        col_prefix = [0] * W
        components = n
        valid = True

        def find(x):
            if parent[x] != x:
                p = parent[x]
                root, d = find(p)
                d ^= parity[x]
                parent[x] = root
                parity[x] = d
                return root, d
            return x, 0

        def unite(a, b, required_xor):
            nonlocal components, valid
            ra, pa = find(a)
            rb, pb = find(b)

            if ra == rb:
                if (pa ^ pb) != required_xor:
                    valid = False
                return

            d = pa ^ pb ^ required_xor
            if size[ra] < size[rb]:
                parent[ra] = rb
                parity[ra] = d
                size[rb] += size[ra]
            else:
                parent[rb] = ra
                parity[rb] = d
                size[ra] += size[rb]
            components -= 1

        for i in range(H):
            s = data[pos]
            pos += 1
            row_prefix = 0

            for j, ch in enumerate(s):
                if ch == 65:  # 'A'
                    row_prefix ^= 1
                    col_prefix[j] ^= 1
                elif valid:  # 'B'
                    # Row variable i and column variable H+j.
                    unite(i, H + j, 1 ^ row_prefix ^ col_prefix[j])

            if row_prefix:
                valid = False

        if not valid or any(col_prefix):
            answers.append("0")
        else:
            answers.append(str(pow(2, components, MOD)))

    sys.stdout.write("\n".join(answers))


if __name__ == "__main__":
    main()

import sys

parent = []
xr = []


def find(x):
    p = parent
    xo = xr

    r = x
    acc = 0
    while p[r] != r:
        acc ^= xo[r]
        r = p[r]

    cur = x
    cur_acc = 0
    while p[cur] != cur:
        nxt = p[cur]
        w = xo[cur]
        p[cur] = r
        xo[cur] = acc ^ cur_acc
        cur_acc ^= w
        cur = nxt

    return r, acc

        invalid = False

        for row in grid:
            row_par = 0
            for j, ch in enumerate(row):
                if ch == 65:  # 'A'
                    row_par ^= 1
                    col_par[j] ^= 1
            if row_par:
                invalid = True

            answers.append("0")
            continue

        n = H + W
        parent = list(range(n))

import sys

MOD = 998244353

def solve():
    input = sys.stdin.buffer.readline
    T_line = input()
    if not T_line:
        return
    T = int(T_line)
    out = []

    for _ in range(T):
        H, W = map(int, input().split())
        V = H + W

        parent = list(range(V))
        size = [1] * V
        parity = [0] * V
        comps = V
        consistent = True
        valid = True

        # col_pref[j] = parity of A in column j above the current row
        col_pref = [0] * W

        for i in range(H):
            s = input().strip()
            # row_pref = parity of A in this row before the current column
            row_pref = 0

            for j, ch in enumerate(s):
                if ch == 65:  # 'A'
                    row_pref ^= 1
                    col_pref[j] ^= 1
                else:  # 'B'
                    if consistent:
                        # equation: r_i xor c_j = 1 xor row_pref xor col_pref[j]
                        k = 1 ^ row_pref ^ col_pref[j]

                        # find row i
                        ra = i
                        xa = 0
                        while parent[ra] != ra:
                            xa ^= parity[ra]
                            ra = parent[ra]

                        # find column j
                        rb = H + j
                        xb = 0
                        while parent[rb] != rb:
                            xb ^= parity[rb]
                            rb = parent[rb]

                        if ra == rb:
                            if (xa ^ xb) != k:
                                consistent = False
                        else:
                            t = k ^ xa ^ xb
                            if size[ra] < size[rb]:
                                parent[ra] = rb
                                parity[ra] = t
                                size[rb] += size[ra]
                            else:
                                parent[rb] = ra
                                parity[rb] = t
                                size[ra] += size[rb]
                            comps -= 1

            if row_pref:
                valid = False

        if valid and any(col_pref):
            valid = False

        if valid and consistent:
            out.append(str(pow(2, comps, MOD)))
        else:
            out.append("0")

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
