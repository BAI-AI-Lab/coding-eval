Count Beautiful Numbers

LeetCode 周赛 441 · 2025-03-15 · 困难

各位数字之积能被各位数字之和整除的正整数称为美丽数；求 [l, r] 内美丽数的个数。

要点：数位 DP：含 0 时积为 0，必满足；否则积为 2ᵃ3ᵇ5ᶜ7ᵈ，状态有限；暴力枚举超时。

from functools import lru_cache


class Solution:
    def beautifulNumbers(self, l: int, r: int) -> int:
        def count_up_to(x: int) -> int:
            if x <= 0:
                return 0

            digits = [int(c) for c in str(x)]
            n = len(digits)

            @lru_cache(None)
            def dfs(
                pos: int,
                digit_sum: int,
                digit_product: int,
                started: bool,
                tight: bool,
            ) -> int:
                if pos == n:
                    if not started:
                        return 0
                    return int(digit_product % digit_sum == 0)

                limit = digits[pos] if tight else 9
                total = 0

                for d in range(limit + 1):
                    next_tight = tight and d == digits[pos]

                    if not started and d == 0:
                        # Leading zero: the number has not started yet.
                        total += dfs(pos + 1, 0, 1, False, next_tight)
                    else:
                        total += dfs(
                            pos + 1,
                            digit_sum + d,
                            digit_product * d,
                            True,
                            next_tight,
                        )

                return total

            return dfs(0, 0, 1, False, True)

        return count_up_to(r) - count_up_to(l - 1)

from functools import lru_cache

class Solution:
    def beautifulNumbers(self, l: int, r: int) -> int:
        def count(n: int) -> int:
            if n <= 0:
                return 0

            digits = list(map(int, str(n)))
            length = len(digits)

            @lru_cache(None)
            def dp(pos: int, digit_sum: int, digit_product: int, tight: bool, started: bool) -> int:
                if pos == length:
                    return 1 if started and digit_product % digit_sum == 0 else 0

                limit = digits[pos] if tight else 9
                total = 0

                for d in range(limit + 1):
                    next_tight = tight and (d == digits[pos])

                    if not started and d == 0:
                        total += dp(pos + 1, digit_sum, digit_product, next_tight, False)
                    elif not started:
                        total += dp(pos + 1, d, d, next_tight, True)
                    else:
                        total += dp(pos + 1, digit_sum + d, digit_product * d, next_tight, True)

                return total

            return dp(0, 0, 1, True, False)

        return count(r) - count(l - 1)

class Solution:
    def beautifulNumbers(self, l: int, r: int) -> int:
        def count_no_zero_upto(n: int) -> int:
            """Count positive integers <= n with no digit 0."""
            if n <= 0:
                return 0
            s = str(n)
            L = len(s)
            total = 0
            for m in range(1, L):
                total += 9 ** m

            for i, ch in enumerate(s):
                digit = ord(ch) - 48
                rem = L - i - 1
                if digit > 0:
                    total += (digit - 1) * (9 ** rem)
                else:
                    break
            else:
                total += 1
            return total

        def build_ways(S: int, max_len: int):
            """
            ways[rem][sum][prod] = number of completions of length rem
            (digits 1..9, no zero) such that final digit sum is S and
            final product is divisible by S (i.e. prod modulo S becomes 0).
            """
            mul = [[(p * d) % S for d in range(10)] for p in range(S)]
            prev = [[0] * S for _ in range(S + 1)]
            prev[S][0] = 1
            ways = [prev]

            for _ in range(1, max_len):
                cur = [[0] * S for _ in range(S + 1)]
                for s in range(S + 1):
                    max_d = min(9, S - s)
                    if max_d <= 0:
                        continue
                    for p in range(S):
                        total = 0
                        for d in range(1, max_d + 1):
                            total += prev[s + d][mul[p][d]]
                        cur[s][p] = total
                ways.append(cur)
                prev = cur
            return ways

        def count_zf_beautiful_upto(n: int, S: int, ways) -> int:
            """Count zero-free beautiful numbers <= n with digit sum S."""
            if n <= 0:
                return 0
            s = str(n)
            L = len(s)
            start_prod = 1 % S
            ans = 0

            # All shorter lengths are automatically <= n.
            for m in range(1, L):
                ans += ways[m][0][start_prod]

            cur_sum = 0
            cur_prod = start_prod

            for i, ch in enumerate(s):
                limit = ord(ch) - 48
                rem = L - i - 1

                # Put a smaller non-zero digit here; the rest can be anything.
                for d in range(1, limit):
                    ns = cur_sum + d
                    if ns <= S:
                        np = (cur_prod * d) % S
                        ans += ways[rem][ns][np]

                # Continue with the exact digit of n.
                if limit == 0:
                    break
                if cur_sum + limit > S:
                    break
                cur_sum += limit
                cur_prod = (cur_prod * limit) % S
            else:
                if cur_sum == S and cur_prod == 0:
                    ans += 1

            return ans

        # Any number containing a zero is beautiful: product = 0.
        total_range = r - l + 1
        zf_range = count_no_zero_upto(r) - count_no_zero_upto(l - 1)
        zero_containing = total_range - zf_range

        max_len = len(str(r))
        max_sum = 9 * max_len

        zf_beautiful_r = 0
        zf_beautiful_lm = 0

        for S in range(1, max_sum + 1):
            ways = build_ways(S, max_len)
            zf_beautiful_r += count_zf_beautiful_upto(r, S, ways)
            zf_beautiful_lm += count_zf_beautiful_upto(l - 1, S, ways)

        zero_free_beautiful = zf_beautiful_r - zf_beautiful_lm
        return zero_containing + zero_free_beautiful
