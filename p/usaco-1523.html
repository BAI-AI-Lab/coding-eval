#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
const int64 MOD = 1000000007;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, A, B;
    int64 D;
    cin >> N >> D;
    cin >> A >> B;

    vector<int64> m(N);
    for (auto &x : m) cin >> x;
    sort(m.begin(), m.end());

    int64 answer = 0;
    auto addSquare = [&](int64 x) {
        int64 r = x % MOD;
        answer = (answer + r * r) % MOD;
    };

    // Only the initially largest A bottles need to be considered active.
    int inactive = N - A;
    for (int i = 0; i < inactive; ++i) addSquare(m[i]);

    vector<int64> a(m.begin() + inactive, m.end());
    vector<int64> pref(A + 1, 0), prefUpper(A + 1, 0);
    vector<int64> prefSq(A + 1, 0), prefUpperSq(A + 1, 0);

    for (int i = 0; i < A; ++i) {
        pref[i + 1] = pref[i] + a[i];
        prefUpper[i + 1] = prefUpper[i] + a[i] + D;

        int64 r1 = a[i] % MOD;
        int64 r2 = (a[i] + D) % MOD;
        prefSq[i + 1] = (prefSq[i] + r1 * r1) % MOD;
        prefUpperSq[i + 1] = (prefUpperSq[i] + r2 * r2) % MOD;
    }

    const int64 target = pref[A] + (int64)(A - B) * D;

    // At level L, bottle i has clamp(L, a[i], a[i]+D) milk.
    // Find the water-filling level for the required total amount.
    auto baseSum = [&](int64 L, int *upIndex = nullptr,
                       int *lowIndex = nullptr) -> int64 {
        int up = upper_bound(a.begin(), a.end(), L - D) - a.begin();
        int low = upper_bound(a.begin(), a.end(), L) - a.begin();

        if (upIndex) *upIndex = up;
        if (lowIndex) *lowIndex = low;

        return prefUpper[up]
             + (int64)(low - up) * L
             + (pref[A] - pref[low]);
    };

    int64 lo = 0, hi = 2000000000LL;
    while (lo < hi) {
        int64 mid = (lo + hi + 1) / 2;
        if (baseSum(mid) <= target) lo = mid;
        else hi = mid - 1;
    }

    int up, low;
    int64 base = baseSum(lo, &up, &low);
    int64 freeCount = low - up;
    int64 raised = target - base; // Number of free bottles at level lo+1.
    int64 atBase = freeCount - raised;

    auto rangeSum = [](const vector<int64> &p, int l, int r) -> int64 {
        return (p[r] - p[l] + MOD) % MOD;
    };

    // Bottles fixed at their upper bounds.
    answer = (answer + rangeSum(prefUpperSq, 0, up)) % MOD;
    // Bottles fixed at their initial amounts.
    answer = (answer + rangeSum(prefSq, low, A)) % MOD;

    int64 r0 = lo % MOD;
    int64 r1 = (lo + 1) % MOD;
    answer = (answer + (atBase % MOD) * r0 % MOD * r0) % MOD;
    answer = (answer + (raised % MOD) * r1 % MOD * r1) % MOD;

    cout << answer << '\n';
    return 0;
}

using namespace std;

using int64 = long long;
const int64 MOD = 1000000007LL;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    int64 D;
    cin >> N >> D;

    int A, B;
    cin >> A >> B;

    vector<int64> m(N);
    for (int i = 0; i < N; i++) cin >> m[i];

    sort(m.begin(), m.end());

    int outsideCount = N - A;

    int64 ans = 0;

    // Bottles outside the largest A never receive any milk.
    for (int i = 0; i < outsideCount; i++) {
        int64 v = m[i] % MOD;
        ans = (ans + v * v) % MOD;
    }

    vector<int64> active;
    active.reserve(A);
    for (int i = outsideCount; i < N; i++) {
        active.push_back(m[i]);
    }

    int64 K = (int64)A - B;      // number incremented after FN steals
    int64 totalInc = K * D;      // total increments among active bottles
    int64 cap = D;               // one bottle can be incremented at most D times

    int64 maxVal = active.back();
    int64 upper = maxVal + D;

    auto used = [&](int64 h) -> int64 {
        int64 sum = 0;
        for (int64 v : active) {
            if (h > v) {
                int64 diff = h - v;
                if (diff > cap) diff = cap;
                sum += diff;
                if (sum > totalInc) return sum;
            }
        }
        return sum;
    };

    int64 lo = 0, hi = upper + 1;
    while (hi - lo > 1) {
        int64 mid = lo + (hi - lo) / 2;
        if (used(mid) <= totalInc) lo = mid;
        else hi = mid;
    }

    int64 H = lo;
    int64 usedH = 0;

    for (int64 v : active) {
        int64 inc = 0;
        if (H > v) {
            inc = H - v;
            if (inc > cap) inc = cap;
        }

        usedH += inc;
        int64 finalVal = v + inc;
        int64 x = finalVal % MOD;
        ans = (ans + x * x) % MOD;
    }

    int64 rem = totalInc - usedH;

    if (rem > 0) {
        int64 add = (2LL * (H % MOD) + 1) % MOD;
        ans = (ans + (rem % MOD) * add) % MOD;
    }

    cout << ans % MOD << '\n';
    return 0;
}

    auto calc = [&](long long h) -> long long {
        long long s = 0;
        for (long long x : b) {
            if (x >= h) s += x;
            else if (x + D <= h) s += x + D;
            else s += h;
        }
        return s;
    };

    long long lo = b.front();
    long long hi = b.back() + D;

    while (lo < hi) {
        long long mid = lo + (hi - lo) / 2;
        if (calc(mid) >= target) hi = mid;
        else lo = mid + 1;
    }

    long long h = lo;
    long long Sh = calc(h);
    long long overshoot = Sh - target;

    const long long MOD = 1000000007LL;

    auto addSquare = [&](long long v, long long &ans) {
        v %= MOD;
        ans = (ans + v * v) % MOD;
    };

    long long ans = 0;

    // Bottles below the top A are never changed.
    for (int i = 0; i < bottom; i++) {
        addSquare(m[i], ans);
    }

    long long activeCount = 0;

    for (long long x : b) {
        if (x + D < h) {
            addSquare(x + D, ans);
        } else if (x >= h) {
            addSquare(x, ans);
        } else {
            activeCount++;
        }

    long long R = overshoot;

    long long hm = h % MOD;
    long long h2 = hm * hm % MOD;

    long long hm1 = (h - 1) % MOD;
    long long hm12 = hm1 * hm1 % MOD;

    long long cntH = (activeCount - R) % MOD;
    long long cntHm1 = R % MOD;

    ans = (ans + cntH * h2) % MOD;

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
constexpr int64 MOD = 1'000'000'007;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    int64 D;
    cin >> N >> D;

    int A, B;
    cin >> A >> B;

    vector<int64> milk(N);
    for (int64 &x : milk) cin >> x;
    sort(milk.begin(), milk.end());

    // Only the largest A bottles can ever be changed.
    vector<int64> active(milk.begin() + (N - A), milk.end());

    const int64 C = A - B;
    const int64 need = C * D;

    // Number of possible increment events with pre-increment value <= level.
    auto countEvents = [&](int64 level) -> int64 {
        int64 result = 0;
        for (int64 x : active) {
            if (level < x) break;
            result += min(D, level - x + 1);
        }
        return result;
    };

    // Find the level of the need-th smallest possible increment event.
    int64 low = active.front();
    int64 high = active.back() + D - 1;

    while (low < high) {
        int64 mid = low + (high - low) / 2;
        if (countEvents(mid) >= need) {
            high = mid;
        } else {
            low = mid + 1;
        }
    }

    const int64 q = low;
    const int64 selectedAtQ = need - countEvents(q - 1);

    auto squareMod = [](int64 x) -> int64 {
        x %= MOD;
        if (x < 0) x += MOD;
        return x * x % MOD;
    };

    int64 answer = 0;

    // The N-A smallest bottles never change.
    for (int i = 0; i < N - A; ++i) {
        answer = (answer + squareMod(milk[i])) % MOD;
    }

    int64 partialCount = 0;

    for (int64 x : active) {
        if (x > q) {
            // None of this bottle's increment events is selected.
            answer = (answer + squareMod(x)) % MOD;
        } else if (x + D - 1 < q) {
            // All D possible increment events are selected.
            answer = (answer + squareMod(x + D)) % MOD;
        } else {
            // This bottle ends with either q or q+1.
            ++partialCount;
        }
    }

    // Exactly selectedAtQ partial bottles receive the event of value q,
    // changing their final amount from q to q+1.
    answer = (answer
              + ((partialCount - selectedAtQ) % MOD) * squareMod(q)) % MOD;
    answer = (answer
              + (selectedAtQ % MOD) * squareMod(q + 1)) % MOD;

    cout << answer << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

constexpr int64 MOD = 1'000'000'007LL;

int64 squareMod(int64 x) {
    x %= MOD;
    return x * x % MOD;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    int64 D;
    cin >> N >> D;

    int A, B;
    cin >> A >> B;

    vector<int64> m(N);
    for (int64 &x : m) cin >> x;
    sort(m.rbegin(), m.rend());

    const int64 K = A - B;
    const int64 M = K * D;

    /*
       Only the initial A largest bottles ever participate.

       For an active bottle initially containing x, its possible marginal
       increment levels are:
           x, x+1, ..., x+D-1.

       After D days, exactly M = K*D marginal increments are chosen.
    */

    auto countLe = [&](int64 z) -> int64 {
        int64 count = 0;
        for (int i = 0; i < A; ++i) {
            if (z < m[i]) continue;
            count += min(D, z - m[i] + 1);
        }
        return count;
    };

    // Find the level of the M-th smallest marginal increment.
    int64 lo = m[A - 1];
    int64 hi = m[0] + D - 1;

    while (lo < hi) {
        int64 mid = lo + (hi - lo) / 2;
        if (countLe(mid) >= M) {
            hi = mid;
        } else {
            lo = mid + 1;
        }
    }

    const int64 z = lo;

    // Number of marginal increments strictly smaller than z.
    int64 below = 0;
    for (int i = 0; i < A; ++i) {
        if (z > m[i]) {
            below += min(D, z - m[i]);
        }
    }

    // Number of increments selected at level z.
    const int64 selectedAtZ = M - below;

    int64 answer = 0;

    for (int i = 0; i < N; ++i) {
        int64 finalAmount;

        if (i < A) {
            const int64 x = m[i];

            if (x > z) {
                // No increments selected for this bottle.
                finalAmount = x;
            } else if (x + D <= z) {
                // All D possible increments were selected.
                finalAmount = x + D;
            } else {
                // Baseline amount; selectedAtZ of these become z+1.
                finalAmount = z;
            }
        } else {
            // These bottles are never filled or stolen from.
            finalAmount = m[i];
        }

        answer = (answer + squareMod(finalAmount)) % MOD;
    }

    // Replacing z by z+1 changes the square sum by 2z+1.
    const int64 delta = (2 * (z % MOD) + 1) % MOD;
    answer = (answer + (selectedAtZ % MOD) * delta) % MOD;

    cout << answer << '\n';
    return 0;
}

using namespace std;

using int64 = long long;
using i128 = __int128_t;

static constexpr int64 MOD = 1'000'000'007LL;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, A, B;
    int64 D;

    cin >> N >> D;
    cin >> A >> B;

    vector<int64> m(N);
    for (int64 &x : m) cin >> x;

    sort(m.begin(), m.end(), greater<int64>());

    const int K = A - B;

    // Farmer John optimally fills these same A bottles every day.
    vector<int64> lower(A), upper(A), finalAmount(A);

    int64 target = 0;
    for (int i = 0; i < A; ++i) {
        lower[i] = m[i];
        upper[i] = m[i] + D;
        target += lower[i];
    }
    target += int64(K) * D;

    auto levelSum = [&](int64 h) -> int64 {
        int64 sum = 0;
        for (int i = 0; i < A; ++i) {
            sum += min(upper[i], max(lower[i], h));
        }
        return sum;
    };

    // Find the water level h such that clamping every selected bottle
    // to h gives a total not exceeding target.
    int64 low = 0;
    int64 high = m[0] + D;

    while (low < high) {
        int64 mid = low + (high - low + 1) / 2;
        if (levelSum(mid) <= target) {
            low = mid;
        } else {
            high = mid - 1;
        }
    }

    const int64 h = low;
    int64 currentSum = 0;

    for (int i = 0; i < A; ++i) {
        finalAmount[i] = min(upper[i], max(lower[i], h));
        currentSum += finalAmount[i];
    }

    // Distribute the remaining units among bottles whose level can
    // be increased from h to h+1.
    int64 remaining = target - currentSum;
    for (int i = 0; i < A && remaining > 0; ++i) {
        if (lower[i] <= h && h < upper[i]) {
            ++finalAmount[i];
            --remaining;
        }
    }

    int64 answer = 0;

    auto addSquare = [&](int64 x) {
        int64 square = static_cast<int64>(static_cast<i128>(x) * x % MOD);
        answer += square;
        if (answer >= MOD) answer -= MOD;
    };

    for (int i = A; i < N; ++i) {
        addSquare(m[i]);
    }
    for (int64 x : finalAmount) {
        addSquare(x);
    }

    cout << answer << '\n';
    return 0;
}
