#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N, M, K, L;
        cin >> N >> M >> K >> L;

        vector<int> flowers(K);
        for (int &x : flowers) cin >> x;

        vector<char> isDestination(N + 1, false);
        for (int i = 0; i < L; ++i) {
            int x;
            cin >> x;
            isDestination[x] = true;
        }

        vector<vector<int>> adj(N + 1);
        vector<pair<int, int>> edges(M);
        for (int i = 0; i < M; ++i) {
            int u, v;
            cin >> u >> v;
            edges[i] = {u, v};
            adj[u].push_back(v);
            adj[v].push_back(u);
        }

        vector<int> dist(N + 1, -1);
        queue<int> q;
        dist[1] = 0;
        q.push(1);

        int maxDist = 0;
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            maxDist = max(maxDist, dist[u]);

            for (int v : adj[u]) {
                if (dist[v] == -1) {
                    dist[v] = dist[u] + 1;
                    q.push(v);
                }
            }
        }

        vector<vector<int>> layers(maxDist + 1);
        for (int v = 1; v <= N; ++v) {
            layers[dist[v]].push_back(v);
        }

        // Directed acyclic graph containing exactly the edges that can be
        // used by shortest paths starting at farm 1.
        vector<vector<int>> dag(N + 1);
        for (auto [u, v] : edges) {
            if (dist[u] + 1 == dist[v]) {
                dag[u].push_back(v);
            } else if (dist[v] + 1 == dist[u]) {
                dag[v].push_back(u);
            }
        }

        sort(flowers.begin(), flowers.end(), [&](int a, int b) {
            return dist[a] < dist[b];
        });

        bool possible = true;
        for (int i = 1; i < K; ++i) {
            if (dist[flowers[i]] == dist[flowers[i - 1]]) {
                possible = false;
                break;
            }
        }

        vector<char> answer(N + 1, false);

        if (possible) {
            vector<int> chain;
            chain.reserve(K + 1);
            chain.push_back(1);
            for (int x : flowers) chain.push_back(x);

            vector<char> forward(N + 1, false);
            vector<char> backward(N + 1, false);

            // Process the portions between consecutive required flowers.
            // Their distance-layer intervals are disjoint, so altogether
            // these loops take O(N + M).
            for (int i = 0; i + 1 < (int)chain.size() && possible; ++i) {
                int a = chain[i];
                int b = chain[i + 1];
                int da = dist[a];
                int db = dist[b];

                forward[a] = true;
                for (int d = da; d < db; ++d) {
                    for (int u : layers[d]) {
                        if (!forward[u]) continue;
                        for (int v : dag[u]) {
                            forward[v] = true;
                        }
                    }
                }

                if (!forward[b]) {
                    possible = false;
                    break;
                }

                for (int d = db; d >= da; --d) {
                    for (int u : layers[d]) {
                        bool canReach = (u == b);

                        if (d < db) {
                            for (int v : dag[u]) {
                                if (backward[v]) {
                                    canReach = true;
                                    break;
                                }
                            }
                        }

                        backward[u] = canReach;
                        if (canReach && forward[u]) {
                            answer[u] = true;
                        }
                    }
                }

                for (int d = da; d <= db; ++d) {
                    for (int u : layers[d]) {
                        forward[u] = false;
                        backward[u] = false;
                    }
                }
            }

            if (possible) {
                // Final portion: from the last required flower (or farm 1)
                // to any reachable destination.
                int a = chain.back();
                int da = dist[a];

                forward[a] = true;
                for (int d = da; d < maxDist; ++d) {
                    for (int u : layers[d]) {
                        if (!forward[u]) continue;
                        for (int v : dag[u]) {
                            forward[v] = true;
                        }
                    }
                }

                for (int d = maxDist; d >= da; --d) {
                    for (int u : layers[d]) {
                        bool canReach = isDestination[u] && forward[u];

                        for (int v : dag[u]) {
                            if (backward[v]) {
                                canReach = true;
                                break;
                            }
                        }

                        backward[u] = canReach;
                        if (canReach && forward[u]) {
                            answer[u] = true;
                        }
                    }
                }

                possible = backward[a];
            }
        }

        string result(N - 1, '0');
        if (possible) {
            for (int v = 2; v <= N; ++v) {
                if (answer[v]) result[v - 2] = '1';
            }
        }

        cout << result << '\n';
    }

    return 0;
}

            edges[i] = {u, v};
            adj[u].push_back(v);
            adj[v].push_back(u);
        }

        q.push(1);

        while (!q.empty()) {

            q.pop();
            for (int u : adj[v]) {

                    q.push(u);
                }
            }

        }

        sort(flowers.begin(), flowers.end(), [&](int a, int b) {
            return dist[a] < dist[b];
        });

        for (int i = 1; i < K; i++) {
            if (dist[flowers[i]] == dist[flowers[i - 1]]) {
                possible = false;
            }
        }

        if (!possible) {
            cout << string(N - 1, '0') << '\n';
            continue;
        }

        vector<vector<int>> out(N + 1), in(N + 1);
        for (auto [u, v] : edges) {
            if (dist[u] + 1 == dist[v]) {
                out[u].push_back(v);
                in[v].push_back(u);
            } else if (dist[v] + 1 == dist[u]) {
                out[v].push_back(u);
                in[u].push_back(v);
            }
        }

        vector<unsigned char> forward(N + 1, 0);
        vector<unsigned char> backward(N + 1, 0);
        vector<unsigned char> good(N + 1, 0);

        auto processBounded = [&](int start, int finish) -> bool {
            int lo = dist[start], hi = dist[finish];

            for (int v : layers[lo]) forward[v] = 0;
            forward[start] = 1;

            for (int d = lo + 1; d <= hi; d++) {
                for (int v : layers[d]) {
                    bool reachable = false;
                    for (int u : in[v]) {
                        if (forward[u]) {
                            reachable = true;
                            break;
                        }
                    }
                    forward[v] = reachable;
                }
            }

            if (!forward[finish]) return false;

            for (int v : layers[hi]) backward[v] = 0;
            backward[finish] = 1;

            for (int d = hi - 1; d >= lo; d--) {
                for (int v : layers[d]) {
                    bool reachable = false;
                    for (int u : out[v]) {
                        if (backward[u]) {
                            reachable = true;
                            break;
                        }
                    }
                    backward[v] = reachable;
                }
            }

            for (int d = lo; d <= hi; d++) {
                for (int v : layers[d]) {
                    if (forward[v] && backward[v]) good[v] = 1;
                }
            }
            return true;
        };

        auto processFinal = [&](int start) -> bool {
            int lo = dist[start];

            for (int v : layers[lo]) forward[v] = 0;
            forward[start] = 1;

            for (int d = lo + 1; d <= maxDist; d++) {
                for (int v : layers[d]) {
                    bool reachable = false;
                    for (int u : in[v]) {
                        if (forward[u]) {
                            reachable = true;
                            break;
                        }
                    }
                    forward[v] = reachable;
                }
            }

            for (int d = lo; d <= maxDist; d++) {
                for (int v : layers[d]) backward[v] = 0;
            }
            for (int x : destinations) {
                if (dist[x] >= lo) backward[x] = 1;
            }

            for (int d = maxDist; d >= lo; d--) {
                for (int v : layers[d]) {
                    for (int u : out[v]) {
                        if (backward[u]) {
                            backward[v] = 1;
                            break;
                        }
                    }
                }
            }

            if (!backward[start]) return false;

            for (int d = lo; d <= maxDist; d++) {
                for (int v : layers[d]) {
                    if (forward[v] && backward[v]) good[v] = 1;
                }
            }
            return true;
        };

        if (K == 0) {
            possible = processFinal(1);
        } else {
            possible = processBounded(1, flowers[0]);
            for (int i = 0; possible && i + 1 < K; i++) {
                possible = processBounded(flowers[i], flowers[i + 1]);
            }
            if (possible) possible = processFinal(flowers.back());
        }

        if (!possible) {
            cout << string(N - 1, '0') << '\n';
            continue;
        }

        for (int x : flowers) good[x] = 1;

        string answer;
        answer.reserve(N - 1);
        for (int v = 2; v <= N; v++) {
            answer.push_back(good[v] ? '1' : '0');
        }
        cout << answer << '\n';
    }

    return 0;
}

            return dist[a] < dist[b];

        for (int x : flowers) required.push_back(x);

        bool possible = canReachDestination[required.back()];

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N, M, K, L;
        cin >> N >> M >> K >> L;

        vector<int> S(K), D(L);
        for (int i = 0; i < K; i++) cin >> S[i];
        for (int i = 0; i < L; i++) cin >> D[i];

        vector<vector<int>> adj(N + 1);
        for (int i = 0; i < M; i++) {
            int u, v;
            cin >> u >> v;
            adj[u].push_back(v);
            adj[v].push_back(u);
        }

        vector<int> dist(N + 1, -1);
        vector<int> order;
        order.reserve(N);

        queue<int> q;
        dist[1] = 0;
        q.push(1);

        while (!q.empty()) {
            int v = q.front();
            q.pop();
            order.push_back(v);

            for (int to : adj[v]) {
                if (dist[to] == -1) {
                    dist[to] = dist[v] + 1;
                    q.push(to);
                }
            }
        }

        vector<char> isDest(N + 1, false);
        for (int v : D) isDest[v] = true;

        sort(S.begin(), S.end(), [&](int a, int b) {
            if (dist[a] != dist[b]) return dist[a] < dist[b];
            return a < b;
        });

        vector<int> idxS(N + 1, 0);
        for (int i = 0; i < K; i++) {
            idxS[S[i]] = i + 1;
        }

        // maxLabel[v] = maximum index of an S vertex (or source as 0)
        // that can reach v in the shortest-path DAG.
        vector<int> maxLabel(N + 1, -1);
        vector<int> inMax(N + 1, -1);
        maxLabel[1] = 0;

        for (int v : order) {
            inMax[v] = maxLabel[v];

            if (idxS[v] > 0) {
                maxLabel[v] = max(maxLabel[v], idxS[v]);
            }

            for (int to : adj[v]) {
                if (dist[to] == dist[v] + 1) {
                    maxLabel[to] = max(maxLabel[to], maxLabel[v]);
                }
            }
        }

        bool valid = true;

        if (K > 0) {
            for (int i = 1; i < K; i++) {
                // S[i - 1] has label i, so it must be able to reach S[i].
                if (inMax[S[i]] < i) {
                    valid = false;
                    break;
                }
            }
        }

        if (!valid) {
            cout << string(N - 1, '0') << '\n';
            continue;
        }

        const int INF = 1e9;

        vector<int> minTo(N + 1, INF);
        vector<char> can(N + 1, false);

        for (int oi = (int)order.size() - 1; oi >= 0; oi--) {
            int v = order[oi];

            int bestMin = (idxS[v] > 0 ? idxS[v] : INF);
            bool reachDest = isDest[v];

            for (int to : adj[v]) {
                if (dist[to] == dist[v] + 1) {
                    bestMin = min(bestMin, minTo[to]);
                    if (can[to]) reachDest = true;
                }
            }

            minTo[v] = bestMin;
            can[v] = reachDest;
        }

        string ans;
        ans.reserve(N - 1);

        if (K == 0) {
            for (int f = 2; f <= N; f++) {
                ans.push_back(can[f] ? '1' : '0');
            }
            cout << ans << '\n';
            continue;
        }

        int B = S.back();

        if (!can[B]) {
            cout << string(N - 1, '0') << '\n';
            continue;
        }

        vector<int> cntDist(N + 1, 0);
        for (int v : S) {
            cntDist[dist[v]]++;
        }

        vector<int> lessCnt(N + 2, 0);
        int running = 0;

        for (int d = 0; d <= N; d++) {
            lessCnt[d] = running;
            running += cntDist[d];
        }

        int distB = dist[B];

        for (int f = 2; f <= N; f++) {
            if (idxS[f] > 0) {
                ans.push_back('1');
            } else if (dist[f] > distB) {
                if (maxLabel[f] >= K && can[f]) {
                    ans.push_back('1');
                } else {
                    ans.push_back('0');
                }
            } else {
                int i = lessCnt[dist[f]];

                if (maxLabel[f] >= i && minTo[f] <= i + 1) {
                    ans.push_back('1');
                } else {
                    ans.push_back('0');
                }
            }
        }

        cout << ans << '\n';
    }

    return 0;
}

struct FastScanner {

        return (unsigned char)buf[idx++];
    }
    int nextInt() {
        int c = gc();
        while (c != EOF && (c < '0' || c > '9')) c = gc();
        int x = 0;
        while (c >= '0' && c <= '9') { x = x * 10 + (c - '0'); c = gc(); }
        return x;
    }
};
