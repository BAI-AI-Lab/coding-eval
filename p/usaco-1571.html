题面

来源：USACO 官网。模型收到的是同一份题面的文字。

Bessie is trying to evade the farmers. The farmers own $N$ ($2 \le N \le 5 \cdot 10^5$) farms with a one way road between the $i$-th farm and the $a_i$-th farm ($1 \le i \le N$, $a_i \neq i$). There are $F$ ($1 \le F \le N$) farmers and the $i$-th farmer is initially stationed at farm $s_i$ ($1 \le s_i \le N$, all $s_i$ unique). At each time step, every farmer takes the road at their current farm and moves to the next. Bessie gets caught if she is ever located at the same farm as any farmer.

Suppose Bessie starts at some farm $b$. At each time step, she has two options: she can either choose to rest (stay at her current farm) or take the road and move to the next farm. If she chooses to move, she moves simultaneously with the farmers. Bessie moves so that she is never caught by any farmer in a finite number of timesteps.

For each starting farm $b$ ($1 \le b \le N$), find the maximum number of times Bessie can choose the rest option if she starts at farm $b$.

INPUT FORMAT (input arrives from the terminal / stdin):

The first line contains $N$ and $F$, the number of farms and the number of farmers.

The second line contains $a_1 \ldots a_N$, the one-way roads going out of each farm.

The third line contains $s_1 \ldots s_F$, the starting locations of each farmer.

OUTPUT FORMAT (print output to the terminal / stdout):

Output $N$ lines, the $b$-th of which must consist of a single integer denoting the maximum number of times Bessie can choose the rest option if she starts at farm $b$. If there is no way for Bessie to ensure she is never caught after a finite number of timesteps, then output $-1$. If Bessie can rest an infinite number of times, then output $-2$.

SAMPLE INPUT:

4 1
2 1 4 3
1

SAMPLE OUTPUT:

-1
0
-2
-2

  • Farm 1: If Bessie starts at a farm with a farmer, then she is caught immediately, and you should output $-1$.
  • Farm 2: Bessie must choose to move at every timestep to avoid being caught by the farmer who starts at farm $1$.
  • Farms 3-4: Bessie can rest an infinite number of times without being caught.

SCORING:

  • Input 2: $N \le 50$
  • Inputs 3-10: $N \le 2000$
  • Inputs 11-20: No additional constraints.

#include <bits/stdc++.h>
using namespace std;

struct Component {
    int len = 0;
    vector<int> safeDist;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, F;
    cin >> N >> F;

    vector<int> nxt(N);
    vector<vector<int>> radj(N);
    vector<int> indeg(N, 0);

    for (int i = 0; i < N; ++i) {
        cin >> nxt[i];
        --nxt[i];
        radj[nxt[i]].push_back(i);
        ++indeg[nxt[i]];
    }

    vector<int> farmers(F);
    for (int &s : farmers) {
        cin >> s;
        --s;
    }

    // Earliest time at which any farmer visits each farm.
    vector<int> dist(N, -1);
    vector<int> queue(N);
    int qh = 0, qt = 0;

    for (int s : farmers) {
        dist[s] = 0;
        queue[qt++] = s;
    }

    while (qh < qt) {
        int v = queue[qh++];
        int u = nxt[v];
        if (dist[u] == -1) {
            dist[u] = dist[v] + 1;
            queue[qt++] = u;
        }
    }

    // Remove all non-cycle vertices; remaining vertices are exactly cycle vertices.
    qh = qt = 0;
    for (int i = 0; i < N; ++i) {
        if (indeg[i] == 0) queue[qt++] = i;
    }

    while (qh < qt) {
        int v = queue[qh++];
        int u = nxt[v];
        if (--indeg[u] == 0) queue[qt++] = u;
    }

    vector<int> comp(N, -1), potential(N, 0);
    vector<Component> components;

    // Assign cycle positions. potential increases by 1 modulo the cycle length
    // whenever any vertex moves along its outgoing edge.
    for (int i = 0; i < N; ++i) {
        if (indeg[i] > 0 && comp[i] == -1) {
            int cid = (int)components.size();
            components.push_back(Component());

            int v = i;
            int pos = 0;
            do {
                comp[v] = cid;
                potential[v] = pos++;
                v = nxt[v];
            } while (v != i);

            components[cid].len = pos;
        }
    }

    // Propagate component IDs and potentials into the trees feeding the cycles.
    vector<int> st;
    for (int i = 0; i < N; ++i) {
        if (indeg[i] <= 0) continue;

        st.push_back(i);
        while (!st.empty()) {
            int v = st.back();
            st.pop_back();

            int cid = comp[v];
            int L = components[cid].len;

            for (int u : radj[v]) {
                if (comp[u] != -1) continue;

                comp[u] = cid;
                potential[u] = potential[v] - 1;
                if (potential[u] < 0) potential[u] += L;
                st.push_back(u);
            }
        }
    }

    // Phases occupied by farmers are unsafe for Bessie's eventual motion.
    for (auto &c : components) c.safeDist.assign(c.len, 0);

    for (int s : farmers) {
        Component &c = components[comp[s]];
        c.safeDist[potential[s]] = -1;
    }

    const int INF = 1e9;

    // For every phase, find the distance forward to the nearest safe phase.
    for (auto &c : components) {
        int safePhase = -1;
        for (int p = 0; p < c.len; ++p) {
            if (c.safeDist[p] == 0) {
                safePhase = p;
                break;
            }
        }

        if (safePhase == -1) {
            fill(c.safeDist.begin(), c.safeDist.end(), INF);
            continue;
        }

        int d = 0;
        for (int step = 1; step < c.len; ++step) {
            int p = safePhase - step;
            if (p < 0) p += c.len;

            if (c.safeDist[p] == 0) {
                d = 0;
            } else {
                c.safeDist[p] = ++d;
            }
        }
    }

    for (int b = 0; b < N; ++b) {
        if (dist[b] == -1) {
            cout << -2 << '\n';
            continue;
        }

        if (dist[b] == 0) {
            cout << -1 << '\n';
            continue;
        }

        int r = dist[b] - 1;
        Component &c = components[comp[b]];

        int phase = potential[b] - (r % c.len);
        if (phase < 0) phase += c.len;

        int delta = c.safeDist[phase];
        if (delta > r) {
            cout << -1 << '\n';
        } else {
            cout << r - delta << '\n';
        }
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, F;
    cin >> N >> F;

    vector<int> nxtNode(N);
    vector<vector<int>> rev(N);
    vector<int> indeg(N, 0);

    for (int i = 0; i < N; ++i) {
        cin >> nxtNode[i];
        --nxtNode[i];
        rev[nxtNode[i]].push_back(i);
        ++indeg[nxtNode[i]];
    }

    vector<int> farmers(F);
    for (int &s : farmers) {
        cin >> s;
        --s;
    }

    // First visit time of any farmer, by multi-source BFS on reversed edges.
    vector<int> dist(N, -1);
    vector<int> bfs;
    bfs.reserve(N);
    for (int s : farmers) {
        if (dist[s] == -1) {
            dist[s] = 0;
            bfs.push_back(s);
        }
    }
    for (size_t head = 0; head < bfs.size(); ++head) {
        int v = bfs[head];
        for (int u : rev[v]) {
            if (dist[u] == -1) {
                dist[u] = dist[v] + 1;
                bfs.push_back(u);
            }
        }
    }

    // Remove trees, leaving precisely the cycle vertices.
    vector<int> q;
    q.reserve(N);
    for (int i = 0; i < N; ++i) {
        if (indeg[i] == 0) q.push_back(i);
    }

    vector<int> removalOrder;
    removalOrder.reserve(N);
    for (size_t head = 0; head < q.size(); ++head) {
        int u = q[head];
        removalOrder.push_back(u);
        int v = nxtNode[u];
        if (--indeg[v] == 0) q.push_back(v);
    }

    vector<int> comp(N, -1), pos(N, -1), depth(N, -1), root(N, -1);
    vector<int> cycleLen;
    int C = 0;

    for (int i = 0; i < N; ++i) {
        if (indeg[i] > 0 && comp[i] == -1) {
            int v = i;
            int len = 0;
            do {
                comp[v] = C;
                pos[v] = len++;
                depth[v] = 0;
                root[v] = v;
                v = nxtNode[v];
            } while (comp[v] == -1);
            cycleLen.push_back(len);
            ++C;
        }
    }

    for (auto it = removalOrder.rbegin(); it != removalOrder.rend(); ++it) {
        int u = *it;
        int v = nxtNode[u];
        comp[u] = comp[v];
        root[u] = root[v];
        depth[u] = depth[v] + 1;
    }

    auto phase = [&](int v) {
        int c = comp[v];
        int L = cycleLen[c];
        int x = pos[root[v]] - depth[v] % L;
        if (x < 0) x += L;
        return x;
    };

    // A delay is eventually safe exactly when its resulting cycle phase
    // is not occupied by any farmer's phase.
    vector<vector<char>> bad(C);
    for (int c = 0; c < C; ++c) {
        bad[c].assign(cycleLen[c], 0);
    }
    for (int s : farmers) {
        bad[comp[s]][phase(s)] = 1;
    }

    // jump[p] = smallest x >= 0 such that phase p+x is safe.
    vector<vector<int>> jump(C);
    const int INF = 1000000000;

    for (int c = 0; c < C; ++c) {
        int L = cycleLen[c];
        jump[c].resize(L);

        bool hasSafe = false;
        for (int p = 0; p < L; ++p) {
            if (!bad[c][p]) {
                hasSafe = true;
                break;
            }
        }

        if (!hasSafe) {
            fill(jump[c].begin(), jump[c].end(), INF);
            continue;
        }

        int cur = L;
        for (int i = 2 * L - 1; i >= 0; --i) {
            int p = i % L;
            if (!bad[c][p]) cur = 0;
            else ++cur;
            if (i < L) jump[c][p] = cur;
        }
    }

    for (int b = 0; b < N; ++b) {
        if (dist[b] == -1) {
            cout << -2 << '\n';
            continue;
        }
        if (dist[b] == 0) {
            cout << -1 << '\n';
            continue;
        }

        int T = dist[b];
        int c = comp[b];
        int L = cycleLen[c];

        int p = phase(b) - (T - 1) % L;
        if (p < 0) p += L;

        int x = jump[c][p];
        if (x < T) cout << T - 1 - x << '\n';
        else cout << -1 << '\n';
    }

    return 0;
}

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, F;
    cin >> N >> F;

    vector<int> a(N);
    vector<int> head(N, -1), to(N), nxt(N);
    vector<int> indeg(N, 0);

    for (int i = 0; i < N; ++i) {
        cin >> a[i];
        --a[i];
        ++indeg[a[i]];
        to[i] = i;
        nxt[i] = head[a[i]];
        head[a[i]] = i;
    }

    vector<int> farmers(F);
    for (int &s : farmers) {
        cin >> s;
        --s;
    }

    // Remove all non-cycle nodes, recording the removal order.
    vector<int> order;
    order.reserve(N);
    queue<int> q;
    for (int i = 0; i < N; ++i) {
        if (indeg[i] == 0) q.push(i);
    }

    while (!q.empty()) {
        int u = q.front();
        q.pop();
        order.push_back(u);
        int v = a[u];
        if (--indeg[v] == 0) q.push(v);
    }

    vector<int> comp(N, -1), height(N, 0), cyclePos(N, 0);
    vector<int> cycleBase, cycleLen;

    int cycleNodes = 0;
    for (int i = 0; i < N; ++i) {
        if (indeg[i] > 0 && comp[i] == -1) {
            int id = (int)cycleLen.size();
            cycleBase.push_back(cycleNodes);

            int v = i, len = 0;
            do {
                comp[v] = id;
                height[v] = 0;
                cyclePos[v] = len++;
                ++cycleNodes;
                v = a[v];
            } while (v != i);

            cycleLen.push_back(len);
        }
    }

    // Propagate cycle information from each node's successor.
    for (int i = (int)order.size() - 1; i >= 0; --i) {
        int u = order[i], v = a[u];
        comp[u] = comp[v];
        height[u] = height[v] + 1;
        cyclePos[u] = cyclePos[v];
    }

    vector<char> blocked(cycleNodes, 0);
    for (int s : farmers) {
        int c = comp[s], L = cycleLen[c];
        int k = (height[s] - cyclePos[s]) % L;
        if (k < 0) k += L;
        blocked[cycleBase[c] + k] = 1;
    }

    // prevSafe[r] is the greatest safe residue at most r.
    vector<int> prevSafe(cycleNodes, -1);
    vector<int> maxSafe(cycleLen.size(), -1);

    for (int c = 0; c < (int)cycleLen.size(); ++c) {
        int last = -1, base = cycleBase[c];
        for (int r = 0; r < cycleLen[c]; ++r) {
            if (!blocked[base + r]) last = r;
            prevSafe[base + r] = last;
        }
        maxSafe[c] = last;
    }

    // Earliest time at which any farmer reaches each farm.
    vector<int> dist(N, -1);
    queue<int> bfs;
    for (int s : farmers) {
        if (dist[s] == -1) {
            dist[s] = 0;
            bfs.push(s);
        }
    }

    while (!bfs.empty()) {
        int u = bfs.front();
        bfs.pop();
        for (int e = head[u]; e != -1; e = nxt[e]) {
            int v = to[e];
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                bfs.push(v);
            }
        }
    }

    for (int b = 0; b < N; ++b) {
        if (dist[b] == -1) {
            cout << -2 << '\n';
            continue;
        }
        if (dist[b] == 0) {
            cout << -1 << '\n';
            continue;
        }

        int c = comp[b], L = cycleLen[c], base = cycleBase[c];
        int x = dist[b] - 1; // Maximum possible number of rests.

        int shift = (cyclePos[b] - height[b]) % L;
        if (shift < 0) shift += L;

        int r = (x - shift) % L;
        if (r < 0) r += L;

        int answer = -1;
        int p = prevSafe[base + r];

        if (p != -1) {
            answer = x - (r - p);
        } else if (maxSafe[c] != -1) {
            answer = x - (r + L - maxSafe[c]);
        }

        if (answer < 0) answer = -1;
        cout << answer << '\n';
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, F;
    cin >> N >> F;

    vector<int> a(N);
    vector<int> indeg(N, 0);

    for (int i = 0; i < N; i++) {
        cin >> a[i];
        --a[i];
        indeg[a[i]]++;
    }

    vector<int> starts(F);
    vector<char> isFarmer(N, false);

    for (int i = 0; i < F; i++) {
        cin >> starts[i];
        --starts[i];
        isFarmer[starts[i]] = true;
    }

    // Peel trees to find cycles.
    queue<int> q;
    for (int i = 0; i < N; i++) {
        if (indeg[i] == 0) q.push(i);
    }

    vector<int> peel;
    peel.reserve(N);

    while (!q.empty()) {
        int v = q.front();
        q.pop();

        peel.push_back(v);

        int p = a[v];
        if (--indeg[p] == 0) {
            q.push(p);
        }
    }

    vector<int> comp(N, -1), depth(N, 0), cpos(N, 0), cyclePos(N, -1);
    vector<int> cycleLen;
    vector<vector<int>> cycles;

    int compCnt = 0;

    // Remaining vertices are cycle vertices.
    for (int i = 0; i < N; i++) {
        if (indeg[i] > 0 && comp[i] == -1) {
            vector<int> cyc;

            int v = i;
            do {
                comp[v] = compCnt;
                depth[v] = 0;
                cyclePos[v] = (int)cyc.size();
                cpos[v] = (int)cyc.size();
                cyc.push_back(v);
                v = a[v];
            } while (v != i);

            int L = (int)cyc.size();
            cycleLen.push_back(L);
            cycles.push_back(cyc);

            compCnt++;
        }
    }

    vector<int> x(N, 0);

    for (int id = 0; id < compCnt; id++) {
        int L = cycleLen[id];
        for (int v : cycles[id]) {
            x[v] = cpos[v] % L;
        }
    }

    // Process non-cycle vertices from near cycle outward.
    for (int idx = (int)peel.size() - 1; idx >= 0; idx--) {
        int v = peel[idx];
        int p = a[v];

        comp[v] = comp[p];
        depth[v] = depth[p] + 1;
        cpos[v] = cpos[p];

        int L = cycleLen[comp[v]];
        int val = (cpos[v] - depth[v]) % L;
        if (val < 0) val += L;
        x[v] = val;
    }

    const int INF = 1e9;

    // mn[v] = minimum depth of a farmer in the in-tree subtree of v.
    vector<int> mn(N, INF);

    for (int v : starts) {
        mn[v] = min(mn[v], depth[v]);
    }

    // Peel order goes from leaves toward cycles, so this aggregates upward.
    for (int v : peel) {
        int p = a[v];
        mn[p] = min(mn[p], mn[v]);
    }

    vector<char> compHasFarmer(compCnt, false);
    vector<vector<int>> farmVerts(compCnt);

    for (int v : starts) {
        int id = comp[v];
        compHasFarmer[id] = true;
        farmVerts[id].push_back(v);
    }

    vector<vector<int>> compVerts(compCnt);

    for (int v = 0; v < N; v++) {
        compVerts[comp[v]].push_back(v);
    }

    vector<int> ans(N, -1);
    const long long INFLL = (long long)4e18;

    for (int id = 0; id < compCnt; id++) {
        int L = cycleLen[id];

        if (!compHasFarmer[id]) {
            for (int v : compVerts[id]) {
                ans[v] = -2;
            }
            continue;
        }

        vector<char> present(L, false);

        for (int v : farmVerts[id]) {
            int xv = x[v] % L;
            if (xv < 0) xv += L;
            present[xv] = true;
        }

        bool allPresent = true;

        for (int i = 0; i < L; i++) {
            if (!present[i]) {
                allPresent = false;
                break;
            }
        }

        vector<int> nextSafe(L, -1);

        if (!allPresent) {
            int nextNot = 2 * L;

            for (int i = 2 * L - 1; i >= 0; i--) {
                int idx = i % L;

                if (!present[idx]) {
                    nextNot = i;
                }

                if (i < L) {
                    nextSafe[i] = nextNot - i;
                }
            }
        }

        // Earliest time a farmer can be at each cycle vertex.
        vector<int> D(L, INF);

        for (int v : farmVerts[id]) {
            int cp = cpos[v];
            int d = depth[v];

            if (d < D[cp]) {
                D[cp] = d;
            }
        }

        vector<int> pref(L, INF), suff(L + 1, INF);

        for (int i = 0; i < L; i++) {
            if (D[i] < INF) {
                pref[i] = D[i] - i;
            }
        }

        for (int i = 1; i < L; i++) {
            pref[i] = min(pref[i], pref[i - 1]);
        }

        for (int i = L - 1; i >= 0; i--) {
            suff[i] = suff[i + 1];

            if (D[i] < INF) {
                suff[i] = min(suff[i], D[i] - i);
            }
        }

        vector<long long> T(L, INFLL);

        for (int p = 0; p < L; p++) {
            long long best = INFLL;

            if (pref[p] < INF) {
                best = min(best, (long long)p + pref[p]);

        int cid = component[v];
        int L = components[cid].length;
        int off = components[cid].offset;

        int x = H - 1;
        int phase = (key[v] - x % L + L) % L;

        int nxt = nextFree[off + phase];
        if (nxt == -1) return -1;

        int delta = (nxt - off) - phase;
        if (delta < 0) delta += L;

        int answer = x - delta;
        return answer >= 0 ? answer : -1;
    };

    for (int v = 0; v < N; ++v) {
        if (depth[v] > 0) {
            // A tree farm is visited by a farmer exactly when that farmer
            // lies in its reverse subtree.
            if (minFarmerDepth[v] == INF) {
                cout << -2 << '\n';
                continue;
            }

            int H = minFarmerDepth[v] - depth[v];
            cout << boundedAnswer(v, H) << '\n';
        } else {
            int cid = component[v];

            if (!components[cid].hasFarmer) {
                cout << -2 << '\n';
                continue;
            }

            int H = dist[components[cid].offset + key[v]];
            cout << boundedAnswer(v, H) << '\n';
        }
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

static inline int mod(int x, int m) {
    x %= m;
    if (x < 0) x += m;
    return x;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, F;
    cin >> N >> F;

    vector<int> to(N);
    vector<int> indegree(N, 0);

    for (int v = 0; v < N; ++v) {
        cin >> to[v];
        --to[v];
        ++indegree[to[v]];
    }

    vector<int> farmers(F);
    for (int &s : farmers) {
        cin >> s;
        --s;
    }

    /*
       nearest[v] = minimum number of steps needed for any farmer
                    to reach v, starting from time 0.
       This is a multi-source BFS following the original outgoing edges.
    */
    vector<int> nearest(N, -1);
    vector<int> bfs;
    bfs.reserve(N);

    for (int s : farmers) {
        nearest[s] = 0;
        bfs.push_back(s);
    }

    for (size_t head = 0; head < bfs.size(); ++head) {
        int v = bfs[head];
        int u = to[v];

        if (nearest[u] == -1) {
            nearest[u] = nearest[v] + 1;
            bfs.push_back(u);
        }
    }

    /*
       Remove all non-cycle vertices.
       peel is ordered from leaves toward cycles, so processing it backwards
       processes each parent before its children.
    */
    vector<int> peel;
    peel.reserve(N);

    for (int v = 0; v < N; ++v) {
        if (indegree[v] == 0) {
            peel.push_back(v);
        }
    }

    for (size_t head = 0; head < peel.size(); ++head) {
        int v = peel[head];
        int u = to[v];

        if (--indegree[u] == 0) {
            peel.push_back(u);
        }
    }

    /*
       label[v] is defined modulo the cycle length of v's component:
           label[to[v]] = label[v] + 1 (mod cycle length).

       On cycle vertices, the labels are all distinct.
    */
    vector<int> component(N, -1);
    vector<int> label(N, 0);
    vector<int> cycleLength;
    vector<int> componentBase;

    int totalCycleLength = 0;

    for (int start = 0; start < N; ++start) {
        if (indegree[start] == 0 || component[start] != -1) {
            continue;
        }

        int cid = static_cast<int>(cycleLength.size());
        int v = start;
        int len = 0;

        do {
            component[v] = cid;
            label[v] = len++;
            v = to[v];
        } while (v != start);

        cycleLength.push_back(len);
        componentBase.push_back(totalCycleLength);
        totalCycleLength += len;
    }

    for (int i = static_cast<int>(peel.size()) - 1; i >= 0; --i) {
        int v = peel[i];
        int parent = to[v];
        int cid = component[parent];

        component[v] = cid;
        label[v] = mod(label[parent] - 1, cycleLength[cid]);
    }

    /*
       For each component, record which cycle labels belong to farmers.
       Labels are stored in one flat array to avoid many small allocations.
    */
    vector<unsigned char> occupied(totalCycleLength, 0);

    for (int s : farmers) {
        int cid = component[s];
        occupied[componentBase[cid] + label[s]] = 1;
    }

    /*
       runLength[p] is the number of consecutive occupied labels starting
       at p and proceeding cyclically upward. If every label is occupied,
       runLength[p] equals the cycle length for every p.
    */
    vector<int> runLength(totalCycleLength, 0);

    for (int cid = 0; cid < static_cast<int>(cycleLength.size()); ++cid) {
        int c = cycleLength[cid];
        int base = componentBase[cid];

        int gap = -1;
        for (int p = 0; p < c; ++p) {
            if (!occupied[base + p]) {
                gap = p;
                break;
            }
        }

        if (gap == -1) {
            for (int p = 0; p < c; ++p) {
                runLength[base + p] = c;
            }
        } else {
            int streak = 0;

            // Scan backward from a missing label. A missing label resets
            // the streak; an occupied label extends it by one.
            for (int step = 1; step < c; ++step) {
                int p = gap - step;
                if (p < 0) p += c;

                int idx = base + p;
                if (occupied[idx]) {
                    ++streak;
                    runLength[idx] = streak;
                } else {
                    streak = 0;
                    runLength[idx] = 0;
                }
            }
        }
    }

    string output;
    output.reserve(static_cast<size_t>(N) * 8);

    for (int b = 0; b < N; ++b) {
        int answer;

        if (nearest[b] == -1) {
            // No farmer ever reaches this farm, so Bessie can rest forever.
            answer = -2;
        } else {
            int limit = nearest[b] - 1;

            if (limit < 0) {
                // A farmer is initially at b.
                answer = -1;
            } else {
                int cid = component[b];
                int c = cycleLength[cid];
                int base = componentBase[cid];

                int x = limit % c;

                /*
                   A total of K rests is cycle-safe exactly when
                       label[b] - K (mod c)
                   is not a farmer's label.

                   First test K = limit, whose residue is x.
                */
                int farmerLabel = mod(label[b] - x, c);

                if (!occupied[base + farmerLabel]) {
                    answer = limit;
                } else {
                    int run = runLength[base + farmerLabel];

                    if (run == c) {
                        // Every residue is forbidden.
                        answer = -1;
                    } else {
                        /*
                           As K decreases from x, the required farmer label
                           increases from farmerLabel. The first safe residue
                           below x is therefore x - run.
                        */
                        int safeResidue = mod(x - run, c);

                        // Largest K <= limit with this safe residue.
                        int delta = mod(limit - safeResidue, c);
                        answer = limit - delta;

                        if (answer < 0) {
                            answer = -1;
                        }
                    }
                }
            }
        }

        output += to_string(answer);
        output.push_back('\n');
    }

    cout << output;
    return 0;
}
