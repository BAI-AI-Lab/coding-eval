const ll INF = (1LL << 62);
const int BLOCK = 16;

struct Func {
    // Values are defined for k = lo, ..., lo + inc.size().
    // value(lo) = base, and inc contains successive differences.
    int lo = -1;
    ll base = 0;
    vector<ll> inc;

    bool empty() const { return lo == -1; }
};

using Matrix = array<Func, 9>;

static Func singletonFunc(int k, ll value) {
    Func f;
    f.lo = k;
    f.base = value;
    return f;
}

static Func convolve(const Func& f, const Func& g) {
    if (f.empty() || g.empty()) return Func();

    Func r;
    r.lo = f.lo + g.lo;
    r.base = f.base + g.base;
    r.inc.resize(f.inc.size() + g.inc.size());
    merge(f.inc.begin(), f.inc.end(),
          g.inc.begin(), g.inc.end(), r.inc.begin());
    return r;
}

static Func pointwiseMin(Func* fs, int n) {
    if (n == 0) return Func();
    if (n == 1) return std::move(fs[0]);

    int lo = INT_MAX, hi = INT_MIN;
    for (int i = 0; i < n; i++) {
        lo = min(lo, fs[i].lo);
        hi = max(hi, fs[i].lo + (int)fs[i].inc.size());
    }

    vector<ll> values(hi - lo + 1, INF);
    for (int i = 0; i < n; i++) {
        const Func& f = fs[i];
        ll value = f.base;
        int at = f.lo - lo;
        for (int j = 0; j <= (int)f.inc.size(); j++) {
            values[at + j] = min(values[at + j], value);
            if (j < (int)f.inc.size()) value += f.inc[j];
        }
    }

    Func r;
    r.lo = lo;
    r.base = values[0];
    r.inc.resize(values.size() - 1);
    for (int i = 1; i < (int)values.size(); i++) {
        r.inc[i - 1] = values[i] - values[i - 1];
    }
    return r;
}

static Matrix combine(const Matrix& a, const Matrix& b, int states) {
    Matrix c;
    for (int leftState = 0; leftState < states; leftState++) {
        for (int rightState = 0; rightState < states; rightState++) {
            Func tmp[3];
            int cnt = 0;
            for (int midState = 0; midState < states; midState++) {
                const Func& f = a[leftState * states + midState];
                const Func& g = b[midState * states + rightState];
                if (!f.empty() && !g.empty()) {
                    tmp[cnt++] = convolve(f, g);
                }
            }
            if (cnt) {
                c[leftState * states + rightState] = pointwiseMin(tmp, cnt);
            }
        }
    }
    return c;
}

static Matrix makeBlock(const vector<ll>& weight, int l, int r, int states) {
    Matrix result;
    int len = r - l;
    int maxK = (len + states - 1) / states;

    ll dp[3][BLOCK + 1], ndp[3][BLOCK + 1];

    for (int initial = 0; initial < states; initial++) {
        for (int s = 0; s < states; s++) {
            for (int k = 0; k <= maxK; k++) dp[s][k] = INF;
        }
        dp[initial][0] = 0;

        for (int pos = l; pos < r; pos++) {
            for (int s = 0; s < states; s++) {
                for (int k = 0; k <= maxK; k++) ndp[s][k] = INF;
            }

            for (int s = 0; s < states; s++) {
                for (int k = 0; k <= maxK; k++) {
                    ll cur = dp[s][k];
                    if (cur == INF) continue;

                    int skipState = (s == 0 ? 0 : s - 1);
                    ndp[skipState][k] = min(ndp[skipState][k], cur);

                    if (s == 0 && k < maxK) {
                        ndp[states - 1][k + 1] =
                            min(ndp[states - 1][k + 1], cur + weight[pos]);
                    }
                }
            }

            for (int s = 0; s < states; s++) {
                for (int k = 0; k <= maxK; k++) dp[s][k] = ndp[s][k];
            }
        }

        for (int ending = 0; ending < states; ending++) {
            int lo = 0;
            while (lo <= maxK && dp[ending][lo] == INF) lo++;
            if (lo > maxK) continue;

            int hi = maxK;
            while (dp[ending][hi] == INF) hi--;

            Func f;
            f.lo = lo;
            f.base = dp[ending][lo];
            for (int k = lo + 1; k <= hi; k++) {
                f.inc.push_back(dp[ending][k] - dp[ending][k - 1]);
            }
            result[initial * states + ending] = std::move(f);
        }
    }
    return result;
}

static Matrix buildDP(const vector<ll>& weight, int l, int r, int states) {
    if (r - l <= BLOCK) return makeBlock(weight, l, r, states);
    int mid = (l + r) / 2;
    Matrix left = buildDP(weight, l, mid, states);
    Matrix right = buildDP(weight, mid, r, states);
    return combine(left, right, states);
}

const int BASE = 24;

int L, N;

struct Funcs {
    vector<ll> f[9];
};

// Inf-convolution of two convex, nondecreasing cost functions.
// Their slopes are merged in increasing order.
vector<ll> convolve(const vector<ll>& a, const vector<ll>& b) {
    if (a.size() == 1) return b;
    if (b.size() == 1) return a;

    vector<ll> res(a.size() + b.size() - 1);
    res[0] = 0;

    size_t i = 1, j = 1;
    for (size_t pos = 1; pos < res.size(); ++pos) {
        ll slope;
        if (i < a.size() &&
            (j == b.size() || a[i] - a[i - 1] <= b[j] - b[j - 1])) {
            slope = a[i] - a[i - 1];
            ++i;
        } else {
            slope = b[j] - b[j - 1];
            ++j;
        }
        res[pos] = res[pos - 1] + slope;
    }
    return res;
}

Funcs brute(int l, int r) {
    Funcs out;

    for (int a = 0; a < L; ++a) {
        for (int b = 0; b < L; ++b) {
            int A = l + a;
            int B = r - b;
            int m = B - A;

            if (m <= 0) {
                out.f[a * L + b] = {0};
                continue;
            }

            int K = m / L;
            ll dp[BASE + 1][BASE + 1];

            for (int i = 0; i <= m; ++i)
                for (int k = 0; k <= K; ++k)
                    dp[i][k] = INF;

            dp[0][0] = 0;

            for (int i = 0; i < m; ++i) {
                for (int k = 0; k <= K; ++k) {
                    if (dp[i][k] == INF) continue;

                    dp[i + 1][k] = min(dp[i + 1][k], dp[i][k]);

                    if (i + L <= m && k < K) {
                        dp[i + L][k + 1] = min(
                            dp[i + L][k + 1],
                            dp[i][k] + weight[A + i]
                        );
                    }
                }
            }

            out.f[a * L + b] =
                vector<ll>(dp[m], dp[m] + K + 1);
        }
    }

    return out;
}

Funcs solve(int l, int r) {
    if (r - l <= BASE) return brute(l, r);

    int mid = (l + r) / 2;
    Funcs left = solve(l, mid);
    Funcs right = solve(mid, r);
    Funcs out;

    for (int a = 0; a < L; ++a) {
        for (int b = 0; b < L; ++b) {
            int A = l + a;
            int B = r - b;
            int K = (B - A) / L;

            // No selected moo crosses mid.
            auto best = convolve(
                left.f[a * L],
                right.f[b]
            );

            if ((int)best.size() < K + 1)
                best.resize(K + 1, INF);

            // Try the unique moo crossing mid, if one is selected.
            for (int p = 1; p < L; ++p) {
                int start = mid - p;
                int q = L - p;

                if (start < A || start + L > B) continue;

                auto cur = convolve(
                    left.f[a * L + p],
                    right.f[q * L + b]
                );

                for (size_t j = 0; j < cur.size(); ++j) {
                    best[j + 1] = min(
                        best[j + 1],
                        cur[j] + weight[start]
                    );
                }
            }

            out.f[a * L + b] = move(best);
        }
    }

    return out;
}

struct Poly {
    int lo = 0;
    vector<ll> v;

    bool empty() const { return v.empty(); }
    int hi() const { return lo + (int)v.size() - 1; }

    ll get(int k) const {
        if (v.empty() || k < lo || k > hi()) return INF;
        return v[k - lo];
    }
};

Poly convolve(const Poly &a, const Poly &b) {
    Poly c;
    c.lo = a.lo + b.lo;

    if (a.v.size() == 1) {
        c.v.resize(b.v.size());
        for (int i = 0; i < (int)b.v.size(); i++)
            c.v[i] = b.v[i] + a.v[0];
        return c;
    }
    if (b.v.size() == 1) {
        c.v.resize(a.v.size());
        for (int i = 0; i < (int)a.v.size(); i++)
            c.v[i] = a.v[i] + b.v[0];
        return c;
    }

    int sa = (int)a.v.size(), sb = (int)b.v.size();
    c.v.resize(sa + sb - 1);

    ll cur = a.v[0] + b.v[0];
    c.v[0] = cur;

    int i = 1, j = 1, out = 1;
    while (i < sa || j < sb) {
        ll da = i < sa ? a.v[i] - a.v[i - 1] : INF;
        ll db = j < sb ? b.v[j] - b.v[j - 1] : INF;

        if (da <= db) {
            cur += da;
            i++;
        } else {
            cur += db;
            j++;
        }
        c.v[out++] = cur;
    }
    return c;
}

Poly polyMin(Poly a, Poly b) {
    if (a.empty()) return b;
    if (b.empty()) return a;

    Poly c;
    c.lo = min(a.lo, b.lo);
    int hi = max(a.hi(), b.hi());
    c.v.resize(hi - c.lo + 1);

    for (int k = c.lo; k <= hi; k++)
        c.v[k - c.lo] = min(a.get(k), b.get(k));

    return c;
}

struct Matrix {

    const Poly &at(int i, int j) const { return p[i * 3 + j]; }
};

int L, S, M;

Matrix combine(const Matrix &a, const Matrix &b) {
    Matrix c;

    for (int i = 0; i < S; i++) {
        for (int j = 0; j < S; j++) {
            for (int mid = 0; mid < S; mid++) {
                const Poly &x = a.at(i, mid);
                const Poly &y = b.at(mid, j);
                if (x.empty() || y.empty()) continue;

                Poly z = convolve(x, y);
                c.at(i, j) = polyMin(std::move(c.at(i, j)), std::move(z));
            }
        }
    }
    return c;
}

A convex sequence is stored as:
  value(lo) = base
  value(k+1) - value(k) = diff[k - lo]

The differences are nondecreasing.
*/
struct Convex {
    int lo = 0;
    int hi = -1;
    ll base = 0;
    vector<ll> diff;

    bool valid() const {
        return hi >= lo;
    }

    static Convex fromDense(vector<ll> values) {
        Convex result;

        int lo = -1, hi = -1;
        for (int i = 0; i < (int)values.size(); ++i) {
            if (values[i] != INF) {
                if (lo == -1) lo = i;
                hi = i;
            }
        }

        if (lo == -1) return result;

        ll base = values[lo];
        int length = hi - lo;

        // Convert costs into successive differences, reusing the allocation.
        ll previous = base;
        for (int i = 0; i < length; ++i) {
            ll current = values[lo + i + 1];
            values[i] = current - previous;
            previous = current;
        }
        values.resize(length);

        result.lo = lo;
        result.hi = hi;
        result.base = base;
        result.diff = std::move(values);
        return result;
    }
};

// Stores values for consecutive feasible cardinalities.
// val[t] is the minimum cost for choosing lo + t intervals.
struct Poly {

struct Matrix {

// Transition for one possible starting position.
//
// State d means that the next d starting positions are forbidden.
// State 0 can either skip the position or choose it.
// Choosing it forbids the next L-1 starting positions.
template <int L>
Matrix<L> make_leaf(ll weight) {
    static_assert(L >= 2);

    Matrix<L> res;

    // Skip.
    res.p[0][0].lo = 0;
    res.p[0][0].val = {0};

    // Choose this starting position.
    res.p[0][L - 1].lo = 1;
    res.p[0][L - 1].val = {weight};

    // A positive blocked state must skip the current position.
    for (int d = 1; d < L; ++d) {
        res.p[d][d - 1].lo = 0;
        res.p[d][d - 1].val = {0};
    }

    return res;
}

// Min-plus polynomial matrix multiplication.
//
// Every coefficient sequence of a matrix entry is convex, so convolution
// of two such sequences is computed by merging their marginal costs.
template <int L>
Matrix<L> multiply(const Matrix<L>& A, const Matrix<L>& B) {
    Matrix<L> C;

    for (int i = 0; i < L; ++i) {
        for (int j = 0; j < L; ++j) {
            int low = INT_INF;
            int high = -1;

            for (int t = 0; t < L; ++t) {
                const Poly& x = A.p[i][t];
                const Poly& y = B.p[t][j];

                if (x.val.empty() || y.val.empty()) continue;

                int xHigh = x.lo + (int)x.val.size() - 1;
                int yHigh = y.lo + (int)y.val.size() - 1;

                low = min(low, x.lo + y.lo);
                high = max(high, xHigh + yHigh);
            }

            if (high == -1) continue;

            Poly& z = C.p[i][j];
            z.lo = low;
            z.val.assign(high - low + 1, INF);

            for (int t = 0; t < L; ++t) {
                const Poly& x = A.p[i][t];
                const Poly& y = B.p[t][j];

                if (x.val.empty() || y.val.empty()) continue;

                int nx = (int)x.val.size();
                int ny = (int)y.val.size();
                int base = x.lo + y.lo - z.lo;

                int ix = 0, iy = 0;
                ll current = x.val[0] + y.val[0];

                while (true) {
                    int index = base + ix + iy;
                    z.val[index] = min(z.val[index], current);

                    if (ix + 1 == nx && iy + 1 == ny) break;

                    ll dx = (ix + 1 < nx)
                                ? x.val[ix + 1] - x.val[ix]
                                : INF;
                    ll dy = (iy + 1 < ny)
                                ? y.val[iy + 1] - y.val[iy]
                                : INF;

                    if (dx <= dy) {
                        current += dx;
                        ++ix;
                    } else {
                        current += dy;
                        ++iy;
                    }
                }
            }
        }
    }

    return C;
}

using ll = long long;
static constexpr ll INF = (1LL << 62);
static constexpr int BASE = 32;

int patternLen;
vector<ll> windowCost;

/*
For a fixed boundary-state problem, val[k] is the minimum cost using
exactly lo + k selected windows.

Every such sequence is discretely convex, so its consecutive differences
are nondecreasing.
*/
struct Poly {
    int lo = 0;
    vector<ll> val;

    bool empty() const {
        return val.empty();
    }

    int hi() const {
        return lo + (int)val.size() - 1;
    }
};

struct Matrix {
    int s;
    array<array<Poly, 3>, 3> p;

    explicit Matrix(int s_ = 0) : s(s_) {}
};

/*
Min-plus convolution of two convex sequences.

If the differences of A and B are sorted, the differences of their
min-plus convolution are obtained by merging those two sorted lists.
*/
Poly minPlusConvolution(const Poly &A, const Poly &B) {
    int na = (int)A.val.size();
    int nb = (int)B.val.size();

    Poly C;
    C.lo = A.lo + B.lo;
    C.val.resize(na + nb - 1);

    if (na == 1) {
        for (int j = 0; j < nb; ++j) {
            C.val[j] = A.val[0] + B.val[j];
        }
        return C;
    }

    if (nb == 1) {
        for (int i = 0; i < na; ++i) {
            C.val[i] = A.val[i] + B.val[0];
        }
        return C;
    }

    C.val[0] = A.val[0] + B.val[0];

    int ia = 1, ib = 1, k = 1;
    while (k < (int)C.val.size()) {
        ll da = (ia < na ? A.val[ia] - A.val[ia - 1] : INF);
        ll db = (ib < nb ? B.val[ib] - B.val[ib - 1] : INF);

        if (da <= db) {
            C.val[k] = C.val[k - 1] + da;
            ++ia;
        } else {
            C.val[k] = C.val[k - 1] + db;
            ++ib;
        }
        ++k;
    }

    return C;
}

/*
Tropical matrix multiplication.

C[i][j] = min_h A[i][h] (*) B[h][j],
where (*) is min-plus convolution.
*/
Matrix multiply(const Matrix &A, const Matrix &B) {
    Matrix C(A.s);

    for (int i = 0; i < A.s; ++i) {
        for (int j = 0; j < A.s; ++j) {
            int low = INT_MAX;
            int high = INT_MIN;

            for (int h = 0; h < A.s; ++h) {
                if (A.p[i][h].empty() || B.p[h][j].empty()) continue;

                low = min(low, A.p[i][h].lo + B.p[h][j].lo);
                high = max(high, A.p[i][h].hi() + B.p[h][j].hi());
            }

            if (low == INT_MAX) continue;

            vector<ll> result(high - low + 1, INF);

            for (int h = 0; h < A.s; ++h) {
                if (A.p[i][h].empty() || B.p[h][j].empty()) continue;

                Poly q = minPlusConvolution(A.p[i][h], B.p[h][j]);
                int offset = q.lo - low;

                for (int x = 0; x < (int)q.val.size(); ++x) {
                    ll &dest = result[offset + x];
                    if (q.val[x] < dest) dest = q.val[x];
                }
            }

            int first = 0;
            while (first < (int)result.size() && result[first] == INF) {
                ++first;
            }
            if (first == (int)result.size()) continue;

            int last = (int)result.size() - 1;
            while (last >= first && result[last] == INF) {
                --last;
            }

            size_t newSize = (size_t)(last - first + 1);
            result.erase(result.begin(), result.begin() + first);
            result.resize(newSize);

            C.p[i][j].lo = low + first;
            C.p[i][j].val = move(result);
        }
    }

    return C;
}

/*
Build the transition product for a small block directly.

State q means that the next q candidate starting positions are blocked.
From state 0 we may skip or select the current window.
From state q > 0 we must skip, and the state decreases by one.
*/
Matrix buildBlock(int l, int r) {

    static constexpr int MAXC = BASE + 2;
    ll buffer1[3][MAXC];
    ll buffer2[3][MAXC];

    ll (*dp)[MAXC] = buffer1;
    ll (*ndp)[MAXC] = buffer2;

    for (int initial = 0; initial < patternLen; ++initial) {
        for (int q = 0; q < patternLen; ++q) {
            fill(dp[q], dp[q] + MAXC, INF);
        }
        dp[initial][0] = 0;

        for (int pos = l; pos <= r; ++pos) {
            for (int q = 0; q < patternLen; ++q) {
                fill(ndp[q], ndp[q] + MAXC, INF);
            }

            for (int q = 0; q < patternLen; ++q) {
                int nextState = (q == 0 ? 0 : q - 1);

                for (int cnt = 0; cnt <= BASE; ++cnt) {
                    if (dp[q][cnt] == INF) continue;

                    ll &skipDest = ndp[nextState][cnt];
                    skipDest = min(skipDest, dp[q][cnt]);

                    if (q == 0) {
                        ll &selectDest = ndp[patternLen - 1][cnt + 1];
                        selectDest = min(selectDest, dp[q][cnt] + windowCost[pos]);
                    }
                }
            }

            swap(dp, ndp);
        }

        for (int q = 0; q < patternLen; ++q) {

    Matrix right = solve(mid + 1, r);

        for (int k = 1; k <= N; ++k) {
            sum += makeM[k - 1];
            cout << sum << '\n';
        }
        return 0;
    }

    int M = N - patternLen + 1;
    windowCost.assign(M, 0);

    for (int i = 0; i < M; ++i) {
        ll cur = 0;

        if (s[i] == 'O') cur += costs[i];

        for (int j = 1; j < patternLen; ++j) {
            if (s[i + j] == 'M') cur += costs[i + j];
        }

        windowCost[i] = cur;
    }

    Matrix result = solve(0, M - 1);

    int maxK = N / patternLen;
    vector<ll> answer(maxK + 1, INF);

    // Any final cooldown state is acceptable.
    for (int state = 0; state < patternLen; ++state) {
        const Poly &p = result.p[0][state];
        for (int x = 0; x < (int)p.val.size(); ++x) {
            int k = p.lo + x;
            if (k <= maxK) {
                answer[k] = min(answer[k], p.val[x]);
            }
        }
    }

    for (int k = 1; k <= maxK; ++k) {
        cout << answer[k] << '\n';
    }

    return 0;
}
